1//! NFQUEUE: packets that the table's `queue` rules hold until the daemon decides.
2//!
3//! A [`Queue`] is bound to one queue number and is read by one thread. Packets are views into the receive buffer, and
4//! verdicts are written into a buffer of their own, so a steady stream of packets allocates nothing.
5

7use std::io::{self, Write as _};

17const NFQNL_MSG_PACKET: u16 = 0;
18const NFQNL_MSG_VERDICT: u16 = 1;
19const NFQNL_MSG_CONFIG: u16 = 2;

21const NFQA_CFG_CMD: u16 = 1;
22const NFQA_CFG_PARAMS: u16 = 2;
23const NFQA_CFG_QUEUE_MAXLEN: u16 = 3;
24const NFQA_CFG_MASK: u16 = 4;
25const NFQA_CFG_FLAGS: u16 = 5;
26const NFQA_CFG_F_FAIL_OPEN: u32 = 0x01;
27const NFQA_CFG_F_UID_GID: u32 = 0x08;
28const NFQNL_CFG_CMD_BIND: u8 = 1;
29const NFQNL_COPY_PACKET: u8 = 2;

33const NFQA_MARK: u16 = 3;
34const NFQA_IFINDEX_OUTDEV: u16 = 6;
35const NFQA_PAYLOAD: u16 = 10;
36const NFQA_UID: u16 = 16;
37const NFQA_GID: u16 = 17;

39const NF_DROP: u32 = 0;
40const NF_ACCEPT: u32 = 1;

42/// Enough of a packet for its IP and transport headers (and IPv6 extension headers in front of them).
43pub const HEADERS: u32 = 256;
44/// Packets the kernel holds for a queue before it drops new ones (never letting them through: see [`Queue::bind`]).
45const MAX_LENGTH: u32 = 4096;
46/// The socket's receive buffer. `SO_RCVBUF` is capped at `net.core.rmem_max`, so `SO_RCVBUFFORCE` is tried first.
47const RECEIVE_BUFFER: usize = 8 << 20;
48/// A datagram from the kernel: headers, small attributes, the copied part of the packet.
49const RECEIVE_CAPACITY: usize = 1 << 16;
50
51/// What happens to a held packet.
52#[derive(Debug, Clone, Copy, PartialEq, Eq)]
53pub enum Verdict {
54    Accept,
55    Drop,
56}
57
58/// A packet the kernel holds, borrowed from the queue's receive buffer.
59#[derive(Debug, Clone, Copy, PartialEq, Eq)]
60pub struct Packet<'a> {
61    /// The id verdicts refer to.
62    pub id: u32,
63    /// Ether type: 0x0800 IPv4, 0x86dd IPv6.
64    pub protocol: u16,
65    pub mark: u32,
66    pub out_interface: Option<u32>,
67    /// The owner of the sending socket, when the kernel knows it (`NFQA_CFG_F_UID_GID`).
68    pub uid: Option<u32>,
69    pub gid: Option<u32>,
70    /// The packet from its IP header on, as much of it as the queue copies.
71    pub payload: &'a [u8],
72}
73
74/// A queue bound to one queue number.
75#[derive(Debug)]
76pub struct Queue {
77    socket: RawSocket,
78    number: u16,
79    receive: Vec<u8>,
80    send: Writer,
81    sequence: u32,
82}
83
84impl Queue {
85    /// Binds queue `number`. When the queue is full the kernel drops new packets: overflow never lets traffic
86    /// through (`NFQA_CFG_F_FAIL_OPEN` is explicitly cleared), so flooding the queue cannot open the firewall.
87    pub fn bind(number: u16) -> io::Result<Self> {
88        Self::bind_copying(number, HEADERS)
89    }
90
91    /// Like [`Queue::bind`], copying up to `copy_range` bytes of each packet (for queues that read payloads).
92    pub fn bind_copying(number: u16, copy_range: u32) -> io::Result<Self> {
93        let mut socket = RawSocket::new(NETLINK_NETFILTER)?;
94        socket.bind_auto()?;
95        socket.connect(&SocketAddr::new(0, 0))?;
96        if socket.set_rx_buf_sz(RECEIVE_BUFFER).is_err() {
97            // Best effort: bursts beyond the default buffer are dropped by the kernel, as with a full queue.
98        }
99        let mut queue = Self {
100            socket,
101            number,
102            receive: Vec::with_capacity(RECEIVE_CAPACITY),
103            send: Writer::new(),
104            sequence: 0,
105        };
106        queue.configure(|w| {
107            // `struct nfqnl_msg_config_cmd`: command, padding, protocol family (ignored since Linux 3.8).
108            let [pf_high, pf_low] = u16::from(AF_UNSPEC).to_be_bytes();
109            w.attribute(NFQA_CFG_CMD, &[NFQNL_CFG_CMD_BIND, 0, pf_high, pf_low]);
110        })?;
111        queue.configure(|w| {
112            // `struct nfqnl_msg_config_params`: copy range, copy mode.
113            let [a, b, c, d] = copy_range.to_be_bytes();
114            w.attribute(NFQA_CFG_PARAMS, &[a, b, c, d, NFQNL_COPY_PACKET]);
115            w.u32_be(NFQA_CFG_QUEUE_MAXLEN, MAX_LENGTH);
116        })?;
117        queue.configure(|w| {
118            w.u32_be(NFQA_CFG_MASK, NFQA_CFG_F_FAIL_OPEN | NFQA_CFG_F_UID_GID);
119            w.u32_be(NFQA_CFG_FLAGS, NFQA_CFG_F_UID_GID);
120        })?;
121        Ok(queue)
122    }
123
124    #[must_use]
125    pub fn number(&self) -> u16 {
126        self.number
127    }
128
129    fn configure(&mut self, attributes: impl FnOnce(&mut Writer)) -> io::Result<()> {
130        self.sequence = self.sequence.wrapping_add(1);
131        self.send.clear();
132        let message = self.send.begin(
133            (NFNL_SUBSYS_QUEUE << 8) | NFQNL_MSG_CONFIG,
134            NLM_F_ACK,
135            self.sequence,
136            AF_UNSPEC,
137            self.number,
138        );
139        attributes(&mut self.send);
140        self.send.finish(message);
141        self.socket.send(self.send.bytes(), 0)?;
142        loop {
143            self.receive.clear();
144            self.socket.recv(&mut self.receive, 0)?;
145            for message in messages(&self.receive) {
146                if message.kind == NLMSG_ERROR && message.sequence == self.sequence {
147                    return match ack_error(message.body) {
148                        0 => Ok(()),
149                        code => Err(io::Error::from_raw_os_error(-code)),
150                    };
151                }
152            }
153        }
154    }
155
156    /// Blocks until the kernel sends packets, and returns them (often one, sometimes a burst).
157    pub fn receive(&mut self) -> io::Result<impl Iterator<Item = Packet<'_>>> {
158        self.receive.clear();
159        self.socket.recv(&mut self.receive, 0)?;
160        Ok(parse(&self.receive))
161    }
162
163    /// A handle that decides this queue's packets from another thread (packets held for a prompt). The kernel takes
164    /// verdicts only from the socket the queue is bound to; the handle writes to a duplicate of its descriptor, which
165    /// is the same socket.
166    pub fn verdict_writer(&self) -> io::Result<VerdictWriter> {
167        Ok(VerdictWriter {
168            socket: File::from(self.socket.as_fd().try_clone_to_owned()?),
169            number: self.number,
170            buffer: Writer::new(),
171        })
172    }
173
174    /// Decides what happens to a held packet.
175    pub fn verdict(&mut self, id: u32, verdict: Verdict) -> io::Result<()> {
176        encode_verdict(&mut self.send, self.number, id, verdict);
177        self.socket.send(self.send.bytes(), 0)?;
178        Ok(())
179    }
180}

182/// Decides a queue's packets from another thread: see [`Queue::verdict_writer`].
183#[derive(Debug)]
184pub struct VerdictWriter {
185    socket: File,
186    number: u16,
187    buffer: Writer,
188}
189
190impl VerdictWriter {
191    #[must_use]
192    pub fn number(&self) -> u16 {
193        self.number
194    }
195
196    pub fn verdict(&mut self, id: u32, verdict: Verdict) -> io::Result<()> {
197        encode_verdict(&mut self.buffer, self.number, id, verdict);
198        // One write is one datagram: one netlink message.
199        let written = (&self.socket).write(self.buffer.bytes())?;
200        if written == self.buffer.bytes().len() {
201            Ok(())
202        } else {
203            Err(io::Error::new(
204                io::ErrorKind::WriteZero,
205                "short netlink write",
206            ))
207        }
208    }
209}

211/// The packets in a received buffer.
212pub fn parse(buffer: &[u8]) -> impl Iterator<Item = Packet<'_>> {
213    messages(buffer)
214        .filter(|message| message.kind == (NFNL_SUBSYS_QUEUE << 8) | NFQNL_MSG_PACKET)
215        .filter_map(|message| {
216            let mut packet = Packet {
217                id: 0,
218                protocol: 0,
219                mark: 0,
220                out_interface: None,
221                uid: None,
222                gid: None,
223                payload: &[],
224            };
225            let mut has_header = false;
226            for (kind, value) in attributes(message.body.get(NFGENMSG_LEN..)?) {
227                match kind {
228                    // `struct nfqnl_msg_packet_hdr`: packet id, ether type, hook.
229                    NFQA_PACKET_HDR => {
230                        packet.id = be_u32(value)?;
231                        packet.protocol = u16::from_be_bytes(value.get(4..6)?.try_into().ok()?);
232                        has_header = true;
233                    }
234                    NFQA_MARK => packet.mark = be_u32(value)?,
235                    NFQA_IFINDEX_OUTDEV => packet.out_interface = be_u32(value),
236                    NFQA_UID => packet.uid = be_u32(value),
237                    NFQA_GID => packet.gid = be_u32(value),
238                    NFQA_PAYLOAD => packet.payload = value,
239                    _ => {}
240                }
241            }
242            has_header.then_some(packet)
243        })
244}

246/// Writes the verdict message for one packet.
247pub fn encode_verdict(writer: &mut Writer, queue: u16, id: u32, verdict: Verdict) {
248    writer.clear();
249    let message = writer.begin(
250        (NFNL_SUBSYS_QUEUE << 8) | NFQNL_MSG_VERDICT,
251        0,
252        0,
253        AF_UNSPEC,
254        queue,
255    );
256    let code = match verdict {
257        Verdict::Accept => NF_ACCEPT,
258        Verdict::Drop => NF_DROP,
259    };
260    // `struct nfqnl_msg_verdict_hdr`: verdict, packet id.
261    let [c0, c1, c2, c3] = code.to_be_bytes();
262    let [i0, i1, i2, i3] = id.to_be_bytes();
263    writer.attribute(NFQA_VERDICT_HDR, &[c0, c1, c2, c3, i0, i1, i2, i3]);
264    writer.finish(message);
265}
266
267fn be_u32(value: &[u8]) -> Option<u32> {
268    Some(u32::from_be_bytes(value.get(..4)?.try_into().ok()?))
269}

276    #[test]
277    fn verdicts_name_the_queue_packet_and_decision() {
278        let mut writer = Writer::new();
279        encode_verdict(&mut writer, 3, 0xabcd, Verdict::Drop);
280        let message = messages(writer.bytes()).next().unwrap();
281        assert_eq!(message.kind, (NFNL_SUBSYS_QUEUE << 8) | NFQNL_MSG_VERDICT);
282        assert_eq!(&message.body[2..4], &3_u16.to_be_bytes());
283        let (kind, header) = attributes(&message.body[NFGENMSG_LEN..]).next().unwrap();
284        assert_eq!(kind, NFQA_VERDICT_HDR);
285        assert_eq!(header, &[0, 0, 0, 0, 0, 0, 0xab, 0xcd]);
286    }

288    #[test]
289    fn parses_packets_from_a_kernel_message() {
290        let mut writer = Writer::new();
291        let message = writer.begin(
292            (NFNL_SUBSYS_QUEUE << 8) | NFQNL_MSG_PACKET,
293            0,
294            0,
295            AF_UNSPEC,
296            0,
297        );
298        writer.attribute(NFQA_PACKET_HDR, &[0, 0, 0, 7, 0x08, 0x00, 3, 0]);
299        writer.u32_be(NFQA_UID, 1001);
300        writer.u32_be(NFQA_IFINDEX_OUTDEV, 2);
301        writer.attribute(NFQA_PAYLOAD, &[0x45, 0, 0, 20]);
302        writer.finish(message);
303        let packets: Vec<_> = parse(writer.bytes()).collect();
304        assert_eq!(
305            packets,
306            vec![Packet {
307                id: 7,
308                protocol: 0x0800,
309                mark: 0,
310                out_interface: Some(2),
311                uid: Some(1001),
312                gid: None,
313                payload: &[0x45, 0, 0, 20],
314            }]
315        );
316    }
