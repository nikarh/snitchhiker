1//! Telling systemd the daemon is ready (`Type=notify`): the `sd_notify` datagram, without libsystemd.
2
3use std::os::linux::net::SocketAddrExt as _;
4use std::os::unix::net::{SocketAddr, UnixDatagram};
5
6/// Sends `state` (`READY=1`, `STOPPING=1`, ...) to the service manager, if it asked for notifications.
7pub fn notify(state: &str) {
8    let Some(path) = std::env::var_os("NOTIFY_SOCKET") else {
9        return;
10    };
11    let path = path.to_string_lossy().into_owned();
12    // A leading `@` is an abstract socket.
13    let address = match path.strip_prefix('@') {
14        Some(name) => SocketAddr::from_abstract_name(name.as_bytes()),
15        None => SocketAddr::from_pathname(&path),
16    };
17    if let (Ok(socket), Ok(address)) = (UnixDatagram::unbound(), address) {
18        let _ = socket.send_to_addr(state.as_bytes(), &address);
19    }
20}
