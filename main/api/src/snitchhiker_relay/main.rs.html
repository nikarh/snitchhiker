1//! `snitchhiker-relay`: the relay a user's nodes and devices meet through when no direct path exists (plan §8.4). It
2//! is iroh's relay server, configured from a TOML file: TLS by ACME, by certificate files, or none behind a reverse
3//! proxy that terminates it; optionally only listed endpoint ids may use it. Traffic through it stays end-to-end
4//! encrypted between node and device: the relay sees endpoint ids and opaque packets.
5

29/// How TLS is done.
30#[derive(Debug, Clone, Copy, Default, PartialEq, Eq, Deserialize)]
31#[serde(rename_all = "kebab-case")]
32enum Tls {
33    /// Plain HTTP: a reverse proxy in front terminates TLS (or a test).
34    #[default]
35    None,
36    /// Certificates from an ACME CA (Let's Encrypt), for `domains`.
37    Acme,
38    /// Certificates from files (`cert`, `key`).
39    Files,
40}
41
42#[derive(Debug, Clone, Deserialize)]
43#[serde(default, deny_unknown_fields)]
44struct Config {
45    /// Where plain HTTP is served (the relay protocol, and ACME's challenges).
46    http_bind: SocketAddr,
47    tls: Tls,
48    /// Where HTTPS is served, with TLS.
49    https_bind: SocketAddr,
50    /// QUIC address discovery (devices learn their public address), with TLS.
51    quic_bind: Option<SocketAddr>,
52    /// The relay's names, for ACME.
53    domains: Vec<String>,
54    /// An e-mail address for the ACME account.
55    contact: String,
56    /// Where ACME's account and certificates are kept.
57    cert_dir: PathBuf,
58    /// Let's Encrypt's staging CA instead of production (for trying the setup).
59    acme_staging: bool,
60    /// Certificate chain and key files (PEM), with `tls = "files"`.
61    cert: PathBuf,
62    key: PathBuf,
63    /// Endpoint ids (nodes and devices) that may use the relay; none: any.
64    allow: Vec<String>,
65    /// Prometheus metrics, if set.
66    metrics_bind: Option<SocketAddr>,
67}
68
69impl Default for Config {
70    fn default() -> Self {
71        Self {
72            http_bind: SocketAddr::from(([0; 16], 80)),
73            tls: Tls::None,
74            https_bind: SocketAddr::from(([0; 16], 443)),
75            quic_bind: None,
76            domains: Vec::new(),
77            contact: String::new(),
78            cert_dir: PathBuf::from("/var/lib/snitchhiker-relay/certs"),
79            acme_staging: false,
80            cert: PathBuf::new(),
81            key: PathBuf::new(),
82            allow: Vec::new(),
83            metrics_bind: None,
84        }
85    }
86}

88#[derive(Debug, thiserror::Error)]
89enum Error {
90    #[error(transparent)]
91    Args(#[from] pico_args::Error),
92    #[error("{0}")]
93    Config(String),
94    #[error("{0}")]
95    Relay(String),

98/// Only the listed endpoint ids.
99#[derive(Debug)]
100struct Allowed(Vec<String>);
101
102impl AccessControl for Allowed {
103    #[expect(
104        clippy::unused_async_trait_impl,
105        reason = "the trait's method is async"
106    )]
107    async fn on_connect(&self, request: &ClientRequest) -> Access {
108        let id = request.endpoint_id().to_string();
109        if self.0.contains(&id) {
110            Access::Allow
111        } else {
112            Access::Deny {
113                reason: Some("not allowed on this relay".to_owned()),
114            }
115        }
116    }
117}

119fn load(path: &Path) -> Result<Config, Error> {
120    let text = match std::fs::read_to_string(path) {
121        Ok(text) => text,
122        Err(e) if e.kind() == std::io::ErrorKind::NotFound => String::new(),
123        Err(e) => return Err(Error::Config(format!("{}: {e}", path.display()))),
124    };
125    toml::from_str(&text).map_err(|e| Error::Config(format!("{}: {e}", path.display())))
126}
127
128fn tls_server_config()
129-> Result<rustls::ConfigBuilder<rustls::ServerConfig, rustls::server::WantsServerCert>, Error> {
130    Ok(rustls::ServerConfig::builder_with_provider(Arc::new(
131        rustls::crypto::ring::default_provider(),
132    ))
133    .with_safe_default_protocol_versions()
134    .map_err(|e| Error::Config(e.to_string()))?
135    .with_no_client_auth())
136}
137
138fn cert_config(config: &Config) -> Result<Option<CertConfig>, Error> {
139    use rustls_pki_types::pem::PemObject as _;
140    Ok(match config.tls {
141        Tls::None => None,
142        Tls::Acme => {
143            if config.domains.is_empty() || config.contact.is_empty() {
144                return Err(Error::Config(
145                    "tls = \"acme\" needs domains and contact".to_owned(),
146                ));
147            }
148            let acme = AcmeConfig::letsencrypt(!config.acme_staging)
149                .domains(config.domains.clone())
150                .contact(vec![format!("mailto:{}", config.contact)])
151                .cache_path(config.cert_dir.clone());
152            Some(CertConfig::LetsEncrypt {
153                acme_config: acme,
154                server_config_builder: tls_server_config()?,
155            })
156        }
157        Tls::Files => {
158            let certs = rustls_pki_types::CertificateDer::pem_file_iter(&config.cert)
159                .and_then(Iterator::collect::<Result<Vec<_>, _>>)
160                .map_err(|e| Error::Config(format!("{}: {e}", config.cert.display())))?;
161            let key = rustls_pki_types::PrivateKeyDer::from_pem_file(&config.key)
162                .map_err(|e| Error::Config(format!("{}: {e}", config.key.display())))?;
163            let server_config = tls_server_config()?
164                .with_single_cert(certs, key)
165                .map_err(|e| Error::Config(e.to_string()))?;
166            Some(CertConfig::Manual { server_config })
167        }
168    })
169}

171async fn run(config: Config) -> Result<(), Error> {
172    let cert = cert_config(&config)?;
173    let mut relay_config = RelayConfig::new(config.http_bind);
174    if !config.allow.is_empty() {
175        relay_config.access = Arc::new(Allowed(config.allow.clone()));
176    }
177    let quic = match (&cert, config.quic_bind) {
178        (Some(CertConfig::Manual { server_config }), Some(bind)) => {
179            let mut quic = QuicConfig::new(bind);
180            quic.server_config = Some(server_config.clone());
181            Some(quic)
182        }
183        (_, Some(_)) => {
184            return Err(Error::Config(
185                "quic_bind needs tls = \"files\" (QUIC needs the certificate at start)".to_owned(),
186            ));
187        }
188        _ => None,
189    };
190    if let Some(cert) = cert {
191        relay_config.tls = Some(relay::TlsConfig::new(config.https_bind, cert));
192    }
193    let mut server_config = ServerConfig::default();
194    server_config.relay = Some(relay_config);
195    server_config.quic = quic;
196    server_config.metrics_addr = config.metrics_bind;
197    let mut server = relay::Server::spawn(server_config)
198        .await
199        .map_err(|e| Error::Relay(e.to_string()))?;
200    info!(
201        http = ?server.http_addr(),
202        https = ?server.https_addr(),
203        allowed = config.allow.len(),
204        "relay serving"
205    );
206    tokio::select! {
207        biased;
208        () = shutdown() => info!("stopping"),
209        _ = server.join() => error!("the relay stopped"),
210    }
211    server
212        .shutdown()
213        .await
214        .map_err(|e| Error::Relay(e.to_string()))
215}

217async fn shutdown() {
218    let Ok(mut term) = tokio::signal::unix::signal(tokio::signal::unix::SignalKind::terminate())
219    else {
220        return std::future::pending().await;
221    };
222    tokio::select! {
223        _ = term.recv() => {}
224        _ = tokio::signal::ctrl_c() => {}
225    }
226}

228fn main() -> ExitCode {
229    tracing_subscriber::fmt()
230        .with_env_filter(
231            tracing_subscriber::EnvFilter::try_from_default_env()
232                .unwrap_or_else(|_| tracing_subscriber::EnvFilter::new("info")),
233        )
234        .init();
235    let mut args = pico_args::Arguments::from_env();
236    if args.contains(["-h", "--help"]) {
237        let _ = std::io::Write::write_all(&mut std::io::stdout(), HELP.as_bytes());
238        return ExitCode::SUCCESS;
239    }
240    if args.contains(["-V", "--version"]) {
241        let _ = std::io::Write::write_all(
242            &mut std::io::stdout(),
243            format!("snitchhiker-relay {}\n", env!("CARGO_PKG_VERSION")).as_bytes(),
244        );
245        return ExitCode::SUCCESS;
246    }
247    let result = (|| -> Result<(), Error> {
248        let path: PathBuf = args
249            .opt_value_from_str("--config")?
250            .unwrap_or_else(|| PathBuf::from("/etc/snitchhiker/relay.toml"));
251        let config = load(&path)?;
252        let runtime = tokio::runtime::Builder::new_multi_thread()
253            .enable_all()
254            .build()
255            .map_err(|e| Error::Relay(e.to_string()))?;
256        runtime.block_on(run(config))
257    })();
258    match result {
259        Ok(()) => ExitCode::SUCCESS,
260        Err(error) => {
261            error!(%error, "snitchhiker-relay");
262            ExitCode::FAILURE
263        }
264    }
265}

271    #[test]
272    fn configuration_reads_and_refuses_unknown_keys() {
273        let config: Config = toml::from_str(
274            "tls = \"acme\"\ndomains = [\"relay.example.org\"]\ncontact = \"a@example.org\"\nallow = [\"x\"]\n",
275        )
276        .unwrap();
277        assert_eq!(config.tls, Tls::Acme);
278        assert_eq!(config.http_bind.port(), 80);
279        assert!(toml::from_str::<Config>("tsl = \"acme\"\n").is_err());
280        assert!(cert_config(&Config::default()).unwrap().is_none());
281    }
