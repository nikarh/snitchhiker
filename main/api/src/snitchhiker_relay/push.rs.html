1//! The push gateway (plan §8.4, §8.5): an iroh endpoint of its own, homed on this relay, that nodes keep a stream open
2//! to (ALPN `snitchhiker-push/1`). On it a node says who to tell (its devices' push targets), who to wake, and why it
3//! stops. The stream ending is the node going away; if it doesn't come back within its `offline_after_secs`, its
4//! devices are told it is offline, and that it is back when it returns.
5//!
6//! What the gateway keeps, in its state file: each node's name, threshold and devices' push targets, so a node that
7//! dies while the gateway restarts is still reported. Nothing about prompts, connections or rules ever reaches it.
8

26/// The `[push]` section of the relay's configuration.
27#[derive(Debug, Clone, Deserialize)]
28#[serde(default, deny_unknown_fields)]
29pub struct Config {
30    /// This relay's own URL, as nodes reach it (`https://relay.example.org/`): the gateway's endpoint is homed here.
31    pub url: String,
32    /// The gateway's key (created on first start): nodes name the gateway by its id (`push_gateway`).
33    pub key: PathBuf,
34    /// Where the nodes' registrations are kept.
35    pub state: PathBuf,
36    /// Node ids that may use the gateway; none: any.
37    pub allow: Vec<String>,
38    /// A Firebase service account key (JSON), to push to the Play build of the app; none: FCM targets are skipped.
39    pub fcm_service_account: Option<PathBuf>,
40    /// Where FCM is (for tests).
41    pub fcm_api: String,
42}
43
44impl Default for Config {
45    fn default() -> Self {
46        Self {
47            url: String::new(),
48            key: PathBuf::from("/var/lib/snitchhiker-relay/push.key"),
49            state: PathBuf::from("/var/lib/snitchhiker-relay/push.json"),
50            allow: Vec::new(),
51            fcm_service_account: None,
52            fcm_api: "https://fcm.googleapis.com".to_owned(),
53        }

57/// A node, as the gateway knows it.
58#[derive(Default)]
59struct Node {
60    registration: v1::GatewayRegister,
61    /// Its streams open now.
62    streams: u32,
63    /// Changes on each connection and disconnection: an offline check is for one disconnection only.
64    generation: u64,
65    /// What it said when it last stopped.
66    stopping: Option<String>,
67    /// Its devices were told it is offline.
68    told_offline: bool,
69}

71pub struct Gateway {

82fn lock<T>(mutex: &Mutex<T>) -> std::sync::MutexGuard<'_, T> {
83    mutex.lock().unwrap_or_else(PoisonError::into_inner)

91impl Gateway {
92    /// Starts the gateway: its endpoint, and the offline checks of the nodes it knew.
93    pub fn start(config: &Config, runtime: tokio::runtime::Handle) -> Result<Arc<Self>, String> {
94        if config.url.is_empty() {
95            return Err("[push] needs url: this relay's URL, as nodes reach it".to_owned());
96        }
97        let fcm = config
98            .fcm_service_account
99            .as_deref()
100            .map(|path| Fcm::load(path, &config.fcm_api).map(Arc::new))
101            .transpose()
102            .map_err(|e| e.to_string())?;
103        let http = reqwest::Client::builder()
104            .timeout(Duration::from_secs(20))
105            .build()
106            .map_err(|e| e.to_string())?;
107        let gateway = Arc::new(Self {
108            nodes: Mutex::new(load(&config.state)),
109            state: config.state.clone(),
110            allow: config.allow.clone(),
111            http,
112            fcm,
113            runtime,
114            listener: Mutex::new(None),
115        });
116        let key = Key::load_or_create(&config.key)
117            .map_err(|e| format!("{}: {e}", config.key.display()))?;
118        let options = Options {
119            relays: vec![config.url.clone()],
120            ..Options::default()
121        };
122        let weak = Arc::downgrade(&gateway);
123        let listener = Listener::bind_for(&key, &options, &[PUSH_ALPN], move |accepted| {
124            if let Some(gateway) = weak.upgrade() {
125                let _ = std::thread::Builder::new()
126                    .name("push-node".to_owned())
127                    .spawn(move || gateway.serve(&accepted));
128            }
129        })
130        .map_err(|e| e.to_string())?;
131        info!(id = %listener.id(), nodes = lock(&gateway.nodes).len(), "push gateway serving");
132        *lock(&gateway.listener) = Some(listener);
133        // The nodes known from before: offline unless they connect in time.
134        let known: Vec<(String, u64)> = lock(&gateway.nodes)
135            .iter()
136            .map(|(id, node)| (id.clone(), node.generation))
137            .collect();
138        for (id, generation) in known {
139            gateway.check_later(&id, generation);
140        }
141        Ok(gateway)
142    }
143
144    /// Closes the gateway's endpoint (outside any runtime: it waits for its own).
145    pub fn stop(&self) {
146        let listener = lock(&self.listener).take();
147        drop(listener);
148    }
149
150    fn serve(self: &Arc<Self>, accepted: &Accepted) {
151        let id = accepted.peer.clone();
152        if !self.allow.is_empty() && !self.allow.contains(&id) {
153            info!(node = %id, "refused a node that isn't allowed on the push gateway");
154            return;
155        }
156        let mut reader = BufReader::new(&accepted.stream);
157        if frame::read_preamble(&mut reader).is_err() {
158            return;
159        }
160        let mut buffer = Vec::new();
161        let mut connected = false;
162        while let Ok(frame) = frame::read::<v1::GatewayFrame>(&mut reader, &mut buffer) {
163            match frame.body {
164                Some(v1::gateway_frame::Body::Register(register)) => {
165                    if !connected {
166                        connected = true;
167                        self.connected(&id);
168                    }
169                    self.register(&id, register);
170                }
171                Some(v1::gateway_frame::Body::Wake(wake)) => {
172                    self.push(&id, &wake.devices, v1::PushKind::Wake, String::new());
173                }
174                Some(v1::gateway_frame::Body::Stopping(stopping)) => {
175                    debug!(node = %id, reason = %stopping.reason, "a node stops");
176                    if let Some(node) = lock(&self.nodes).get_mut(&id) {
177                        node.stopping = Some(stopping.reason);
178                    }
179                }
180                None => {}
181            }
182        }
183        if connected {
184            self.disconnected(&id);
185        }
186    }
187
188    fn connected(&self, id: &str) {
189        let back = {
190            let mut nodes = lock(&self.nodes);
191            let node = nodes.entry(id.to_owned()).or_default();
192            node.streams += 1;
193            node.generation += 1;
194            node.stopping = None;
195            std::mem::take(&mut node.told_offline)
196        };
197        info!(node = %id, "a node connected to the push gateway");
198        if back {
199            self.push_all(id, v1::PushKind::Online, String::new());
200        }
201    }
202
203    fn register(&self, id: &str, register: v1::GatewayRegister) {
204        {
205            let mut nodes = lock(&self.nodes);
206            if register.devices.is_empty() {
207                // Nobody to tell: the node is forgotten (until it registers devices again).
208                if let Some(node) = nodes.get_mut(id) {
209                    node.registration = v1::GatewayRegister::default();
210                }
211            } else if let Some(node) = nodes.get_mut(id) {
212                node.registration = register;
213            }
214        }
215        self.save();
216    }
217
218    fn disconnected(self: &Arc<Self>, id: &str) {
219        let generation = {
220            let mut nodes = lock(&self.nodes);
221            let Some(node) = nodes.get_mut(id) else {
222                return;
223            };
224            node.streams = node.streams.saturating_sub(1);
225            if node.streams > 0 {
226                return;
227            }
228            node.generation += 1;
229            node.generation
230        };
231        info!(node = %id, "a node left the push gateway");
232        self.check_later(id, generation);
233    }
234
235    /// After the node's threshold: if it is still gone (the same disconnection), its devices are told.
236    fn check_later(self: &Arc<Self>, id: &str, generation: u64) {
237        let after = lock(&self.nodes)
238            .get(id)
239            .map_or(120, |n| n.registration.offline_after_secs);
240        let after = Duration::from_secs(u64::from(if after == 0 { 120 } else { after }));
241        let gateway = Arc::clone(self);
242        let id = id.to_owned();
243        self.runtime.spawn(async move {
244            tokio::time::sleep(after).await;
245            let reason = {
246                let mut nodes = lock(&gateway.nodes);
247                let Some(node) = nodes.get_mut(&id) else {
248                    return;
249                };
250                if node.generation != generation || node.streams > 0 || node.told_offline {
251                    return;
252                }
253                node.told_offline = true;
254                node.stopping.clone().unwrap_or_default()
255            };
256            info!(node = %id, "a node is offline");
257            gateway.push_all(&id, v1::PushKind::Offline, reason);
258        });
259    }
260
261    fn push_all(&self, node: &str, kind: v1::PushKind, reason: String) {
262        let devices: Vec<String> = lock(&self.nodes).get(node).map_or_default(|n| {
263            n.registration
264                .devices
265                .iter()
266                .map(|d| d.id.clone())
267                .collect()
268        });
269        self.push(node, &devices, kind, reason);
270    }
271
272    /// Pushes to the node's `devices` (those it registered).
273    fn push(&self, node: &str, devices: &[String], kind: v1::PushKind, reason: String) {
274        let (name, targets) = {
275            let nodes = lock(&self.nodes);
276            let Some(known) = nodes.get(node) else {
277                return;
278            };
279            let targets: Vec<(String, v1::PushTarget)> = known
280                .registration
281                .devices
282                .iter()
283                .filter(|d| devices.contains(&d.id))
284                .filter_map(|d| Some((d.id.clone(), d.target.clone()?)))
285                .collect();
286            (known.registration.node_name.clone(), targets)
287        };
288        let payload = v1::PushMessage {
289            node: node.to_owned(),
290            kind: kind.into(),
291            reason,
292            node_name: name,
293        }
294        .encode_to_vec();
295        for (device, target) in targets {
296            let delivery = deliver(self.http.clone(), self.fcm.clone(), target, payload.clone());
297            self.runtime.spawn(async move {
298                match delivery.await {
299                    Ok(()) => debug!(%device, ?kind, "pushed"),
300                    Err(error) => warn!(%device, %error, "pushing to a device"),
301                }
302            });
303        }
304    }
305
306    fn save(&self) {
307        let saved: HashMap<String, String> = lock(&self.nodes)
308            .iter()
309            .filter(|(_, node)| !node.registration.devices.is_empty())
310            .map(|(id, node)| {
311                (
312                    id.clone(),
313                    STANDARD.encode(node.registration.encode_to_vec()),
314                )
315            })
316            .collect();
317        let written = serde_json::to_vec(&saved)
318            .map_err(std::io::Error::other)
319            .and_then(|bytes| write_private(&self.state, &bytes));
320        if let Err(error) = written {
321            warn!(%error, path = %self.state.display(), "saving the push registrations");
322        }
323    }
324}

326/// Sends one push: encrypted to a `UnifiedPush` endpoint, or through FCM.
327async fn deliver(
328    http: reqwest::Client,
329    fcm: Option<Arc<Fcm>>,
330    target: v1::PushTarget,
331    payload: Vec<u8>,
332) -> Result<(), String> {
333    match target.kind {
334        Some(v1::push_target::Kind::UnifiedPush(up)) => {
335            let body = webpush::encrypt(
336                &webpush::Keys {
337                    p256dh: &up.p256dh,
338                    auth: &up.auth,
339                },
340                &payload,
341            )
342            .map_err(|e| e.to_string())?;
343            http.post(&up.endpoint)
344                .header("content-encoding", "aes128gcm")
345                .header("content-type", "application/octet-stream")
346                // Kept a day by the push server for a device that is off; delivered at once to one that is on.
347                .header("ttl", "86400")
348                .header("urgency", "high")
349                .body(body)
350                .send()
351                .await
352                .and_then(reqwest::Response::error_for_status)
353                .map_err(|e| e.to_string())?;
354            Ok(())
355        }
356        Some(v1::push_target::Kind::Fcm(fcm_token)) => match fcm {
357            Some(fcm) => fcm
358                .send(&http, &fcm_token.token, &payload)
359                .await
360                .map_err(|e| e.to_string()),
361            None => Err("an FCM target, and no FCM service account".to_owned()),
362        },
363        None => Ok(()),
364    }
365}

367/// Writes a file only its owner reads, replacing the old one at once.
368fn write_private(path: &Path, bytes: &[u8]) -> std::io::Result<()> {
369    use std::io::Write as _;
370    use std::os::unix::fs::OpenOptionsExt as _;
371    let temporary = path.with_extension("tmp");
372    let mut file = std::fs::OpenOptions::new()
373        .write(true)
374        .create(true)
375        .truncate(true)
376        .mode(0o600)
377        .open(&temporary)?;
378    file.write_all(bytes)?;
379    file.sync_all()?;
380    std::fs::rename(&temporary, path)
381}

383fn load(path: &Path) -> HashMap<String, Node> {
384    let Ok(bytes) = std::fs::read(path) else {
385        return HashMap::new();
386    };
387    let saved: HashMap<String, String> = serde_json::from_slice(&bytes).unwrap_or_default();
388    saved
389        .into_iter()
390        .filter_map(|(id, encoded)| {
391            let registration =
392                v1::GatewayRegister::decode(STANDARD.decode(encoded).ok()?.as_slice()).ok()?;
393            Some((
394                id,
395                Node {
396                    registration,
397                    ..Node::default()
398                },
399            ))
400        })
401        .collect()
