1//! `snitchhiker-mobile`: the Rust core of the Android app (decision 0011). It holds the app's connections to its
2//! nodes (iroh, as the desktop app's remote ones) and pairs with nodes; requests and notices cross as the protocol's
3//! own messages, encoded. This file is only the JNI glue of `core`: the static methods of the app's
4//! `dev.nikarh.snitchhiker.app.Native`. Errors and panics become Java `RuntimeException`s.
5
6#![expect(
7    clippy::must_use_candidate,
8    reason = "JNI entry points: the JVM calls them, and uses what they return"
9)]

19/// An application error, for Java.
20fn fail(env: &mut Env<'_>, message: &str) -> jni::errors::Error {
21    let _ = env.throw_new(
22        jni::jni_str!("java/lang/RuntimeException"),
23        jni::strings::JNIString::new(message),
24    );
25    jni::errors::Error::JavaException
26}
27
28fn bytes(env: &Env<'_>, array: &JByteArray<'_>) -> jni::errors::Result<Vec<u8>> {
29    env.convert_byte_array(array)
30}
31
32fn handle(value: jlong) -> u64 {
33    u64::try_from(value).unwrap_or(0)
34}
35
36fn jlong_of(value: u64) -> jlong {
37    jlong::try_from(value).unwrap_or(jlong::MAX)
38}

50#[jni_mangle("dev.nikarh.snitchhiker.app.Native")]
51pub extern "system" fn key_id<'local>(
52    mut unowned: EnvUnowned<'local>,
53    _class: JClass<'local>,
54    key: JByteArray<'local>,
55) -> JString<'local> {
56    unowned
57        .with_env(|env| {
58            let key = bytes(env, &key)?;
59            match core::key_id(&key) {
60                Ok(id) => JString::from_str(env, id),
61                Err(e) => Err(fail(env, &e)),
62            }
63        })
64        .resolve::<ThrowRuntimeExAndDefault>()
65}

67#[jni_mangle("dev.nikarh.snitchhiker.app.Native")]
68pub extern "system" fn pair<'local>(
69    mut unowned: EnvUnowned<'local>,
70    _class: JClass<'local>,
71    key: JByteArray<'local>,
72    link_or_code: JString<'local>,
73    name: JString<'local>,
74    relays: JObjectArray<'local>,
75) -> JString<'local> {
76    unowned
77        .with_env(|env| {
78            let key = bytes(env, &key)?;
79            let link_or_code = link_or_code.try_to_string(env)?;
80            let name = name.try_to_string(env)?;
81            let mut list = Vec::new();
82            for index in 0..relays.len(env)? {
83                let element = relays.get_element(env, index)?;
84                let item = JString::cast_local(env, element)?;
85                list.push(item.try_to_string(env)?);
86            }
87            match core::pair(&key, &link_or_code, &name, &list) {
88                Ok(node) => JString::from_str(env, node),
89                Err(e) => Err(fail(env, &e)),
90            }
91        })
92        .resolve::<ThrowRuntimeExAndDefault>()
93}

95#[jni_mangle("dev.nikarh.snitchhiker.app.Native")]
96pub extern "system" fn connect<'local>(
97    mut unowned: EnvUnowned<'local>,
98    _class: JClass<'local>,
99    key: JByteArray<'local>,
100    node: JString<'local>,
101) -> jlong {
102    unowned
103        .with_env(|env| {
104            let key = bytes(env, &key)?;
105            let node = node.try_to_string(env)?;
106            core::connect(&key, &node)
107                .map(jlong_of)
108                .map_err(|e| fail(env, &e))
109        })
110        .resolve::<ThrowRuntimeExAndDefault>()
111}

161/// The next notice, or null when none came within the time.
162#[jni_mangle("dev.nikarh.snitchhiker.app.Native")]
163pub extern "system" fn next_notice<'local>(
164    mut unowned: EnvUnowned<'local>,
165    _class: JClass<'local>,
166    connection: jlong,
167    timeout_ms: jlong,
168) -> JByteArray<'local> {
169    unowned
170        .with_env(|env| {
171            match core::next_notice(handle(connection), u64::try_from(timeout_ms).unwrap_or(0)) {
172                Ok(Some(notice)) => env.byte_array_from_slice(&notice),
173                Ok(None) => Ok(JByteArray::null()),
174                Err(e) => Err(fail(env, &e)),
175            }
176        })
177        .resolve::<ThrowRuntimeExAndDefault>()
178}
