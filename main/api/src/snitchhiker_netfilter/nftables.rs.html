1//! Snitchhiker's `nf_tables` tables, written as netlink batches: the daemon's (which queues new connections to it),
2//! and the system firewall's (the base packet filter the user configures).
3//!
4//! Only our own table is ever created, changed, read or deleted: nothing here lists the ruleset or touches another
5//! table, so another firewall's large sets cost us nothing (a lesson from libnftables, which reads every interval set
6//! of every table before adding a set element).
7

119/// What the daemon's table looks like.
120#[derive(Debug, Clone, PartialEq, Eq)]
121pub struct Table {
122    pub name: String,
123    /// The first NFQUEUE number new flows are queued to.
124    pub queue: u16,
125    /// How many queues, from `queue` up; flows are spread over them by hash.
126    pub queues: u16,
127    /// Whether packets pass while no process is bound to the queues (the daemon isn't running): fail-open. Queue
128    /// overflow never lets packets pass either way.
129    pub bypass: bool,
130    /// Packets whose mark has these bits set are the daemon's own and are not queued.
131    pub own_mark: u32,
132    /// The daemon's own cgroup (v2), when it has one to itself (a systemd service): its level in the hierarchy and
133    /// its id. Packets of sockets in it are the daemon's own and are not queued.
134    pub own_cgroup: Option<Cgroup>,
135    /// A queue that DNS answers (UDP from port 53) are copied to, to learn the names of addresses; `None` for none.
136    /// It only observes: it always passes packets while nobody is bound to it.
137    pub dns_queue: Option<u16>,
138    /// Interfaces whose forwarded new flows are queued too: the bridges of containers and VMs, whose traffic never
139    /// passes the `output` hook. None: forwarded traffic isn't looked at.
140    pub forward_from: Vec<Interface>,
141}
142

175/// Writes the batch that (re)creates the table: in one transaction it is created, deleted (with whatever an older
176/// daemon or another tool left in it) and created again with its chain and rules, so packets see either the old
177/// table or the new one, never neither.
178pub fn encode_install(writer: &mut Writer, table: &Table, sequence: u32) -> Batch {
179    writer.clear();
180    let mut batch = Builder::start(writer, sequence);
181    batch.table(NFT_MSG_NEWTABLE, NLM_F_CREATE, &table.name);
182    batch.table(NFT_MSG_DELTABLE, 0, &table.name);
183    batch.table(NFT_MSG_NEWTABLE, NLM_F_CREATE, &table.name);
184    batch.base_chain(
185        &table.name,
186        "output",
187        "route",
188        NF_INET_LOCAL_OUT,
189        NF_IP_PRI_MANGLE,
190    );
191
192    // Our own traffic (the relay connection, pushes) is never queued.
193    batch.rule(&table.name, "output", |rule| {
194        rule.meta(NFT_META_MARK);
195        rule.bitwise(4, &table.own_mark.to_ne_bytes());
196        rule.cmp(NFT_CMP_NEQ, &0_u32.to_ne_bytes());
197        rule.accept();
198    });
199    // And whatever its cgroup's sockets send: the sockets a library opened, which it can't mark.
200    if let Some(cgroup) = table.own_cgroup {
201        batch.rule(&table.name, "output", |rule| {
202            rule.socket_cgroup(cgroup.level);
203            rule.cmp(NFT_CMP_EQ, &cgroup.id.to_ne_bytes());
204            rule.accept();
205        });
206    }
207    // New TCP connections: packets with only SYN of FIN, SYN, RST and ACK set. `ct state new` would also catch
208    // out-of-order packets of established connections.
209    batch.rule(&table.name, "output", |rule| {
210        rule.meta(NFT_META_L4PROTO);
211        rule.cmp(NFT_CMP_EQ, &[IPPROTO_TCP]);
212        rule.payload(NFT_PAYLOAD_TRANSPORT_HEADER, TCP_FLAGS_OFFSET, 1);
213        rule.bitwise(1, &[TCP_FIN_SYN_RST_ACK]);
214        rule.cmp(NFT_CMP_EQ, &[TCP_SYN]);
215        rule.queue(table);
216    });
217    // Everything else: the first packet of a flow, and related ones (ICMP errors, FTP data and the like).
218    batch.rule(&table.name, "output", |rule| {
219        rule.meta(NFT_META_L4PROTO);
220        rule.cmp(NFT_CMP_NEQ, &[IPPROTO_TCP]);
221        rule.ct(NFT_CT_STATE);
222        rule.bitwise(4, &(CT_STATE_NEW | CT_STATE_RELATED).to_ne_bytes());
223        rule.cmp(NFT_CMP_NEQ, &0_u32.to_ne_bytes());
224        rule.queue(table);
225    });
226    if !table.forward_from.is_empty() {
227        // Before NAT (postrouting) changes the source: the tuple is the one the sender's namespace recorded.
228        batch.base_chain(
229            &table.name,
230            "forward",
231            "filter",
232            NF_INET_FORWARD,
233            NF_IP_PRI_MANGLE,
234        );
235        for interface in &table.forward_from {
236            batch.rule(&table.name, "forward", |rule| {
237                rule.interface(NFT_META_IIFNAME, interface);
238                rule.meta(NFT_META_L4PROTO);
239                rule.cmp(NFT_CMP_EQ, &[IPPROTO_TCP]);
240                rule.payload(NFT_PAYLOAD_TRANSPORT_HEADER, TCP_FLAGS_OFFSET, 1);
241                rule.bitwise(1, &[TCP_FIN_SYN_RST_ACK]);
242                rule.cmp(NFT_CMP_EQ, &[TCP_SYN]);
243                rule.queue(table);
244            });
245            batch.rule(&table.name, "forward", |rule| {
246                rule.interface(NFT_META_IIFNAME, interface);
247                rule.meta(NFT_META_L4PROTO);
248                rule.cmp(NFT_CMP_NEQ, &[IPPROTO_TCP]);
249                rule.ct(NFT_CT_STATE);
250                rule.bitwise(4, &(CT_STATE_NEW | CT_STATE_RELATED).to_ne_bytes());
251                rule.cmp(NFT_CMP_NEQ, &0_u32.to_ne_bytes());
252                rule.queue(table);
253            });
254        }
255    }
256    if let Some(dns_queue) = table.dns_queue {
257        batch.base_chain(
258            &table.name,
259            "input",
260            "filter",
261            NF_INET_LOCAL_IN,
262            NF_IP_PRI_MANGLE,
263        );
264        batch.rule(&table.name, "input", |rule| {
265            rule.meta(NFT_META_L4PROTO);
266            rule.cmp(NFT_CMP_EQ, &[IPPROTO_UDP]);
267            rule.payload(NFT_PAYLOAD_TRANSPORT_HEADER, 0, 2);
268            rule.cmp(NFT_CMP_EQ, &DNS_PORT.to_be_bytes());
269            rule.queue_to(dns_queue, 1, true);
270        });
271    }
272    batch.end()
273}

293/// An address prefix: 4 or 16 bytes, and how many leading bits count.
294#[derive(Debug, Clone, PartialEq, Eq)]
295pub struct Prefix {
296    pub address: Vec<u8>,
297    pub length: u8,
298}
299
300impl Prefix {
301    fn v6(&self) -> bool {
302        self.address.len() == 16
303    }
304
305    /// The prefix's mask, and the address masked by it.
306    fn masked(&self) -> (Vec<u8>, Vec<u8>) {
307        let mut bits = u32::from(self.length);
308        let mask: Vec<u8> = self
309            .address
310            .iter()
311            .map(|_| {
312                let take = bits.min(8);
313                bits -= take;
314                if take == 0 { 0 } else { 0xff_u8 << (8 - take) }
315            })
316            .collect();
317        let network = self.address.iter().zip(&mask).map(|(a, m)| a & m).collect();
318        (mask, network)
319    }
320}

370/// Writes the batch that (re)creates the system table, in one transaction as [`encode_install`] does.
371pub fn encode_system(writer: &mut Writer, table: &SystemTable, sequence: u32) -> Batch {
372    writer.clear();
373    let mut batch = Builder::start(writer, sequence);
374    batch.table(NFT_MSG_NEWTABLE, NLM_F_CREATE, &table.name);
375    batch.table(NFT_MSG_DELTABLE, 0, &table.name);
376    batch.table(NFT_MSG_NEWTABLE, NLM_F_CREATE, &table.name);
377    for (chain, hook, drop) in [
378        (SystemChain::Input, NF_INET_LOCAL_IN, table.input_drop),
379        (SystemChain::Forward, NF_INET_FORWARD, table.forward_drop),
380        (SystemChain::Output, NF_INET_LOCAL_OUT, false),
381    ] {
382        batch.filter_chain(
383            &table.name,
384            chain.name(),
385            hook,
386            if drop { NF_DROP } else { NF_ACCEPT },
387        );
388    }
389    for rule in &table.rules {
390        batch.rule(&table.name, rule.chain.name(), |e| e.system_rule(rule));
391    }
392    batch.end()
393}

395/// Writes a request for table `name` (only that one is read): the kernel answers with it, or with `ENOENT`. Returns the
396/// request's sequence number.
397pub fn encode_get_table(writer: &mut Writer, name: &str, sequence: u32) -> u32 {
398    writer.clear();
399    let message = writer.begin(
400        (NFNL_SUBSYS_NFTABLES << 8) | NFT_MSG_GETTABLE,
401        NLM_F_ACK,
402        sequence,
403        NFPROTO_INET,
404        0,
405    );
406    writer.string(NFTA_TABLE_NAME, name);
407    writer.finish(message);
408    sequence
409}

420/// Writes a batch: the begin marker, numbered messages, the end marker.
421struct Builder<'a> {
422    writer: &'a mut Writer,
423    first: u32,
424    sequence: u32,
425}
426
427impl<'a> Builder<'a> {
428    fn start(writer: &'a mut Writer, sequence: u32) -> Self {
429        let begin = writer.begin(
430            NFNL_MSG_BATCH_BEGIN,
431            0,
432            sequence,
433            AF_UNSPEC,
434            NFNL_SUBSYS_NFTABLES,
435        );
436        writer.finish(begin);
437        Self {
438            writer,
439            first: sequence.wrapping_add(1),
440            sequence,
441        }
442    }
443
444    fn message(&mut self, kind: u16, flags: u16) -> crate::netlink::Open {
445        self.sequence = self.sequence.wrapping_add(1);
446        self.writer.begin(
447            (NFNL_SUBSYS_NFTABLES << 8) | kind,
448            flags | NLM_F_ACK,
449            self.sequence,
450            NFPROTO_INET,
451            0,
452        )
453    }
454
455    fn table(&mut self, kind: u16, flags: u16, name: &str) {
456        let message = self.message(kind, flags);
457        self.writer.string(NFTA_TABLE_NAME, name);
458        self.writer.finish(message);
459    }
460
461    fn base_chain(&mut self, table: &str, name: &str, kind: &str, hook: u32, priority: i32) {
462        let message = self.message(NFT_MSG_NEWCHAIN, NLM_F_CREATE);
463        self.writer.string(NFTA_CHAIN_TABLE, table);
464        self.writer.string(NFTA_CHAIN_NAME, name);
465        let nested = self.writer.nest(NFTA_CHAIN_HOOK);
466        self.writer.u32_be(NFTA_HOOK_HOOKNUM, hook);
467        self.writer
468            .attribute(NFTA_HOOK_PRIORITY, &priority.to_be_bytes());
469        self.writer.end(nested);
470        self.writer.u32_be(NFTA_CHAIN_POLICY, NF_ACCEPT);
471        self.writer.string(NFTA_CHAIN_TYPE, kind);
472        self.writer.finish(message);
473    }
474
475    fn filter_chain(&mut self, table: &str, name: &str, hook: u32, policy: u32) {
476        let message = self.message(NFT_MSG_NEWCHAIN, NLM_F_CREATE);
477        self.writer.string(NFTA_CHAIN_TABLE, table);
478        self.writer.string(NFTA_CHAIN_NAME, name);
479        let nested = self.writer.nest(NFTA_CHAIN_HOOK);
480        self.writer.u32_be(NFTA_HOOK_HOOKNUM, hook);
481        self.writer
482            .attribute(NFTA_HOOK_PRIORITY, &NF_IP_PRI_FILTER.to_be_bytes());
483        self.writer.end(nested);
484        self.writer.u32_be(NFTA_CHAIN_POLICY, policy);
485        self.writer.string(NFTA_CHAIN_TYPE, "filter");
486        self.writer.finish(message);
487    }
488
489    fn rule(&mut self, table: &str, chain: &str, expressions: impl FnOnce(&mut Expressions<'_>)) {
490        let message = self.message(NFT_MSG_NEWRULE, NLM_F_CREATE | NLM_F_APPEND);
491        self.writer.string(NFTA_RULE_TABLE, table);
492        self.writer.string(NFTA_RULE_CHAIN, chain);
493        let list = self.writer.nest(NFTA_RULE_EXPRESSIONS);
494        expressions(&mut Expressions {
495            writer: self.writer,
496        });
497        self.writer.end(list);
498        self.writer.finish(message);
499    }
500
501    fn end(self) -> Batch {
502        let last = self.sequence;
503        let end = self.writer.begin(
504            NFNL_MSG_BATCH_END,
505            0,
506            last.wrapping_add(1),
507            AF_UNSPEC,
508            NFNL_SUBSYS_NFTABLES,
509        );
510        self.writer.finish(end);
511        Batch {
512            first: self.first,
513            last,
514            next: last.wrapping_add(2),
515        }
516    }
517}

519/// The expressions of one rule. They all work on register 1, except the verdicts.
520struct Expressions<'a> {
521    writer: &'a mut Writer,
522}
523
524impl Expressions<'_> {
525    fn expression(&mut self, name: &str, data: impl FnOnce(&mut Writer)) {
526        let element = self.writer.nest(NFTA_LIST_ELEM);
527        self.writer.string(NFTA_EXPR_NAME, name);
528        let nested = self.writer.nest(NFTA_EXPR_DATA);
529        data(self.writer);
530        self.writer.end(nested);
531        self.writer.end(element);
532    }
533
534    fn meta(&mut self, key: u32) {
535        self.expression("meta", |w| {
536            w.u32_be(NFTA_META_KEY, key);
537            w.u32_be(NFTA_META_DREG, NFT_REG_1);
538        });
539    }
540
541    /// Register 1 = the id of the cgroup (v2) at `level` above the packet's socket.
542    fn socket_cgroup(&mut self, level: u32) {
543        self.expression("socket", |w| {
544            w.u32_be(NFTA_SOCKET_KEY, NFT_SOCKET_CGROUPV2);
545            w.u32_be(NFTA_SOCKET_DREG, NFT_REG_1);
546            w.u32_be(NFTA_SOCKET_LEVEL, level);
547        });
548    }
549
550    fn ct(&mut self, key: u32) {
551        self.expression("ct", |w| {
552            w.u32_be(NFTA_CT_KEY, key);
553            w.u32_be(NFTA_CT_DREG, NFT_REG_1);
554        });
555    }
556
557    fn payload(&mut self, base: u32, offset: u32, length: u32) {
558        self.expression("payload", |w| {
559            w.u32_be(NFTA_PAYLOAD_DREG, NFT_REG_1);
560            w.u32_be(NFTA_PAYLOAD_BASE, base);
561            w.u32_be(NFTA_PAYLOAD_OFFSET, offset);
562            w.u32_be(NFTA_PAYLOAD_LEN, length);
563        });
564    }
565
566    /// Register 1 = register 1 & `mask` (the mask is `length` bytes, in the register's byte order).
567    fn bitwise(&mut self, length: u32, mask: &[u8]) {
568        self.expression("bitwise", |w| {
569            w.u32_be(NFTA_BITWISE_SREG, NFT_REG_1);
570            w.u32_be(NFTA_BITWISE_DREG, NFT_REG_1);
571            w.u32_be(NFTA_BITWISE_LEN, length);
572            let nested = w.nest(NFTA_BITWISE_MASK);
573            w.attribute(NFTA_DATA_VALUE, mask);
574            w.end(nested);
575            let nested = w.nest(NFTA_BITWISE_XOR);
576            w.attribute(
577                NFTA_DATA_VALUE,
578                [0; 16].get(..mask.len()).unwrap_or_default(),
579            );
580            w.end(nested);
581        });
582    }
583
584    fn cmp(&mut self, op: u32, value: &[u8]) {
585        self.expression("cmp", |w| {
586            w.u32_be(NFTA_CMP_SREG, NFT_REG_1);
587            w.u32_be(NFTA_CMP_OP, op);
588            let nested = w.nest(NFTA_CMP_DATA);
589            w.attribute(NFTA_DATA_VALUE, value);
590            w.end(nested);
591        });
592    }
593
594    fn accept(&mut self) {
595        self.verdict(NF_ACCEPT);
596    }
597
598    fn verdict(&mut self, code: u32) {
599        self.expression("immediate", |w| {
600            w.u32_be(NFTA_IMMEDIATE_DREG, NFT_REG_VERDICT);
601            let data = w.nest(NFTA_IMMEDIATE_DATA);
602            let verdict = w.nest(NFTA_DATA_VERDICT);
603            w.u32_be(NFTA_VERDICT_CODE, code);
604            w.end(verdict);
605            w.end(data);
606        });
607    }
608
609    /// Register 1 within the ports, inclusive.
610    fn ports(&mut self, offset: u32, (low, high): (u16, u16)) {
611        self.payload(NFT_PAYLOAD_TRANSPORT_HEADER, offset, 2);
612        if low == high {
613            self.cmp(NFT_CMP_EQ, &low.to_be_bytes());
614        } else {
615            self.cmp(NFT_CMP_GTE, &low.to_be_bytes());
616            self.cmp(NFT_CMP_LTE, &high.to_be_bytes());
617        }
618    }
619
620    fn address(&mut self, prefix: &Prefix, source: bool) {
621        let (offset, length) = match (prefix.v6(), source) {
622            (false, true) => (12, 4),
623            (false, false) => (16, 4),
624            (true, true) => (8, 16),
625            (true, false) => (24, 16),
626        };
627        self.payload(NFT_PAYLOAD_NETWORK_HEADER, offset, length);
628        let (mask, network) = prefix.masked();
629        if usize::from(prefix.length) < mask.len() * 8 {
630            self.bitwise(length, &mask);
631        }
632        self.cmp(NFT_CMP_EQ, &network);
633    }
634
635    fn interface(&mut self, key: u32, interface: &Interface) {
636        self.meta(key);
637        let mut name = interface.name.as_bytes().to_vec();
638        if !interface.prefix {
639            name.resize(IFNAMSIZ, 0);
640        }
641        self.cmp(NFT_CMP_EQ, &name);
642    }
643
644    fn system_rule(&mut self, rule: &SystemRule) {
645        let family = rule
646            .source
647            .as_ref()
648            .or(rule.destination.as_ref())
649            .map(Prefix::v6);
650        if let Some(v6) = family {
651            self.meta(NFT_META_NFPROTO);
652            self.cmp(NFT_CMP_EQ, &[if v6 { NFPROTO_IPV6 } else { NFPROTO_IPV4 }]);
653        }
654        if let Some(interface) = &rule.in_interface {
655            self.interface(NFT_META_IIFNAME, interface);
656        }
657        if let Some(interface) = &rule.out_interface {
658            self.interface(NFT_META_OIFNAME, interface);
659        }
660        if let Some(prefix) = &rule.source {
661            self.address(prefix, true);
662        }
663        if let Some(prefix) = &rule.destination {
664            self.address(prefix, false);
665        }
666        if let Some(protocol) = rule.protocol {
667            self.meta(NFT_META_L4PROTO);
668            self.cmp(NFT_CMP_EQ, &[protocol]);
669        }
670        if let Some(ports) = rule.source_ports {
671            self.ports(0, ports);
672        }
673        if let Some(ports) = rule.destination_ports {
674            self.ports(2, ports);
675        }
676        if let Some(icmp_type) = rule.icmp_type {
677            self.payload(NFT_PAYLOAD_TRANSPORT_HEADER, 0, 1);
678            self.cmp(NFT_CMP_EQ, &[icmp_type]);
679        }
680        if rule.ct_state != 0 {
681            self.ct(NFT_CT_STATE);
682            self.bitwise(4, &rule.ct_state.to_ne_bytes());
683            self.cmp(NFT_CMP_NEQ, &0_u32.to_ne_bytes());
684        }
685        match &rule.target {
686            SystemTarget::Accept => self.verdict(NF_ACCEPT),
687            SystemTarget::Drop => self.verdict(NF_DROP),
688            SystemTarget::Reject => {
689                let tcp = rule.protocol == Some(IPPROTO_TCP);
690                self.expression("reject", |w| {
691                    if tcp {
692                        w.u32_be(NFTA_REJECT_TYPE, NFT_REJECT_TCP_RST);
693                    } else {
694                        w.u32_be(NFTA_REJECT_TYPE, NFT_REJECT_ICMPX_UNREACH);
695                        w.attribute(NFTA_REJECT_ICMP_CODE, &[NFT_REJECT_ICMPX_PORT_UNREACH]);
696                    }
697                });
698            }
699            SystemTarget::Log(prefix) => {
700                self.expression("log", |w| w.string(NFTA_LOG_PREFIX, prefix));
701            }
702            SystemTarget::Queue {
703                number,
704                total,
705                bypass,
706            } => self.queue_to(*number, *total, *bypass),
707        }
708    }
709
710    fn queue(&mut self, table: &Table) {
711        self.queue_to(table.queue, table.queues, table.bypass);
712    }
713
714    fn queue_to(&mut self, number: u16, total: u16, bypass: bool) {
715        self.expression("queue", |w| {
716            w.u16_be(NFTA_QUEUE_NUM, number);
717            w.u16_be(NFTA_QUEUE_TOTAL, total.max(1));
718            w.u16_be(
719                NFTA_QUEUE_FLAGS,
720                if bypass { NFT_QUEUE_FLAG_BYPASS } else { 0 },
721            );
722        });
723    }
724}

865/// The nftables changes of the namespace it was opened in, as the kernel announces them (the `NFNLGRP_NFTABLES`
866/// multicast group): for noticing at once that a table was deleted, without asking again and again.
867#[derive(Debug)]
868pub struct Monitor {
869    socket: netlink_sys::Socket,
870    receive: Vec<u8>,
871}
872
873const NFNLGRP_NFTABLES: u32 = 7;
874
875impl Monitor {
876    pub fn open() -> std::io::Result<Self> {
877        let mut socket = netlink_sys::Socket::new(netlink_sys::protocols::NETLINK_NETFILTER)?;
878        socket.bind_auto()?;
879        socket.add_membership(NFNLGRP_NFTABLES)?;
880        Ok(Self {
881            socket,
882            receive: Vec::with_capacity(1 << 16),
883        })
884    }
885
886    /// Reads what the kernel announced; whether the table `name` was deleted.
887    pub fn deleted(&mut self, name: &str) -> std::io::Result<bool> {
888        self.deleted_any(&[name])
889    }
890
891    /// Reads what the kernel announced; whether any of the tables `names` was deleted.
892    pub fn deleted_any(&mut self, names: &[&str]) -> std::io::Result<bool> {
893        self.receive.clear();
894        self.socket.recv(&mut self.receive, 0)?;
895        let mut deleted = false;
896        for message in crate::netlink::messages(&self.receive) {
897            if message.kind != (NFNL_SUBSYS_NFTABLES << 8) | NFT_MSG_DELTABLE {
898                continue;
899            }
900            let body = message
901                .body
902                .get(crate::netlink::NFGENMSG_LEN..)
903                .unwrap_or_default();
904            deleted |= crate::netlink::attributes(body).any(|(kind, value)| {
905                let value = value.strip_suffix(&[0]).unwrap_or(value);
906                kind == NFTA_TABLE_NAME && names.iter().any(|name| value == name.as_bytes())
907            });
908        }
909        Ok(deleted)
910    }
911}
912
913impl std::os::fd::AsFd for Monitor {
914    fn as_fd(&self) -> std::os::fd::BorrowedFd<'_> {
915        self.socket.as_fd()
916    }
