1//! Local sockets through `sock_diag`: listing the live connections (`SOCK_DIAG_BY_FAMILY` dumps), and closing one by its
2//! addresses (`SOCK_DESTROY`), which is how a rejected connection fails at once instead of timing out: the program's
3//! `connect()` or next send returns `ECONNABORTED`.
4
5use std::io;
6use std::net::IpAddr;
7
8use netlink_sys::{Socket as RawSocket, SocketAddr, protocols::NETLINK_SOCK_DIAG};
9
10use crate::netlink::{NLM_F_ACK, NLM_F_REQUEST, NLMSG_DONE, NLMSG_ERROR, ack_error, messages};
11
12const SOCK_DIAG_BY_FAMILY: u16 = 20;
13const SOCK_DESTROY: u16 = 21;
14const NLM_F_DUMP: u16 = 0x300;
15/// TCP connections that have, or are getting, a peer: established, opening and closing; not listening sockets nor
16/// `TIME_WAIT` ones (no process holds those).
17const TCP_LIVE: u32 =
18    (1 << 1) | (1 << 2) | (1 << 3) | (1 << 4) | (1 << 5) | (1 << 8) | (1 << 9) | (1 << 11);
19/// Connected UDP sockets (established, in the kernel's terms).
20const UDP_CONNECTED: u32 = 1 << 1;
21const AF_INET: u8 = 2;
22const AF_INET6: u8 = 10;
23/// Every TCP state: the socket is found whatever state it is in.
24const ALL_STATES: u32 = 0xffff_ffff;
25const NO_COOKIE: u32 = 0xffff_ffff;
26
27/// A local socket, as its packets show it.
28#[derive(Debug, Clone, Copy, PartialEq, Eq)]
29pub struct SocketId {
30    pub protocol: u8,
31    pub local: IpAddr,
32    pub local_port: u16,
33    pub remote: IpAddr,
34    pub remote_port: u16,
35}
36
37/// A live socket, as `sock_diag` lists it.
38#[derive(Debug, Clone, Copy, PartialEq, Eq)]
39pub struct LiveSocket {
40    pub id: SocketId,
41    /// The kernel's state number (TCP's: 1 established, 2 SYN sent, ...).
42    pub state: u8,
43    pub uid: u32,
44    pub inode: u32,
45    /// The kernel's socket cookie: what the attribution programs count bytes by.
46    pub cookie: u64,
47}
48
49/// A TCP or UDP state's name.
50#[must_use]
51pub fn state_name(protocol: u8, state: u8) -> &'static str {
52    match (protocol, state) {
53        (17, 1) => "connected",
54        (_, 1) => "established",
55        (_, 2) => "opening",
56        (_, 3) => "accepted",
57        (_, 4 | 5 | 8 | 9 | 11) => "closing",
58        _ => "other",
59    }
60}

62/// A `sock_diag` netlink socket.
63#[derive(Debug)]
64pub struct SockDiag {
65    socket: RawSocket,
66    request: Vec<u8>,
67    receive: Vec<u8>,
68    sequence: u32,
69}
70
71impl SockDiag {
72    pub fn open() -> io::Result<Self> {
73        let mut socket = RawSocket::new(NETLINK_SOCK_DIAG)?;
74        socket.bind_auto()?;
75        socket.connect(&SocketAddr::new(0, 0))?;
76        Ok(Self {
77            socket,
78            request: Vec::with_capacity(128),
79            receive: Vec::with_capacity(4096),
80            sequence: 0,
81        })
82    }
83
84    /// The live TCP connections and connected UDP sockets, IPv4 and IPv6.
85    pub fn list(&mut self) -> io::Result<Vec<LiveSocket>> {
86        let mut found = Vec::new();
87        for (protocol, states) in [(6_u8, TCP_LIVE), (17, UDP_CONNECTED)] {
88            for family in [AF_INET, AF_INET6] {
89                self.sequence = self.sequence.wrapping_add(1);
90                encode_dump(&mut self.request, family, protocol, states, self.sequence);
91                self.socket.send(&self.request, 0)?;
92                'receive: loop {
93                    self.receive.clear();
94                    self.receive.reserve(64 * 1024);
95                    self.socket.recv(&mut self.receive, 0)?;
96                    for message in messages(&self.receive) {
97                        if message.sequence != self.sequence {
98                            continue;
99                        }
100                        match message.kind {
101                            NLMSG_DONE => break 'receive,
102                            NLMSG_ERROR => match ack_error(message.body) {
103                                0 => break 'receive,
104                                code => return Err(io::Error::from_raw_os_error(-code)),
105                            },
106                            SOCK_DIAG_BY_FAMILY => {
107                                if let Some(socket) = decode(message.body, protocol) {
108                                    found.push(socket);
109                                }
110                            }
111                            _ => {}
112                        }
113                    }
114                }
115            }
116        }
117        Ok(found)
118    }
119
120    /// Closes the socket; `NotFound` when there is none (it may have closed already).
121    pub fn destroy(&mut self, id: &SocketId) -> io::Result<()> {
122        self.sequence = self.sequence.wrapping_add(1);
123        encode_destroy(&mut self.request, id, self.sequence);
124        self.socket.send(&self.request, 0)?;
125        loop {
126            self.receive.clear();
127            self.socket.recv(&mut self.receive, 0)?;
128            for message in messages(&self.receive) {
129                if message.kind == NLMSG_ERROR && message.sequence == self.sequence {
130                    return match ack_error(message.body) {
131                        0 => Ok(()),
132                        code => Err(io::Error::from_raw_os_error(-code)),
133                    };
134                }
135            }
136        }
137    }
138}

140/// Writes a `SOCK_DIAG_BY_FAMILY` dump request for one family, protocol and set of states.
141pub fn encode_dump(out: &mut Vec<u8>, family: u8, protocol: u8, states: u32, sequence: u32) {
142    out.clear();
143    out.extend_from_slice(&0_u32.to_ne_bytes());
144    out.extend_from_slice(&SOCK_DIAG_BY_FAMILY.to_ne_bytes());
145    out.extend_from_slice(&(NLM_F_REQUEST | NLM_F_DUMP).to_ne_bytes());
146    out.extend_from_slice(&sequence.to_ne_bytes());
147    out.extend_from_slice(&0_u32.to_ne_bytes());
148    out.extend_from_slice(&[family, protocol, 0, 0]);
149    out.extend_from_slice(&states.to_ne_bytes());
150    // An empty inet_diag_sockid: every socket.
151    out.extend_from_slice(&[0_u8; 40]);
152    out.extend_from_slice(&0_u32.to_ne_bytes());
153    out.extend_from_slice(&NO_COOKIE.to_ne_bytes());
154    out.extend_from_slice(&NO_COOKIE.to_ne_bytes());
155    let length = u32::try_from(out.len()).unwrap_or(u32::MAX);
156    if let Some(field) = out.get_mut(0..4) {
157        field.copy_from_slice(&length.to_ne_bytes());
158    }
159}

161/// Reads a `struct inet_diag_msg`.
162fn decode(body: &[u8], protocol: u8) -> Option<LiveSocket> {
163    let family = *body.first()?;
164    let state = *body.get(1)?;
165    let port = |at: usize| Some(u16::from_be_bytes(body.get(at..at + 2)?.try_into().ok()?));
166    let address = |at: usize| -> Option<IpAddr> {
167        let bytes: [u8; 16] = body.get(at..at + 16)?.try_into().ok()?;
168        Some(if family == AF_INET {
169            IpAddr::from([bytes[0], bytes[1], bytes[2], bytes[3]])
170        } else {
171            IpAddr::from(bytes)
172        })
173    };
174    let word = |at: usize| Some(u32::from_ne_bytes(body.get(at..at + 4)?.try_into().ok()?));
175    let id = SocketId {
176        protocol,
177        local_port: port(4)?,
178        remote_port: port(6)?,
179        local: address(8)?,
180        remote: address(24)?,
181    };
182    let cookie = u64::from(word(44)?) | (u64::from(word(48)?) << 32);
183    Some(LiveSocket {
184        id,
185        state,
186        uid: word(64)?,
187        inode: word(68)?,
188        cookie,
189    })
190}

192/// Writes the `SOCK_DESTROY` request: a netlink header and `struct inet_diag_req_v2`.
193pub fn encode_destroy(out: &mut Vec<u8>, id: &SocketId, sequence: u32) {
194    out.clear();
195    let family = if id.local.is_ipv6() {
196        AF_INET6
197    } else {
198        AF_INET
199    };
200    let address = |address: IpAddr| -> [u8; 16] {
201        let mut bytes = [0_u8; 16];
202        match address {
203            IpAddr::V4(v4) => {
204                if let Some(head) = bytes.get_mut(..4) {
205                    head.copy_from_slice(&v4.octets());
206                }
207            }
208            IpAddr::V6(v6) => bytes = v6.octets(),
209        }
210        bytes
211    };
212    // Header, filled in below.
213    out.extend_from_slice(&0_u32.to_ne_bytes());
214    out.extend_from_slice(&SOCK_DESTROY.to_ne_bytes());
215    out.extend_from_slice(&(NLM_F_REQUEST | NLM_F_ACK).to_ne_bytes());
216    out.extend_from_slice(&sequence.to_ne_bytes());
217    out.extend_from_slice(&0_u32.to_ne_bytes());
218    // inet_diag_req_v2: family, protocol, extensions, padding, states.
219    out.extend_from_slice(&[family, id.protocol, 0, 0]);
220    out.extend_from_slice(&ALL_STATES.to_ne_bytes());
221    // inet_diag_sockid: ports and addresses in network order, interface, cookie.
222    out.extend_from_slice(&id.local_port.to_be_bytes());
223    out.extend_from_slice(&id.remote_port.to_be_bytes());
224    out.extend_from_slice(&address(id.local));
225    out.extend_from_slice(&address(id.remote));
226    out.extend_from_slice(&0_u32.to_ne_bytes());
227    out.extend_from_slice(&NO_COOKIE.to_ne_bytes());
228    out.extend_from_slice(&NO_COOKIE.to_ne_bytes());
229    let length = u32::try_from(out.len()).unwrap_or(u32::MAX);
230    if let Some(field) = out.get_mut(0..4) {
231        field.copy_from_slice(&length.to_ne_bytes());
232    }
233}

239    #[test]
240    fn reads_an_inet_diag_message() {
241        let mut body = vec![AF_INET, 1, 0, 0];
242        body.extend_from_slice(&40000_u16.to_be_bytes());
243        body.extend_from_slice(&443_u16.to_be_bytes());
244        let mut local = [0_u8; 16];
245        local[..4].copy_from_slice(&[192, 0, 2, 1]);
246        let mut remote = [0_u8; 16];
247        remote[..4].copy_from_slice(&[198, 51, 100, 7]);
248        body.extend_from_slice(&local);
249        body.extend_from_slice(&remote);
250        body.extend_from_slice(&0_u32.to_ne_bytes());
251        body.extend_from_slice(&7_u32.to_ne_bytes());
252        body.extend_from_slice(&1_u32.to_ne_bytes());
253        // expires, rqueue, wqueue, uid, inode
254        for value in [0_u32, 0, 0, 1000, 4242] {
255            body.extend_from_slice(&value.to_ne_bytes());
256        }
257        let socket = decode(&body, 6).unwrap();
258        assert_eq!(socket.id.local, "192.0.2.1".parse::<IpAddr>().unwrap());
259        assert_eq!(socket.id.remote_port, 443);
260        assert_eq!((socket.state, socket.uid, socket.inode), (1, 1000, 4242));
261        assert_eq!(socket.cookie, 7 | (1 << 32));
262        assert_eq!(state_name(6, 1), "established");
263    }
264
265    #[test]
266    fn writes_an_inet_diag_request() {
267        let mut out = Vec::new();
268        encode_destroy(
269            &mut out,
270            &SocketId {
271                protocol: 6,
272                local: "192.0.2.1".parse().unwrap(),
273                local_port: 40000,
274                remote: "198.51.100.7".parse().unwrap(),
275                remote_port: 443,
276            },
277            9,
278        );
279        // 16 bytes of header, 8 of request, 48 of socket id.
280        assert_eq!(out.len(), 72);
281        let message = messages(&out).next().unwrap();
282        assert_eq!(message.kind, SOCK_DESTROY);
283        assert_eq!(message.sequence, 9);
284        let body = message.body;
285        assert_eq!(&body[0..2], &[AF_INET, 6]);
286        assert_eq!(&body[8..10], &40000_u16.to_be_bytes());
287        assert_eq!(&body[10..12], &443_u16.to_be_bytes());
288        assert_eq!(&body[12..16], &[192, 0, 2, 1]);
289        assert_eq!(&body[28..32], &[198, 51, 100, 7]);
290    }
