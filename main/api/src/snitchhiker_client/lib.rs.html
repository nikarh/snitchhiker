1//! A blocking client of a Snitchhiker node over its local sockets: the handshake, requests (answered by id) and
2//! subscription notices (queued while a request waits for its answer).
3//!
4//! Before saying anything, a client checks it is talking to the node: the socket is in a directory only root can
5//! change, and the process serving it runs as root. A process that bound a socket somewhere else, or got one into a
6//! directory it may write, can't pose as the node to answer the client's requests or feed it prompts.
7
8#[cfg(feature = "cache")]
9pub mod cache;

28#[derive(Debug, thiserror::Error)]
29pub enum Error {
30    #[error("{0}")]
31    Io(#[from] std::io::Error),
32    #[error("the node refused the connection: {0}")]
33    Refused(String),
34    #[error("the node answered something unexpected: {0}")]
35    Protocol(&'static str),
36    #[error("not a Snitchhiker node's socket: {0}")]
37    NotTheNode(String),
38    #[error("{message}")]
39    Remote {
40        code: v1::ErrorCode,
41        message: String,
42    },
43}
44
45/// The socket and every directory above it belong to root, and no one else may write to those directories: no other
46/// user can have put the socket there, or swap it for theirs.
47fn only_root_can_change(path: &Path) -> Result<(), Error> {
48    const GROUP_OR_OTHER_WRITE: u32 = 0o022;
49    for (index, entry) in path.ancestors().enumerate() {
50        if entry.as_os_str().is_empty() {
51            continue;
52        }
53        let meta = std::fs::symlink_metadata(entry)
54            .map_err(|e| Error::NotTheNode(format!("{}: {e}", entry.display())))?;
55        if meta.uid() != 0 {
56            return Err(Error::NotTheNode(format!(
57                "{} doesn't belong to root",
58                entry.display()
59            )));
60        }
61        // The socket's own mode says who may connect, not who may replace it.
62        if index > 0 && meta.mode() & GROUP_OR_OTHER_WRITE != 0 {
63            return Err(Error::NotTheNode(format!(
64                "others than root may write to {}",
65                entry.display()
66            )));
67        }
68    }
69    Ok(())
70}

72pub struct Client {
73    reader: BufReader<UnixStream>,
74    writer: BufWriter<UnixStream>,
75    buffer: Vec<u8>,
76    next_id: u64,
77    notices: VecDeque<v1::Notice>,
78    welcome: v1::Welcome,
79    /// The node's strings this connection was sent with columnar blocks, by id.
80    strings: HashMap<u32, String>,
81}
82
83impl Client {
84    /// Connects to the node's socket at `path`, introducing itself as `app` (`cli`, `desktop`, ...) at `version`.
85    pub fn connect(path: &Path, app: &str, version: &str) -> Result<Self, Error> {
86        only_root_can_change(path)?;
87        let stream = UnixStream::connect(path)?;
88        let peer = getsockopt(&stream, sockopt::PeerCredentials)
89            .map_err(|e| Error::Io(std::io::Error::from(e)))?;
90        if peer.uid() != 0 {
91            return Err(Error::NotTheNode(format!(
92                "{} is served by uid {}, not root",
93                path.display(),
94                peer.uid()
95            )));
96        }
97        let mut writer = BufWriter::new(stream.try_clone()?);
98        let mut reader = BufReader::new(stream);
99        let mut buffer = Vec::new();
100        frame::write_preamble(&mut writer)?;
101        let hello = v1::ClientFrame {
102            body: Some(v1::client_frame::Body::Hello(v1::Hello {
103                minor: MINOR,
104                app: app.to_owned(),
105                app_version: version.to_owned(),
106                capabilities: vec!["prompts".to_owned()],
107            })),
108        };
109        frame::write(&mut writer, &hello, &mut buffer)?;
110        writer.flush()?;
111        let answer: v1::ServerFrame = frame::read(&mut reader, &mut buffer)?;
112        match answer.body {
113            Some(v1::server_frame::Body::Welcome(welcome)) => Ok(Self {
114                reader,
115                writer,
116                buffer,
117                next_id: 0,
118                notices: VecDeque::new(),
119                strings: HashMap::new(),
120                welcome,
121            }),
122            Some(v1::server_frame::Body::Refusal(refusal)) => Err(Error::Refused(refusal.reason)),
123            _ => Err(Error::Protocol("no welcome")),
124        }
125    }
126
127    #[must_use]
128    pub fn welcome(&self) -> &v1::Welcome {
129        &self.welcome
130    }
131
132    /// Sends a request and waits for its answer; notices arriving meanwhile are kept for [`Client::notice`]. A node's
133    /// error is `Error::Remote`.
134    pub fn call(&mut self, method: v1::request::Method) -> Result<v1::response::Result, Error> {
135        self.next_id += 1;
136        let id = self.next_id;
137        let request = v1::ClientFrame {
138            body: Some(v1::client_frame::Body::Request(v1::Request {
139                id,
140                method: Some(method),
141            })),
142        };
143        frame::write(&mut self.writer, &request, &mut self.buffer)?;
144        self.writer.flush()?;
145        loop {
146            let frame = self.read()?;
147            match frame.body {
148                Some(v1::server_frame::Body::Response(response)) if response.id == id => {
149                    return match response.result {
150                        Some(v1::response::Result::Error(error)) => Err(Error::Remote {
151                            code: error.code(),
152                            message: error.message,
153                        }),
154                        Some(result) => Ok(result),
155                        None => Err(Error::Protocol("an empty answer")),
156                    };
157                }
158                Some(v1::server_frame::Body::Notice(notice)) => self.notices.push_back(notice),
159                Some(v1::server_frame::Body::Refusal(refusal)) => {
160                    return Err(Error::Refused(refusal.reason));
161                }
162                _ => {}
163            }
164        }
165    }
166
167    /// The next subscription notice, waiting for one (up to the read timeout, if set).
168    pub fn notice(&mut self) -> Result<v1::Notice, Error> {
169        if let Some(notice) = self.notices.pop_front() {
170            return Ok(notice);
171        }
172        loop {
173            let frame = self.read()?;
174            match frame.body {
175                Some(v1::server_frame::Body::Notice(notice)) => return Ok(notice),
176                Some(v1::server_frame::Body::Refusal(refusal)) => {
177                    return Err(Error::Refused(refusal.reason));
178                }
179                _ => {}
180            }
181        }
182    }
183
184    /// The next frame, with the dictionary entries of a block it carries learned.
185    fn read(&mut self) -> Result<v1::ServerFrame, Error> {
186        let frame: v1::ServerFrame = frame::read(&mut self.reader, &mut self.buffer)?;
187        let block = match &frame.body {
188            Some(
189                v1::server_frame::Body::Response(v1::Response {
190                    result: Some(v1::response::Result::EventBlock(block)),
191                    ..
192                })
193                | v1::server_frame::Body::Notice(v1::Notice {
194                    body: Some(v1::notice::Body::EventBlock(block)),
195                    ..
196                }),
197            ) => Some(block),
198            _ => None,
199        };
200        if let Some(dictionary) = block.and_then(|b| b.dictionary.as_ref()) {
201            for (id, value) in dictionary.ids.iter().zip(&dictionary.values) {
202                self.strings.insert(*id, value.clone());
203            }
204        }
205        Ok(frame)
206    }
207
208    /// A string of the node's dictionary, by the id a columnar block uses.
209    #[must_use]
210    pub fn string(&self, id: u32) -> Option<&str> {
211        self.strings.get(&id).map(String::as_str)
212    }
213
214    /// How long reads wait (for [`Client::notice`]); `None` for ever.
215    pub fn set_read_timeout(&self, timeout: Option<Duration>) -> Result<(), Error> {
216        Ok(self.reader.get_ref().set_read_timeout(timeout)?)
217    }
218
219    /// Subscribes to a topic; returns the subscription's id.
220    pub fn subscribe(&mut self, topic: v1::Topic) -> Result<u64, Error> {
221        self.subscribe_after(topic, None)
222    }
223
224    /// Subscribes to a topic, for events after the sequence number `after` first (the last event seen before a
225    /// reconnect): they are queued as notices before this returns. Returns the subscription's id.
226    pub fn subscribe_after(&mut self, topic: v1::Topic, after: Option<i64>) -> Result<u64, Error> {
227        self.subscribe_with(topic, after, false)
228    }
229
230    /// Subscribes to a topic, from `after` (see [`Client::subscribe_after`]); `columnar` asks for events as blocks.
231    pub fn subscribe_with(
232        &mut self,
233        topic: v1::Topic,
234        after: Option<i64>,
235        columnar: bool,
236    ) -> Result<u64, Error> {
237        match self.call(v1::request::Method::Subscribe(v1::Subscribe {
238            topic: topic.into(),
239            after,
240            columnar,
241        }))? {
242            v1::response::Result::Subscribed(subscribed) => Ok(subscribed.subscription),
243            _ => Err(Error::Protocol("not a subscription")),
244        }
245    }
246}

252    #[test]
253    fn a_socket_outside_roots_directories_is_not_the_node() {
254        let dir = std::env::temp_dir().join(format!("snitchhiker-client-{}", std::process::id()));
255        std::fs::create_dir_all(&dir).unwrap();
256        let path = dir.join("client.sock");
257        let _listener = std::os::unix::net::UnixListener::bind(&path).unwrap();
258        let result = Client::connect(&path, "test", "0");
259        std::fs::remove_dir_all(&dir).unwrap();
260        assert!(
261            matches!(result, Err(Error::NotTheNode(_))),
262            "{:?}",
263            result.err()
264        );
265    }
