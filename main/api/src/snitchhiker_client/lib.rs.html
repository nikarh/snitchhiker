1//! A blocking client of a Snitchhiker node over its local sockets: the handshake, requests (answered by id) and
2//! subscription notices (queued while a request waits for its answer).
3//!
4//! Before saying anything, a client checks it is talking to the node: the socket is in a directory only root can
5//! change, and the process serving it runs as root. A process that bound a socket somewhere else, or got one into a
6//! directory it may write, can't pose as the node to answer the client's requests or feed it prompts.
7
8use std::collections::VecDeque;
9use std::io::{BufReader, BufWriter, Write as _};

11use std::os::unix::net::UnixStream;
12use std::path::Path;
13use std::time::Duration;

19/// The node's sockets.
20pub const CLIENT_SOCKET: &str = "/run/snitchhiker/client.sock";
21pub const ADMIN_SOCKET: &str = "/run/snitchhiker/admin.sock";

23#[derive(Debug, thiserror::Error)]
24pub enum Error {
25    #[error("{0}")]
26    Io(#[from] std::io::Error),
27    #[error("the node refused the connection: {0}")]
28    Refused(String),
29    #[error("the node answered something unexpected: {0}")]
30    Protocol(&'static str),
31    #[error("not a Snitchhiker node's socket: {0}")]
32    NotTheNode(String),
33    #[error("{message}")]
34    Remote {
35        code: v1::ErrorCode,
36        message: String,
37    },
38}
39
40/// The socket and every directory above it belong to root, and no one else may write to those directories: no other
41/// user can have put the socket there, or swap it for theirs.
42fn only_root_can_change(path: &Path) -> Result<(), Error> {
43    const GROUP_OR_OTHER_WRITE: u32 = 0o022;
44    for (index, entry) in path.ancestors().enumerate() {
45        if entry.as_os_str().is_empty() {
46            continue;
47        }
48        let meta = std::fs::symlink_metadata(entry)
49            .map_err(|e| Error::NotTheNode(format!("{}: {e}", entry.display())))?;
50        if meta.uid() != 0 {
51            return Err(Error::NotTheNode(format!(
52                "{} doesn't belong to root",
53                entry.display()
54            )));
55        }
56        // The socket's own mode says who may connect, not who may replace it.
57        if index > 0 && meta.mode() & GROUP_OR_OTHER_WRITE != 0 {
58            return Err(Error::NotTheNode(format!(
59                "others than root may write to {}",
60                entry.display()
61            )));
62        }
63    }
64    Ok(())
65}

67pub struct Client {
68    reader: BufReader<UnixStream>,
69    writer: BufWriter<UnixStream>,
70    buffer: Vec<u8>,
71    next_id: u64,
72    notices: VecDeque<v1::Notice>,
73    welcome: v1::Welcome,
74}
75
76impl Client {
77    /// Connects to the node's socket at `path`, introducing itself as `app` (`cli`, `desktop`, ...) at `version`.
78    pub fn connect(path: &Path, app: &str, version: &str) -> Result<Self, Error> {
79        only_root_can_change(path)?;
80        let stream = UnixStream::connect(path)?;
81        let peer = getsockopt(&stream, sockopt::PeerCredentials)
82            .map_err(|e| Error::Io(std::io::Error::from(e)))?;
83        if peer.uid() != 0 {
84            return Err(Error::NotTheNode(format!(
85                "{} is served by uid {}, not root",
86                path.display(),
87                peer.uid()
88            )));
89        }
90        let mut writer = BufWriter::new(stream.try_clone()?);
91        let mut reader = BufReader::new(stream);
92        let mut buffer = Vec::new();
93        frame::write_preamble(&mut writer)?;
94        let hello = v1::ClientFrame {
95            body: Some(v1::client_frame::Body::Hello(v1::Hello {
96                minor: MINOR,
97                app: app.to_owned(),
98                app_version: version.to_owned(),
99                capabilities: vec!["prompts".to_owned()],
100            })),
101        };
102        frame::write(&mut writer, &hello, &mut buffer)?;
103        writer.flush()?;
104        let answer: v1::ServerFrame = frame::read(&mut reader, &mut buffer)?;
105        match answer.body {
106            Some(v1::server_frame::Body::Welcome(welcome)) => Ok(Self {
107                reader,
108                writer,
109                buffer,
110                next_id: 0,
111                notices: VecDeque::new(),
112                welcome,
113            }),
114            Some(v1::server_frame::Body::Refusal(refusal)) => Err(Error::Refused(refusal.reason)),
115            _ => Err(Error::Protocol("no welcome")),
116        }
117    }
118
119    #[must_use]
120    pub fn welcome(&self) -> &v1::Welcome {
121        &self.welcome
122    }
123
124    /// Sends a request and waits for its answer; notices arriving meanwhile are kept for [`Client::notice`]. A node's
125    /// error is `Error::Remote`.
126    pub fn call(&mut self, method: v1::request::Method) -> Result<v1::response::Result, Error> {
127        self.next_id += 1;
128        let id = self.next_id;
129        let request = v1::ClientFrame {
130            body: Some(v1::client_frame::Body::Request(v1::Request {
131                id,
132                method: Some(method),
133            })),
134        };
135        frame::write(&mut self.writer, &request, &mut self.buffer)?;
136        self.writer.flush()?;
137        loop {
138            let frame: v1::ServerFrame = frame::read(&mut self.reader, &mut self.buffer)?;
139            match frame.body {
140                Some(v1::server_frame::Body::Response(response)) if response.id == id => {
141                    return match response.result {
142                        Some(v1::response::Result::Error(error)) => Err(Error::Remote {
143                            code: error.code(),
144                            message: error.message,
145                        }),
146                        Some(result) => Ok(result),
147                        None => Err(Error::Protocol("an empty answer")),
148                    };
149                }
150                Some(v1::server_frame::Body::Notice(notice)) => self.notices.push_back(notice),
151                Some(v1::server_frame::Body::Refusal(refusal)) => {
152                    return Err(Error::Refused(refusal.reason));
153                }
154                _ => {}
155            }
156        }
157    }
158
159    /// The next subscription notice, waiting for one (up to the read timeout, if set).
160    pub fn notice(&mut self) -> Result<v1::Notice, Error> {
161        if let Some(notice) = self.notices.pop_front() {
162            return Ok(notice);
163        }
164        loop {
165            let frame: v1::ServerFrame = frame::read(&mut self.reader, &mut self.buffer)?;
166            match frame.body {
167                Some(v1::server_frame::Body::Notice(notice)) => return Ok(notice),
168                Some(v1::server_frame::Body::Refusal(refusal)) => {
169                    return Err(Error::Refused(refusal.reason));
170                }
171                _ => {}
172            }
173        }
174    }
175
176    /// How long reads wait (for [`Client::notice`]); `None` for ever.
177    pub fn set_read_timeout(&self, timeout: Option<Duration>) -> Result<(), Error> {
178        Ok(self.reader.get_ref().set_read_timeout(timeout)?)
179    }
180
181    /// Subscribes to a topic; returns the subscription's id.
182    pub fn subscribe(&mut self, topic: v1::Topic) -> Result<u64, Error> {
183        self.subscribe_after(topic, None)
184    }
185
186    /// Subscribes to a topic, for events after the sequence number `after` first (the last event seen before a
187    /// reconnect): they are queued as notices before this returns. Returns the subscription's id.
188    pub fn subscribe_after(&mut self, topic: v1::Topic, after: Option<i64>) -> Result<u64, Error> {
189        match self.call(v1::request::Method::Subscribe(v1::Subscribe {
190            topic: topic.into(),
191            after,
192        }))? {
193            v1::response::Result::Subscribed(subscribed) => Ok(subscribed.subscription),
194            _ => Err(Error::Protocol("not a subscription")),
195        }
196    }
197}

203    #[test]
204    fn a_socket_outside_roots_directories_is_not_the_node() {
205        let dir = std::env::temp_dir().join(format!("snitchhiker-client-{}", std::process::id()));
206        std::fs::create_dir_all(&dir).unwrap();
207        let path = dir.join("client.sock");
208        let _listener = std::os::unix::net::UnixListener::bind(&path).unwrap();
209        let result = Client::connect(&path, "test", "0");
210        std::fs::remove_dir_all(&dir).unwrap();
211        assert!(
212            matches!(result, Err(Error::NotTheNode(_))),
213            "{:?}",
214            result.err()
215        );
216    }
