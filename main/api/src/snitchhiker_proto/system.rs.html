9fn protocol_name(protocol: Option<u8>) -> String {
10    match protocol {
11        None => String::new(),
12        Some(TCP) => "tcp".to_owned(),
13        Some(UDP) => "udp".to_owned(),
14        Some(ICMP) => "icmp".to_owned(),
15        Some(ICMPV6) => "icmpv6".to_owned(),
16        Some(n) => n.to_string(),
17    }
18}
19
20fn ports(range: Option<(u16, u16)>) -> String {
21    match range {
22        None => String::new(),
23        Some((low, high)) if low == high => low.to_string(),
24        Some((low, high)) => format!("{low}-{high}"),
25    }
26}

28/// The firewall as the protocol has it.
29#[must_use]
30pub fn firewall(firewall: &Firewall, reverts_at_ms: u64) -> v1::SystemFirewall {
31    let policy = |p: ChainPolicy| match p {
32        ChainPolicy::Accept => v1::ChainPolicy::Accept,
33        ChainPolicy::Drop => v1::ChainPolicy::Drop,
34    };
35    v1::SystemFirewall {
36        enabled: firewall.enabled,
37        input: policy(firewall.input).into(),
38        forward: policy(firewall.forward).into(),
39        rules: firewall
40            .rules
41            .iter()
42            .map(|rule| v1::SystemRule {
43                enabled: rule.enabled,
44                description: rule.description.clone(),
45                chain: match rule.chain {
46                    Chain::Input => v1::SystemChain::Input,
47                    Chain::Forward => v1::SystemChain::Forward,
48                    Chain::Output => v1::SystemChain::Output,
49                }
50                .into(),
51                protocol: protocol_name(rule.protocol),
52                source: rule.source.iter().map(ToString::to_string).collect(),
53                destination: rule.destination.iter().map(ToString::to_string).collect(),
54                source_ports: ports(rule.source_ports),
55                destination_ports: ports(rule.destination_ports),
56                in_interface: rule.in_interface.clone().unwrap_or_default(),
57                out_interface: rule.out_interface.clone().unwrap_or_default(),
58                ct_state: rule.ct_state.iter().map(|s| s.name().to_owned()).collect(),
59                icmp_types: rule
60                    .icmp_types
61                    .iter()
62                    .map(|t| t.name().to_owned())
63                    .collect(),
64                target: match rule.target {
65                    Target::Accept => v1::SystemTarget::Accept,
66                    Target::Drop => v1::SystemTarget::Drop,
67                    Target::Reject => v1::SystemTarget::Reject,
68                    Target::Log => v1::SystemTarget::Log,
69                    Target::Queue => v1::SystemTarget::Queue,
70                }
71                .into(),
72            })
73            .collect(),
74        reverts_at_ms,
75    }
76}

78fn parse_ports(text: &str, n: usize) -> Result<Option<(u16, u16)>, String> {
79    let text = text.trim();
80    if text.is_empty() {
81        return Ok(None);
82    }
83    let bad = || format!("rule {n}: `{text}` isn't a port or a range");
84    let (low, high) = text.split_once('-').unwrap_or((text, text));
85    Ok(Some((
86        low.trim().parse().map_err(|_| bad())?,
87        high.trim().parse().map_err(|_| bad())?,
88    )))
89}

91/// The firewall a client sent, checked.
92pub fn firewall_from(wire: &v1::SystemFirewall) -> Result<Firewall, String> {
93    let policy = |p: i32| match v1::ChainPolicy::try_from(p) {
94        Ok(v1::ChainPolicy::Drop) => ChainPolicy::Drop,
95        _ => ChainPolicy::Accept,
96    };
97    let mut rules = Vec::with_capacity(wire.rules.len());
98    for (i, rule) in wire.rules.iter().enumerate() {
99        let n = i + 1;
100        let nets = |list: &[String]| {
101            list.iter()
102                .filter(|t| !t.trim().is_empty())
103                .map(|text| {
104                    system::parse_network(text)
105                        .map_err(|_| format!("rule {n}: `{text}` isn't an address or a network"))
106                })
107                .collect::<Result<Vec<_>, _>>()
108        };
109        let protocol = match rule.protocol.trim().to_ascii_lowercase().as_str() {
110            "" => None,
111            "tcp" => Some(TCP),
112            "udp" => Some(UDP),
113            "icmp" => Some(ICMP),
114            "icmpv6" => Some(ICMPV6),
115            other => Some(
116                other
117                    .parse()
118                    .map_err(|_| format!("rule {n}: `{other}` isn't a protocol"))?,
119            ),
120        };
121        let interface = |text: &str| {
122            let text = text.trim();
123            (!text.is_empty()).then(|| text.to_owned())
124        };
125        rules.push(Rule {
126            enabled: rule.enabled,
127            description: rule.description.trim().to_owned(),
128            chain: match v1::SystemChain::try_from(rule.chain) {
129                Ok(v1::SystemChain::Forward) => Chain::Forward,
130                Ok(v1::SystemChain::Output) => Chain::Output,
131                _ => Chain::Input,
132            },
133            protocol,
134            source: nets(&rule.source)?,
135            destination: nets(&rule.destination)?,
136            source_ports: parse_ports(&rule.source_ports, n)?,
137            destination_ports: parse_ports(&rule.destination_ports, n)?,
138            in_interface: interface(&rule.in_interface),
139            out_interface: interface(&rule.out_interface),
140            ct_state: rule
141                .ct_state
142                .iter()
143                .map(|s| {
144                    CtState::parse(s)
145                        .ok_or_else(|| format!("rule {n}: `{s}` isn't a connection state"))
146                })
147                .collect::<Result<_, _>>()?,
148            icmp_types: rule
149                .icmp_types
150                .iter()
151                .map(|t| {
152                    IcmpType::parse(t).ok_or_else(|| format!("rule {n}: `{t}` isn't an ICMP type"))
153                })
154                .collect::<Result<_, _>>()?,
155            target: match v1::SystemTarget::try_from(rule.target) {
156                Ok(v1::SystemTarget::Drop) => Target::Drop,
157                Ok(v1::SystemTarget::Reject) => Target::Reject,
158                Ok(v1::SystemTarget::Log) => Target::Log,
159                Ok(v1::SystemTarget::Queue) => Target::Queue,
160                _ => Target::Accept,
161            },
162        });
163    }
164    let firewall = Firewall {
165        enabled: wire.enabled,
166        input: policy(wire.input),
167        forward: policy(wire.forward),
168        rules,
169    };
170    firewall.validate().map_err(|e| e.to_string())?;
171    Ok(firewall)
172}

182        let mut bad = firewall(&defaults, 0);
183        bad.rules[2].destination_ports = "http".to_owned();
184        assert!(firewall_from(&bad).unwrap_err().contains("rule 3"));
