1//! Rules as people write them: the TOML (or JSON) of rule files, exports and managed `rules.d` files.
2//!
3//! ```toml
4//! [folder]
5//! name = "Claude Code"
6//! exe = ["glob:{home}/.local/share/claude/versions/*/claude"]   # a scope: added to every rule below
7//!
8//! [[rule]]
9//! name = "API"
10//! action = "allow"
11//! host = ["api.example.com", "*.example.org"]
12//! port = ["443"]
13//! ```
14//!
15//! Patterns are exact unless prefixed with `glob:` or `re:`. Ports are numbers or `low-high` ranges; protocols are
16//! `tcp`, `udp`, `icmp`, `icmpv6` or numbers.
17

154#[derive(Debug, Clone, PartialEq, Eq, thiserror::Error)]
155pub enum SpecError {
156    #[error("{0}")]
157    Host(#[from] crate::destination::HostPatternError),
158    #[error("{0}")]
159    Network(#[from] crate::destination::NetworkError),
160    #[error("invalid port `{0}`: a number or low-high")]
161    Port(String),
162    #[error("invalid protocol `{0}`: tcp, udp, icmp, icmpv6 or a number")]
163    Protocol(String),
164    #[error("invalid session `{0}`: pid@start")]
165    Session(String),
166    #[error("invalid label `{0}`: key=value")]
167    Label(String),
168    #[error("rule `{0}` has no conditions: it would match every connection")]
169    NoConditions(String),
170}

176/// A pattern as written: `glob:`, `re:` or exact.
177#[must_use]
178pub fn pattern(text: &str) -> PatternSource {
179    if let Some(glob) = text.strip_prefix("glob:") {
180        PatternSource::Glob(glob.to_owned())
181    } else if let Some(regex) = text.strip_prefix("re:") {
182        PatternSource::Regex(regex.to_owned())
183    } else {
184        PatternSource::Exact(text.to_owned())
185    }
186}
187
188fn pattern_text(pattern: &PatternSource) -> String {
189    match pattern {
190        PatternSource::Exact(text) => text.clone(),
191        PatternSource::Glob(text) => format!("glob:{text}"),
192        PatternSource::Regex(text) => format!("re:{text}"),
193    }
194}
195
196fn port_range(text: &str) -> Result<(u16, u16), SpecError> {
197    let error = || SpecError::Port(text.to_owned());
198    let (low, high) = text.split_once('-').unwrap_or((text, text));
199    let low: u16 = low.trim().parse().map_err(|_| error())?;
200    let high: u16 = high.trim().parse().map_err(|_| error())?;
201    if low > high {
202        return Err(error());
203    }
204    Ok((low, high))
205}
206
207fn protocol(text: &str) -> Result<u8, SpecError> {
208    match text {
209        "tcp" => Ok(6),
210        "udp" => Ok(17),
211        "icmp" => Ok(1),
212        "icmpv6" => Ok(58),
213        other => other
214            .parse()
215            .map_err(|_| SpecError::Protocol(text.to_owned())),
216    }
217}

219impl Conditions {
220    /// The model's conditions.
221    pub fn to_conditions(&self) -> Result<Vec<Condition>, SpecError> {
222        let patterns = |texts: &[String]| texts.iter().map(|t| pattern(t)).collect::<Vec<_>>();
223        let mut conditions = Vec::new();
224        if !self.exe.is_empty() {
225            conditions.push(Condition::Exe(patterns(&self.exe)));
226        }
227        if !self.app.is_empty() {
228            conditions.push(Condition::App(patterns(&self.app)));
229        }
230        if !self.cmdline.is_empty() {
231            conditions.push(Condition::Cmdline(patterns(&self.cmdline)));
232        }
233        if !self.parent.is_empty() {
234            conditions.push(Condition::Parent(patterns(&self.parent)));
235        }
236        if !self.uid.is_empty() {
237            conditions.push(Condition::Uid(self.uid.clone()));
238        }
239        if !self.host.is_empty() {
240            let hosts = self
241                .host
242                .iter()
243                .map(|h| HostPattern::try_from(h.clone()))
244                .collect::<Result<_, _>>()?;
245            conditions.push(Condition::Host(hosts));
246        }
247        if !self.net.is_empty() {
248            let nets = self
249                .net
250                .iter()
251                .map(|n| Network::try_from(n.clone()))
252                .collect::<Result<_, _>>()?;
253            conditions.push(Condition::Net(nets));
254        }
255        if !self.port.is_empty() {
256            let ports = self
257                .port
258                .iter()
259                .map(|p| port_range(p))
260                .collect::<Result<_, _>>()?;
261            conditions.push(Condition::Port(ports));
262        }
263        if !self.protocol.is_empty() {
264            let protocols = self
265                .protocol
266                .iter()
267                .map(|p| protocol(p))
268                .collect::<Result<_, _>>()?;
269            conditions.push(Condition::Protocol(protocols));
270        }
271        if let Some(context) = self.context {
272            conditions.push(Condition::Context(context));
273        }
274        if !self.session.is_empty() {
275            let roots = self
276                .session
277                .iter()
278                .map(|s| s.parse().map_err(|()| SpecError::Session(s.clone())))
279                .collect::<Result<_, _>>()?;
280            conditions.push(Condition::Session(roots));
281        }
282        self.identity_conditions(&mut conditions)?;
283        Ok(conditions)
284    }
285
286    /// The conditions on a container's, a VM's or a namespace's identity.
287    fn identity_conditions(&self, conditions: &mut Vec<Condition>) -> Result<(), SpecError> {
288        let patterns = |texts: &[String]| texts.iter().map(|t| pattern(t)).collect::<Vec<_>>();
289        let c = &self.container;
290        if !c.name.is_empty() {
291            conditions.push(Condition::ContainerName(patterns(&c.name)));
292        }
293        if !c.image.is_empty() {
294            conditions.push(Condition::ContainerImage(patterns(&c.image)));
295        }
296        if !c.compose.is_empty() {
297            conditions.push(Condition::ContainerCompose(patterns(&c.compose)));
298        }
299        if !c.label.is_empty() {
300            let labels = c
301                .label
302                .iter()
303                .map(|l| {
304                    let (key, value) = l
305                        .split_once('=')
306                        .ok_or_else(|| SpecError::Label(l.clone()))?;
307                    Ok((key.to_owned(), pattern(value)))
308                })
309                .collect::<Result<_, SpecError>>()?;
310            conditions.push(Condition::ContainerLabel(labels));
311        }
312        if !c.runtime.is_empty() {
313            conditions.push(Condition::ContainerRuntime(c.runtime.clone()));
314        }
315        if !c.owner.is_empty() {
316            conditions.push(Condition::ContainerOwner(c.owner.clone()));
317        }
318        if !c.id.is_empty() {
319            conditions.push(Condition::ContainerId(c.id.clone()));
320        }
321        if !self.vm.name.is_empty() {
322            conditions.push(Condition::VmName(patterns(&self.vm.name)));
323        }
324        if !self.vm.uuid.is_empty() {
325            conditions.push(Condition::VmUuid(self.vm.uuid.clone()));
326        }
327        if !self.vm.hypervisor.is_empty() {
328            conditions.push(Condition::VmHypervisor(self.vm.hypervisor.clone()));
329        }
330        if !self.netns.is_empty() {
331            conditions.push(Condition::NetnsName(patterns(&self.netns)));
332        }
333        if !self.list.is_empty() {
334            conditions.push(Condition::List(self.list.clone()));
335        }
336        if !self.sha256.is_empty() {
337            conditions.push(Condition::Sha256(
338                self.sha256.iter().map(|h| h.to_ascii_lowercase()).collect(),
339            ));
340        }
341        if let Some(writable) = self.writable {
342            conditions.push(Condition::Writable(writable));
343        }
344        Ok(())
345    }

347    /// The written form of model conditions (negations can't be written in this form and are left out).
348    #[must_use]
349    pub fn from_conditions(conditions: &[Condition]) -> Self {
350        let mut spec = Self::default();
351        for condition in conditions {
352            match condition {
353                Condition::Exe(p) => spec.exe.extend(p.iter().map(pattern_text)),
354                Condition::App(p) => spec.app.extend(p.iter().map(pattern_text)),
355                Condition::Cmdline(p) => spec.cmdline.extend(p.iter().map(pattern_text)),
356                Condition::Parent(p) => spec.parent.extend(p.iter().map(pattern_text)),
357                Condition::Uid(u) => spec.uid.extend(u),
358                Condition::Host(h) => spec.host.extend(h.iter().map(ToString::to_string)),
359                Condition::Net(n) => spec.net.extend(n.iter().map(ToString::to_string)),
360                Condition::Port(p) => spec.port.extend(p.iter().map(|&(low, high)| {
361                    if low == high {
362                        low.to_string()
363                    } else {
364                        format!("{low}-{high}")
365                    }
366                })),
367                Condition::Protocol(p) => spec.protocol.extend(p.iter().map(|&n| match n {
368                    6 => "tcp".to_owned(),
369                    17 => "udp".to_owned(),
370                    1 => "icmp".to_owned(),
371                    58 => "icmpv6".to_owned(),
372                    n => n.to_string(),
373                })),
374                Condition::Context(c) => spec.context = Some(*c),
375                Condition::Session(roots) => {
376                    spec.session.extend(roots.iter().map(ToString::to_string));
377                }
378                Condition::ContainerName(p) => {
379                    spec.container.name.extend(p.iter().map(pattern_text));
380                }
381                Condition::ContainerImage(p) => {
382                    spec.container.image.extend(p.iter().map(pattern_text));
383                }
384                Condition::ContainerCompose(p) => {
385                    spec.container.compose.extend(p.iter().map(pattern_text));
386                }
387                Condition::ContainerLabel(labels) => spec.container.label.extend(
388                    labels
389                        .iter()
390                        .map(|(k, v)| format!("{k}={}", pattern_text(v))),
391                ),
392                Condition::ContainerRuntime(r) => spec.container.runtime.extend(r.iter().cloned()),
393                Condition::ContainerOwner(u) => spec.container.owner.extend(u),
394                Condition::ContainerId(i) => spec.container.id.extend(i.iter().cloned()),
395                Condition::VmName(p) => spec.vm.name.extend(p.iter().map(pattern_text)),
396                Condition::VmUuid(u) => spec.vm.uuid.extend(u.iter().cloned()),
397                Condition::VmHypervisor(h) => spec.vm.hypervisor.extend(h.iter().cloned()),
398                Condition::NetnsName(p) => spec.netns.extend(p.iter().map(pattern_text)),
399                Condition::List(names) => spec.list.extend(names.iter().cloned()),
400                Condition::Sha256(hashes) => spec.sha256.extend(hashes.iter().cloned()),
401                Condition::Writable(writable) => spec.writable = Some(*writable),
402                Condition::Not(_) => {}
403            }
404        }
405        spec
406    }
407}

409impl RuleSpec {
410    /// The model's rule, with `id` and in `folder`. A rule without any condition is refused: it would decide every
411    /// connection, which a typo shouldn't do.
412    pub fn to_rule(&self, id: u64, folder: Option<u64>) -> Result<Rule, SpecError> {
413        let conditions = self.conditions.to_conditions()?;
414        if conditions.is_empty() {
415            return Err(SpecError::NoConditions(self.name.clone()));
416        }
417        Ok(Rule {
418            id,
419            folder,
420            name: self.name.clone(),
421            enabled: self.enabled,
422            action: self.action,
423            tier: self.tier,
424            expiry: Expiry::Never,
425            log: true,
426            notes: self.notes.clone(),
427            conditions,
428        })
429    }
430
431    #[must_use]
432    pub fn from_rule(rule: &Rule) -> Self {
433        Self {
434            name: rule.name.clone(),
435            action: rule.action,
436            tier: rule.tier,
437            enabled: rule.enabled,
438            notes: rule.notes.clone(),
439            conditions: Conditions::from_conditions(&rule.conditions),
440        }
441    }

483    #[test]
484    fn reads_a_rule_file() {
485        let file: RuleFile = toml::from_str(FILE).unwrap();
486        let folder = file.folder.as_ref().unwrap();
487        assert_eq!(folder.scope.to_conditions().unwrap().len(), 1);
488        let api = file.rule[0].to_rule(1, Some(9)).unwrap();
489        assert_eq!(api.action, Action::Allow);
490        assert_eq!(
491            api.conditions[1],
492            Condition::Port(vec![(443, 443), (8000, 8080)])
493        );
494        let imds = file.rule[1].to_rule(2, None).unwrap();
495        assert_eq!(imds.tier, Tier::Override);
496        assert_eq!(imds.conditions[1], Condition::Context(ContextKind::Any));
497        let web = file.rule[2].to_rule(3, None).unwrap();
498        assert!(
499            web.conditions
500                .contains(&Condition::ContainerCompose(vec![PatternSource::Exact(
501                    "shop/web".to_owned()
502                )]))
503        );
504        assert!(web.conditions.contains(&Condition::ContainerLabel(vec![(
505            "tier".to_owned(),
506            PatternSource::Exact("front".to_owned())
507        )])));
508        let vm = file.rule[3].to_rule(4, None).unwrap();
509        assert!(
510            vm.conditions
511                .contains(&Condition::VmName(vec![PatternSource::Glob(
