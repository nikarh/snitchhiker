1//! Rules as people write them: the TOML (or JSON) of rule files, exports and managed `rules.d` files.
2//!
3//! ```toml
4//! [folder]
5//! name = "Claude Code"
6//! exe = ["glob:{home}/.local/share/claude/versions/*/claude"]   # a scope: added to every rule below
7//!
8//! [[rule]]
9//! name = "API"
10//! action = "allow"
11//! host = ["api.example.com", "*.example.org"]
12//! port = ["443"]
13//! ```
14//!
15//! Patterns are exact unless prefixed with `glob:` or `re:`. Ports are numbers or `low-high` ranges; protocols are
16//! `tcp`, `udp`, `icmp`, `icmpv6` or numbers.
17

40    #[serde(default, skip_serializing_if = "Vec::is_empty")]

86#[derive(Debug, Clone, PartialEq, Eq, thiserror::Error)]
87pub enum SpecError {
88    #[error("{0}")]
89    Host(#[from] crate::destination::HostPatternError),
90    #[error("{0}")]
91    Network(#[from] crate::destination::NetworkError),
92    #[error("invalid port `{0}`: a number or low-high")]
93    Port(String),
94    #[error("invalid protocol `{0}`: tcp, udp, icmp, icmpv6 or a number")]
95    Protocol(String),
96    #[error("invalid session `{0}`: pid@start")]
97    Session(String),
98    #[error("rule `{0}` has no conditions: it would match every connection")]
99    NoConditions(String),
100}

106fn pattern(text: &str) -> PatternSource {
107    if let Some(glob) = text.strip_prefix("glob:") {
108        PatternSource::Glob(glob.to_owned())
109    } else if let Some(regex) = text.strip_prefix("re:") {
110        PatternSource::Regex(regex.to_owned())
111    } else {
112        PatternSource::Exact(text.to_owned())
113    }
114}
115
116fn pattern_text(pattern: &PatternSource) -> String {
117    match pattern {
118        PatternSource::Exact(text) => text.clone(),
119        PatternSource::Glob(text) => format!("glob:{text}"),
120        PatternSource::Regex(text) => format!("re:{text}"),
121    }
122}
123
124fn port_range(text: &str) -> Result<(u16, u16), SpecError> {
125    let error = || SpecError::Port(text.to_owned());
126    let (low, high) = text.split_once('-').unwrap_or((text, text));
127    let low: u16 = low.trim().parse().map_err(|_| error())?;
128    let high: u16 = high.trim().parse().map_err(|_| error())?;
129    if low > high {
130        return Err(error());
131    }
132    Ok((low, high))
133}
134
135fn protocol(text: &str) -> Result<u8, SpecError> {
136    match text {
137        "tcp" => Ok(6),
138        "udp" => Ok(17),
139        "icmp" => Ok(1),
140        "icmpv6" => Ok(58),
141        other => other
142            .parse()
143            .map_err(|_| SpecError::Protocol(text.to_owned())),
144    }
145}

147impl Conditions {
148    /// The model's conditions.
149    pub fn to_conditions(&self) -> Result<Vec<Condition>, SpecError> {
150        let patterns = |texts: &[String]| texts.iter().map(|t| pattern(t)).collect::<Vec<_>>();
151        let mut conditions = Vec::new();
152        if !self.exe.is_empty() {
153            conditions.push(Condition::Exe(patterns(&self.exe)));
154        }
155        if !self.app.is_empty() {
156            conditions.push(Condition::App(patterns(&self.app)));
157        }
158        if !self.cmdline.is_empty() {
159            conditions.push(Condition::Cmdline(patterns(&self.cmdline)));
160        }
161        if !self.parent.is_empty() {
162            conditions.push(Condition::Parent(patterns(&self.parent)));
163        }
164        if !self.uid.is_empty() {
165            conditions.push(Condition::Uid(self.uid.clone()));
166        }
167        if !self.host.is_empty() {
168            let hosts = self
169                .host
170                .iter()
171                .map(|h| HostPattern::try_from(h.clone()))
172                .collect::<Result<_, _>>()?;
173            conditions.push(Condition::Host(hosts));
174        }
175        if !self.net.is_empty() {
176            let nets = self
177                .net
178                .iter()
179                .map(|n| Network::try_from(n.clone()))
180                .collect::<Result<_, _>>()?;
181            conditions.push(Condition::Net(nets));
182        }
183        if !self.port.is_empty() {
184            let ports = self
185                .port
186                .iter()
187                .map(|p| port_range(p))
188                .collect::<Result<_, _>>()?;
189            conditions.push(Condition::Port(ports));
190        }
191        if !self.protocol.is_empty() {
192            let protocols = self
193                .protocol
194                .iter()
195                .map(|p| protocol(p))
196                .collect::<Result<_, _>>()?;
197            conditions.push(Condition::Protocol(protocols));
198        }
199        if let Some(context) = self.context {
200            conditions.push(Condition::Context(context));
201        }
202        if !self.session.is_empty() {
203            let roots = self
204                .session
205                .iter()
206                .map(|s| s.parse().map_err(|()| SpecError::Session(s.clone())))
207                .collect::<Result<_, _>>()?;
208            conditions.push(Condition::Session(roots));
209        }
210        Ok(conditions)
211    }
212
213    /// The written form of model conditions (negations can't be written in this form and are left out).
214    #[must_use]
215    pub fn from_conditions(conditions: &[Condition]) -> Self {
216        let mut spec = Self::default();
217        for condition in conditions {
218            match condition {
219                Condition::Exe(p) => spec.exe.extend(p.iter().map(pattern_text)),
220                Condition::App(p) => spec.app.extend(p.iter().map(pattern_text)),
221                Condition::Cmdline(p) => spec.cmdline.extend(p.iter().map(pattern_text)),
222                Condition::Parent(p) => spec.parent.extend(p.iter().map(pattern_text)),
223                Condition::Uid(u) => spec.uid.extend(u),
224                Condition::Host(h) => spec.host.extend(h.iter().map(ToString::to_string)),
225                Condition::Net(n) => spec.net.extend(n.iter().map(ToString::to_string)),
226                Condition::Port(p) => spec.port.extend(p.iter().map(|&(low, high)| {
227                    if low == high {
228                        low.to_string()
229                    } else {
230                        format!("{low}-{high}")
231                    }
232                })),
233                Condition::Protocol(p) => spec.protocol.extend(p.iter().map(|&n| match n {
234                    6 => "tcp".to_owned(),
235                    17 => "udp".to_owned(),
236                    1 => "icmp".to_owned(),
237                    58 => "icmpv6".to_owned(),
238                    n => n.to_string(),
239                })),
240                Condition::Context(c) => spec.context = Some(*c),
241                Condition::Session(roots) => {
242                    spec.session.extend(roots.iter().map(ToString::to_string));
243                }
244                Condition::Not(_) => {}
245            }
246        }
247        spec
248    }
249}

251impl RuleSpec {
252    /// The model's rule, with `id` and in `folder`. A rule without any condition is refused: it would decide every
253    /// connection, which a typo shouldn't do.
254    pub fn to_rule(&self, id: u64, folder: Option<u64>) -> Result<Rule, SpecError> {
255        let conditions = self.conditions.to_conditions()?;
256        if conditions.is_empty() {
257            return Err(SpecError::NoConditions(self.name.clone()));
258        }
259        Ok(Rule {
260            id,
261            folder,
262            name: self.name.clone(),
263            enabled: self.enabled,
264            action: self.action,
265            tier: self.tier,
266            expiry: Expiry::Never,
267            log: true,
268            notes: self.notes.clone(),
269            conditions,
270        })
271    }
272
273    #[must_use]
274    pub fn from_rule(rule: &Rule) -> Self {
275        Self {
276            name: rule.name.clone(),
277            action: rule.action,
278            tier: rule.tier,
279            enabled: rule.enabled,
280            notes: rule.notes.clone(),
281            conditions: Conditions::from_conditions(&rule.conditions),
282        }
283    }
284}

287mod tests {
288    use super::*;
289
290    const FILE: &str = r#"
291[folder]
292name = "Claude Code"
293exe = ["glob:{home}/.local/share/claude/versions/*/claude"]
294
295[[rule]]
296name = "API"
297action = "allow"
298host = ["api.example.com", "*.example.org"]
299port = ["443", "8000-8080"]
300protocol = ["tcp"]
301
302[[rule]]
303name = "Never the metadata service"
304action = "deny"
305tier = "override"
306net = ["imds"]
307context = "any"
308"#;
309
310    #[test]
311    fn reads_a_rule_file() {
312        let file: RuleFile = toml::from_str(FILE).unwrap();
313        let folder = file.folder.as_ref().unwrap();
314        assert_eq!(folder.scope.to_conditions().unwrap().len(), 1);
315        let api = file.rule[0].to_rule(1, Some(9)).unwrap();
316        assert_eq!(api.action, Action::Allow);
317        assert_eq!(
318            api.conditions[1],
319            Condition::Port(vec![(443, 443), (8000, 8080)])
320        );
321        let imds = file.rule[1].to_rule(2, None).unwrap();
322        assert_eq!(imds.tier, Tier::Override);
323        assert_eq!(imds.conditions[1], Condition::Context(ContextKind::Any));
324    }
325
326    #[test]
327    fn round_trips_through_the_model() {
328        let file: RuleFile = toml::from_str(FILE).unwrap();
329        for spec in &file.rule {
330            let rule = spec.to_rule(1, None).unwrap();
331            assert_eq!(&RuleSpec::from_rule(&rule), spec);
332        }
333    }
334
335    #[test]
336    fn refuses_rules_matching_everything_and_bad_values() {
337        let empty: RuleFile =
338            toml::from_str("[[rule]]\nname = \"x\"\naction = \"allow\"\n").unwrap();
339        assert!(matches!(
340            empty.rule[0].to_rule(1, None),
341            Err(SpecError::NoConditions(_))
342        ));
343        let bad: RuleFile =
344            toml::from_str("[[rule]]\nname = \"x\"\naction = \"allow\"\nport = [\"9-1\"]\n")
345                .unwrap();
346        assert!(bad.rule[0].to_rule(1, None).is_err());
347        assert!(
348            toml::from_str::<RuleFile>(
349                "[[rule]]\nname = \"x\"\naction = \"allow\"\nhots = [\"a\"]\n"
350            )
351            .is_err()
352        );
353    }
