1//! The lint and dedupe report over a rule set: duplicates, rules another one makes redundant, allow rules a deny always
2//! beats, expired rules, and, from what the node knows, rules never hit, on binaries any user can replace, or on
3//! binaries that are gone (with the pattern that would cover their new version).
4//!
5//! Every finding is certain: a rule is called redundant only when the other rule matches everything it matches, which
6//! is decided on the conditions as written (an exact value inside a pattern, a network inside a network, a port range
7//! inside a range), never guessed from names.
8
9use crate::destination::{HostPattern, Network};
10use crate::generalize;
11use crate::pattern::{HOME, Pattern, PatternSource};
12use crate::rule::{Action, Condition, ContextKind, Expiry, Rule, RuleId};

14/// What the node knows that the rules don't.
15pub trait NodeFacts {
16    /// Whether the file exists.
17    fn exists(&self, path: &str) -> bool;
18    /// Whether a user other than root can replace the file (it, or a directory above it, is writable by one).
19    fn writable_by_user(&self, path: &str) -> bool;
20    /// Whether some file matches the glob (`*` within one path component).
21    fn exists_like(&self, glob: &str) -> bool;
22    /// When the rule last decided a connection, and when it was last changed (Unix ms).
23    fn last_hit_ms(&self, rule: RuleId) -> Option<u64>;
24    fn changed_ms(&self, rule: RuleId) -> Option<u64>;
25}
26
27#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
28pub enum Kind {
29    /// Matches exactly what `other` matches, with the same action.
30    Duplicate,
31    /// `other` matches everything this one does, with the same action, at the same tier or above.
32    Redundant,
33    /// An allow rule that a deny or reject (`other`) always beats.
34    Shadowed,
35    /// Its time is up; it no longer applies.
36    Expired,
37    /// It decided no connection in the days asked about.
38    NeverHit,
39    /// An allow rule on a binary that a user other than root can replace.
40    WritableBinary,
41    /// Its binary is gone; `suggestion` covers the one that took its place, if there is one.
42    MissingBinary,
43}
44
45#[derive(Debug, Clone, PartialEq, Eq)]
46pub struct Finding {
47    pub kind: Kind,
48    pub rule: RuleId,
49    pub other: Option<RuleId>,
50    /// A pattern to replace the rule's program with.
51    pub suggestion: Option<PatternSource>,
52}

54/// A rule as the lint sees it: with its folders' scopes, and whether it is in effect (it and its folders enabled).
55pub struct Linted<'a> {
56    pub rule: &'a Rule,
57    /// Its own conditions and its folders' scopes.
58    pub conditions: Vec<&'a Condition>,
59    pub effective: bool,
60    /// Locked rules can't be changed: they make others redundant but aren't reported themselves.
61    pub locked: bool,
62}
63
64/// The report: findings in rule order.
65pub fn lint(
66    rules: &[Linted<'_>],
67    facts: &dyn NodeFacts,
68    now_ms: u64,
69    unused_days: u32,
70) -> Vec<Finding> {
71    let mut findings = Vec::new();
72    let finding = |kind, rule: &Rule, other: Option<&Rule>| Finding {
73        kind,
74        rule: rule.id,
75        other: other.map(|o| o.id),
76        suggestion: None,
77    };
78    for (i, a) in rules.iter().enumerate() {
79        if a.locked {
80            continue;
81        }
82        let rule = a.rule;
83        if let Expiry::At(at) = rule.expiry
84            && at <= now_ms
85        {
86            findings.push(finding(Kind::Expired, rule, None));
87            continue;
88        }
89        if !a.effective {
90            continue;
91        }
92        // Redundancy and shadowing, against every other rule in effect.
93        let mut found = None;
94        for (j, b) in rules.iter().enumerate() {
95            if i == j || !b.effective || matches!(b.rule.expiry, Expiry::At(at) if at <= now_ms) {
96                continue;
97            }
98            let (other, mine) = (b.rule, rule);
99            if !covers(&b.conditions, &a.conditions) {
100                continue;
101            }
102            if other.action == mine.action && other.tier >= mine.tier {
103                let mutual = other.tier == mine.tier && covers(&a.conditions, &b.conditions);
104                // Of two duplicates, the newer one is reported (unless the older one is locked).
105                if mutual && !b.locked && other.id > mine.id {
106                    continue;
107                }
108                found = Some(finding(
109                    if mutual {
110                        Kind::Duplicate
111                    } else {
112                        Kind::Redundant
113                    },
114                    mine,
115                    Some(other),
116                ));
117                break;
118            }
119            if mine.action == Action::Allow && other.action.blocks() && other.tier >= mine.tier {
120                found = Some(finding(Kind::Shadowed, mine, Some(other)));
121                break;
122            }
123        }
124        findings.extend(found);
125        // What the node knows.
126        let since = now_ms.saturating_sub(u64::from(unused_days) * 86_400_000);
127        if unused_days > 0
128            && facts
129                .changed_ms(rule.id)
130                .is_some_and(|changed| changed < since)
131            && facts.last_hit_ms(rule.id).is_none_or(|hit| hit < since)
132        {
133            findings.push(finding(Kind::NeverHit, rule, None));
134        }
135        for path in exact_exes(rule) {
136            if !facts.exists(path) {
137                let generalized = generalize::executable(path, None);
138                let suggestion = match &generalized {
139                    PatternSource::Glob(glob) if facts.exists_like(glob) => Some(generalized),
140                    _ => None,
141                };
142                findings.push(Finding {
143                    suggestion,
144                    ..finding(Kind::MissingBinary, rule, None)
145                });
146            } else if rule.action == Action::Allow && facts.writable_by_user(path) {
147                findings.push(finding(Kind::WritableBinary, rule, None));
148            }
149        }
150    }
151    findings
152}

154/// A rule's programs given as exact paths (not under `{home}`: whose home it is depends on the connection).
155fn exact_exes(rule: &Rule) -> impl Iterator<Item = &str> {
156    rule.conditions.iter().flat_map(|c| match c {
157        Condition::Exe(patterns) => patterns
158            .iter()
159            .filter_map(|p| match p {
160                PatternSource::Exact(path) if !path.starts_with(HOME) && path.starts_with('/') => {
161                    Some(path.as_str())
162                }
163                _ => None,
164            })
165            .collect::<Vec<_>>(),
166        _ => Vec::new(),
167    })
168}

170/// Whether a rule with conditions `broad` matches every connection one with `narrow` does: each of `broad`'s
171/// conditions is implied by one of `narrow`'s. A rule without a context condition is on the host.
172#[must_use]
173pub fn covers(broad: &[&Condition], narrow: &[&Condition]) -> bool {
174    let host = Condition::Context(ContextKind::Host);
175    let context = |conditions: &[&Condition]| {
176        !conditions
177            .iter()
178            .any(|c| matches!(c, Condition::Context(_)))
179    };
180    let narrow_host = context(narrow).then_some(&host);
181    let broad_host = context(broad).then_some(&host);
182    broad.iter().copied().chain(broad_host).all(|b| {
183        narrow
184            .iter()
185            .copied()
186            .chain(narrow_host)
187            .any(|n| implies(n, b))
188    })
189}
190
191/// Whether every connection meeting `narrow` meets `broad`.
192fn implies(narrow: &Condition, broad: &Condition) -> bool {
193    let all = |narrow: &[PatternSource], broad: &[PatternSource]| {
194        narrow
195            .iter()
196            .all(|n| broad.iter().any(|b| pattern_covers(b, n)))
197    };
198    match (narrow, broad) {
199        (Condition::Exe(n), Condition::Exe(b))
200        | (Condition::App(n), Condition::App(b))
201        | (Condition::Cmdline(n), Condition::Cmdline(b))
202        | (Condition::Parent(n), Condition::Parent(b)) => all(n, b),
203        (Condition::Uid(n), Condition::Uid(b)) => n.iter().all(|u| b.contains(u)),
204        (Condition::Protocol(n), Condition::Protocol(b)) => n.iter().all(|p| b.contains(p)),
205        (Condition::Port(n), Condition::Port(b)) => n
206            .iter()
207            .all(|(low, high)| b.iter().any(|(l, h)| l <= low && high <= h)),
208        (Condition::Host(n), Condition::Host(b)) => {
209            n.iter().all(|n| b.iter().any(|b| host_covers(b, n)))
210        }
211        (Condition::Net(n), Condition::Net(b)) => {
212            n.iter().all(|n| b.iter().any(|b| net_covers(b, n)))
213        }
214        (Condition::Context(n), Condition::Context(b)) => n == b || *b == ContextKind::Any,
215        (n @ Condition::Not(_), b @ Condition::Not(_)) => n == b,
216        _ => false,
217    }
218}

220fn pattern_covers(broad: &PatternSource, narrow: &PatternSource) -> bool {
221    if broad == narrow {
222        return true;
223    }
224    match narrow {
225        // An exact value inside a pattern; with `{home}` on both sides or neither.
226        PatternSource::Exact(value) => {
227            let broad_home = match broad {
228                PatternSource::Exact(t) | PatternSource::Glob(t) | PatternSource::Regex(t) => {
229                    t.starts_with(HOME)
230                }
231            };
232            match value.strip_prefix(HOME) {
233                Some(rest) if broad_home => Pattern::compile(broad)
234                    .is_ok_and(|p| p.matches(&format!("/h{rest}"), Some("/h"))),
235                None if !broad_home => {
236                    Pattern::compile(broad).is_ok_and(|p| p.matches(value, None))
237                }
238                _ => false,
239            }
240        }
241        PatternSource::Glob(_) | PatternSource::Regex(_) => false,
242    }
243}
244
245fn host_covers(broad: &HostPattern, narrow: &HostPattern) -> bool {
246    match narrow {
247        _ if broad == narrow => true,
248        HostPattern::Exact(name) => broad.matches(name),
249        HostPattern::Subdomains(suffix) => match broad {
250            HostPattern::Subdomains(wider) => {
251                suffix.len() >= wider.len()
252                    && suffix
253                        .get(suffix.len() - wider.len()..)
254                        .is_some_and(|tail| tail.eq_ignore_ascii_case(wider))
255            }
256            HostPattern::Exact(_) | HostPattern::Regex(_) => false,
257        },
258        HostPattern::Regex(_) => false,
259    }
260}
261
262fn net_covers(broad: &Network, narrow: &Network) -> bool {
263    match (broad, narrow) {
264        (Network::Net(b), Network::Net(n)) => b.contains(n),
265        _ => broad == narrow,
266    }
267}

269impl Kind {
270    /// A short explanation, for people.
271    #[must_use]
272    pub fn describe(self) -> &'static str {
273        match self {
274            Self::Duplicate => "duplicates another rule",
275            Self::Redundant => {
276                "another rule already matches everything it does, with the same action"
277            }
278            Self::Shadowed => "a deny or reject rule always beats it",
279            Self::Expired => "its time is up",
280            Self::NeverHit => "decided no connection in the days looked at",
281            Self::WritableBinary => "allows a binary that a user other than root can replace",
282            Self::MissingBinary => "its binary is gone",
283        }
284    }
285}

294    #[derive(Default)]
295    struct Facts {
296        files: Vec<&'static str>,
297        writable: Vec<&'static str>,
298        hits: HashMap<RuleId, u64>,
299    }
300
301    impl NodeFacts for Facts {
302        fn exists(&self, path: &str) -> bool {
303            self.files.contains(&path)
304        }
305        fn writable_by_user(&self, path: &str) -> bool {
306            self.writable.contains(&path)
307        }
308        fn exists_like(&self, glob: &str) -> bool {
309            let pattern = Pattern::compile(&PatternSource::Glob(glob.to_owned())).unwrap();
310            self.files.iter().any(|f| pattern.matches(f, None))
311        }
312        fn last_hit_ms(&self, rule: RuleId) -> Option<u64> {
313            self.hits.get(&rule).copied()
314        }
315        fn changed_ms(&self, _: RuleId) -> Option<u64> {
316            Some(0)
317        }
318    }

320    fn rule(id: RuleId, action: Action, conditions: Vec<Condition>) -> Rule {
321        Rule {
322            id,
323            folder: None,
324            name: format!("r{id}"),
325            enabled: true,
326            action,
327            tier: Tier::Normal,
328            expiry: Expiry::Never,
329            log: true,
330            notes: String::new(),
331            conditions,
332        }
333    }
334
335    fn exe(path: &str) -> Condition {
336        Condition::Exe(vec![PatternSource::Exact(path.to_owned())])
337    }
338
339    fn host(pattern: &str) -> Condition {
340        Condition::Host(vec![HostPattern::try_from(pattern.to_owned()).unwrap()])
341    }
342
343    fn report(rules: &[Rule], facts: &Facts) -> Vec<(Kind, RuleId, Option<RuleId>)> {
344        let linted: Vec<Linted<'_>> = rules
345            .iter()
346            .map(|r| Linted {
347                rule: r,
348                conditions: r.conditions.iter().collect(),
349                effective: r.enabled,
350                locked: false,
351            })
352            .collect();
353        lint(&linted, facts, 10 * 86_400_000, 0)
354            .into_iter()
355            .map(|f| (f.kind, f.rule, f.other))
356            .collect()
357    }

359    #[test]
360    fn duplicates_redundant_and_shadowed_rules_are_found() {
361        let curl = "/usr/bin/curl";
362        let facts = Facts {
363            files: vec![curl],
364            ..Facts::default()
365        };
366        let rules = vec![
367            rule(1, Action::Allow, vec![exe(curl), host("*.example.com")]),
368            // The same as 1.
369            rule(2, Action::Allow, vec![host("*.example.com"), exe(curl)]),
370            // Inside 1: a subdomain of example.com.
371            rule(
372                3,
373                Action::Allow,
374                vec![
375                    exe(curl),
376                    host("api.example.com"),
377                    Condition::Port(vec![(443, 443)]),
378                ],
379            ),
380            // Not inside 1: another program.
381            rule(
382                4,
383                Action::Allow,
384                vec![exe("/usr/bin/wget"), host("api.example.com")],
385            ),
386            // A deny on all of example.com for curl beats 5 at the same tier.
387            rule(5, Action::Deny, vec![exe(curl), host("*.example.org")]),
388            rule(6, Action::Allow, vec![exe(curl), host("www.example.org")]),
389        ];
390        assert_eq!(
391            report(&rules, &facts),
392            vec![
393                (Kind::Duplicate, 2, Some(1)),
394                (Kind::Redundant, 3, Some(1)),
395                (Kind::MissingBinary, 4, None),
396                (Kind::Shadowed, 6, Some(5)),
397            ]
398        );
399    }

414    #[test]
415    fn coverage_is_decided_on_the_conditions() {
416        fn c(conditions: &[Condition]) -> Vec<&Condition> {
417            conditions.iter().collect()
418        }
419        let glob = Condition::Exe(vec![PatternSource::Glob("/opt/app/*/bin".to_owned())]);
420        assert!(covers(
421            &c(std::slice::from_ref(&glob)),
422            &c(&[exe("/opt/app/1.2/bin")])
423        ));
424        assert!(!covers(
425            &c(&[exe("/opt/app/1.2/bin")]),
426            &c(std::slice::from_ref(&glob))
427        ));
428        // Globs aren't compared with each other: only identical ones count.
429        let wider = Condition::Exe(vec![PatternSource::Glob("/opt/**".to_owned())]);
430        assert!(!covers(&c(&[wider]), &c(&[glob])));
431        let net = |n: &str| Condition::Net(vec![Network::try_from(n.to_owned()).unwrap()]);
432        assert!(covers(&c(&[net("10.0.0.0/8")]), &c(&[net("10.1.0.0/16")])));
433        assert!(!covers(&c(&[net("10.1.0.0/16")]), &c(&[net("10.0.0.0/8")])));
434        let ports = |r: Vec<(u16, u16)>| Condition::Port(r);
435        assert!(covers(
436            &c(&[ports(vec![(1, 1024)])]),
437            &c(&[ports(vec![(80, 80), (443, 443)])])
438        ));
439        // No context means the host; `any` covers every context.
440        let any = Condition::Context(ContextKind::Any);
441        let container = Condition::Context(ContextKind::Container);
442        assert!(covers(&c(std::slice::from_ref(&any)), &c(&[])));
443        assert!(!covers(&c(&[]), &c(std::slice::from_ref(&container))));
444        assert!(covers(&c(&[any]), &c(&[container])));
445        // The empty rule matches everything on the host.
446        assert!(covers(&c(&[]), &c(&[exe("/x")])));
447        let home = Condition::Exe(vec![PatternSource::Glob("{home}/.local/bin/*".to_owned())]);
448        let homed = Condition::Exe(vec![PatternSource::Exact(
449            "{home}/.local/bin/tool".to_owned(),
450        )]);
451        assert!(covers(&c(std::slice::from_ref(&home)), &c(&[homed])));
452        assert!(!covers(
453            &c(&[home]),
454            &c(&[exe("/home/ada/.local/bin/tool")])
455        ));
456    }

458    #[test]
459    fn what_the_node_knows_is_reported() {
460        let old = "/home/ada/.local/share/claude/versions/2.0.31/claude";
461        let new = "/home/ada/.local/share/claude/versions/2.0.40/claude";
462        let facts = Facts {
463            files: vec![new, "/home/ada/bin/tool"],
464            writable: vec!["/home/ada/bin/tool"],
465            hits: HashMap::from([(3, 9 * 86_400_000)]),
466        };
467        let mut expired = rule(4, Action::Allow, vec![host("x.org")]);
468        expired.expiry = Expiry::At(1);
469        let rules = [
470            rule(1, Action::Allow, vec![exe(old)]),
471            rule(2, Action::Allow, vec![exe("/home/ada/bin/tool")]),
472            rule(3, Action::Deny, vec![host("y.org")]),
473            expired,
474        ];
475        let linted: Vec<Linted<'_>> = rules
476            .iter()
477            .map(|r| Linted {
478                rule: r,
479                conditions: r.conditions.iter().collect(),
480                effective: true,
481                locked: false,
482            })
483            .collect();
484        let found = lint(&linted, &facts, 10 * 86_400_000, 7);
485        let kinds: Vec<(Kind, RuleId)> = found.iter().map(|f| (f.kind, f.rule)).collect();
486        assert_eq!(
487            kinds,
488            vec![
489                (Kind::NeverHit, 1),
490                (Kind::MissingBinary, 1),
491                (Kind::NeverHit, 2),
492                (Kind::WritableBinary, 2),
493                (Kind::Expired, 4),
494            ]
495        );
496        assert_eq!(
497            found.get(1).and_then(|f| f.suggestion.clone()),
498            Some(PatternSource::Glob(
499                "/home/ada/.local/share/claude/versions/*/claude".to_owned()
500            ))
501        );
502    }
