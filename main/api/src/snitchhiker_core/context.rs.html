1//! Who a connection's sender is beyond the host (plan §5.10): a container's identity, a VM's, a network namespace's
2//! name. Known exactly or not at all: a field that couldn't be read is empty, and conditions on it don't match.
3
4/// A container, as rules see it.
5#[derive(Debug, Clone, Default, PartialEq, Eq, Hash)]
6pub struct Container {
7    /// `docker`, `podman`, `containerd`, `cri-o`, `nspawn`, `lxc`.
8    pub runtime: String,
9    pub id: String,
10    pub name: String,
11    /// The image reference it was created from (`docker.io/library/node:22`).
12    pub image: String,
13    /// The Compose project and service: `project/service`.
14    pub compose: Option<String>,
15    pub labels: Vec<(String, String)>,
16    /// For a rootless runtime: the user it runs as.
17    pub owner: Option<u32>,
18    /// A process of it holds `CAP_NET_ADMIN` in its namespace: it can change its own network (and remove the table
19    /// the daemon keeps there, which is put back). Not part of the identity: [`Container::key`] leaves it out.
20    pub net_admin: bool,
21}
22
23impl Container {
24    /// The identity as one line, as the history keeps it: runtime, id, name, image, Compose service and owner,
25    /// tab-separated (none of them holds a tab). Labels are for rules only and aren't kept.
26    #[must_use]
27    pub fn key(&self) -> String {
28        format!(
29            "{}\t{}\t{}\t{}\t{}\t{}",
30            self.runtime,
31            self.id,
32            self.name,
33            self.image,
34            self.compose.as_deref().unwrap_or_default(),
35            self.owner.map_or_else(String::new, |uid| uid.to_string())
36        )
37    }
38
39    /// The identity a [`Container::key`] wrote.
40    #[must_use]
41    pub fn from_key(key: &str) -> Self {
42        let mut parts = key.split('\t');
43        let mut next = || parts.next().unwrap_or_default().to_owned();
44        let (runtime, id, name, image, compose, owner) =
45            (next(), next(), next(), next(), next(), next());
46        Self {
47            runtime,
48            id,
49            name,
50            image,
51            compose: (!compose.is_empty()).then_some(compose),
52            labels: Vec::new(),
53            owner: owner.parse().ok(),
54            net_admin: false,
55        }
56    }
57
58    /// How people know it: the Compose service, or the name, with the image.
59    #[must_use]
60    pub fn label(&self) -> String {
61        let name = self.compose.as_deref().unwrap_or(&self.name);
62        if self.image.is_empty() {
63            name.to_owned()
64        } else {
65            format!("{name} ({})", self.image)
66        }
67    }
68}

70/// A virtual machine, as rules see it.
71#[derive(Debug, Clone, Default, PartialEq, Eq, Hash)]
72pub struct Vm {
73    /// `qemu`, `firecracker`, ...
74    pub hypervisor: String,
75    pub name: String,
76    pub uuid: String,
77}

79#[cfg(test)]
80mod tests {
81    use super::*;
82
83    #[test]
84    fn keys_round_trip_without_labels() {
85        let container = Container {
86            runtime: "podman".to_owned(),
87            id: "abc".to_owned(),
88            name: "db".to_owned(),
89            image: "docker.io/library/postgres:17".to_owned(),
90            compose: Some("shop/db".to_owned()),
91            labels: vec![("a".to_owned(), "b".to_owned())],
92            owner: Some(1001),
93            net_admin: false,
94        };
95        assert_eq!(
96            Container::from_key(&container.key()),
97            Container {
98                labels: Vec::new(),
99                ..container
100            }
101        );
102        assert_eq!(Container::from_key("docker\tid\tname\t\t\t").compose, None);
103    }
