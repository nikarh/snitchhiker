1//! Turning an executable's path into a pattern that survives updates: path segments that look like versions or
2//! content hashes become `*`, and the user's home becomes `{home}`.
3//!
4//! `/home/ada/.local/share/claude/versions/2.0.31/claude` becomes `{home}/.local/share/claude/versions/*/claude`;
5//! `/nix/store/0c7…-firefox-133.0/bin/firefox` becomes `/nix/store/*-firefox-*/bin/firefox`. A path with nothing to
6//! generalise stays exact.
7
8use crate::pattern::{HOME, PatternSource};
9
10/// What changes from one version of a program to the next, within a path segment.
11fn is_version(text: &str) -> bool {
12    // `v12`: a v and digits.
13    if let Some(digits) = text.strip_prefix('v')
14        && !digits.is_empty()
15        && digits.bytes().all(|b| b.is_ascii_digit())
16    {
17        return true;
18    }
19    let text = text.strip_prefix('v').unwrap_or(text);
20    let mut parts = text.split(['.', '-', '_', '+']);
21    let first = parts.next().unwrap_or_default();
22    !first.is_empty()
23        && first.bytes().all(|b| b.is_ascii_digit())
24        && text.contains(['.', '-', '_', '+']).then_some(()).is_some()
25        && parts.all(|part| !part.is_empty() && part.bytes().all(|b| b.is_ascii_alphanumeric()))
26}
27
28/// A content hash: long, hexadecimal or Nix's base32, with digits and letters mixed.
29fn is_hash(text: &str) -> bool {
30    text.len() >= 12
31        && text
32            .bytes()
33            .all(|b| b.is_ascii_digit() || b.is_ascii_lowercase())
34        && text.bytes().any(|b| b.is_ascii_digit())
35        && text.bytes().any(|b| b.is_ascii_lowercase())
36}
37
38/// One path segment, generalised: versions and hashes, alone or as `name-1.2.3` / `hash-name` parts.
39fn segment(segment: &str) -> String {
40    if is_version(segment) || is_hash(segment) {
41        return "*".to_owned();
42    }
43    // AppImages are mounted at /tmp/.mount_<up to 6 characters of the name><6 random characters>.
44    if let Some(name) = segment.strip_prefix(".mount_")
45        && name.len() > 6
46        && let Some(stem) = name.get(..name.len() - 6)
47    {
48        return format!(".mount_{stem}*");
49    }
50    // Dash-separated parts (`firefox-133.0`, `0c7…-firefox-133.0`).
51    let parts: Vec<&str> = segment.split('-').collect();
52    if parts.len() > 1 {
53        let mut changed = false;
54        let mut out: Vec<&str> = Vec::with_capacity(parts.len());
55        let mut index = 0;
56        while let Some(part) = parts.get(index) {
57            // A version may itself contain dashes (`1.2.3-beta`): the rest of the segment from here.
58            let rest = parts.get(index..).unwrap_or_default().join("-");
59            if index > 0 && is_version(&rest) {
60                out.push("*");
61                changed = true;
62                break;
63            }
64            if is_version(part) || is_hash(part) {
65                out.push("*");
66                changed = true;
67            } else {
68                out.push(part);
69            }
70            index += 1;
71        }
72        if changed {
73            return out.join("-");
74        }
75    }
76    segment.to_owned()
77}

79/// The pattern for an executable run by a user whose home is `home`.
80#[must_use]
81pub fn executable(path: &str, home: Option<&str>) -> PatternSource {
82    let (prefix, rest) = match home.map(|h| h.trim_end_matches('/')) {
83        Some(home)
84            if !home.is_empty()
85                && path.starts_with(home)
86                && path.get(home.len()..).is_some_and(|r| r.starts_with('/')) =>
87        {
88            (HOME, path.get(home.len()..).unwrap_or_default())
89        }
90        _ => ("", path),
91    };
92    let generalised: Vec<String> = rest.split('/').map(segment).collect();
93    let pattern = format!("{prefix}{}", generalised.join("/"));
94    if pattern.contains('*') {
95        PatternSource::Glob(pattern)
96    } else {
97        PatternSource::Exact(pattern)
98    }
99}

101#[cfg(test)]
102mod tests {
103    use super::*;
104
105    fn glob(text: &str) -> PatternSource {
106        PatternSource::Glob(text.to_owned())
107    }
108
109    #[test]
110    fn version_directories_become_wildcards() {
111        assert_eq!(
112            executable(
113                "/home/ada/.local/share/claude/versions/2.0.31/claude",
114                Some("/home/ada")
115            ),
116            glob("{home}/.local/share/claude/versions/*/claude")
117        );
118        assert_eq!(
119            executable("/opt/app/v12/bin/app", None),
120            glob("/opt/app/*/bin/app")
121        );
122    }
123
124    #[test]
125    fn versions_inside_names_become_wildcards() {
126        assert_eq!(
127            executable(
128                "/home/ada/.local/share/JetBrains/Toolbox/apps/idea/2026.2.1/bin/idea",
129                Some("/home/ada")
130            ),
131            glob("{home}/.local/share/JetBrains/Toolbox/apps/idea/*/bin/idea")
132        );
133        assert_eq!(
134            executable("/usr/lib/firefox-133.0b2/firefox", None),
135            glob("/usr/lib/firefox-*/firefox")
136        );
137    }
138
139    #[test]
140    fn nix_store_hashes_and_appimage_mounts() {
141        assert_eq!(
142            executable(
143                "/nix/store/0c7k2xzq9yrd1wv5mzh8cl3x9a0b1c2d-firefox-133.0/bin/firefox",
144                None
145            ),
146            glob("/nix/store/*-firefox-*/bin/firefox")
147        );
148        assert_eq!(
149            executable("/tmp/.mount_ObsidiXa8kTq/obsidian", None),
150            glob("/tmp/.mount_Obsidi*/obsidian")
151        );
152    }
153
154    #[test]
155    fn stable_paths_stay_exact() {
156        assert_eq!(
157            executable("/usr/bin/python3.13", None),
158            PatternSource::Exact("/usr/bin/python3.13".to_owned())
159        );
160        assert_eq!(
161            executable("/usr/bin/curl", Some("/home/ada")),
162            PatternSource::Exact("/usr/bin/curl".to_owned())
163        );
164        assert_eq!(
165            executable("/home/ada/bin/tool", Some("/home/ada")),
166            PatternSource::Exact("{home}/bin/tool".to_owned())
167        );
168    }
169
170    #[test]
171    fn a_home_prefix_must_end_at_a_directory() {
172        assert_eq!(
173            executable("/home/adams/bin/x", Some("/home/ada")),
174            PatternSource::Exact("/home/adams/bin/x".to_owned())
175        );
176    }
