1//! Deciding a connection: rules compiled into an immutable, indexed set, evaluated without allocating.
2//!
3//! Precedence is fixed and explainable:
4//! 1. the highest tier with any matching rule decides (`override`, then `normal`, then `fallback`);
5//! 2. within it, a blocking rule (deny, reject) beats an allowing one;
6//! 3. among those, the most specific rule is the one reported (ties: the lowest id).
7//!
8//! A rule applies only while it and every folder above it are enabled and it hasn't expired. Folder scopes are added
9//! to every rule below them. A rule that says nothing about the context applies to the host only, so allowing
10//! `/usr/bin/curl` doesn't allow `curl` in every container.
11

25/// What is known about a connection when it is decided. Everything is borrowed: building one allocates nothing.
26#[derive(Debug, Clone, Copy)]
27pub struct Facts<'a> {
28    pub exe: &'a str,
29    /// Stable identities of the program, if any are known (`pkg:firefox`, `unit:sshd.service`).
30    pub apps: &'a [String],
31    pub cmdline: &'a str,
32    /// Executables of the ancestor processes, nearest first.
33    pub ancestors: &'a [String],
34    /// The process and its ancestors, nearest first, for `session` conditions.
35    pub lineage: &'a [ProcessRef],
36    pub uid: u32,
37    /// The user's home directory, for `{home}` patterns.
38    pub home: Option<&'a str>,
39    /// Names known for the destination address (as resolved by this process first).
40    pub hosts: &'a [std::sync::Arc<str>],
41    pub address: IpAddr,
42    pub port: u16,
43    pub protocol: u8,
44    pub context: ContextKind,
45    /// The container's identity, for a container's connection (when it could be read).
46    pub container: Option<&'a crate::context::Container>,
47    pub vm: Option<&'a crate::context::Vm>,
48    /// The network namespace's name, for one with a name.
49    pub netns: Option<&'a str>,
50    /// Unix time in milliseconds, for expiring rules.
51    pub now_ms: u64,
52    /// The executable's SHA-256 (lowercase hex), when a rule asks for it ([`RuleSet::needs_sha256`]).
53    pub sha256: Option<&'a str>,
54    /// Whether a user other than root can change the executable.
55    pub writable: bool,
56}

117/// A host pattern, ready to match: names and suffixes as they are, regular expressions compiled (case-insensitive,
118/// over the whole name).
119#[derive(Debug, Clone)]
120enum HostMatcher {
121    Pattern(HostPattern),
122    Regex(Pattern),
123}
124
125impl HostMatcher {
126    fn compile(pattern: &HostPattern) -> Result<Self, PatternError> {
127        Ok(match pattern {
128            HostPattern::Regex(regex) => Self::Regex(Pattern::compile(
129                &crate::pattern::PatternSource::Regex(format!("(?i){regex}")),
130            )?),
131            other => Self::Pattern(other.clone()),
132        })
133    }
134
135    fn matches(&self, host: &str) -> bool {
136        match self {
137            Self::Pattern(pattern) => pattern.matches(host),
138            Self::Regex(regex) => regex.matches(host.strip_suffix('.').unwrap_or(host), None),
139        }
140    }
141
142    fn specificity(&self) -> u32 {
143        match self {
144            Self::Pattern(pattern) => pattern.specificity(),
145            Self::Regex(_) => 0,
146        }
147    }
148}

160/// The rules that apply, compiled and indexed. Built once per change and shared read-only by the deciding threads.
161///
162/// Each rule is filed under its most selective key: the host names it is limited to (lowercase; `*.example.com`
163/// under `example.com`), else the exact programs, else the ports; the rest are checked for every connection. A
164/// connection looks only at the rules under its own names (and the domains above them), its program and its port.
165#[derive(Debug, Clone, Default)]
166pub struct RuleSet {
167    rules: Vec<CompiledRule>,
168    by_host: HashMap<Box<str>, Vec<u32>>,
169    /// Every rule in `by_host`, for a name with capitals (looked up without allocating a lowercase copy).
170    host_indexed: Vec<u32>,
171    by_exe: HashMap<Box<str>, Vec<u32>>,
172    /// Rules whose conditions name destination ports, by port.
173    by_port: HashMap<u16, Vec<u32>>,
174    /// Rules that apply to any connection, as far as the index can tell.
175    rest: Vec<u32>,
176    /// Whether a rule names a SHA-256: only then are executables hashed.
177    sha256: bool,
178}
179
180/// Where a rule is filed.
181enum Key {
182    Hosts(Vec<Box<str>>),
183    Exes(Vec<Box<str>>),
184    Ports(Vec<u16>),
185    Rest,
186}
187
188/// A rule's most selective key, from its positive conditions.
189fn key(conditions: &[Compiled]) -> Key {
190    let hosts = conditions.iter().find_map(|c| match c {
191        Compiled::Host(matchers) if !matchers.is_empty() => matchers
192            .iter()
193            .map(|m| match m {
194                HostMatcher::Pattern(HostPattern::Exact(name)) => Some(
195                    name.trim_end_matches('.')
196                        .to_ascii_lowercase()
197                        .into_boxed_str(),
198                ),
199                HostMatcher::Pattern(HostPattern::Subdomains(suffix)) => Some(
200                    suffix
201                        .trim_start_matches('.')
202                        .trim_end_matches('.')
203                        .to_ascii_lowercase()
204                        .into_boxed_str(),
205                ),
206                _ => None,
207            })
208            .collect::<Option<Vec<_>>>(),
209        _ => None,
210    });
211    if let Some(hosts) = hosts {
212        return Key::Hosts(hosts);
213    }
214    let exes = conditions.iter().find_map(|c| match c {
215        Compiled::Exe(patterns) if !patterns.is_empty() => patterns
216            .iter()
217            .map(|p| p.exact().map(Box::from))
218            .collect::<Option<Vec<_>>>(),
219        _ => None,
220    });
221    if let Some(exes) = exes {
222        return Key::Exes(exes);
223    }
224    indexed_ports(conditions).map_or(Key::Rest, Key::Ports)
225}

227/// A port condition with at most this many ports is indexed by port; wider ones are checked for every connection.
228const MAX_INDEXED_PORTS: usize = 64;

230impl RuleSet {
231    /// Compiles the rules that apply: enabled, in enabled folders, with their folders' scopes.
232    pub fn compile(folders: &[Folder], rules: &[Rule]) -> Result<Self, CompileError> {
233        Self::compile_with(folders, rules, &Lists::new())
234    }
235
236    /// [`RuleSet::compile`], with the blocklists `list` conditions name.
237    pub fn compile_with(
238        folders: &[Folder],
239        rules: &[Rule],
240        lists: &Lists,
241    ) -> Result<Self, CompileError> {
242        let folder_by_id: HashMap<FolderId, &Folder> = folders.iter().map(|f| (f.id, f)).collect();
243        let mut set = Self::default();
244        for rule in rules {
245            if !rule.enabled {
246                continue;
247            }
248            let Some(scopes) = folder_chain(rule.folder, &folder_by_id)? else {
249                continue;
250            };
251            let mut conditions = Vec::new();
252            let mut has_context = false;
253            for condition in scopes
254                .iter()
255                .flat_map(|scope| scope.iter())
256                .chain(&rule.conditions)
257            {
258                if let Some(implied) = crate::rule::implied_context(condition) {
259                    if !matches!(condition, Condition::Context(_)) {
260                        conditions.push(Compiled::Context(implied));
261                    }
262                    has_context = true;
263                }
264                conditions.push(compile_condition(condition, lists).map_err(|error| {
265                    CompileError::Pattern {
266                        rule: rule.id,
267                        error,
268                    }
269                })?);
270            }
271            if !has_context {
272                conditions.push(Compiled::Context(ContextKind::Host));
273            }
274            set.sha256 |= conditions.iter().any(names_sha256);
275            let specificity = conditions.iter().map(specificity).sum();
276            let index = u32::try_from(set.rules.len()).unwrap_or(u32::MAX);
277            match key(&conditions) {
278                Key::Hosts(hosts) => {
279                    for host in hosts {
280                        set.by_host.entry(host).or_default().push(index);
281                    }
282                    set.host_indexed.push(index);
283                }
284                Key::Exes(exes) => {
285                    for exe in exes {
286                        set.by_exe.entry(exe).or_default().push(index);
287                    }
288                }
289                Key::Ports(ports) => {
290                    for port in ports {
291                        set.by_port.entry(port).or_default().push(index);
292                    }
293                }
294                Key::Rest => set.rest.push(index),
295            }
296            set.rules.push(CompiledRule {
297                id: rule.id,
298                action: rule.action,
299                tier: rule.tier,
300                expires_ms: match rule.expiry {
301                    Expiry::At(at) => Some(at),
302                    Expiry::Never | Expiry::UntilRestart => None,
303                },
304                specificity,
305                conditions,
306            });
307        }
308        Ok(set)
309    }

327    /// The rules that may match: those filed under the connection's names and the domains above them, its program
328    /// and its port, and the rest. A rule may come more than once (under two of the names).
329    fn candidates<'s>(&'s self, facts: &'s Facts<'_>) -> impl Iterator<Item = &'s CompiledRule> {
330        let bucket = |list: Option<&'s Vec<u32>>| list.map_or(&[][..], Vec::as_slice).iter();
331        let by_host = facts.hosts.iter().flat_map(move |host| {
332            let host: &str = host.strip_suffix('.').unwrap_or(host);
333            let lowercase = !host.bytes().any(|b| b.is_ascii_uppercase());
334            // A name with capitals: every rule filed under a name (rare: resolvers answer in lowercase).
335            let all = if lowercase {
336                &[][..]
337            } else {
338                self.host_indexed.as_slice()
339            };
340            let names = std::iter::successors(lowercase.then_some(host), |name| {
341                name.split_once('.')
342                    .map(|(_, parent)| parent)
343                    .filter(|p| !p.is_empty())
344            });
345            all.iter()
346                .chain(names.flat_map(move |name| bucket(self.by_host.get(name))))
347        });
348        by_host
349            .chain(bucket(self.by_exe.get(facts.exe)))
350            .chain(bucket(self.by_port.get(&facts.port)))
351            .chain(&self.rest)
352            .filter_map(|&index| self.rules.get(index as usize))
353    }

355    /// The decision for a connection, or `None` when no rule matches (the policy then decides: ask, allow, deny).
356    #[must_use]
357    pub fn decide(&self, facts: &Facts<'_>) -> Option<Decision> {
358        let mut best: Option<&CompiledRule> = None;
359        for rule in self.candidates(facts) {
360            if !rule.matches(facts) {
361                continue;
362            }
363            if best.is_none_or(|best| outranks(rule, best)) {
364                best = Some(rule);
365            }
366        }
367        best.map(|rule| Decision {
368            rule: rule.id,
369            action: rule.action,
370            tier: rule.tier,
371        })
372    }

374    /// Every rule that matches, in precedence order: the first is the one [`RuleSet::decide`] picks.
375    #[must_use]
376    pub fn explain(&self, facts: &Facts<'_>) -> Vec<Match> {
377        let mut matched: Vec<&CompiledRule> = self
378            .candidates(facts)
379            .filter(|r| r.matches(facts))
380            .collect();
381        // A rule filed under two of the connection's names comes twice.
382        matched.sort_by_key(|r| std::ptr::from_ref(*r));
383        matched.dedup_by_key(|r| std::ptr::from_ref(*r));
384        matched.sort_by(|a, b| {
385            if outranks(a, b) {
386                std::cmp::Ordering::Less
387            } else if outranks(b, a) {
388                std::cmp::Ordering::Greater
389            } else {
390                std::cmp::Ordering::Equal
391            }
392        });
393        matched
394            .into_iter()
395            .map(|rule| Match {
396                rule: rule.id,
397                action: rule.action,
398                tier: rule.tier,
399                specificity: rule.specificity,
400            })
401            .collect()
402    }
403}

405/// Whether `a` takes precedence over `b`.
406fn outranks(a: &CompiledRule, b: &CompiledRule) -> bool {
407    (
408        a.tier,
409        a.action.blocks(),
410        a.specificity,
411        std::cmp::Reverse(a.id),
412    ) > (
413        b.tier,
414        b.action.blocks(),
415        b.specificity,
416        std::cmp::Reverse(b.id),
417    )
418}

420impl CompiledRule {
421    fn matches(&self, facts: &Facts<'_>) -> bool {
422        self.expires_ms.is_none_or(|at| facts.now_ms < at)
423            && self.conditions.iter().all(|c| condition_matches(c, facts))
424    }
425}

427fn condition_matches(condition: &Compiled, facts: &Facts<'_>) -> bool {
428    let any =
429        |patterns: &[Pattern], value: &str| patterns.iter().any(|p| p.matches(value, facts.home));
430    match condition {
431        Compiled::Exe(patterns) => any(patterns, facts.exe),
432        Compiled::App(patterns) => facts.apps.iter().any(|app| any(patterns, app.as_str())),
433        Compiled::Cmdline(patterns) => any(patterns, facts.cmdline),
434        Compiled::Parent(patterns) => facts
435            .ancestors
436            .iter()
437            .any(|exe| any(patterns, exe.as_str())),
438        Compiled::Uid(uids) => uids.contains(&facts.uid),
439        Compiled::Host(patterns) => facts
440            .hosts
441            .iter()
442            .any(|host| patterns.iter().any(|p| p.matches(host))),
443        Compiled::Net(nets) => nets.iter().any(|net| net.contains(&facts.address)),
444        Compiled::Port(ranges) => ranges
445            .iter()
446            .any(|&(low, high)| (low..=high).contains(&facts.port)),
447        Compiled::Protocol(protocols) => protocols.contains(&facts.protocol),
448        Compiled::Context(ContextKind::Any) => true,
449        Compiled::Context(kind) => *kind == facts.context,
450        Compiled::Session(roots) => facts.lineage.iter().any(|p| roots.contains(p)),
451        Compiled::ContainerName(patterns) => {
452            facts.container.is_some_and(|c| any(patterns, &c.name))
453        }
454        Compiled::ContainerImage(patterns) => {
455            facts.container.is_some_and(|c| any(patterns, &c.image))
456        }
457        Compiled::ContainerCompose(patterns) => facts
458            .container
459            .and_then(|c| c.compose.as_deref())
460            .is_some_and(|compose| any(patterns, compose)),
461        Compiled::ContainerLabel(labels) => facts.container.is_some_and(|c| {
462            labels.iter().any(|(key, value)| {
463                c.labels
464                    .iter()
465                    .any(|(k, v)| k == key && value.matches(v, None))
466            })
467        }),
468        Compiled::ContainerRuntime(runtimes) => facts
469            .container
470            .is_some_and(|c| runtimes.iter().any(|r| r == &c.runtime)),
471        Compiled::ContainerOwner(uids) => facts
472            .container
473            .and_then(|c| c.owner)
474            .is_some_and(|owner| uids.contains(&owner)),
475        // Ids as `docker ps` shortens them match too.
476        Compiled::ContainerId(ids) => facts.container.is_some_and(|c| {
477            !c.id.is_empty() && ids.iter().any(|id| c.id.starts_with(id.as_str()))
478        }),
479        Compiled::VmName(patterns) => facts.vm.is_some_and(|vm| any(patterns, &vm.name)),
480        Compiled::VmUuid(uuids) => facts
481            .vm
482            .is_some_and(|vm| uuids.iter().any(|u| u.eq_ignore_ascii_case(&vm.uuid))),
483        Compiled::VmHypervisor(names) => facts
484            .vm
485            .is_some_and(|vm| names.iter().any(|n| n == &vm.hypervisor)),
486        Compiled::NetnsName(patterns) => facts.netns.is_some_and(|name| any(patterns, name)),
487        Compiled::Sha256(hashes) => facts
488            .sha256
489            .is_some_and(|hash| hashes.iter().any(|h| h.eq_ignore_ascii_case(hash))),
490        Compiled::Writable(writable) => facts.writable == *writable,
491        Compiled::List(lists) => lists.iter().any(|list| {
492            list.has_address(facts.address) || facts.hosts.iter().any(|h| list.has_host(h))
493        }),
494        Compiled::Not(inner) => !condition_matches(inner, facts),
495    }
496}

498/// The scopes of a rule's folders, outermost first; `None` when one of them is disabled.
499fn folder_chain<'a>(
500    mut folder: Option<FolderId>,
501    folders: &HashMap<FolderId, &'a Folder>,
502) -> Result<Option<Vec<&'a [Condition]>>, CompileError> {
503    let mut scopes = Vec::new();
504    let mut seen = Vec::new();
505    while let Some(id) = folder {
506        if seen.contains(&id) {
507            return Err(CompileError::FolderCycle { folder: id });
508        }
509        seen.push(id);
510        // A rule in a folder that no longer exists doesn't apply.
511        let Some(current) = folders.get(&id) else {
512            return Ok(None);
513        };
514        if !current.enabled {
515            return Ok(None);
516        }
517        scopes.push(current.scope.as_slice());
518        folder = current.parent;
519    }
520    scopes.reverse();
521    Ok(Some(scopes))
522}

532fn compile_condition(condition: &Condition, lists: &Lists) -> Result<Compiled, PatternError> {
533    let patterns =
534        |sources: &[crate::pattern::PatternSource]| -> Result<Vec<Pattern>, PatternError> {
535            sources.iter().map(Pattern::compile).collect()
536        };
537    Ok(match condition {
538        Condition::Exe(p) => Compiled::Exe(patterns(p)?),
539        Condition::App(p) => Compiled::App(patterns(p)?),
540        Condition::Cmdline(p) => Compiled::Cmdline(patterns(p)?),
541        Condition::Parent(p) => Compiled::Parent(patterns(p)?),
542        Condition::Uid(uids) => Compiled::Uid(uids.clone()),
543        Condition::Host(hosts) => Compiled::Host(
544            hosts
545                .iter()
546                .map(HostMatcher::compile)
547                .collect::<Result<_, _>>()?,
548        ),
549        Condition::Net(networks) => Compiled::Net(networks.iter().flat_map(expand).collect()),
550        Condition::Port(ranges) => Compiled::Port(ranges.clone()),
551        Condition::Protocol(protocols) => Compiled::Protocol(protocols.clone()),
552        Condition::Context(kind) => Compiled::Context(*kind),
553        Condition::Session(roots) => Compiled::Session(roots.clone()),
554        Condition::ContainerName(p) => Compiled::ContainerName(patterns(p)?),
555        Condition::ContainerImage(p) => Compiled::ContainerImage(patterns(p)?),
556        Condition::ContainerCompose(p) => Compiled::ContainerCompose(patterns(p)?),
557        Condition::ContainerLabel(labels) => Compiled::ContainerLabel(
558            labels
559                .iter()
560                .map(|(key, value)| Ok((key.clone(), Pattern::compile(value)?)))
561                .collect::<Result<_, PatternError>>()?,
562        ),
563        Condition::ContainerRuntime(runtimes) => Compiled::ContainerRuntime(runtimes.clone()),
564        Condition::ContainerOwner(uids) => Compiled::ContainerOwner(uids.clone()),
565        Condition::ContainerId(ids) => Compiled::ContainerId(ids.clone()),
566        Condition::VmName(p) => Compiled::VmName(patterns(p)?),
567        Condition::VmUuid(uuids) => Compiled::VmUuid(uuids.clone()),
568        Condition::VmHypervisor(names) => Compiled::VmHypervisor(names.clone()),
569        Condition::NetnsName(p) => Compiled::NetnsName(patterns(p)?),
570        Condition::Sha256(hashes) => Compiled::Sha256(hashes.clone()),
571        Condition::Writable(writable) => Compiled::Writable(*writable),
572        Condition::List(names) => Compiled::List(
573            names
574                .iter()
575                .filter_map(|name| lists.get(name.as_str()).cloned())
576                .collect(),
577        ),
578        Condition::Not(inner) => Compiled::Not(Box::new(compile_condition(inner, lists)?)),
579    })
580}

582fn expand(network: &Network) -> Vec<IpNet> {
583    match network {
584        Network::Net(net) => vec![*net],
585        Network::Alias(index) => ALIASES
586            .get(*index)
587            .into_iter()
588            .flat_map(|(_, nets)| nets.iter().filter_map(|net| net.parse().ok()))
589            .collect(),
590    }

593fn specificity(condition: &Compiled) -> u32 {
594    match condition {
595        Compiled::Exe(p) | Compiled::App(p) | Compiled::Cmdline(p) | Compiled::Parent(p) => {
596            p.iter().map(Pattern::specificity).min().unwrap_or(0) * 10
597        }
598        Compiled::Host(p) => p.iter().map(HostMatcher::specificity).min().unwrap_or(0),
599        Compiled::Net(nets) => nets
600            .iter()
601            .map(|n| u32::from(n.prefix_len()))
602            .min()
603            .unwrap_or(0),
604        Compiled::Port(_) | Compiled::Uid(_) => 5,
605        Compiled::Protocol(_) | Compiled::Writable(_) => 1,
606        // Who in the container ranks as a program does; the container itself, between the program and nothing.
607        Compiled::ContainerName(p)
608        | Compiled::ContainerImage(p)
609        | Compiled::ContainerCompose(p)
610        | Compiled::VmName(p)
611        | Compiled::NetnsName(p) => p.iter().map(Pattern::specificity).min().unwrap_or(0) * 5,
612        Compiled::ContainerId(_) | Compiled::VmUuid(_) => 50,
613        Compiled::ContainerLabel(_) => 20,
614        // Like a domain and everything under it.
615        Compiled::List(_) => 3,
616        // As exact as a program's path.
617        Compiled::Sha256(_) => 30,
618        Compiled::ContainerRuntime(_) | Compiled::ContainerOwner(_) | Compiled::VmHypervisor(_) => {
619            2
620        }
621        Compiled::Context(_) | Compiled::Session(_) | Compiled::Not(_) => 0,
622    }
623}

625/// The ports a rule is indexed under: those of a (positive) port condition naming few enough of them.
626fn indexed_ports(conditions: &[Compiled]) -> Option<Vec<u16>> {
627    conditions.iter().find_map(|condition| match condition {
628        Compiled::Port(ranges) => {
629            let count: usize = ranges
630                .iter()
631                .map(|&(low, high)| usize::from(high.saturating_sub(low)) + 1)
632                .sum();
633            (count <= MAX_INDEXED_PORTS)
634                .then(|| ranges.iter().flat_map(|&(low, high)| low..=high).collect())
635        }
636        _ => None,
637    })
638}

739        let mut rules = Vec::new();
740        let mut id = 0;
741        let mut next = |action, conditions| {
742            id += 1;
743            rule(id, action, conditions)
744        };
745        for i in 0..40_u16 {
746            let action = if i % 3 == 0 {
747                Action::Deny
748            } else {
749                Action::Allow
750            };
751            rules.push(next(
752                action,
753                vec![host(&format!("*.s{}.example.com", i % 7))],
754            ));
755            rules.push(next(
756                action,
757                vec![
758                    host(&format!("api.s{}.example.com", i % 5)),
759                    exe("/usr/bin/curl"),
760                ],
761            ));
762            rules.push(next(action, vec![exe(&format!("/usr/bin/app{}", i % 6))]));
763            rules.push(next(
764                action,
765                vec![Condition::Port(vec![(8000 + i % 4, 8000 + i % 4)])],
766            ));
767            rules.push(next(action, vec![host(r"re:.*\.s3\.example\.com")]));
768            rules.push(next(
769                action,
770                vec![Condition::Not(Box::new(host("x.example.com")))],
771            ));
772            rules.push(next(
773                action,
774                vec![Condition::Net(vec![
775                    Network::try_from("lan".to_owned()).unwrap(),
776                ])],
777            ));
778        }
779        let set = RuleSet::compile(&[], &rules).unwrap();
780        let everything = |facts: &Facts<'_>| {
781            let mut best: Option<&CompiledRule> = None;
782            for rule in set.rules.iter().filter(|r| r.matches(facts)) {
783                if best.is_none_or(|b| outranks(rule, b)) {
784                    best = Some(rule);
785                }
786            }
787            best.map(|r| (r.id, r.action))
788        };
789        let names = [
790            "api.s1.example.com",
791            "API.S2.Example.com",
792            "s3.example.com",
793            "deep.a.s4.example.com.",
794            "x.example.com",
795            "example.com",
796            "elsewhere.org",
797        ];
798        for exe_path in ["/usr/bin/curl", "/usr/bin/app2", "/usr/bin/other"] {
799            for name in names {
800                for port in [443, 8001, 8003] {
801                    let f = facts(exe_path, &[name], port);
802                    assert_eq!(
803                        set.decide(&f).map(|d| (d.rule, d.action)),
804                        everything(&f),
805                        "{exe_path} {name} {port}"
806                    );
807                    assert_eq!(
808                        set.explain(&f).first().map(|m| (m.rule, m.action)),
809                        everything(&f)
810                    );
811                }
812            }
813            let f = facts(exe_path, &[], 8002);
814            assert_eq!(set.decide(&f).map(|d| (d.rule, d.action)), everything(&f));
815        }
816    }
