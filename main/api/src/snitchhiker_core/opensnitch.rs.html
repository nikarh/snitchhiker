1//! Importing OpenSnitch's rules (`/etc/opensnitchd/rules/*.json`, its documented format) as a rule file.
2//!
3//! Each rule converts to one that matches exactly what OpenSnitch's matched, or is reported with the reason it can't:
4//!
5//! - `simple` operators compare whole values, case-insensitively unless `sensitive`; `regexp` ones match anywhere in
6//!   the value (Go's `MatchString`); `network` ones match a CIDR range; `list` ones need all of theirs to hold, as
7//!   Snitchhiker's conditions do.
8//! - Host names compare without case in Snitchhiker anyway. A `simple` path or command line is taken as the exact
9//!   text even when not `sensitive` (OpenSnitch's UI writes every condition so; Linux's paths have case); case-
10//!   insensitive regular expressions keep `(?i)`, and unanchored ones are wrapped so they still match anywhere.
11//! - The patterns OpenSnitch's UI writes become Snitchhiker's own forms: `^(|.*\.)example\.com$` the host names
12//!   `example.com` and `*.example.com`; a list of names `^(a.example.com|b.example.com)$` those names; a list of addresses `^(10.1.1.5|10.1.1.6)$` those addresses; an address's first
13//!   numbers and `.*` (`192\.168\.1\..*`) their network. Such a pattern without anchors is taken as meant: anchored.
14//! - The same condition twice (OpenSnitch's UI writes some so) counts once.
15//! - `precedence` rules go in the override tier; others in the normal one.
16//! - Temporary rules (any `duration` but `always`) are left out: they were never meant to last.
17//! - Operands Snitchhiker has no condition for (environment variables, interfaces, process ids, hashes, block lists)
18//!   are reported, not approximated.
19//!
20//! OpenSnitch can't tell containers apart, so its rules matched their processes too. Imported rules apply to the host
21//! only, as every Snitchhiker rule without a context condition does; the report says so.
22

28// OpenSnitch writes `null` for what a rule doesn't use (`"list": null` in every simple rule): `nullable` takes it as
29// the default.
30#[derive(Debug, Deserialize)]
31struct Rule {
32    name: String,
33    #[serde(default = "yes", deserialize_with = "enabled")]
34    enabled: bool,
35    #[serde(default, deserialize_with = "nullable")]
36    precedence: bool,
37    action: String,
38    #[serde(default, deserialize_with = "nullable")]
39    duration: String,
40    operator: Operator,
41}
42
43#[derive(Debug, Deserialize)]
44struct Operator {
45    #[serde(rename = "type")]
46    kind: String,
47    #[serde(default, deserialize_with = "nullable")]
48    operand: String,
49    #[serde(default, deserialize_with = "nullable")]
50    data: String,
51    #[serde(default, deserialize_with = "nullable")]
52    sensitive: bool,
53    #[serde(default, deserialize_with = "nullable")]
54    list: Vec<Operator>,
55}

57const fn yes() -> bool {
58    true
59}

61fn nullable<'de, D, T>(deserializer: D) -> Result<T, D::Error>
62where
63    D: serde::Deserializer<'de>,
64    T: Deserialize<'de> + Default,
65{
66    Ok(Option::<T>::deserialize(deserializer)?.unwrap_or_default())
67}

69fn enabled<'de, D: serde::Deserializer<'de>>(deserializer: D) -> Result<bool, D::Error> {
70    Ok(Option::<bool>::deserialize(deserializer)?.unwrap_or(true))
71}

73/// The import: the rules that convert, and why each other one doesn't.
74#[derive(Debug, Default)]
75pub struct Imported {
76    pub file: RuleFile,
77    /// (rule name or file, reason).
78    pub skipped: Vec<(String, String)>,
79}

81/// Converts OpenSnitch rule files, given as (file name, JSON), into a rule file with a folder named `folder`.
82#[must_use]
83pub fn import(files: &[(String, String)], folder: &str) -> Imported {
84    let mut imported = Imported {
85        file: RuleFile {
86            folder: Some(FolderSpec {
87                name: folder.to_owned(),
88                scope: Conditions::default(),
89            }),
90            rule: Vec::new(),
91        },
92        skipped: Vec::new(),
93    };
94    for (name, json) in files {
95        let rule: Rule = match serde_json::from_str(json) {
96            Ok(rule) => rule,
97            Err(e) => {
98                imported
99                    .skipped
100                    .push((name.clone(), format!("not an OpenSnitch rule: {e}")));
101                continue;
102            }
103        };
104        // A rule that converts but isn't valid (a regular expression Snitchhiker's engine doesn't take, say) is
105        // reported like one that doesn't convert: one such rule mustn't make the whole import fail.
106        match convert(&rule).and_then(|spec| valid(&spec).map(|()| spec)) {
107            Ok(spec) => imported.file.rule.push(spec),
108            Err(reason) => imported.skipped.push((rule.name, reason)),
109        }
110    }
111    imported
112}

114/// Whether the rule is one the daemon takes: its conditions parse and its patterns compile (even when it is off).
115fn valid(spec: &RuleSpec) -> Result<(), String> {
116    let mut rule = spec.to_rule(0, None).map_err(|e| e.to_string())?;
117    rule.enabled = true;
118    crate::engine::RuleSet::compile(&[], &[rule])
119        .map(|_| ())
120        .map_err(|e| e.to_string())
121}

123fn convert(rule: &Rule) -> Result<RuleSpec, String> {
124    if !(rule.duration.is_empty() || rule.duration == "always") {
125        return Err(format!("a temporary rule ({})", rule.duration));
126    }
127    let action = match rule.action.as_str() {
128        "allow" => Action::Allow,
129        "deny" => Action::Deny,
130        "reject" => Action::Reject,
131        other => return Err(format!("unknown action `{other}`")),
132    };
133    let mut conditions = Conditions::default();
134    let operators: Vec<&Operator> = if rule.operator.kind == "list" {
135        rule.operator.list.iter().collect()
136    } else {
137        vec![&rule.operator]
138    };
139    for operator in operators {
140        add(&mut conditions, operator)?;
141    }
142    if conditions == Conditions::default() {
143        return Err("it matches every connection".to_owned());
144    }
145    Ok(RuleSpec {
146        name: rule.name.clone(),
147        action,
148        tier: if rule.precedence {
149            Tier::Override
150        } else {
151            Tier::Normal
152        },
153        enabled: rule.enabled,
154        notes: "Imported from OpenSnitch".to_owned(),
155        conditions,
156    })
157}

159/// A regular expression matching what OpenSnitch's `regexp` matched: anywhere in the value, unless anchored.
160fn unanchored(regex: &str, sensitive: bool) -> String {
161    let anchored = regex.starts_with('^') && regex.ends_with('$') && !regex.ends_with("\\$");
162    let body = if anchored {
163        regex.to_owned()
164    } else {
165        format!(".*(?:{regex}).*")
166    };
167    if sensitive {
168        format!("re:{body}")
169    } else {
170        format!("re:(?i){body}")
171    }
172}

174/// A pattern matching what OpenSnitch's `simple` or `regexp` operator matched, for a text value. A `simple` one is
175/// the exact text even when not `sensitive`: OpenSnitch's UI writes every condition so, and Linux paths and command
176/// lines have case.
177fn text_pattern(operator: &Operator) -> Result<String, String> {
178    match operator.kind.as_str() {
179        "simple" => Ok(operator.data.clone()),
180        "regexp" => Ok(unanchored(&operator.data, operator.sensitive)),
181        other => Err(format!("`{other}` on {}", operator.operand)),
182    }
183}

185/// Puts one operator's values into a condition. Two operators on the same operand both have to hold, which a list
186/// (any of its values) can't say, unless they allow the same values: OpenSnitch's UI writes some conditions twice.
187fn put<T: PartialEq>(field: &mut Vec<T>, values: Vec<T>, operand: &str) -> Result<(), String> {
188    if field.is_empty() {
189        *field = values;
190        Ok(())
191    } else if field.len() == values.len() && values.iter().all(|v| field.contains(v)) {
192        Ok(())
193    } else {
194        Err(format!("two different conditions on {operand}"))
195    }
196}

198/// Adds one operator's condition.
199fn add(conditions: &mut Conditions, operator: &Operator) -> Result<(), String> {
200    let simple = operator.kind == "simple";
201    let operand = operator.operand.as_str();
202    match operand {
203        "process.path" => put(&mut conditions.exe, vec![text_pattern(operator)?], operand)?,
204        "process.command" => put(
205            &mut conditions.cmdline,
206            vec![text_pattern(operator)?],
207            operand,
208        )?,
209        "process.parent.path" => {
210            put(
211                &mut conditions.parent,
212                vec![text_pattern(operator)?],
213                operand,
214            )?;
215        }
216        "dest.host" => {
217            let hosts = match operator.kind.as_str() {
218                "simple" => vec![operator.data.trim_end_matches('.').to_ascii_lowercase()],
219                // Host names match without case anyway.
220                "regexp" => host_shape(&operator.data)
221                    .unwrap_or_else(|| vec![unanchored(&operator.data, true)]),
222                other => return Err(format!("`{other}` on dest.host")),
223            };
224            put(&mut conditions.host, hosts, operand)?;
225        }
226        // An address, a network, or a range that is exactly one network (`127.0.0.0-127.255.255.255`, as
227        // OpenSnitch's own rules write loopback).
228        "dest.ip" | "dest.network" if simple || operator.kind == "network" => {
229            let net = crate::system::parse_network(&operator.data)
230                .map_err(|e| format!("{operand}: {e}"))?
231                .to_string();
232            put(&mut conditions.net, vec![net], operand)?;
233        }
234        "dest.ip" | "dest.network" if operator.kind == "regexp" => {
235            let nets = address_shape(&operator.data).ok_or_else(|| {
236                format!(
237                    "{operand} `{}` is neither a list of addresses nor an address's first numbers followed by `.*`",
238                    operator.data
239                )
240            })?;
241            put(&mut conditions.net, nets, operand)?;
242        }
243        "dest.port" if simple => put(&mut conditions.port, vec![operator.data.clone()], operand)?,
244        "user.id" if simple => {
245            let uid = operator
246                .data
247                .parse()
248                .map_err(|_| format!("user.id `{}` isn't a number", operator.data))?;
249            put(&mut conditions.uid, vec![uid], operand)?;
250        }
251        "protocol" if simple => {
252            let protocol = match operator.data.to_ascii_lowercase().as_str() {
253                "tcp" | "tcp6" => "tcp",
254                "udp" | "udp6" => "udp",
255                "icmp" => "icmp",
256                "icmp6" | "icmpv6" => "icmpv6",
257                other => return Err(format!("protocol `{other}`")),
258            };
259            put(&mut conditions.protocol, vec![protocol.to_owned()], operand)?;
260        }
261        "source.ip" | "source.network" | "source.port" => {
262            let loopback = operator
263                .data
264                .parse::<std::net::IpAddr>()
265                .is_ok_and(|ip| ip.is_loopback());
266            return Err(if loopback {
267                "Snitchhiker doesn't match on the connection's local address; connections to loopback are allowed \
268                 by the System folder's Loopback rule"
269                    .to_owned()
270            } else {
271                format!("Snitchhiker doesn't match on the connection's local address ({operand})")
272            });
273        }
274        operand => {
275            return Err(format!(
276                "Snitchhiker has no condition for `{}` on {operand}",
277                operator.kind
278            ));
279        }
280    }
281    Ok(())
282}

284/// The pattern without OpenSnitch's anchors. Patterns of the shapes below that lack them are taken as anchored: an
285/// unanchored `172\.18\..*` also matched `10.172.18.1`, which no one meant.
286fn strip_anchors(regex: &str) -> &str {
287    let regex = regex.strip_prefix('^').unwrap_or(regex);
288    match regex.strip_suffix('$') {
289        Some(rest) if !rest.ends_with('\\') => rest,
290        _ => regex,
291    }
292}

294/// Literal text of a pattern made of letters, digits, `-`, `_`, `:` and dots (`\.`, or `.` meant as one); `None` for
295/// anything else.
296fn literal(text: &str) -> Option<String> {
297    let mut out = String::with_capacity(text.len());
298    let mut chars = text.chars();
299    while let Some(c) = chars.next() {
300        match c {
301            '\\' if chars.next()? == '.' => out.push('.'),
302            '.' => out.push('.'),
303            c if c.is_ascii_alphanumeric() || matches!(c, '-' | '_' | ':') => out.push(c),
304            _ => return None,
305        }
306    }
307    (!out.is_empty()).then_some(out)
308}

310/// The host names of a `dest.host` pattern OpenSnitch's UI writes: `^(|.*\.)example\.com$` (the name and its
311/// subdomains), `^(.*\.)?example\.com$`, `^.*\.example\.com$` (subdomains only), `^example\.com$`, or a list of
312/// names, `^(a.example.com|b.example.com)$`.
313fn host_shape(regex: &str) -> Option<Vec<String>> {
314    let body = strip_anchors(regex);
315    if let Some(names) = body.strip_prefix('(').and_then(|b| b.strip_suffix(')'))
316        && !names.contains('(')
317        && names.split('|').all(|name| !name.is_empty())
318    {
319        let mut hosts: Vec<String> = Vec::new();
320        for name in names.split('|') {
321            for host in one_host(name)? {
322                if !hosts.contains(&host) {
323                    hosts.push(host);
324                }
325            }
326        }
327        return Some(hosts);
328    }
329    one_host(body)
330}
331
332fn one_host(body: &str) -> Option<Vec<String>> {
333    let (apex, subdomains, name) = if let Some(name) = body
334        .strip_prefix("(|.*\\.)")
335        .or_else(|| body.strip_prefix("(.*\\.|)"))
336        .or_else(|| body.strip_prefix("(.*\\.)?"))
337    {
338        (true, true, name)
339    } else if let Some(name) = body.strip_prefix(".*\\.") {
340        (false, true, name)
341    } else {
342        (true, false, body)
343    };
344    let name = literal(name)?.to_ascii_lowercase();
345    if name.contains(':') || name.starts_with('.') || name.ends_with('.') || name.contains("..") {
346        return None;
347    }
348    let mut hosts = Vec::new();
349    if apex {
350        hosts.push(name.clone());
351    }
352    if subdomains {
353        hosts.push(format!("*.{name}"));
354    }
355    Some(hosts)
356}

358/// The networks of a `dest.ip` pattern of the shapes people write: a list of addresses, `^(10.1.1.5|10.1.1.6)$`, or
359/// an IPv4 address's first numbers followed by anything, `192\.168\.1\..*` (`192.168.1.0/24`).
360fn address_shape(regex: &str) -> Option<Vec<String>> {
361    let body = strip_anchors(regex);
362    for any in [".*", ".+", "\\d+", "[0-9]+", "\\d{1,3}", "[0-9]{1,3}"] {
363        if let Some(prefix) = body.strip_suffix(any) {
364            let prefix = literal(prefix)?;
365            let octets: Vec<u8> = prefix
366                .strip_suffix('.')?
367                .split('.')
368                .map(|octet| octet.parse().ok())
369                .collect::<Option<_>>()?;
370            if octets.is_empty() || octets.len() > 3 {
371                return None;
372            }
373            let mut address = [0u8; 4];
374            for (slot, octet) in address.iter_mut().zip(&octets) {
375                *slot = *octet;
376            }
377            let length = u8::try_from(octets.len() * 8).ok()?;
378            let net = ipnet::Ipv4Net::new(address.into(), length).ok()?;
379            return Some(vec![net.to_string()]);
380        }
381    }
382    let body = body
383        .strip_prefix('(')
384        .and_then(|b| b.strip_suffix(')'))
385        .unwrap_or(body);
386    let mut nets: Vec<String> = Vec::new();
387    for alternative in body.split('|') {
388        let address: std::net::IpAddr = literal(alternative)?.parse().ok()?;
389        let net = ipnet::IpNet::from(address).to_string();
390        if !nets.contains(&net) {
391            nets.push(net);
392        }
393    }
394    Some(nets)
395}

546                       {"type": "simple", "operand": "process.path", "data": "/usr/bin/docker-proxy", "sensitive": true}"#,
547                ),
548                rule(
549                    "addresses",
550                    r#"{"type": "regexp", "operand": "dest.ip", "data": "^(10.1.1.176|10.9.1.76|10.9.1.76)$"}"#,
551                ),
552                rule(
553                    "other",
554                    r#"{"type": "regexp", "operand": "dest.ip", "data": "^10\\.[0-5]\\..*"}"#,
555                ),
556                rule(
557                    "different",
558                    r#"{"type": "simple", "operand": "dest.host", "data": "a.com"},
559                       {"type": "simple", "operand": "dest.host", "data": "b.com"}"#,
560                ),
561                rule(
562                    "source",
563                    r#"{"type": "simple", "operand": "source.ip", "data": "127.0.0.1"},
564                       {"type": "simple", "operand": "dest.ip", "data": "127.0.0.1"}"#,
565                ),
566            ],
567            "OpenSnitch",
568        );
569        let conditions: Vec<(&str, &Conditions)> = imported
570            .file
571            .rule
572            .iter()
573            .map(|r| (r.name.as_str(), &r.conditions))
574            .collect();
575        let [
576            ("domain", domain),
577            ("subdomains", subdomains),
578            ("names", names),
579            ("prefix", prefix),
580            ("addresses", addresses),
581        ] = conditions.as_slice()
582        else {
583            panic!("{conditions:?} {:?}", imported.skipped);
584        };
585        assert_eq!(domain.host, vec!["us.sentry.io", "*.us.sentry.io"]);
586        assert_eq!(subdomains.host, vec!["*.example.com"]);
587        assert_eq!(
588            names.host,
589            vec!["time.cloudflare.com", "0.arch.pool.ntp.org"]
590        );
591        assert_eq!(prefix.net, vec!["172.18.0.0/16"]);
592        assert_eq!(prefix.exe, vec!["/usr/bin/docker-proxy"]);
593        assert_eq!(addresses.net, vec!["10.1.1.176/32", "10.9.1.76/32"]);
594        let skipped: Vec<&str> = imported.skipped.iter().map(|(n, _)| n.as_str()).collect();
595        assert_eq!(skipped, vec!["other", "different", "source"]);
596        assert!(imported.skipped[2].1.contains("Loopback rule"));
597    }
