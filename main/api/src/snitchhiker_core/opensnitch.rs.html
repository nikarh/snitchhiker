1//! Importing OpenSnitch's rules (`/etc/opensnitchd/rules/*.json`, its documented format) as a rule file.
2//!
3//! Each rule converts to one that matches exactly what OpenSnitch's matched, or is reported with the reason it can't:
4//!
5//! - `simple` operators compare whole values, case-insensitively unless `sensitive`; `regexp` ones match anywhere in
6//!   the value (Go's `MatchString`); `network` ones match a CIDR range; `list` ones need all of theirs to hold, as
7//!   Snitchhiker's conditions do.
8//! - Host names compare without case in Snitchhiker anyway. Other case-insensitive values become `(?i)` regular
9//!   expressions; unanchored regular expressions are wrapped so they still match anywhere.
10//! - `precedence` rules go in the override tier; others in the normal one.
11//! - Temporary rules (any `duration` but `always`) are left out: they were never meant to last.
12//! - Operands Snitchhiker has no condition for (environment variables, interfaces, process ids, hashes, block lists)
13//!   are reported, not approximated.
14//!
15//! OpenSnitch can't tell containers apart, so its rules matched their processes too. Imported rules apply to the host
16//! only, as every Snitchhiker rule without a context condition does; the report says so.
17
18use serde::Deserialize;

20use crate::rule::{Action, Tier};
21use crate::spec::{Conditions, FolderSpec, RuleFile, RuleSpec};

23#[derive(Debug, Deserialize)]
24struct Rule {
25    name: String,
26    #[serde(default = "yes")]
27    enabled: bool,
28    #[serde(default)]
29    precedence: bool,
30    action: String,
31    #[serde(default)]
32    duration: String,
33    operator: Operator,
34}
35
36#[derive(Debug, Deserialize)]
37struct Operator {
38    #[serde(rename = "type")]
39    kind: String,
40    #[serde(default)]
41    operand: String,
42    #[serde(default)]
43    data: String,
44    #[serde(default)]
45    sensitive: bool,
46    #[serde(default)]
47    list: Vec<Operator>,
48}
49
50const fn yes() -> bool {
51    true
52}

54/// The import: the rules that convert, and why each other one doesn't.
55#[derive(Debug, Default)]
56pub struct Imported {
57    pub file: RuleFile,
58    /// (rule name or file, reason).
59    pub skipped: Vec<(String, String)>,
60}
61
62/// Converts OpenSnitch rule files, given as (file name, JSON), into a rule file with a folder named `folder`.
63#[must_use]
64pub fn import(files: &[(String, String)], folder: &str) -> Imported {
65    let mut imported = Imported {
66        file: RuleFile {
67            folder: Some(FolderSpec {
68                name: folder.to_owned(),
69                scope: Conditions::default(),
70            }),
71            rule: Vec::new(),
72        },
73        skipped: Vec::new(),
74    };
75    for (name, json) in files {
76        let rule: Rule = match serde_json::from_str(json) {
77            Ok(rule) => rule,
78            Err(e) => {
79                imported
80                    .skipped
81                    .push((name.clone(), format!("not an OpenSnitch rule: {e}")));
82                continue;
83            }
84        };
85        match convert(&rule) {
86            Ok(spec) => imported.file.rule.push(spec),
87            Err(reason) => imported.skipped.push((rule.name, reason)),
88        }
89    }
90    imported
91}

93fn convert(rule: &Rule) -> Result<RuleSpec, String> {
94    if !(rule.duration.is_empty() || rule.duration == "always") {
95        return Err(format!("a temporary rule ({})", rule.duration));
96    }
97    let action = match rule.action.as_str() {
98        "allow" => Action::Allow,
99        "deny" => Action::Deny,
100        "reject" => Action::Reject,
101        other => return Err(format!("unknown action `{other}`")),
102    };
103    let mut conditions = Conditions::default();
104    let operators: Vec<&Operator> = if rule.operator.kind == "list" {
105        rule.operator.list.iter().collect()
106    } else {
107        vec![&rule.operator]
108    };
109    for operator in operators {
110        add(&mut conditions, operator)?;
111    }
112    if conditions == Conditions::default() {
113        return Err("it matches every connection".to_owned());
114    }
115    Ok(RuleSpec {
116        name: rule.name.clone(),
117        action,
118        tier: if rule.precedence {
119            Tier::Override
120        } else {
121            Tier::Normal
122        },
123        enabled: rule.enabled,
124        notes: "Imported from OpenSnitch".to_owned(),
125        conditions,
126    })
127}

129/// A regular expression matching what OpenSnitch's `regexp` matched: anywhere in the value, unless anchored.
130fn unanchored(regex: &str, sensitive: bool) -> String {
131    let anchored = regex.starts_with('^') && regex.ends_with('$') && !regex.ends_with("\\$");
132    let body = if anchored {
133        regex.to_owned()
134    } else {
135        format!(".*(?:{regex}).*")
136    };
137    if sensitive {
138        format!("re:{body}")
139    } else {
140        format!("re:(?i){body}")
141    }
142}
143
144/// A pattern matching what OpenSnitch's `simple` or `regexp` operator matched, for a text value.
145fn text_pattern(operator: &Operator) -> Result<String, String> {
146    match operator.kind.as_str() {
147        "simple" if operator.sensitive => Ok(operator.data.clone()),
148        "simple" => Ok(format!("re:(?i){}", regex_escape(&operator.data))),
149        "regexp" => Ok(unanchored(&operator.data, operator.sensitive)),
150        other => Err(format!("`{other}` on {}", operator.operand)),
151    }
152}
153
154fn regex_escape(text: &str) -> String {
155    let mut escaped = String::with_capacity(text.len());
156    for c in text.chars() {
157        if "\\.+*?()|[]{}^$#&-~".contains(c) {
158            escaped.push('\\');
159        }
160        escaped.push(c);
161    }
162    escaped
163}
164
165/// Adds one operator's condition; two on the same operand would need both to hold, which Snitchhiker's lists can't say.
166fn add(conditions: &mut Conditions, operator: &Operator) -> Result<(), String> {
167    let once = |values: &Vec<String>| {
168        if values.is_empty() {
169            Ok(())
170        } else {
171            Err(format!("two conditions on {}", operator.operand))
172        }
173    };
174    let simple = operator.kind == "simple";
175    match operator.operand.as_str() {
176        "process.path" => {
177            once(&conditions.exe)?;
178            conditions.exe.push(text_pattern(operator)?);
179        }
180        "process.command" => {
181            once(&conditions.cmdline)?;
182            conditions.cmdline.push(text_pattern(operator)?);
183        }
184        "process.parent.path" => {
185            once(&conditions.parent)?;
186            conditions.parent.push(text_pattern(operator)?);
187        }
188        "dest.host" => {
189            once(&conditions.host)?;
190            conditions.host.push(match operator.kind.as_str() {
191                "simple" => operator.data.trim_end_matches('.').to_ascii_lowercase(),
192                // Host names match without case anyway.
193                "regexp" => unanchored(&operator.data, true),
194                other => return Err(format!("`{other}` on dest.host")),
195            });
196        }
197        "dest.ip" if simple => {
198            once(&conditions.net)?;
199            conditions.net.push(operator.data.clone());
200        }
201        "dest.network" => {
202            once(&conditions.net)?;
203            conditions.net.push(operator.data.clone());
204        }
205        "dest.port" if simple => {
206            once(&conditions.port)?;
207            conditions.port.push(operator.data.clone());
208        }
209        "user.id" if simple => {
210            if !conditions.uid.is_empty() {
211                return Err("two conditions on user.id".to_owned());
212            }
213            conditions.uid.push(
214                operator
215                    .data
216                    .parse()
217                    .map_err(|_| format!("user.id `{}` isn't a number", operator.data))?,
218            );
219        }
220        "protocol" if simple => {
221            once(&conditions.protocol)?;
222            let protocol = match operator.data.to_ascii_lowercase().as_str() {
223                "tcp" | "tcp6" => "tcp",
224                "udp" | "udp6" => "udp",
225                "icmp" => "icmp",
226                "icmp6" | "icmpv6" => "icmpv6",
227                other => return Err(format!("protocol `{other}`")),
228            };
229            conditions.protocol.push(protocol.to_owned());
230        }
231        operand => {
232            return Err(format!(
233                "Snitchhiker has no condition for `{}` on {operand}",
234                operator.kind
235            ));
236        }
237    }
238    Ok(())
239}

245    fn one(json: &str) -> Imported {
246        import(&[("rule.json".to_owned(), json.to_owned())], "OpenSnitch")

249    #[test]
250    fn a_list_rule_converts_exactly() {
251        let imported = one(
252            r#"{"name": "allow-curl-example", "enabled": true, "precedence": true, "action": "allow",
253                "duration": "always", "operator": {"type": "list", "operand": "list", "list": [
254                  {"type": "simple", "operand": "process.path", "data": "/usr/bin/curl", "sensitive": true},
255                  {"type": "regexp", "operand": "dest.host", "data": "^(.*\\.)?example\\.com$"},
256                  {"type": "simple", "operand": "dest.port", "data": "443"},
257                  {"type": "simple", "operand": "user.id", "data": "1000"},
258                  {"type": "simple", "operand": "protocol", "data": "tcp6"}]}}"#,
259        );
260        assert!(imported.skipped.is_empty(), "{:?}", imported.skipped);
261        let rule = &imported.file.rule[0];
262        assert_eq!(rule.tier, Tier::Override);
263        assert_eq!(rule.conditions.exe, vec!["/usr/bin/curl"]);
264        assert_eq!(rule.conditions.host, vec![r"re:^(.*\.)?example\.com$"]);
265        assert_eq!(rule.conditions.port, vec!["443"]);
266        assert_eq!(rule.conditions.uid, vec![1000]);
267        assert_eq!(rule.conditions.protocol, vec!["tcp"]);
268        // And it is a valid rule.
269        assert!(rule.to_rule(0, None).is_ok());
270    }
271
272    #[test]
273    fn case_and_anchoring_keep_their_meaning() {
274        let imported = one(r#"{"name": "a", "action": "deny", "duration": "always",
275                "operator": {"type": "simple", "operand": "process.path", "data": "/opt/App.bin"}}"#);
276        assert_eq!(
277            imported.file.rule[0].conditions.exe,
278            vec![r"re:(?i)/opt/App\.bin"]
279        );
280        let imported = one(r#"{"name": "b", "action": "deny", "duration": "always",
281                "operator": {"type": "regexp", "operand": "process.command", "data": "--remote-debugging", "sensitive": true}}"#);
282        assert_eq!(
283            imported.file.rule[0].conditions.cmdline,
284            vec!["re:.*(?:--remote-debugging).*"]
285        );
286    }

288    #[test]
289    fn what_cant_be_expressed_is_reported() {
290        let imported = import(
291            &[
292                (
293                    "temp.json".to_owned(),
294                    r#"{"name": "temp", "action": "allow", "duration": "30m",
295                        "operator": {"type": "simple", "operand": "dest.port", "data": "80"}}"#
296                        .to_owned(),
297                ),
298                (
299                    "env.json".to_owned(),
300                    r#"{"name": "env", "action": "allow", "duration": "always",
301                        "operator": {"type": "simple", "operand": "process.env.FOO", "data": "1"}}"#
302                        .to_owned(),
303                ),
304                ("broken.json".to_owned(), "{".to_owned()),
305            ],
306            "OpenSnitch",
307        );
308        assert_eq!(imported.file.rule, Vec::new());
309        let names: Vec<&str> = imported.skipped.iter().map(|(n, _)| n.as_str()).collect();
310        assert_eq!(names, vec!["temp", "env", "broken.json"]);
311    }
