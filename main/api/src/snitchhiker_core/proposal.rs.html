1//! Shaping agents' proposals (plan §5.11). An agent only suggests: what reaches a person is a rule made from what the
2//! kernel saw where it can be (the program of the blocked connections, not the one the agent typed), refused when it
3//! would allow every program everywhere, and marked wherever it is broad, so that approving it needs a second look
4//! (and step-up).

15/// A rule as an agent proposed it.
16#[derive(Debug, Clone, Default, PartialEq, Eq, Serialize, Deserialize)]
17pub struct Proposed {
18    /// The program's path; empty for any.
19    #[serde(default)]
20    pub exe: String,
21    /// Host names (`*.` for subdomains), addresses or networks; empty for any.
22    #[serde(default)]
23    pub hosts: Vec<String>,
24    /// Ports or ranges; empty for any.
25    #[serde(default)]
26    pub ports: Vec<String>,
27    /// tcp, udp; empty for any.
28    #[serde(default)]
29    pub protocol: String,
30    #[serde(default)]
31    pub context: Option<ContextKind>,
32}

34/// A connection the node blocked or asked about, from the proposing agent's own processes.

36pub struct Observation {

38    pub host: Option<String>,
39    pub address: IpAddr,
40    pub port: u16,

43/// Why a shaped rule is broad: approving it takes step-up.
44#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
45#[serde(tag = "risk", content = "detail", rename_all = "snake_case")]
46pub enum Risk {
47    /// Any destination.
48    AnyHost,
49    /// A wildcard on a public suffix (`*.com`, `*.co.uk`): every site under it.
50    PublicSuffix(String),
51    /// Any program.
52    AnyProgram,
53    /// Any port, with a broad destination.
54    AnyPort,
55    /// A network wider than a /24 (a /48 for IPv6).
56    WideNetwork(String),
57    /// The LAN, the cloud metadata service and the like.
58    SensitiveAlias(String),
59    /// A binary any user can replace, with a broad destination.
60    WritableBinary(String),
61    /// No blocked connection of the agent's matches it: the agent guessed.
62    Speculative,
63}
64
65impl Risk {
66    /// For people.
67    #[must_use]
68    pub fn describe(&self) -> String {
69        match self {
70            Self::AnyHost => "any destination".to_owned(),
71            Self::PublicSuffix(suffix) => format!("every site under the public suffix {suffix}"),
72            Self::AnyProgram => "any program".to_owned(),
73            Self::AnyPort => "any port".to_owned(),
74            Self::WideNetwork(net) => format!("the wide network {net}"),
75            Self::SensitiveAlias(alias) => format!("the sensitive network `{alias}`"),
76            Self::WritableBinary(exe) => format!("{exe}, which a user other than root can replace"),
77            Self::Speculative => "nothing the agent tried matches it".to_owned(),
78        }
79    }
80}

82#[derive(Debug, Clone, PartialEq, Eq, thiserror::Error)]
83pub enum Refusal {
84    #[error("a rule for any program to any destination")]
85    Everything,
86    #[error("{0}")]
87    Invalid(String),
88    #[error("too long: {0}")]
89    TooLong(&'static str),
90}

92/// A proposed rule, shaped: its conditions, the program it was grounded in (when observations named one), and its
93/// risks.
94#[derive(Debug, Clone, PartialEq, Eq)]
95pub struct Shaped {
96    pub conditions: Conditions,
97    pub risks: Vec<Risk>,
98}
99
100/// Limits on what an agent sends.
101const MAX_HOSTS: usize = 32;
102const MAX_PORTS: usize = 16;
103const MAX_TEXT: usize = 253;
104
105/// Whether `suffix` (`com`, `co.uk`) is a public suffix: anything under it is someone else's.
106fn public_suffix(suffix: &str) -> bool {
107    psl::suffix_str(suffix).is_some_and(|s| s.eq_ignore_ascii_case(suffix))
108}
109
110fn check_limits(proposed: &Proposed) -> Result<(), Refusal> {
111    if proposed.hosts.len() > MAX_HOSTS {
112        return Err(Refusal::TooLong("hosts"));
113    }
114    if proposed.ports.len() > MAX_PORTS {
115        return Err(Refusal::TooLong("ports"));
116    }
117    if proposed.exe.len() > 4096
118        || proposed
119            .hosts
120            .iter()
121            .chain(&proposed.ports)
122            .any(|t| t.len() > MAX_TEXT)
123    {
124        return Err(Refusal::TooLong("a value"));
125    }
126    Ok(())
127}

129/// The destinations into `conditions`; whether any is broad.
130fn destinations(
131    hosts: &[String],
132    conditions: &mut Conditions,
133    risks: &mut Vec<Risk>,
134) -> Result<bool, Refusal> {
135    let mut broad = false;
136    for host in hosts.iter().map(|h| h.trim()) {
137        if host.starts_with("re:") || host.contains(['/', ' ']) && host.parse::<IpNet>().is_err() {
138            return Err(Refusal::Invalid(format!(
139                "`{host}`: a host name, `*.` and a name, an address or a network"
140            )));
141        }
142        if host.is_empty() || host == "*" {
143            broad = true;
144            risks.push(Risk::AnyHost);
145            continue;
146        }
147        if let Some((_, nets)) = ALIASES.iter().find(|(alias, _)| *alias == host) {
148            let _ = nets;
149            broad = true;
150            risks.push(Risk::SensitiveAlias(host.to_owned()));
151            conditions.net.push(host.to_owned());
152            continue;
153        }
154        if let Ok(net) = host
155            .parse::<IpNet>()
156            .or_else(|_| host.parse::<IpAddr>().map(IpNet::from))
157        {
158            let wide = match net {
159                IpNet::V4(n) => n.prefix_len() < 24,
160                IpNet::V6(n) => n.prefix_len() < 48,
161            };
162            if wide {
163                broad = true;
164                risks.push(Risk::WideNetwork(net.to_string()));
165            }
166            if ALIASES
167                .iter()
168                .filter(|(alias, _)| matches!(*alias, "lan" | "imds" | "loopback"))
169                .flat_map(|(alias, nets)| nets.iter().map(move |n| (*alias, *n)))
170                .any(|(_, n)| {
171                    n.parse::<IpNet>()
172                        .is_ok_and(|n| n.contains(&net.addr()) || net.contains(&n.addr()))
173                })
174            {
175                broad = true;
176                risks.push(Risk::SensitiveAlias(net.to_string()));
177            }
178            // A single address stays as written; a network as the network.
179            conditions.net.push(if host.parse::<IpAddr>().is_ok() {
180                host.to_owned()
181            } else {
182                net.to_string()
183            });
184            continue;
185        }
186        if let Some(suffix) = host.strip_prefix("*.") {
187            if public_suffix(suffix) {
188                broad = true;
189                risks.push(Risk::PublicSuffix(suffix.to_owned()));
190            }
191        } else if host.contains('*') {
192            return Err(Refusal::Invalid(format!(
193                "`{host}`: a wildcard only as `*.` and a name"
194            )));
195        }
196        conditions.host.push(host.to_ascii_lowercase());
197    }
198    Ok(broad)
199}

201/// Whether the observed connection `o` is one the shaped rule (`conditions`, and the program the agent named, if
202/// any) would have let through.
203#[must_use]
204pub fn grounds(o: &Observation, conditions: &Conditions, exe: &str) -> bool {
205    let host_ok = conditions.host.is_empty() && conditions.net.is_empty()
206        || o.host.as_deref().is_some_and(|h| {
207            conditions.host.iter().any(|p| {
208                p.strip_prefix("*.").map_or_else(
209                    || h.eq_ignore_ascii_case(p),
210                    |suffix| {
211                        h.len() > suffix.len()
212                            && h.to_ascii_lowercase().ends_with(&format!(".{suffix}"))
213                    },
214                )
215            })
216        })
217        || conditions.net.iter().any(|n| {
218            n.parse::<IpNet>()
219                .or_else(|_| n.parse::<IpAddr>().map(IpNet::from))
220                .is_ok_and(|n| n.contains(&o.address))
221        });
222    let port_ok = conditions.port.is_empty() || conditions.port.iter().any(|p| {
223        let (a, b) = p.split_once('-').unwrap_or((p, p));
224        matches!((a.parse::<u16>(), b.parse::<u16>()), (Ok(a), Ok(b)) if (a..=b).contains(&o.port))
225    });
226    let exe_ok = exe.is_empty() || o.exe == exe;
227    host_ok && port_ok && exe_ok
228}

230/// Shapes `proposed`, from the agent's `observed` connections; `writable` says whether a user other than root can
231/// replace a binary.
232pub fn shape(
233    proposed: &Proposed,
234    observed: &[Observation],
235    writable: &dyn Fn(&str) -> bool,
236) -> Result<Shaped, Refusal> {
237    check_limits(proposed)?;
238    let mut risks = Vec::new();
239    let mut conditions = Conditions {
240        context: proposed.context,
241        ..Conditions::default()
242    };
243    let mut broad = destinations(&proposed.hosts, &mut conditions, &mut risks)?;
244    if proposed.hosts.is_empty() {
245        broad = true;
246        risks.push(Risk::AnyHost);
247    }
248    // Ports and protocol.
249    for port in proposed.ports.iter().map(|p| p.trim()) {
250        let valid = port.split_once('-').map_or_else(
251            || port.parse::<u16>().is_ok(),
252            |(a, b)| a.parse::<u16>().is_ok() && b.parse::<u16>().is_ok(),
253        );
254        if !valid {
255            return Err(Refusal::Invalid(format!("`{port}`: a port or a range")));
256        }
257        conditions.port.push(port.to_owned());
258    }
259    if proposed.ports.is_empty() && broad {
260        risks.push(Risk::AnyPort);
261    }
262    match proposed.protocol.trim().to_ascii_lowercase().as_str() {
263        "" => {}
264        p @ ("tcp" | "udp") => conditions.protocol.push(p.to_owned()),
265        other => return Err(Refusal::Invalid(format!("`{other}`: tcp or udp"))),
266    }
267    // The program: what the kernel saw for these destinations, before what the agent says (which may be a symlink
268    // or a launcher of the program that connected). The agent's program decides only between several seen.
269    let to_these: Vec<&Observation> = observed
270        .iter()
271        .filter(|o| grounds(o, &conditions, ""))
272        .collect();
273    let named = proposed.exe.trim();
274    let mut programs: Vec<&str> = to_these.iter().map(|o| o.exe.as_str()).collect();
275    programs.sort_unstable();
276    programs.dedup();
277    let exe = if let [only] = programs.as_slice() {
278        (*only).to_owned()
279    } else {
280        named.to_owned()
281    };
282    if !to_these.iter().any(|o| o.exe == exe) {
283        risks.push(Risk::Speculative);
284    }
285    if exe.is_empty() {
286        if broad {
287            return Err(Refusal::Everything);
288        }
289        risks.push(Risk::AnyProgram);
290    } else {
291        if !exe.starts_with('/') {
292            return Err(Refusal::Invalid(format!(
293                "`{exe}`: a program's path from the root"
294            )));
295        }
296        if broad && writable(&exe) {
297            risks.push(Risk::WritableBinary(exe.clone()));
298        }
299        conditions.exe.push(exe);
300    }
301    Ok(Shaped { conditions, risks })
302}

317    fn proposed(exe: &str, hosts: &[&str], ports: &[&str]) -> Proposed {
318        Proposed {
319            exe: exe.to_owned(),
320            hosts: hosts.iter().map(|h| (*h).to_owned()).collect(),
321            ports: ports.iter().map(|p| (*p).to_owned()).collect(),
322            ..Proposed::default()
323        }
324    }

328    #[test]
329    fn a_narrow_proposal_is_grounded_in_what_was_blocked() {
330        let observed = [seen("/home/ada/.cargo/bin/cargo", "static.crates.io", 443)];
331        // The agent names a program loosely; the kernel's is used.
332        let shaped = shape(
333            &proposed("", &["static.crates.io", "crates.io"], &["443"]),
334            &observed,
335            NEVER,
336        )
337        .unwrap();
338        assert_eq!(shaped.conditions.exe, vec!["/home/ada/.cargo/bin/cargo"]);
339        assert_eq!(
340            shaped.conditions.host,
341            vec!["static.crates.io", "crates.io"]
342        );
343        assert_eq!(shaped.risks, vec![]);
344    }

346    #[test]
347    fn the_program_is_the_one_the_kernel_saw() {
348        // A symlink (or a launcher) named: the program that connected is used.
349        let observed = [seen("/usr/bin/python3.13", "pypi.org", 443)];
350        let shaped = shape(
351            &proposed("/usr/bin/python3", &["pypi.org"], &["443"]),
352            &observed,
353            NEVER,
354        )
355        .unwrap();
356        assert_eq!(shaped.conditions.exe, vec!["/usr/bin/python3.13"]);
357        assert_eq!(shaped.risks, vec![]);
358        // Two programs reached it: the agent's word picks one of them, never a third.
359        let observed = [
360            seen("/usr/bin/curl", "pypi.org", 443),
361            seen("/usr/bin/python3.13", "pypi.org", 443),
362        ];
363        let pick =
364            |exe: &str| shape(&proposed(exe, &["pypi.org"], &["443"]), &observed, NEVER).unwrap();
365        assert_eq!(pick("/usr/bin/curl").conditions.exe, vec!["/usr/bin/curl"]);
366        assert_eq!(pick("/usr/bin/curl").risks, vec![]);
367        assert_eq!(pick("/usr/bin/wget").risks, vec![Risk::Speculative]);
368        // An address is kept as written, and grounds like a network.
369        let observed = [Observation {
370            exe: "/usr/bin/curl".to_owned(),
371            host: None,
372            address: "192.0.2.7".parse().unwrap(),
373            port: 80,
374        }];
375        let shaped = shape(&proposed("", &["192.0.2.7"], &["80"]), &observed, NEVER).unwrap();
376        assert_eq!(shaped.conditions.net, vec!["192.0.2.7"]);
377        assert_eq!(shaped.conditions.exe, vec!["/usr/bin/curl"]);
378        assert_eq!(shaped.risks, vec![]);
379    }

381    #[test]
382    fn broad_and_guessed_proposals_are_marked() {
383        let observed = [seen("/usr/bin/node", "registry.npmjs.org", 443)];
384        let risks = |p: Proposed| shape(&p, &observed, NEVER).unwrap().risks;
385        assert_eq!(
386            risks(proposed("/usr/bin/node", &["*.com"], &[])),
387            vec![
388                Risk::PublicSuffix("com".to_owned()),
389                Risk::AnyPort,
390                Risk::Speculative
391            ]
392        );
393        assert_eq!(
394            risks(proposed("/usr/bin/node", &["10.0.0.0/8"], &["443"])),
395            vec![
396                Risk::WideNetwork("10.0.0.0/8".to_owned()),
397                Risk::SensitiveAlias("10.0.0.0/8".to_owned()),
398                Risk::Speculative
399            ]
400        );
401        assert_eq!(
402            risks(proposed("/usr/bin/node", &["imds"], &["80"])),
403            vec![Risk::SensitiveAlias("imds".to_owned()), Risk::Speculative]
404        );
405        assert_eq!(
406            risks(proposed("", &["api.example.com"], &["443"])),
407            vec![Risk::Speculative, Risk::AnyProgram]
408        );
409        // `*.npmjs.org` is under a registrable domain: not broad.
410        assert_eq!(
411            risks(proposed("/usr/bin/node", &["*.npmjs.org"], &["443"])),
412            vec![]
413        );
414        let writable: &dyn Fn(&str) -> bool = &|_| true;
415        assert!(
416            shape(&proposed("/tmp/x", &["*"], &["443"]), &[], writable)
417                .unwrap()
418                .risks
419                .contains(&Risk::WritableBinary("/tmp/x".to_owned()))
420        );
421    }
