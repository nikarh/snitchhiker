1//! What the daemon knows about the process behind a connection, read from `/proc` and cached.
2//!
3//! Entries are keyed by pid *and* start time (from the attribution programs, so a hit reads nothing from `/proc`);
4//! on a miss the process in `/proc` must have exactly that start time, so a pid reused by a new process is never
5//! given the old one's facts.
6
7use std::collections::HashMap;
8use std::fs;
9use std::os::unix::fs::MetadataExt as _;
10use std::path::Path;
11use std::sync::Arc;
12
13use snitchhiker_core::rule::{ContextKind, ProcessRef};

15/// The unit of the start time in `/proc/<pid>/stat`: clock ticks of `USER_HZ`, 100 on Linux (fixed for
16/// userspace whatever the kernel's tick rate).
17pub const NANOS_PER_TICK: u64 = 1_000_000_000 / 100;
18/// How many ancestors are recorded.
19const MAX_ANCESTORS: usize = 16;
20/// Entries kept before the oldest are dropped.
21const CAPACITY: usize = 4096;
22
23/// A process, as rules see it.
24#[derive(Debug, Clone, PartialEq, Eq)]
25pub struct ProcessInfo {
26    pub pid: u32,
27    pub exe: Arc<str>,
28    pub cmdline: String,
29    /// Executables of the ancestors, nearest first.
30    pub ancestors: Vec<String>,
31    /// The process and its ancestors, nearest first.
32    pub lineage: Vec<ProcessRef>,
33    pub context: ContextKind,
34    /// Stable identities of the program (`pkg:curl`, `unit:sshd.service`): see `apps`. None in a container.
35    pub apps: Vec<String>,
36    /// Whether a user other than root can change the executable (or, on the host, the directories it is in).
37    pub writable: bool,
38}

40/// The host's namespaces, to tell containers and other namespaces from the host.
41#[derive(Debug, Clone, Copy, PartialEq, Eq)]
42pub struct HostNamespaces {
43    pub pid: u64,
44    pub user: u64,
45    pub net: u64,
46}
47
48impl HostNamespaces {
49    /// The namespaces of process 1, the host's init.
50    pub fn read() -> std::io::Result<Self> {
51        Ok(Self {
52            pid: fs::metadata("/proc/1/ns/pid")?.ino(),
53            user: fs::metadata("/proc/1/ns/user")?.ino(),
54            net: fs::metadata("/proc/1/ns/net")?.ino(),
55        })
56    }
57}

59/// Process facts by (pid, start time), with home directories by uid.
60///
61/// The user isn't among the facts: it comes from the attribution programs, as it was when the connection was opened.
62/// The owner of `/proc/<pid>` is no substitute: the kernel makes it root for any process that isn't dumpable, and a
63/// process can make itself so.
64#[derive(Debug)]
65pub struct ProcessCache {
66    host: HostNamespaces,
67    packages: Option<Arc<crate::apps::Packages>>,
68    entries: HashMap<(u32, u64), (u64, Arc<ProcessInfo>)>,
69    homes: HashMap<u32, Option<Arc<str>>>,
70    clock: u64,
71}
72
73impl ProcessCache {
74    #[must_use]
75    pub fn new(host: HostNamespaces) -> Self {
76        Self::with_packages(host, None)
77    }
78
79    /// A cache that also names programs by their package (`pkg:`) from `packages`.
80    #[must_use]
81    pub fn with_packages(
82        host: HostNamespaces,
83        packages: Option<Arc<crate::apps::Packages>>,
84    ) -> Self {
85        Self {
86            host,
87            packages,
88            entries: HashMap::with_capacity(CAPACITY),
89            homes: HashMap::new(),
90            clock: 0,
91        }
92    }
93
94    /// The process `pid` that started at `start_boottime` (nanoseconds since boot), if it still exists.
95    pub fn get(&mut self, pid: u32, start_boottime: u64) -> Option<Arc<ProcessInfo>> {
96        let start = start_boottime / NANOS_PER_TICK;
97        self.clock += 1;
98        if let Some((used, info)) = self.entries.get_mut(&(pid, start)) {
99            *used = self.clock;
100            return Some(Arc::clone(info));
101        }
102        // A different process may have the pid now.
103        if start_time(pid)? != start {
104            return None;
105        }
106        let info = Arc::new(self.read(pid, start)?);
107        if self.entries.len() >= CAPACITY {
108            self.evict();
109        }
110        self.entries
111            .insert((pid, start), (self.clock, Arc::clone(&info)));
112        Some(info)
113    }
114
115    /// The home directory of `uid`, from the password database (remembered: users don't move often).
116    pub fn home(&mut self, uid: u32) -> Option<Arc<str>> {
117        self.homes
118            .entry(uid)
119            .or_insert_with(|| home_of(uid).map(Arc::from))
120            .clone()
121    }
122
123    fn read(&self, pid: u32, start: u64) -> Option<ProcessInfo> {
124        let exe = fs::read_link(format!("/proc/{pid}/exe")).ok()?;
125        let exe = Arc::from(exe.to_string_lossy());
126        let cmdline = fs::read(format!("/proc/{pid}/cmdline")).map_or_else(
127            |_| String::new(),
128            |bytes| {
129                bytes
130                    .split(|&b| b == 0)
131                    .filter(|part| !part.is_empty())
132                    .map(String::from_utf8_lossy)
133                    .collect::<Vec<_>>()
134                    .join(" ")
135            },
136        );
137        let (ancestors, mut lineage) = ancestors(pid);
138        lineage.insert(0, ProcessRef { pid, start });
139        let context = context_of(pid, self.host);
140        let apps = match (&self.packages, context) {
141            (Some(packages), ContextKind::Host) => crate::apps::identities(
142                packages,
143                &exe,
144                &fs::read_to_string(format!("/proc/{pid}/cgroup")).unwrap_or_default(),
145            ),
146            _ => Vec::new(),
147        };
148        let writable = if context == ContextKind::Host {
149            crate::lint::writable_by_user(Path::new(&*exe))
150        } else {
151            writable_by_others_path(&format!("/proc/{pid}/exe"))
152        };
153        Some(ProcessInfo {
154            pid,
155            exe,
156            cmdline,
157            ancestors,
158            lineage,
159            context,
160            apps,
161            writable,
162        })
163    }
164
165    /// Drops the least recently used half.
166    fn evict(&mut self) {
167        let mut uses: Vec<u64> = self.entries.values().map(|(used, _)| *used).collect();
168        uses.sort_unstable();
169        let cutoff = uses.get(uses.len() / 2).copied().unwrap_or(0);
170        self.entries.retain(|_, (used, _)| *used > cutoff);
171    }
172}

174/// The process's start time (field 22 of `/proc/<pid>/stat`, in clock ticks since boot).
175#[must_use]
176pub fn start_time(pid: u32) -> Option<u64> {
177    stat_field(pid, 22)
178}

180#[must_use]
181pub fn parent(pid: u32) -> Option<u32> {
182    stat_field(pid, 4).and_then(|ppid| u32::try_from(ppid).ok())
183}

185/// A numeric field of `/proc/<pid>/stat` (1-based, as in proc(5)). The command name (field 2) may contain spaces
186/// and parentheses, so fields are counted from its closing parenthesis.
187fn stat_field(pid: u32, field: usize) -> Option<u64> {
188    let stat = fs::read_to_string(format!("/proc/{pid}/stat")).ok()?;
189    let after_name = stat.get(stat.rfind(')')? + 1..)?;
190    after_name
191        .split_whitespace()
192        .nth(field.checked_sub(3)?)?
193        .parse()
194        .ok()
195}

197/// The executables of `pid`'s ancestors, and the ancestors themselves, nearest first.
198#[must_use]
199pub fn lineage(pid: u32) -> (Vec<String>, Vec<ProcessRef>) {
200    ancestors(pid)
201}
202
203fn ancestors(pid: u32) -> (Vec<String>, Vec<ProcessRef>) {
204    let mut chain = Vec::new();
205    let mut lineage = Vec::new();
206    let mut current = pid;
207    while lineage.len() < MAX_ANCESTORS {
208        let Some(ppid) = parent(current) else { break };
209        if ppid == 0 || ppid == current {
210            break;
211        }
212        let Some(start) = start_time(ppid) else { break };
213        lineage.push(ProcessRef { pid: ppid, start });
214        if let Ok(exe) = fs::read_link(format!("/proc/{ppid}/exe")) {
215            chain.push(exe.to_string_lossy().into_owned());
216        }
217        current = ppid;
218    }
219    (chain, lineage)
220}

222fn context_of(pid: u32, host: HostNamespaces) -> ContextKind {
223    let ns = |kind: &str| {
224        fs::metadata(format!("/proc/{pid}/ns/{kind}"))
225            .ok()
226            .map(|m| m.ino())
227    };
228    // A container has a process namespace of its own, or a user namespace (a rootless one, or anything a user made
229    // with unshare: in one, a user can mount another binary over an allowed path). A mount namespace alone is no
230    // container: making one takes privileges outside a user namespace, and systemd gives its sandboxed services one
231    // (ProtectSystem, PrivateTmp); they are the host's.
232    match (ns("pid"), ns("user"), ns("net")) {
233        (Some(pid), _, _) if pid != host.pid => ContextKind::Container,
234        (_, Some(user), _) if user != host.user => ContextKind::Container,
235        (_, _, Some(net)) if net != host.net => ContextKind::Netns,
236        _ => ContextKind::Host,
237    }
238}

240/// The name of `uid`, from the password database.
241#[must_use]
242pub fn user_of(uid: u32) -> Option<String> {
243    let passwd = fs::read_to_string("/etc/passwd").ok()?;
244    passwd.lines().find_map(|line| {
245        let mut fields = line.split(':');
246        let name = fields.next()?;
247        let entry_uid: u32 = fields.nth(1)?.parse().ok()?;
248        (entry_uid == uid).then(|| name.to_owned())
249    })
250}

259/// Executables' SHA-256s by file version (device, inode, size, modification time): each binary is read once.
260static HASHES: std::sync::Mutex<Option<HashMap<FileVersion, Arc<str>>>> =
261    std::sync::Mutex::new(None);
262/// A file's device, inode, size and modification time.
263type FileVersion = (u64, u64, u64, i64);
264/// Versions remembered at most.
265const HASHED: usize = 1024;
266
267/// The SHA-256 (lowercase hex) of the executable process `pid` runs, read through `/proc/<pid>/exe` (the very file,
268/// wherever it is).
269#[must_use]
270pub fn sha256_of(pid: u32) -> Option<Arc<str>> {
271    use sha2::Digest as _;
272    use std::io::Read as _;
273    let path = format!("/proc/{pid}/exe");
274    let meta = fs::metadata(&path).ok()?;
275    let version = (meta.dev(), meta.ino(), meta.size(), meta.mtime());
276    let hashes = HASHES
277        .lock()
278        .unwrap_or_else(std::sync::PoisonError::into_inner);
279    if let Some(hash) = hashes.as_ref().and_then(|h| h.get(&version)) {
280        return Some(Arc::clone(hash));
281    }
282    drop(hashes);
283    let mut file = fs::File::open(&path).ok()?;
284    let mut hasher = sha2::Sha256::new();
285    let mut buffer = vec![0_u8; 64 * 1024];
286    loop {
287        let n = file.read(&mut buffer).ok()?;
288        if n == 0 {
289            break;
290        }
291        hasher.update(buffer.get(..n)?);
292    }
293    let hex = hasher
294        .finalize()
295        .iter()
296        .fold(String::with_capacity(64), |mut hex, b| {
297            use std::fmt::Write as _;
298            let _ = write!(hex, "{b:02x}");
299            hex
300        });
301    let hash: Arc<str> = Arc::from(hex.as_str());
302    let mut hashes = HASHES
303        .lock()
304        .unwrap_or_else(std::sync::PoisonError::into_inner);
305    let hashes = hashes.get_or_insert_with(HashMap::new);
306    if hashes.len() >= HASHED {
307        hashes.clear();
308    }
309    hashes.insert(version, Arc::clone(&hash));
310    Some(hash)
311}

313/// Whether a user other than root can replace the file at `path` (it is theirs, or group or world writable).
314#[must_use]
315pub fn writable_by_others_path(path: &str) -> bool {
316    fs::metadata(path).is_ok_and(|meta| meta.uid() != 0 || meta.mode() & 0o022 != 0)
317}

333fn uid_in(passwd: &str, name: &str) -> Option<u32> {
334    passwd.lines().find_map(|line| {
335        let mut fields = line.split(':');
336        (fields.next()? == name).then_some(())?;
337        fields.nth(1)?.parse().ok()
338    })
339}
340
341fn home_in(passwd: &str, uid: u32) -> Option<String> {
342    passwd.lines().find_map(|line| {
343        let mut fields = line.split(':');
344        let entry_uid: u32 = fields.nth(2)?.parse().ok()?;
345        let home = fields.nth(2)?;
346        (entry_uid == uid && Path::new(home).is_absolute()).then(|| home.to_owned())
347    })
348}

354    #[test]
355    fn reads_homes_from_passwd() {
356        let passwd = "root:x:0:0::/root:/bin/bash\nada:x:1000:1000:Ada:/home/ada:/bin/zsh\nweird:x:5:5::relative:/bin/sh\n";
357        assert_eq!(home_in(passwd, 1000).as_deref(), Some("/home/ada"));
358        assert_eq!(home_in(passwd, 0).as_deref(), Some("/root"));
359        assert_eq!(home_in(passwd, 5), None);
360        assert_eq!(home_in(passwd, 42), None);
361        assert_eq!(uid_in(passwd, "ada"), Some(1000));
362        assert_eq!(uid_in(passwd, "nobody"), None);
363    }

365    #[test]
366    fn reads_this_process_once_and_checks_its_start_time() {
367        let host = HostNamespaces::read().unwrap_or(HostNamespaces {
368            pid: 0,
369            user: 0,
370            net: 0,
371        });
372        let mut cache = ProcessCache::new(host);
373        let pid = std::process::id();
374        let start = start_time(pid).unwrap() * NANOS_PER_TICK;
375        let me = cache.get(pid, start).unwrap();
376        assert_eq!(me.pid, pid);
377        assert!(me.exe.contains("snitchhiker_daemon") || me.exe.contains("snitchhiker-daemon"));
378        let again = cache.get(pid, start).unwrap();
379        assert!(Arc::ptr_eq(&me, &again));
380        // Another start time is another process: not this one.
381        assert!(cache.get(pid, start + NANOS_PER_TICK).is_none());
382    }
