1//! Who is on the other end of a unix socket, from facts only the kernel and root can establish, and whether it is one
2//! of Snitchhiker's own client binaries.
3//!
4//! Nothing a client sends counts: no tokens, no passwords. A client is trusted when the kernel says it runs a
5//! root-owned binary that root registered as trusted, with the client group that only that binary's setgid bit gives,
6//! nobody tracing it, and memory no other process of its user may read.
7

19/// What the kernel says about a connected peer.
20#[derive(Debug, Clone, PartialEq, Eq)]
21pub struct Peer {
22    pub pid: u32,
23    /// Real and effective user ids.
24    pub uid: u32,
25    pub euid: u32,
26    /// Real and effective group ids, and the supplementary groups.
27    pub gid: u32,
28    pub egid: u32,
29    pub groups: Vec<u32>,
30    /// The executable, as the kernel resolved it.
31    pub exe: PathBuf,
32    /// Device and inode of the executable the process runs (not of whatever is at `exe` now).
33    pub exe_id: (u64, u64),
34    /// The process tracing it (`ptrace`), 0 for none.
35    pub tracer: u32,
36    /// Whether other processes of its user may read its memory and attach to it. Set-id executables start
37    /// non-dumpable; the kernel then makes the files under `/proc/<pid>` root's.
38    pub dumpable: bool,
39}

41/// A client binary root registered as trusted.
42#[derive(Debug, Clone, PartialEq, Eq)]
43pub struct TrustedExe {
44    pub path: PathBuf,
45    /// The SHA-256 of the binary, when it was registered with `snitchhikerd trust-client`: a binary changed in place
46    /// is then refused until it is registered again.
47    pub sha256: Option<[u8; 32]>,
48}

50/// What a peer must be to be a client.
51#[derive(Debug, Clone)]
52pub struct ClientPolicy {
53    /// The group the client binaries are setgid to; users are not members of it.
54    pub group: u32,
55    pub trusted: Vec<TrustedExe>,
56}

58#[derive(Debug, thiserror::Error, PartialEq, Eq)]
59pub enum Refusal {
60    #[error("the process doesn't have the client group")]
61    NotInGroup,
62    #[error("{0} is not a trusted client binary")]
63    UntrustedExe(PathBuf),
64    #[error("{0}: {1}")]
65    UnsafeExe(PathBuf, &'static str),
66    #[error("the process is being traced by {0}")]
67    Traced(u32),
68    #[error("other processes may read the process's memory (it isn't running set-id)")]
69    Dumpable,
70    #[error("{0} changed since it was registered (snitchhikerd trust-client registers it again)")]
71    Changed(PathBuf),
72}

74#[derive(Debug, thiserror::Error)]
75pub enum Error {
76    #[error("the kernel can't tell the peer's process (SO_PEERPIDFD needs Linux 6.5): {0}")]
77    NoPidfd(io::Error),
78    #[error("the peer exited")]
79    Exited,
80    #[error("reading /proc/{pid}: {source}")]
81    Proc { pid: u32, source: io::Error },
82}

84/// Identifies the process connected to `stream`. The process is held by a pidfd, and checked to still be alive after
85/// everything was read, so its pid can't have been reused by another process in between. The pidfd is returned: it
86/// tells when that process exits.
87pub fn identify(stream: &UnixStream) -> Result<(Peer, OwnedFd), Error> {
88    let pidfd = getsockopt(stream, sockopt::PeerPidfd).map_err(|e| Error::NoPidfd(e.into()))?;
89    let pid = pidfd_pid(&pidfd)?;
90    let proc = |path: &str| -> Result<String, Error> {
91        fs::read_to_string(format!("/proc/{pid}/{path}"))
92            .map_err(|source| Error::Proc { pid, source })
93    };
94    let status = proc("status")?;
95    let ids = |field: &str| -> Vec<u32> {
96        status
97            .lines()
98            .find_map(|line| line.strip_prefix(field))
99            .map_or_else(Vec::new, |rest| {
100                rest.split_whitespace()
101                    .filter_map(|n| n.parse().ok())
102                    .collect()
103            })
104    };
105    let uids = ids("Uid:");
106    let gids = ids("Gid:");
107    let exe_path = format!("/proc/{pid}/exe");
108    let exe = fs::read_link(&exe_path).map_err(|source| Error::Proc { pid, source })?;
109    let exe_meta = fs::metadata(&exe_path).map_err(|source| Error::Proc { pid, source })?;
110    // For a non-dumpable process the kernel makes the files under /proc/<pid> root's (the directory keeps the
111    // process's owner).
112    let proc_owner = fs::metadata(format!("/proc/{pid}/mem"))
113        .map_err(|source| Error::Proc { pid, source })?
114        .uid();
115    let uid = uids.first().copied().unwrap_or(u32::MAX);
116    let peer = Peer {
117        pid,
118        uid,
119        euid: uids.get(1).copied().unwrap_or(u32::MAX),
120        gid: gids.first().copied().unwrap_or(u32::MAX),
121        egid: gids.get(1).copied().unwrap_or(u32::MAX),
122        groups: ids("Groups:"),
123        exe,
124        exe_id: (exe_meta.dev(), exe_meta.ino()),
125        tracer: ids("TracerPid:").first().copied().unwrap_or(0),
126        dumpable: uid == 0 || proc_owner != 0,
127    };
128    if exited(&pidfd) {
129        return Err(Error::Exited);
130    }
131    Ok((peer, pidfd))
132}

134/// Ends `stream` when the process of `pidfd` exits, even if another process holds the connection (it was passed on, or
135/// inherited): a client's rights are its process's, not its socket's. Dropping the returned guard stops watching.
136pub fn end_with_process(pidfd: OwnedFd, stream: &UnixStream) -> io::Result<io::PipeWriter> {
137    let stream = stream.try_clone()?;
138    let (stop, guard) = io::pipe()?;
139    std::thread::Builder::new()
140        .name("api-client-watch".to_owned())
141        .spawn(move || {
142            let mut fds = [
143                PollFd::new(pidfd.as_fd(), PollFlags::POLLIN),
144                PollFd::new(stop.as_fd(), PollFlags::POLLIN),
145            ];
146            loop {
147                match poll(&mut fds, PollTimeout::NONE) {
148                    Err(nix::errno::Errno::EINTR) => {}
149                    Err(_) => return,
150                    Ok(_) => break,
151                }
152            }
153            if exited(&pidfd) {
154                let _ = stream.shutdown(std::net::Shutdown::Both);
155            }
156        })?;
157    Ok(guard)
158}

160/// The pid a pidfd refers to, from `/proc/self/fdinfo`.
161fn pidfd_pid(pidfd: &OwnedFd) -> Result<u32, Error> {
162    let fd = pidfd.as_raw_fd();
163    let info =
164        fs::read_to_string(format!("/proc/self/fdinfo/{fd}")).map_err(|source| Error::Proc {
165            pid: std::process::id(),
166            source,
167        })?;
168    let pid: i64 = info
169        .lines()
170        .find_map(|line| line.strip_prefix("Pid:"))
171        .and_then(|n| n.trim().parse().ok())
172        .unwrap_or(-1);
173    u32::try_from(pid).map_err(|_| Error::Exited)
174}

176/// A pidfd becomes readable when its process exits.
177fn exited(pidfd: &OwnedFd) -> bool {
178    let mut fds = [PollFd::new(pidfd.as_fd(), PollFlags::POLLIN)];
179    matches!(poll(&mut fds, PollTimeout::ZERO), Ok(n) if n > 0)
180}

182/// Whether `peer` is one of the trusted client binaries, running as it should.
183pub fn verify(peer: &Peer, policy: &ClientPolicy) -> Result<(), Refusal> {
184    if peer.egid != policy.group && !peer.groups.contains(&policy.group) {
185        return Err(Refusal::NotInGroup);
186    }
187    let trusted = policy
188        .trusted
189        .iter()
190        .find(|trusted| {
191            fs::metadata(&trusted.path).is_ok_and(|meta| (meta.dev(), meta.ino()) == peer.exe_id)
192        })
193        .ok_or_else(|| Refusal::UntrustedExe(peer.exe.clone()))?;
194    root_owned_and_unwritable(&trusted.path)?;
195    if let Some(expected) = trusted.sha256
196        && !digest_is(&trusted.path, peer.exe_id, &expected)
197    {
198        return Err(Refusal::Changed(trusted.path.clone()));
199    }
200    if peer.tracer != 0 {
201        return Err(Refusal::Traced(peer.tracer));
202    }
203    if peer.dumpable {
204        return Err(Refusal::Dumpable);
205    }
206    Ok(())
207}

209/// What identifies a file's contents without reading them: an inode whose size, modification and change times are the
210/// same. Only the kernel sets the change time, so a write can't hide behind it.
211type ContentKey = (u64, u64, u64, i64, i64, i64, i64);
212
213/// Digests of client binaries, so a client's binary is read once, not on every connection.
214static DIGESTS: Mutex<Vec<(ContentKey, [u8; 32])>> = Mutex::new(Vec::new());
215/// Digests remembered; a few binaries are clients, more is churn from upgrades.
216const DIGESTS_KEPT: usize = 16;
217
218/// Whether the file at `path`, which must be the inode `exe_id` (the one the process runs), has the digest `expected`.
219fn digest_is(path: &Path, exe_id: (u64, u64), expected: &[u8; 32]) -> bool {
220    let Ok(mut file) = fs::File::open(path) else {
221        return false;
222    };
223    let Ok(meta) = file.metadata() else {
224        return false;
225    };
226    if (meta.dev(), meta.ino()) != exe_id {
227        return false;
228    }
229    let key = (
230        meta.dev(),
231        meta.ino(),
232        meta.size(),
233        meta.mtime(),
234        meta.mtime_nsec(),
235        meta.ctime(),
236        meta.ctime_nsec(),
237    );
238    let mut digests = DIGESTS.lock().unwrap_or_else(PoisonError::into_inner);
239    if let Some((_, digest)) = digests.iter().find(|(k, _)| *k == key) {
240        return digest == expected;
241    }
242    let Ok(digest) = sha256(&mut file) else {
243        return false;
244    };
245    if digests.len() >= DIGESTS_KEPT {
246        digests.remove(0);
247    }
248    digests.push((key, digest));
249    digest == *expected
250}

252/// The SHA-256 of what `input` holds.
253pub fn sha256(input: &mut impl io::Read) -> io::Result<[u8; 32]> {
254    use sha2::Digest as _;
255    let mut hasher = sha2::Sha256::new();
256    let mut buffer = vec![0_u8; 1 << 16];
257    loop {
258        let read = input.read(&mut buffer)?;
259        if read == 0 {
260            break;
261        }
262        hasher.update(buffer.get(..read).unwrap_or_default());
263    }
264    Ok(hasher.finalize().into())
265}

267/// The file and every directory above it belong to root and only root may write to them, so no other user can
268/// replace the binary.
269pub fn root_owned_and_unwritable(path: &Path) -> Result<(), Refusal> {
270    const GROUP_OR_OTHER_WRITE: u32 = 0o022;
271    for ancestor in path.ancestors() {
272        if ancestor.as_os_str().is_empty() {
273            continue;
274        }
275        let meta = fs::symlink_metadata(ancestor)
276            .map_err(|_| Refusal::UnsafeExe(path.to_owned(), "can't be inspected"))?;
277        if meta.uid() != 0 {
278            return Err(Refusal::UnsafeExe(
279                ancestor.to_owned(),
280                "is not owned by root",
281            ));
282        }
283        if meta.mode() & GROUP_OR_OTHER_WRITE != 0 && !meta.file_type().is_symlink() {
284            // The sticky bit doesn't make a directory safe here: its owner (root) is fine, but a world-writable
285            // directory could hold a replacement we would then be told about by path.
286            return Err(Refusal::UnsafeExe(
287                ancestor.to_owned(),
288                "is writable by others than root",
289            ));
290        }
291    }
292    Ok(())
293}
