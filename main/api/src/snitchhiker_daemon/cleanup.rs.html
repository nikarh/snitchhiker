1//! `snitchhikerd cleanup`: what happens to the daemon's tables after it stopped (the service runs it after every
2//! stop, `ExecStopPost`, with systemd's `SERVICE_RESULT` saying how the stop went).
3//!
4//! - With `fail = "open"`, run by hand, or with a configuration that doesn't read: they are removed.
5//! - With `fail = "closed"`, after a clean stop: they stay (the daemon kept them; nothing gets out until it is back).
6//! - With `fail = "closed"`, after a crash: they stay for a restart, unless the daemon crashed [`CRASHES`] times in
7//!   [`WINDOW`]: a daemon that can't stay up mustn't keep the machine offline. Then they are removed after each crash
8//!   until the crashes are that far apart again.
9
10use std::path::{Path, PathBuf};
11use std::time::Duration;
12
13use crate::config::{Config, Fail};
14
15/// Crashes in [`WINDOW`] after which the tables are removed even with `fail = "closed"`.
16pub const CRASHES: usize = 3;
17/// How far back crashes count.
18pub const WINDOW: Duration = Duration::from_secs(300);
19
20/// What to do with the tables.
21#[derive(Debug, Clone, Copy, PartialEq, Eq)]
22pub enum Outcome {
23    Remove,
24    Keep,
25}
26
27/// Decides, given the configuration (`None`: it doesn't read), systemd's `SERVICE_RESULT` (`None`: run by hand), the
28/// times of the crashes before (seconds), and now. Returns what to do and the crashes to remember.
29#[must_use]
30pub fn decide(
31    fail: Option<Fail>,
32    result: Option<&str>,
33    mut crashes: Vec<u64>,
34    now: u64,
35) -> (Outcome, Vec<u64>) {
36    let (Some(Fail::Closed), Some(result)) = (fail, result) else {
37        return (Outcome::Remove, Vec::new());
38    };
39    if result == "success" {
40        return (Outcome::Keep, Vec::new());
41    }
42    crashes.retain(|&at| at <= now && now - at < WINDOW.as_secs());
43    crashes.push(now);
44    let outcome = if crashes.len() >= CRASHES {
45        Outcome::Remove
46    } else {
47        Outcome::Keep
48    };
49    (outcome, crashes)
50}
51
52/// Where the crashes are remembered: next to the database.
53fn record(config: &Config) -> PathBuf {
54    config
55        .database
56        .parent()
57        .unwrap_or_else(|| Path::new("/"))
58        .join("crashes")
59}
60
61fn read(path: &Path) -> Vec<u64> {
62    std::fs::read_to_string(path)
63        .unwrap_or_default()
64        .lines()
65        .filter_map(|line| line.trim().parse().ok())
66        .collect()
67}
68
69fn write(path: &Path, crashes: &[u64]) {
70    let result = if crashes.is_empty() {
71        match std::fs::remove_file(path) {
72            Err(e) if e.kind() == std::io::ErrorKind::NotFound => Ok(()),
73            other => other,
74        }
75    } else {
76        let mut text = String::new();
77        for at in crashes {
78            text.push_str(&at.to_string());
79            text.push('\n');
80        }
81        std::fs::write(path, text)
82    };
83    if let Err(error) = result {
84        tracing::warn!(%error, path = %path.display(), "remembering the daemon's crashes");
85    }
86}

88/// Runs the decision for the configuration at `config_path`; true when nothing failed.
89#[must_use]
90pub fn run(config_path: &Path) -> bool {
91    let config = match Config::load(config_path) {
92        Ok(config) => Some(config),
93        Err(error) => {
94            tracing::warn!(%error, "reading the configuration: removing the tables");
95            None
96        }
97    };
98    let result = std::env::var("SERVICE_RESULT").ok();
99    let now = std::time::SystemTime::now()
100        .duration_since(std::time::UNIX_EPOCH)
101        .map_or(0, |d| d.as_secs());
102    let path = config.as_ref().map(record);
103    let before = path.as_deref().map_or_else(Vec::new, read);
104    let (outcome, crashes) = decide(
105        config.as_ref().map(|c| c.fail),
106        result.as_deref(),
107        before,
108        now,
109    );
110    if let Some(path) = &path {
111        write(path, &crashes);
112    }
113    match outcome {
114        Outcome::Keep => {
115            tracing::info!(
116                result = result.as_deref().unwrap_or_default(),
117                crashes = crashes.len(),
118                "fail = closed: the tables stay until the daemon is back"
119            );
120            true
121        }
122        Outcome::Remove => {
123            if crashes.len() >= CRASHES {
124                tracing::warn!(
125                    crashes = crashes.len(),
126                    "the daemon keeps crashing: removing its tables despite fail = closed, so the machine isn't cut off"
127                );
128            }
129            crate::runtime::remove_tables(&snitchhiker_netfilter::nftables::Table::default().name)
130        }
131    }
132}

134#[cfg(test)]
135mod tests {
136    use super::*;
137
138    #[test]
139    fn open_by_hand_or_unreadable_removes() {
140        let removes = |fail, result| decide(fail, result, vec![], 1000).0 == Outcome::Remove;
141        assert!(removes(Some(Fail::Open), Some("success")));
142        assert!(removes(Some(Fail::Open), Some("signal")));
143        assert!(removes(Some(Fail::Closed), None));
144        assert!(removes(None, Some("success")));
145    }
146
147    #[test]
148    fn closed_keeps_after_a_clean_stop_and_forgets_crashes() {
149        assert_eq!(
150            decide(Some(Fail::Closed), Some("success"), vec![990, 995], 1000),
151            (Outcome::Keep, vec![])
152        );
153    }
154
155    #[test]
156    fn closed_removes_once_crashes_come_too_often() {
157        let closed = Some(Fail::Closed);
158        let (first, crashes) = decide(closed, Some("signal"), vec![], 1000);
159        let (second, crashes) = decide(closed, Some("core-dump"), crashes, 1010);
160        let (third, crashes) = decide(closed, Some("exit-code"), crashes, 1020);
161        assert_eq!(
162            (first, second, third),
163            (Outcome::Keep, Outcome::Keep, Outcome::Remove)
164        );
165        // The next crash in the window removes them too.
166        assert_eq!(
167            decide(closed, Some("signal"), crashes, 1030).0,
168            Outcome::Remove
169        );
170    }
171
172    #[test]
173    fn old_crashes_dont_count() {
174        let (outcome, crashes) = decide(Some(Fail::Closed), Some("signal"), vec![100, 200], 1000);
175        assert_eq!((outcome, crashes), (Outcome::Keep, vec![1000]));
176    }
177}
