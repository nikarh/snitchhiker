19/// Makes the kernel's table what `firewall` says: installed when it is on, removed when it is off.
20pub fn apply(firewall: &Firewall, config: &Config) -> std::io::Result<()> {
21    let mut socket = Socket::open()?;
22    let mut writer = Writer::new();
23    let batch = if firewall.enabled {
24        encode_system(&mut writer, &table(firewall, config), 1)
25    } else {
26        encode_remove(&mut writer, TABLE, 1)
27    };
28    socket.transact(&writer, batch.first, batch.last)
29}
30
31/// Removes the table (the daemon stops with `fail = open`).
32pub fn remove() -> std::io::Result<()> {
33    let mut socket = Socket::open()?;
34    let mut writer = Writer::new();
35    let batch = encode_remove(&mut writer, TABLE, 1);
36    socket.transact(&writer, batch.first, batch.last)
37}

39fn prefix(net: &IpNet) -> Prefix {
40    Prefix {
41        address: match net.addr() {
42            std::net::IpAddr::V4(a) => a.octets().to_vec(),
43            std::net::IpAddr::V6(a) => a.octets().to_vec(),
44        },
45        length: net.prefix_len(),
46    }
47}
48
49fn interface(name: Option<&String>) -> Option<Interface> {
50    name.map(|name| Interface {
51        name: name.trim_end_matches('*').to_owned(),
52        prefix: name.ends_with('*'),
53    })
54}

56/// The kernel's rules for the firewall: those a drop policy implies, then each rule that is on, once for each pair of
57/// source and destination of one family and each ICMP type.
58#[must_use]
59pub fn table(firewall: &Firewall, config: &Config) -> SystemTable {
60    let mut rules = Vec::new();
61    for rule in system::implied(firewall)
62        .iter()
63        .chain(firewall.rules.iter().filter(|r| r.enabled))
64    {
65        let sources: Vec<Option<&IpNet>> = if rule.source.is_empty() {
66            vec![None]
67        } else {
68            rule.source.iter().map(Some).collect()
69        };
70        let destinations: Vec<Option<&IpNet>> = if rule.destination.is_empty() {
71            vec![None]
72        } else {
73            rule.destination.iter().map(Some).collect()
74        };
75        let v6 = |net: Option<&IpNet>| net.map(|n| matches!(n, IpNet::V6(_)));
76        let icmp: Vec<Option<u8>> = if rule.icmp_types.is_empty() {
77            vec![None]
78        } else {
79            rule.icmp_types
80                .iter()
81                .map(|t| Some(t.number(rule.protocol == Some(ICMPV6))))
82                .collect()
83        };
84        for source in &sources {
85            for destination in &destinations {
86                if let (Some(a), Some(b)) = (v6(*source), v6(*destination))
87                    && a != b
88                {
89                    continue;
90                }
91                for icmp_type in &icmp {
92                    rules.push(SystemRule {
93                        chain: match rule.chain {
94                            Chain::Input => SystemChain::Input,
95                            Chain::Forward => SystemChain::Forward,
96                            Chain::Output => SystemChain::Output,
97                        },
98                        protocol: rule.protocol,
99                        source: source.map(prefix),
100                        destination: destination.map(prefix),
101                        source_ports: rule.source_ports,
102                        destination_ports: rule.destination_ports,
103                        in_interface: interface(rule.in_interface.as_ref()),
104                        out_interface: interface(rule.out_interface.as_ref()),
105                        ct_state: rule.ct_state.iter().fold(0, |bits, s| bits | s.bit()),
106                        icmp_type: *icmp_type,
107                        target: match rule.target {
108                            Target::Accept => SystemTarget::Accept,
109                            Target::Drop => SystemTarget::Drop,
110                            Target::Reject => SystemTarget::Reject,
111                            Target::Log => SystemTarget::Log(format!(
112                                "snitchhiker {}: ",
113                                if rule.description.is_empty() {
114                                    "system"
115                                } else {
116                                    &rule.description
117                                }
118                                .chars()
119                                .take(100)
120                                .collect::<String>()
121                            )),
122                            Target::Queue => SystemTarget::Queue {
123                                number: config.queue,
124                                total: config.queues,
125                                bypass: config.fail == Fail::Open,
126                            },
127                        },
128                    });
129                }
130            }
131        }
132    }
133    SystemTable {
134        name: TABLE.to_owned(),
135        input_drop: firewall.input == ChainPolicy::Drop,
136        forward_drop: firewall.forward == ChainPolicy::Drop,
137        rules,
138    }
139}

152/// Whether going from `before` to `after` lets in what was kept out: a dangerous action.
153#[must_use]
154pub fn loosened(before: &Firewall, after: &Firewall) -> bool {
155    (before.enabled && !after.enabled)
156        || (before.enabled
157            && before.input == ChainPolicy::Drop
158            && after.input == ChainPolicy::Accept)
159}

161/// How a change reads in the history.
162#[must_use]
163pub fn describe(before: &Firewall, after: &Firewall) -> String {
164    let mut parts = Vec::new();
165    if before.enabled != after.enabled {
166        parts.push(
167            if after.enabled {
168                "turned on"
169            } else {
170                "turned off"
171            }
172            .to_owned(),
173        );
174    }
175    let name = |p: ChainPolicy| match p {
176        ChainPolicy::Accept => "accept",
177        ChainPolicy::Drop => "drop",
178    };
179    if before.input != after.input {
180        parts.push(format!(
181            "input {} → {}",
182            name(before.input),
183            name(after.input)
184        ));
185    }
186    if before.forward != after.forward {
187        parts.push(format!(
188            "forward {} → {}",
189            name(before.forward),
190            name(after.forward)
191        ));
192    }
193    if before.rules != after.rules {
194        let on = after.rules.iter().filter(|r| r.enabled).count();
195        let mut text = format!("{} rules", after.rules.len());
196        let _ = write!(text, " ({on} on)");
197        parts.push(text);
198    }
199    if parts.is_empty() {
200        "system firewall unchanged".to_owned()
201    } else {
202        format!("system firewall: {}", parts.join(", "))
203    }
204}

206#[cfg(test)]
207mod tests {
208    use super::*;
209    use snitchhiker_core::system::Rule;

211    #[test]
212    fn the_wire_form_round_trips() {
213        let firewall = system::defaults();
214        assert_eq!(from_wire(&wire(&firewall, 0)).unwrap(), firewall);
215        let mut bad = wire(&firewall, 0);
216        bad.rules[2].destination_ports = "http".to_owned();
217        assert!(from_wire(&bad).unwrap_err().contains("rule 3"));
218    }
219
220    #[test]
221    fn rules_expand_per_family_and_icmp_type() {
222        let firewall = Firewall {
223            enabled: true,
224            input: ChainPolicy::Drop,
225            ..system::defaults()
226        };
227        let table = table(&firewall, &Config::default());
228        assert!(table.input_drop);
229        // Three implied, then three ICMP types for each of ICMP and ICMPv6.
230        assert_eq!(table.rules.len(), 3 + 3 + 3);
231        let mixed = Firewall {
232            rules: vec![Rule {
233                source: vec![
234                    "10.0.0.0/8".parse().unwrap(),
235                    "2001:db8::/32".parse().unwrap(),
236                ],
237                destination: vec!["192.168.1.1/32".parse().unwrap()],
238                ..Rule::default()
239            }],
240            ..firewall
241        };
242        // Only the IPv4 source pairs with the IPv4 destination.
243        assert_eq!(table_rules(&mixed), 3 + 1);
244    }
245
246    fn table_rules(firewall: &Firewall) -> usize {
247        table(firewall, &Config::default()).rules.len()
248    }
249
250    #[test]
251    fn turning_it_off_or_opening_input_is_loosening() {
252        let on = Firewall {
253            enabled: true,
254            input: ChainPolicy::Drop,
255            ..system::defaults()
256        };
257        assert!(loosened(
258            &on,
259            &Firewall {
260                enabled: false,
261                ..on.clone()
262            }
263        ));
264        assert!(loosened(
265            &on,
266            &Firewall {
267                input: ChainPolicy::Accept,
268                ..on.clone()
269            }
270        ));
271        assert!(!loosened(&system::defaults(), &on));
272        assert_eq!(
273            describe(&system::defaults(), &on),
274            "system firewall: turned on, input accept → drop"
275        );
276    }
