1//! Turning what the daemon saw (connections no rule decided) into rules to review: one rule per program,
2//! destination and port, where it came from, with the program's path generalised so updates don't break it. A
3//! container's connections get rules for that container (its Compose service, else its name, else its image), a VM's
4//! for that VM, a named namespace's for it: a rule learned from a container never applies to the host.
5
6use std::collections::{BTreeMap, HashMap};
7
8use snitchhiker_core::generalize;
9use snitchhiker_core::pattern::PatternSource;
10use snitchhiker_core::rule::Action;
11use snitchhiker_core::spec::{Conditions, FolderSpec, RuleFile, RuleSpec};
12
13use crate::store::Undecided;

15/// The rules that would have decided `seen`, in a folder named `folder`. `home_of` gives a user's home directory.
16pub fn suggest(
17    seen: &[Undecided],
18    folder: &str,
19    home_of: impl Fn(u32) -> Option<String>,
20) -> RuleFile {
21    let mut homes: HashMap<u32, Option<String>> = HashMap::new();
22    // (where, exe pattern, destination, port, protocol) -> how many connections; ordered for a stable file.
23    let mut groups: BTreeMap<(Where, String, String, bool, u16, u8), usize> = BTreeMap::new();
24    for connection in seen {
25        let place = Where::of(connection);
26        // A program the node couldn't know: on the host no rule could name it; elsewhere (a VM's traffic, a rootless
27        // container's first connection) the rule names the place alone.
28        if connection.port == 0 || (connection.exe.is_empty() && place == Where::Host) {
29            continue;
30        }
31        let home = homes
32            .entry(connection.uid)
33            .or_insert_with(|| home_of(connection.uid));
34        let exe = match generalize::executable(&connection.exe, home.as_deref()) {
35            _ if connection.exe.is_empty() => String::new(),
36            PatternSource::Glob(glob) => format!("glob:{glob}"),
37            PatternSource::Exact(exact) => exact,
38            PatternSource::Regex(regex) => format!("re:{regex}"),
39        };
40        let (destination, is_host) = connection.host.clone().map_or_else(
41            || (connection.address.to_string(), false),
42            |host| (host, true),
43        );
44        *groups
45            .entry((
46                place,
47                exe,
48                destination,
49                is_host,
50                connection.port,
51                connection.protocol,
52            ))
53            .or_default() += 1;
54    }
55    let rule = groups
56        .into_iter()
57        .map(
58            |((place, exe, destination, is_host, port, protocol), count)| {
59                let program = match exe.rsplit('/').next() {
60                    Some(name) if !name.is_empty() => name.to_owned(),
61                    _ => "anything".to_owned(),
62                };
63                let mut conditions = Conditions {
64                    exe: if exe.is_empty() { vec![] } else { vec![exe] },
65                    host: if is_host {
66                        vec![destination.clone()]
67                    } else {
68                        vec![]
69                    },
70                    net: if is_host {
71                        vec![]
72                    } else {
73                        vec![destination.clone()]
74                    },
75                    port: vec![port.to_string()],
76                    protocol: vec![match protocol {
77                        6 => "tcp".to_owned(),
78                        17 => "udp".to_owned(),
79                        other => other.to_string(),
80                    }],
81                    ..Conditions::default()
82                };
83                let within = place.apply(&mut conditions);
84                RuleSpec {
85                    name: format!("{within}{program} to {destination}:{port}"),
86                    action: Action::Allow,
87                    tier: snitchhiker_core::rule::Tier::Normal,
88                    enabled: true,
89                    notes: format!("Learned from {count} connection(s)."),
90                    conditions,
91                }
92            },
93        )
94        .collect();
95    RuleFile {
96        folder: Some(FolderSpec {
97            name: folder.to_owned(),
98            scope: Conditions::default(),
99        }),
100        rule,
101    }
102}

104/// Where connections came from, as rules name it.
105#[derive(Debug, Clone, PartialEq, Eq, PartialOrd, Ord)]
106enum Where {
107    Host,
108    /// A container named by `container.compose`, `.name` or `.image`: (which, value).
109    Container(&'static str, String),
110    /// A container nothing names (its identity couldn't be read): any container.
111    AnyContainer,
112    Vm(String),
113    Netns(String),
114    /// Another context with no name.
115    Unnamed(u8),
116}
117
118impl Where {
119    fn of(connection: &Undecided) -> Self {
120        let identity = connection
121            .container
122            .as_deref()
123            .map(snitchhiker_core::context::Container::from_key);
124        match (connection.context, identity) {
125            (0, _) => Self::Host,
126            (1, Some(c)) => {
127                if let Some(compose) = c.compose {
128                    Self::Container("compose", compose)
129                } else if !c.name.is_empty() {
130                    Self::Container("name", c.name)
131                } else if !c.image.is_empty() {
132                    Self::Container("image", c.image)
133                } else {
134                    Self::AnyContainer
135                }
136            }
137            (1, None) => Self::AnyContainer,
138            (2, Some(c)) if !c.name.is_empty() => Self::Vm(c.name),
139            (3, Some(c)) if c.runtime == "netns" && !c.name.is_empty() => Self::Netns(c.name),
140            (other, _) => Self::Unnamed(other),
141        }
142    }
143
144    /// Limits `conditions` to this place; the prefix of the rule's name.
145    fn apply(&self, conditions: &mut Conditions) -> String {
146        use snitchhiker_core::rule::ContextKind;
147        match self {
148            Self::Host => String::new(),
149            Self::Container(which, value) => {
150                let field = match *which {
151                    "compose" => &mut conditions.container.compose,
152                    "name" => &mut conditions.container.name,
153                    _ => &mut conditions.container.image,
154                };
155                *field = vec![value.clone()];
156                format!("in {value}: ")
157            }
158            Self::AnyContainer => {
159                conditions.context = Some(ContextKind::Container);
160                "in a container: ".to_owned()
161            }
162            Self::Vm(name) => {
163                conditions.vm.name = vec![name.clone()];
164                format!("in {name}: ")
165            }
166            Self::Netns(name) => {
167                conditions.netns = vec![name.clone()];
168                format!("in {name}: ")
169            }
170            Self::Unnamed(code) => {
171                conditions.context = Some(match code {
172                    2 => ContextKind::Vm,
173                    3 => ContextKind::Netns,
174                    _ => ContextKind::Container,
175                });
176                "elsewhere: ".to_owned()
177            }
178        }
179    }
180}

182#[cfg(test)]
183mod tests {
184    use super::*;
185
186    fn seen(exe: &str, host: Option<&str>, address: &str, port: u16) -> Undecided {
187        Undecided {
188            exe: exe.to_owned(),
189            host: host.map(str::to_owned),
190            uid: 1000,
191            protocol: 6,
192            address: address.parse().unwrap(),
193            port,
194            context: 0,
195            container: None,
196        }
197    }
198
199    #[test]
200    fn containers_vms_and_namespaces_get_rules_of_their_own() {
201        let mut web = seen(
202            "/usr/local/bin/node",
203            Some("api.example.com"),
204            "198.51.100.7",
205            443,
206        );
207        web.context = 1;
208        web.container = Some("docker\tabc\tshop-web-1\tnode:22\tshop/web\t".to_owned());
209        let mut vm = seen("", Some("deb.debian.org"), "198.51.100.8", 80);
210        vm.context = 2;
211        vm.exe = String::new();
212        vm.container = Some("qemu\tuuid\tbuilder\t\t\t".to_owned());
213        let mut ns = seen("/usr/bin/curl", None, "203.0.113.9", 443);
214        ns.context = 3;
215        ns.container = Some("netns\t\tvpn\t\t\t".to_owned());
216        let host = seen(
217            "/usr/local/bin/node",
218            Some("api.example.com"),
219            "198.51.100.7",
220            443,
221        );
222        let file = suggest(&[web, vm, ns, host], "Learned", |_| None);
223        let names: Vec<&str> = file.rule.iter().map(|r| r.name.as_str()).collect();
224        assert_eq!(names.len(), 4, "{names:?}");
225        let by_name = |prefix: &str| {
226            file.rule
227                .iter()
228                .find(|r| r.name.starts_with(prefix))
229                .unwrap()
230        };
231        assert_eq!(
232            by_name("in shop/web").conditions.container.compose,
233            vec!["shop/web"]
234        );
235        assert_eq!(by_name("in builder").conditions.vm.name, vec!["builder"]);
236        assert_eq!(by_name("in builder").conditions.exe, Vec::<String>::new());
237        assert_eq!(by_name("in vpn").conditions.netns, vec!["vpn"]);
238        let on_host = by_name("node to");
239        assert!(
240            on_host.conditions.container.compose.is_empty() && on_host.conditions.context.is_none()
241        );
242    }

244    #[test]
245    fn one_rule_per_program_destination_and_port() {
246        let file = suggest(
247            &[
248                seen(
249                    "/home/ada/.local/share/claude/versions/2.0.31/claude",
250                    Some("api.example.com"),
251                    "198.51.100.7",
252                    443,
253                ),
254                seen(
255                    "/home/ada/.local/share/claude/versions/2.0.32/claude",
256                    Some("api.example.com"),
257                    "198.51.100.8",
258                    443,
259                ),
260                seen("/usr/bin/curl", None, "203.0.113.9", 80),
261                seen("", None, "203.0.113.9", 80),
262            ],
263            "Learned",
264            |_| Some("/home/ada".to_owned()),
265        );
266        assert_eq!(file.rule.len(), 2);
267        let claude = file
268            .rule
269            .iter()
270            .find(|r| r.name.starts_with("claude"))
271            .unwrap();
272        assert_eq!(
273            claude.conditions.exe,
274            vec!["glob:{home}/.local/share/claude/versions/*/claude"]
275        );
276        assert_eq!(claude.conditions.host, vec!["api.example.com"]);
277        assert_eq!(claude.notes, "Learned from 2 connection(s).");
278        let curl = file
279            .rule
280            .iter()
281            .find(|r| r.name.starts_with("curl"))
282            .unwrap();
283        assert_eq!(curl.conditions.net, vec!["203.0.113.9"]);
284        // The suggestions are valid rules.
285        for spec in &file.rule {
286            spec.to_rule(1, None).unwrap();
287        }
288    }
