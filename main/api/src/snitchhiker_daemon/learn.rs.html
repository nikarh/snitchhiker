1//! Turning what the daemon saw (connections no rule decided) into rules to review: one rule per program,
2//! destination and port, with the program's path generalised so updates don't break it.
3
4use std::collections::{BTreeMap, HashMap};
5
6use snitchhiker_core::generalize;
7use snitchhiker_core::pattern::PatternSource;
8use snitchhiker_core::rule::Action;
9use snitchhiker_core::spec::{Conditions, FolderSpec, RuleFile, RuleSpec};
10
11use crate::store::Undecided;
12
13/// The rules that would have decided `seen`, in a folder named `folder`. `home_of` gives a user's home directory.
14pub fn suggest(
15    seen: &[Undecided],
16    folder: &str,
17    home_of: impl Fn(u32) -> Option<String>,
18) -> RuleFile {
19    let mut homes: HashMap<u32, Option<String>> = HashMap::new();
20    // (exe pattern, destination, port, protocol) -> how many connections; ordered for a stable file.
21    let mut groups: BTreeMap<(String, String, bool, u16, u8), usize> = BTreeMap::new();
22    for connection in seen {
23        if connection.exe.is_empty() || connection.port == 0 {
24            continue;
25        }
26        let home = homes
27            .entry(connection.uid)
28            .or_insert_with(|| home_of(connection.uid));
29        let exe = match generalize::executable(&connection.exe, home.as_deref()) {
30            PatternSource::Glob(glob) => format!("glob:{glob}"),
31            PatternSource::Exact(exact) => exact,
32            PatternSource::Regex(regex) => format!("re:{regex}"),
33        };
34        let (destination, is_host) = connection.host.clone().map_or_else(
35            || (connection.address.to_string(), false),
36            |host| (host, true),
37        );
38        *groups
39            .entry((
40                exe,
41                destination,
42                is_host,
43                connection.port,
44                connection.protocol,
45            ))
46            .or_default() += 1;
47    }
48    let rule = groups
49        .into_iter()
50        .map(|((exe, destination, is_host, port, protocol), count)| {
51            let program = exe.rsplit('/').next().unwrap_or(&exe).to_owned();
52            RuleSpec {
53                name: format!("{program} to {destination}:{port}"),
54                action: Action::Allow,
55                tier: snitchhiker_core::rule::Tier::Normal,
56                enabled: true,
57                notes: format!("Learned from {count} connection(s)."),
58                conditions: Conditions {
59                    exe: vec![exe],
60                    host: if is_host {
61                        vec![destination.clone()]
62                    } else {
63                        vec![]
64                    },
65                    net: if is_host { vec![] } else { vec![destination] },
66                    port: vec![port.to_string()],
67                    protocol: vec![match protocol {
68                        6 => "tcp".to_owned(),
69                        17 => "udp".to_owned(),
70                        other => other.to_string(),
71                    }],
72                    ..Conditions::default()
73                },
74            }
75        })
76        .collect();
77    RuleFile {
78        folder: Some(FolderSpec {
79            name: folder.to_owned(),
80            scope: Conditions::default(),
81        }),
82        rule,
83    }
84}

86#[cfg(test)]
87mod tests {
88    use super::*;
89
90    fn seen(exe: &str, host: Option<&str>, address: &str, port: u16) -> Undecided {
91        Undecided {
92            exe: exe.to_owned(),
93            host: host.map(str::to_owned),
94            uid: 1000,
95            protocol: 6,
96            address: address.parse().unwrap(),
97            port,
98        }
99    }
100
101    #[test]
102    fn one_rule_per_program_destination_and_port() {
103        let file = suggest(
104            &[
105                seen(
106                    "/home/ada/.local/share/claude/versions/2.0.31/claude",
107                    Some("api.example.com"),
108                    "198.51.100.7",
109                    443,
110                ),
111                seen(
112                    "/home/ada/.local/share/claude/versions/2.0.32/claude",
113                    Some("api.example.com"),
114                    "198.51.100.8",
115                    443,
116                ),
117                seen("/usr/bin/curl", None, "203.0.113.9", 80),
118                seen("", None, "203.0.113.9", 80),
119            ],
120            "Learned",
121            |_| Some("/home/ada".to_owned()),
122        );
123        assert_eq!(file.rule.len(), 2);
124        let claude = file
125            .rule
126            .iter()
127            .find(|r| r.name.starts_with("claude"))
128            .unwrap();
129        assert_eq!(
130            claude.conditions.exe,
131            vec!["glob:{home}/.local/share/claude/versions/*/claude"]
132        );
133        assert_eq!(claude.conditions.host, vec!["api.example.com"]);
134        assert_eq!(claude.notes, "Learned from 2 connection(s).");
135        let curl = file
136            .rule
137            .iter()
138            .find(|r| r.name.starts_with("curl"))
139            .unwrap();
140        assert_eq!(curl.conditions.net, vec!["203.0.113.9"]);
141        // The suggestions are valid rules.
142        for spec in &file.rule {
143            spec.to_rule(1, None).unwrap();
144        }
145    }
146}
