1//! The running daemon: the table, the attribution programs, one deciding thread per queue, and the writer.
2//!
3//! The packet path: receive → parse the headers → look the flow's owner up (eBPF) → the process's facts (cached) →
4//! the rules (an immutable set, swapped atomically on change) → verdict, and an event to the writer's queue. Nothing
5//! on it waits for the database, and a full event queue drops the event (counted), never the verdict.
6

39/// Events waiting for the writer; beyond this they are dropped (and counted).
40const EVENT_QUEUE: usize = 65_536;
41/// Events written per transaction at most.
42const BATCH: usize = 1024;
43/// How long the writer waits to fill a batch.
44const BATCH_WAIT: Duration = Duration::from_millis(100);
45/// How much of a DNS answer is copied: a whole UDP answer, as large as resolvers send without EDNS tricks.
46const DNS_COPY: u32 = 4096;
47const DNS_PORT: u16 = 53;
48/// How often byte counts are written to the history.
49const BYTES_EVERY: Duration = Duration::from_secs(5);
50/// How often old history is pruned, and how many rows per chunk.
51const PRUNE_EVERY: Duration = Duration::from_secs(600);
52const PRUNE_CHUNK: u32 = 10_000;

68/// Why an answer to a prompt was refused.
69#[derive(Debug, thiserror::Error)]
70pub enum AnswerError {
71    #[error("no such prompt (answered already, or timed out)")]
72    NotFound,
73    #[error("{0}")]
74    Invalid(String),
75    #[error(transparent)]
76    Runtime(#[from] Error),
77}

79fn placement_error(error: crate::placement::Error) -> AnswerError {
80    match error {
81        crate::placement::Error::Store(e) => AnswerError::Runtime(Error::from(e)),
82        other => AnswerError::Invalid(other.to_string()),
83    }

125    /// Opens the store, compiles the rules, installs the table, loads the attribution programs and starts the
126    /// deciding threads and the writer.
127    pub fn start(config: Config) -> Result<Arc<Self>, Error> {
128        if let Some(parent) = config.database.parent() {
129            std::fs::create_dir_all(parent)?;
130        }
131        let store = Store::open(&config.database)?;
132        let rules = compile(&config, &store)?;
133        info!(rules = rules.len(), "rules loaded");
134        // The programs first: from the moment the table is in place, every new flow's owner must be known.
135        let attribution = Attribution::load()?;
136        let table = Table {
137            queue: config.queue,
138            queues: config.queues,
139            bypass: config.fail == Fail::Open,
140            dns_queue: config.dns_queue(),
141            ..Table::default()
142        };
143        install(&table)?;
144        info!(queues = config.queues, mode = ?config.mode, "table installed");
145
146        // Until-restart rules from before don't apply any more.
147        let mut store = store;
148        store.delete_until_restart()?;
149        let (sender, receiver) = sync_channel(EVENT_QUEUE);
150        let store = Arc::new(Mutex::new(store));
151        let mut queues = Vec::with_capacity(usize::from(config.queues));
152        let mut writers = Vec::with_capacity(usize::from(config.queues));
153        for offset in 0..config.queues {
154            let queue = Queue::bind(config.queue + offset)?;
155            writers.push(queue.verdict_writer()?);
156            queues.push(queue);
157        }
158        let policy = Effective::of(
159            &config,
160            &store
161                .lock()
162                .unwrap_or_else(PoisonError::into_inner)
163                .policy()?,
164        );
165        let broker = Broker::new(writers, Duration::from_secs(policy.prompt_timeout_secs));
166        let shared = Arc::new(Shared {
167            rules: ArcSwap::from_pointee(rules),
168            attribution,
169            names: RwLock::new(Names::default()),
170            netns: u32::try_from(std::fs::metadata("/proc/self/ns/net")?.ino()).unwrap_or(0),
171            host: HostNamespaces::read()?,
172            events: sender,
173            dropped_events: AtomicU64::new(0),
174            unknown: Arc::from(""),
175            decided: AtomicU64::new(0),
176            hub: Arc::new(Hub::default()),
177            broker,
178            alert_rules_changed: AtomicU64::new(0),
179            policy: ArcSwap::from_pointee(policy),
180            config,
181        });
182        for (offset, queue) in queues.into_iter().enumerate() {
183            let number = queue.number();
184            let shared = Arc::clone(&shared);
185            thread::Builder::new()
186                .name(format!("queue-{number}"))
187                .spawn(move || {
188                    if let Err(error) = decide_loop(&shared, queue, offset) {
189                        error!(%error, queue = number, "deciding thread stopped");
190                    }
191                })?;
192        }
193        if let Some(number) = shared.config.dns_queue() {
194            let queue = Queue::bind_copying(number, DNS_COPY)?;
195            let shared = Arc::clone(&shared);
196            thread::Builder::new()
197                .name("dns".to_owned())
198                .spawn(move || {
199                    if let Err(error) = dns_loop(&shared, queue) {
200                        error!(%error, "learning names stopped");
201                    }
202                })?;
203        }
204        {
205            let store = Arc::clone(&store);
206            let shared = Arc::clone(&shared);
207            thread::Builder::new()
208                .name("writer".to_owned())
209                .spawn(move || write_loop(&shared, &store, &receiver))?;
210        }
211        check_marker(
212            &shared,
213            &store.lock().unwrap_or_else(PoisonError::into_inner),
214        );
215        // The system firewall as last set: a rule the kernel refuses mustn't keep the daemon from deciding.
216        let firewall = store
217            .lock()
218            .unwrap_or_else(PoisonError::into_inner)
219            .system_firewall()?;
220        if let Err(error) = crate::system::apply(&firewall, &shared.config) {
221            error!(%error, "installing the system firewall");
222        }
223        let daemon = Arc::new(Self {
224            shared,
225            store,
226            table,
227            trial: Mutex::new(Trial::default()),
228        });
229        daemon.spawn_prompt_timeouts()?;
230        daemon.spawn_table_watch()?;
231        Ok(daemon)
232    }

234    /// Every few seconds, checks that the tables are still there: one removed by someone else is put back, with an alert.
235    fn spawn_table_watch(self: &Arc<Self>) -> std::io::Result<()> {
236        let daemon = Arc::clone(self);
237        thread::Builder::new()
238            .name("table-watch".to_owned())
239            .spawn(move || {
240                loop {
241                    thread::sleep(TABLE_WATCH_EVERY);
242                    daemon.check_tables();
243                }
244            })?;
245        Ok(())
246    }

248    fn check_tables(&self) {
249        let missing = |name: &str| {
250            let mut writer = Writer::new();
251            let sequence = encode_get_table(&mut writer, name, 1);
252            Socket::open()
253                .and_then(|mut socket| socket.transact(&writer, sequence, sequence))
254                .is_err_and(|e| e.raw_os_error() == Some(2))
255        };
256        if missing(&self.table.name) {
257            let put_back = install(&self.table);
258            self.raise_table_removed("the daemon's", &put_back);
259        }
260        let firewall = self.store().system_firewall();
261        if firewall.as_ref().is_ok_and(|f| f.enabled) && missing(crate::system::TABLE) {
262            let put_back = self
263                .apply_system()
264                .map_err(|e| std::io::Error::other(e.to_string()));
265            self.raise_table_removed("the system firewall's", &put_back);
266        }
267    }

269    fn raise_table_removed(&self, which: &str, put_back: &std::io::Result<()>) {
270        crate::alerts::raise(
271            &self.shared.hub,
272            &self.store(),
273            &crate::alerts::NewAlert {
274                kind: "table_removed",
275                title: format!("Something removed {which} nftables table"),
276                body: match put_back {
277                    Ok(()) => {
278                        "It was put back. Another firewall tool may be flushing the whole ruleset."
279                            .to_owned()
280                    }
281                    Err(error) => format!("Putting it back failed: {error}"),
282                },
283                rule: None,
284            },
285        );

300    /// Prompts nobody answered in time are decided by the policy's `on_timeout`.
301    fn spawn_prompt_timeouts(self: &Arc<Self>) -> std::io::Result<()> {
302        let daemon = Arc::clone(self);
303        thread::Builder::new()
304            .name("prompt-timeouts".to_owned())
305            .spawn(move || {
306                loop {
307                    for pending in daemon.shared.broker.wait_expired() {
308                        let action = daemon.shared.policy.load().on_timeout;
309                        daemon.resolve(&pending, action, Source::Timeout, None, "", true);
310                    }
311                }
312            })?;
313        Ok(())
314    }

327    /// The live TCP connections and connected UDP sockets, each with its program as the attribution programs
328    /// recorded it when it opened (unknown for inbound sockets, and for ones older than the daemon).
329    pub fn connections(&self) -> std::io::Result<Vec<v1::LiveConnection>> {
330        let sockets = SockDiag::open()?.list()?;
331        let bytes: HashMap<u64, (u64, u64)> = self
332            .shared
333            .attribution
334            .socket_bytes()
335            .map(|(cookie, sent, read)| (cookie, (sent, read)))
336            .collect();
337        let mut processes = ProcessCache::new(self.shared.host);
338        let names = self
339            .shared
340            .names
341            .read()
342            .unwrap_or_else(PoisonError::into_inner);
343        let now = now_ms();
344        let plain = |address: std::net::IpAddr| match address {
345            std::net::IpAddr::V6(v6) => v6.to_ipv4_mapped().map_or(address, std::net::IpAddr::V4),
346            v4 @ std::net::IpAddr::V4(_) => v4,
347        };
348        Ok(sockets
349            .iter()
350            .map(|socket| {
351                let remote = plain(socket.id.remote);
352                let key = FlowKey {
353                    netns: self.shared.netns,
354                    protocol: socket.id.protocol,
355                    source_port: socket.id.local_port,
356                    destination: remote,
357                    destination_port: socket.id.remote_port,
358                };
359                let owner = self.shared.attribution.owner(&key);
360                let process = owner.and_then(|o| processes.get(o.pid, o.start_boottime));
361                let pid = owner.map_or(0, |o| o.pid);
362                let (sent, read) = bytes.get(&socket.cookie).copied().unwrap_or_default();
363                v1::LiveConnection {
364                    protocol: u32::from(socket.id.protocol),
365                    local: convert::address(plain(socket.id.local)),
366                    local_port: u32::from(socket.id.local_port),
367                    remote: convert::address(remote),
368                    remote_port: u32::from(socket.id.remote_port),
369                    state: snitchhiker_netfilter::sockdiag::state_name(
370                        socket.id.protocol,
371                        socket.state,
372                    )
373                    .to_owned(),
374                    uid: owner.map_or(socket.uid, |o| o.uid),
375                    pid,
376                    exe: process
377                        .as_ref()
378                        .map_or_else(String::new, |p| p.exe.to_string()),
379                    host: names
380                        .lookup(pid, remote, now)
381                        .first()
382                        .map_or_else(String::new, ToString::to_string),
383                    bytes_out: sent,
384                    bytes_in: read,
385                    context: convert::context(
386                        process.as_ref().map_or(ContextKind::Host, |p| p.context),
387                    )
388                    .into(),
389                }
390            })
391            .collect())
392    }

394    /// Why the recorded connection `seq` is decided as it is, by today's rules; `None` if there is no such event.
395    pub fn explain(&self, seq: i64) -> Result<Option<v1::Explanation>, Error> {
396        let Some(stored) = self.store().event(seq)? else {
397            return Ok(None);
398        };
399        let rules = self.shared.rules.load();
400        let hosts: Vec<Arc<str>> = stored.host.iter().map(|h| Arc::from(h.as_str())).collect();
401        let home = crate::process::home_of(stored.uid);
402        let facts = Facts {
403            exe: &stored.exe,
404            apps: &[],
405            cmdline: "",
406            ancestors: &[],
407            uid: stored.uid,
408            home: home.as_deref(),
409            hosts: &hosts,
410            address: stored.address,
411            port: stored.port,
412            protocol: stored.protocol,
413            context: match stored.context {
414                1 => ContextKind::Container,
415                2 => ContextKind::Vm,
416                3 => ContextKind::Netns,
417                _ => ContextKind::Host,
418            },
419            now_ms: now_ms(),
420        };
421        let matches = rules.explain(&facts);
422        // Names and whether a match depends on what the history doesn't keep.
423        let store = self.store();
424        let managed = crate::managed::Managed::load(
425            self.shared.config.default_rules,
426            &self.shared.config.rules_dir,
427        )
428        .map_err(Error::from)?;
429        // Rules on command lines or parents can't be checked against the history: their absence from the matches says
430        // nothing.
431        let needs_more = |r: &Rule| {
432            r.enabled
433                && r.conditions
434                    .iter()
435                    .any(|c| matches!(c, Condition::Cmdline(_) | Condition::Parent(_)))
436        };
437        let partial = store.rules()?.iter().any(needs_more) || managed.rules.iter().any(needs_more);
438        let mut explained = Vec::new();
439        for m in &matches {
440            let rule = match store.rule(m.rule)? {
441                Some(rule) => Some(rule),
442                None => managed.rules.iter().find(|r| r.id == m.rule).cloned(),
443            };
444            explained.push(v1::ExplainedRule {
445                id: m.rule,
446                name: rule.map_or_else(String::new, |r| r.name),
447                action: convert::action(m.action).into(),
448                tier: convert::tier(m.tier).into(),
449                specificity: m.specificity,
450            });
451        }
452        Ok(Some(v1::Explanation {
453            event: Some(crate::api::event(&stored)),
454            rules: explained,
455            partial,
456        }))
457    }

464    /// Answers a prompt: makes a rule of the chosen suggestion (or `conditions`) unless `duration` is once, then
465    /// decides every connection waiting on it. Returns the rule made, if any.
466    pub fn answer_prompt(
467        &self,
468        id: u64,
469        action: Action,
470        duration: v1::Duration,
471        suggestion: usize,
472        conditions: Option<&v1::Conditions>,
473        folder: Option<FolderId>,
474        by: &str,
475    ) -> Result<Option<RuleId>, AnswerError> {
476        let now = now_ms();
477        let expiry = match duration {
478            v1::Duration::Once => None,
479            v1::Duration::UntilRestart => Some(Expiry::UntilRestart),
480            v1::Duration::Hour => Some(Expiry::At(now + 3_600_000)),
481            v1::Duration::HalfDay => Some(Expiry::At(now + 43_200_000)),
482            v1::Duration::Forever | v1::Duration::Unspecified => Some(Expiry::Never),
483        };
484        // The rule is checked before the prompt is taken: a refused answer leaves it waiting.
485        let rule = match expiry {
486            None => None,
487            Some(expiry) => Some(
488                self.shared
489                    .broker
490                    .peek(id, |pending| -> Result<(Rule, String), AnswerError> {
491                        let (label, spec) = match conditions {
492                            Some(conditions) => (
493                                "Answer to a prompt".to_owned(),
494                                convert::conditions_from(conditions),
495                            ),
496                            None => pending
497                                .suggestions
498                                .get(suggestion)
499                                .map(|s| (s.label.clone(), s.conditions.clone()))
500                                .ok_or_else(|| {
501                                    AnswerError::Invalid(format!("no suggestion {suggestion}"))
502                                })?,
503                        };
504                        let conditions = spec
505                            .to_conditions()
506                            .map_err(|e| AnswerError::Invalid(e.to_string()))?;
507                        if conditions.is_empty() {
508                            return Err(AnswerError::Invalid(
509                                "the answer's rule has no conditions".to_owned(),
510                            ));
511                        }
512                        let program = pending
513                            .prompt
514                            .exe
515                            .rsplit('/')
516                            .next()
517                            .unwrap_or_default()
518                            .to_owned();
519                        let rule = Rule {
520                            id: 0,
521                            folder: None,
522                            name: label,
523                            enabled: true,
524                            action,
525                            tier: Tier::Normal,
526                            expiry,
527                            log: true,
528                            notes: String::new(),
529                            conditions,
530                        };
531                        Ok((rule, program))
532                    })
533                    .ok_or(AnswerError::NotFound)??,
534            ),
535        };
536        // Into the chosen folder, or the program's own.
537        let rule = match rule {
538            Some((rule, program)) => {
539                // A folder the rule can't go in leaves the prompt waiting.
540                crate::placement::check(&self.store(), folder).map_err(placement_error)?;
541                Some((rule, program))
542            }
543            None => None,
544        };
545        let pending = self.shared.broker.take(id).ok_or(AnswerError::NotFound)?;
546        let rule_id = match rule {
547            Some((mut rule, program)) => {
548                let id = {
549                    let mut store = self.store();
550                    let batch = store.batch(by, now).map_err(Error::from)?;
551                    store
552                        .atomically(|store| {
553                            crate::placement::place(store, &batch, &mut rule, folder, &program)?;
554                            Ok(store.put_rule(&rule, &batch)?)
555                        })
556                        .map_err(placement_error)?
557                };
558                self.reload_rules()?;
559                Some(id)
560            }
561            None => None,
562        };
563        self.resolve(&pending, action, Source::Prompt, rule_id, by, false);
564        Ok(rule_id)
565    }

567    /// Decides a prompt's connections, records the decision, and tells the clients.
568    fn resolve(
569        &self,
570        pending: &Pending,
571        action: Action,
572        source: Source,
573        rule: Option<RuleId>,
574        by: &str,
575        timed_out: bool,
576    ) {
577        let verdict = if action == Action::Allow {
578            Verdict::Accept
579        } else {
580            Verdict::Drop
581        };
582        self.shared.broker.release(pending, verdict);
583        let mut event = pending.event.clone();
584        event.time_ms = now_ms();
585        event.action = action;
586        event.source = source;
587        event.rule = rule;
588        if self.shared.events.try_send(event).is_err() {
589            self.shared.dropped_events.fetch_add(1, Ordering::Relaxed);
590        }
591        self.shared.hub.publish(
592            v1::Topic::Prompts,
593            &v1::notice::Body::PromptResolved(v1::PromptResolved {
594                id: pending.prompt.id,
595                action: convert::action(action).into(),
596                by: by.to_owned(),
597                rule,
598                timed_out,
599            }),
600        );
601    }

611    /// Replaces the system firewall with `firewall` (checked), recording it as `app`'s change. With `revert_after`,
612    /// the previous one comes back then unless [`Daemon::confirm_system`] is called first.
613    pub fn set_system(
614        self: &Arc<Self>,
615        firewall: &snitchhiker_core::system::Firewall,
616        app: &str,
617        revert_after: Option<Duration>,
618    ) -> Result<u64, Error> {
619        let before = self.store().system_firewall()?;
620        // The kernel first: a table it refuses is never recorded.
621        crate::system::apply(firewall, &self.shared.config)?;
622        let what = crate::system::describe(&before, firewall);
623        let recorded = {
624            let mut store = self.store();
625            store.batch(app, now_ms()).and_then(|batch| {
626                store.atomically(|s| s.put_system_firewall(firewall, &what, &batch))
627            })
628        };
629        if let Err(error) = recorded {
630            let _ = crate::system::apply(&before, &self.shared.config);
631            return Err(error.into());
632        }
633        let mut trial = self.trial.lock().unwrap_or_else(PoisonError::into_inner);
634        trial.generation += 1;
635        trial.pending = None;
636        let Some(after) = revert_after else {
637            return Ok(0);
638        };
639        let reverts_at_ms = now_ms() + u64::try_from(after.as_millis()).unwrap_or(u64::MAX);
640        trial.pending = Some((reverts_at_ms, before));
641        let generation = trial.generation;
642        drop(trial);
643        let daemon = Arc::clone(self);
644        thread::Builder::new()
645            .name("system-trial".to_owned())
646            .spawn(move || {
647                thread::sleep(after);
648                daemon.revert_system(generation);
649            })?;
650        Ok(reverts_at_ms)
651    }
652
653    /// Reverts trial `generation` if it is still waiting.
654    fn revert_system(self: &Arc<Self>, generation: u64) {
655        let previous = {
656            let mut trial = self.trial.lock().unwrap_or_else(PoisonError::into_inner);
657            if trial.generation != generation {
658                return;
659            }
660            trial.pending.take()
661        };
662        if let Some((_, previous)) = previous {
663            warn!("the system firewall change wasn't confirmed in time: reverting it");
664            if let Err(error) = self.set_system(&previous, "daemon (not confirmed)", None) {
665                error!(%error, "reverting the system firewall");
666            }
667        }
668    }
669
670    /// Confirms the change on trial; false when none waits.
671    pub fn confirm_system(&self) -> bool {
672        let mut trial = self.trial.lock().unwrap_or_else(PoisonError::into_inner);
673        trial.generation += 1;
674        trial.pending.take().is_some()
675    }
676
677    /// When the change on trial reverts (Unix ms); 0 for none.
678    pub fn system_reverts_at(&self) -> u64 {
679        self.trial
680            .lock()
681            .unwrap_or_else(PoisonError::into_inner)
682            .pending
683            .as_ref()
684            .map_or(0, |(at, _)| *at)
685    }
686
687    /// Applies the policy as the store has it (after a client changed it, or an undo).
688    pub fn reload_policy(&self) -> Result<Effective, Error> {
689        let overrides = self
690            .store
691            .lock()
692            .unwrap_or_else(PoisonError::into_inner)
693            .policy()?;
694        let effective = Effective::of(&self.shared.config, &overrides);
695        self.shared
696            .broker
697            .set_timeout(Duration::from_secs(effective.prompt_timeout_secs));
698        self.shared.policy.store(Arc::new(effective));
699        Ok(effective)
700    }
701
702    pub fn reload_rules(&self) -> Result<usize, Error> {
703        let rules = {
704            let store = self.store.lock().unwrap_or_else(PoisonError::into_inner);
705            compile(&self.shared.config, &store)?
706        };
707        let count = rules.len();
708        self.shared.rules.store(Arc::new(rules));
709        self.shared.hub.publish(
710            v1::Topic::Rules,
711            &v1::notice::Body::RulesChanged(v1::RulesChanged {}),
712        );
713        Ok(count)
714    }
715
716    /// Stops: with `fail = open` the table is removed, so traffic flows freely; with `fail = closed` it stays, so
717    /// nothing new gets out until the daemon is back.
718    pub fn stop(&self) {
719        let _ = std::fs::remove_file(config_marker(&self.shared.config));
720        if self.shared.config.fail == Fail::Open {
721            if let Err(error) = remove(&self.table.name) {
722                warn!(%error, "removing the table");
723            }
724            if let Err(error) = crate::system::remove() {
725                warn!(%error, "removing the system firewall");
726            }
727        }
728        info!(
729            decided = self.shared.decided.load(Ordering::Relaxed),
730            dropped_events = self.shared.dropped_events.load(Ordering::Relaxed),
731            "stopped"
732        );
733    }
734}

760/// Learns names from DNS answers: only those to sockets that sent a query to that resolver, and remembered for the
761/// process that sent it. Every packet passes; this queue only observes.
762fn dns_loop(shared: &Shared, mut queue: Queue) -> Result<(), Error> {
763    let mut ids: Vec<u32> = Vec::with_capacity(64);
764    loop {
765        ids.clear();
766        for queued in queue.receive()? {
767            ids.push(queued.id);
768            let Some(headers) = packet::parse(queued.payload) else {
769                continue;
770            };
771            let Some(answer) = packet::udp_payload(queued.payload, &headers).and_then(parse_answer)
772            else {
773                continue;
774            };
775            let query = FlowKey {
776                netns: shared.netns,
777                protocol: packet::IPPROTO_UDP,
778                source_port: headers.destination_port,
779                destination: headers.source,
780                destination_port: DNS_PORT,
781            };
782            // An answer nobody asked for teaches nothing.
783            let Some(asker) = shared.attribution.owner(&query) else {
784                continue;
785            };
786            shared
787                .names
788                .write()
789                .unwrap_or_else(PoisonError::into_inner)
790                .learn(asker.pid, &answer, now_ms());
791        }
792        for &id in &ids {
793            queue.verdict(id, Verdict::Accept)?;
794        }
795    }
796}

798fn decide_loop(shared: &Shared, mut queue: Queue, offset: usize) -> Result<(), Error> {
799    let mut processes = ProcessCache::new(shared.host);
800    let mut verdicts: Vec<(u32, Verdict)> = Vec::with_capacity(64);
801    let mut rejected: Vec<SocketId> = Vec::with_capacity(8);
802    let mut sockdiag: Option<SockDiag> = None;
803    loop {
804        verdicts.clear();
805        rejected.clear();
806        for packet in queue.receive()? {
807            let (verdict, reject) = decide(shared, &mut processes, &packet, offset);
808            // No verdict: the packet waits for a prompt's answer.
809            if let Some(verdict) = verdict {
810                verdicts.push((packet.id, verdict));
811            }
812            rejected.extend(reject);
813        }
814        for &(id, verdict) in &verdicts {
815            queue.verdict(id, verdict)?;
816        }
817        // After the drop: the program's socket is closed, so it fails at once instead of retrying.
818        for id in &rejected {
819            if sockdiag.is_none() {
820                sockdiag = SockDiag::open()
821                    .inspect_err(|error| warn!(%error, "sock_diag: rejected connections will time out instead"))
822                    .ok();
823            }
824            if let Some(diag) = sockdiag.as_mut()
825                && let Err(error) = diag.destroy(id)
826                && error.kind() != std::io::ErrorKind::NotFound
827            {
828                warn!(%error, "closing a rejected socket");
829            }
830        }
831    }
832}

870/// Whether a connection no rule decides is held for a prompt: the policy asks, the daemon enforces, and some client
871/// listens for prompts.
872fn asks(shared: &Shared) -> bool {
873    let policy = shared.policy.load();

875        && policy.mode == Mode::Enforce
876        && shared.hub.has_subscribers(v1::Topic::Prompts)
877}

879/// What a policy does without asking (`ask` falls back to `no_client`).
880fn policy_action(policy: Policy, no_client: Action) -> Action {
881    match policy {
882        Policy::Allow => Action::Allow,
883        Policy::Deny => Action::Deny,
884        Policy::Reject => Action::Reject,
885        Policy::Ask => no_client,
886    }

889/// The verdict for one packet (none: it is held for a prompt), recording what happened, and the socket to close for a
890/// rejected connection.
891fn decide(
892    shared: &Shared,
893    processes: &mut ProcessCache,
894    queued: &Packet<'_>,
895    queue: usize,
896) -> (Option<Verdict>, Option<SocketId>) {
897    shared.decided.fetch_add(1, Ordering::Relaxed);
898    let Some(headers) = packet::parse(queued.payload) else {
899        // Not IPv4 or IPv6: nothing rules speak about.
900        return (Some(Verdict::Accept), None);
901    };
902    let key = FlowKey {
903        netns: shared.netns,
904        protocol: headers.protocol,
905        source_port: headers.source_port,
906        destination: headers.destination,
907        destination_port: headers.destination_port,
908    };
909    let owner = shared.attribution.owner(&key);
910    let process = owner.and_then(|owner| processes.get(owner.pid, owner.start_boottime));
911    let now = now_ms();
912    // The user as the kernel recorded it when the socket was opened (the queue's uid is the socket's owner, for
913    // packets the attribution programs didn't see).
914    let uid = owner.map(|o| o.uid).or(queued.uid).unwrap_or(u32::MAX);
915    let home = process.as_ref().and_then(|_| processes.home(uid));
916    let pid = owner.map_or(0, |o| o.pid);
917    let hosts = shared
918        .names
919        .read()
920        .unwrap_or_else(PoisonError::into_inner)
921        .lookup(pid, headers.destination, now);
922    let empty: &[String] = &[];
923    let facts = Facts {
924        exe: process.as_ref().map_or("", |p| &*p.exe),
925        apps: empty,
926        cmdline: process.as_ref().map_or("", |p| p.cmdline.as_str()),
927        ancestors: process.as_ref().map_or(empty, |p| p.ancestors.as_slice()),
928        uid,
929        home: home.as_deref(),
930        hosts: &hosts,
931        address: headers.destination,
932        port: headers.destination_port,
933        protocol: headers.protocol,
934        context: process.as_ref().map_or(ContextKind::Host, |p| p.context),
935        now_ms: now,
936    };
937    let exe = process
938        .as_ref()
939        .map_or_else(|| Arc::clone(&shared.unknown), |p| Arc::clone(&p.exe));
940    let mut event = undecided_event(&facts, exe, pid, owner.map(|o| o.cookie));
941    let rules = shared.rules.load();
942    let (action, source, rule) = if let Some(decision) = rules.decide(&facts) {
943        (decision.action, Source::Rule, Some(decision.rule))
944    } else {
945        if asks(shared) {
946            hold(
947                shared,
948                &facts,
949                &event,
950                Held {
951                    queue,
952                    id: queued.id,
953                },
954            );
955            return (None, None);
956        }
957        let policy = shared.policy.load();
958        let action = policy_action(policy.no_match, policy.no_client);
959        (action, Source::Policy, None)
960    };
961    let (verdict, source) = match shared.policy.load().mode {
962        Mode::Monitor => (Verdict::Accept, Source::Monitor),
963        Mode::Enforce => (
964            if action == Action::Allow {
965                Verdict::Accept
966            } else {
967                Verdict::Drop
968            },
969            source,
970        ),
971    };
972    event.action = action;
973    event.source = source;
974    event.rule = rule;
975    if let Err(TrySendError::Full(_) | TrySendError::Disconnected(_)) =
976        shared.events.try_send(event)
977    {
978        shared.dropped_events.fetch_add(1, Ordering::Relaxed);
979    }
980    let reject = (verdict == Verdict::Drop && action == Action::Reject).then_some(SocketId {
981        protocol: headers.protocol,
982        local: headers.source,
983        local_port: headers.source_port,
984        remote: headers.destination,
985        remote_port: headers.destination_port,
986    });
987    (Some(verdict), reject)
988}

990/// Holds a packet for the prompt of its program and destination, opening (and announcing) one if none is open.
991fn hold(shared: &Shared, facts: &Facts<'_>, event: &Event, packet: Held) {
992    let destination = facts
993        .hosts
994        .first()
995        .map_or_else(|| facts.address.to_string(), ToString::to_string);
996    let key = PromptKey {
997        exe: Arc::clone(&event.exe),
998        uid: facts.uid,
999        destination,
1000        port: facts.port,
1001        protocol: facts.protocol,
1002    };
1003    let opened = shared.broker.hold(key, packet, facts.now_ms, || {
1004        let suggestions = snitchhiker_core::suggest::suggestions(
1005            facts.exe,
1006            facts.home,
1007            facts.hosts,
1008            facts.address,
1009            facts.port,
1010            facts.protocol,
1011        );
1012        NewPrompt {
1013            prompt: v1::Prompt {
1014                exe: facts.exe.to_owned(),
1015                cmdline: facts.cmdline.to_owned(),
1016                uid: facts.uid,
1017                user: crate::process::user_of(facts.uid).unwrap_or_default(),
1018                pid: event.pid,
1019                hosts: facts.hosts.iter().map(ToString::to_string).collect(),
1020                address: convert::address(facts.address),
1021                port: u32::from(facts.port),
1022                protocol: u32::from(facts.protocol),
1023                context: convert::context(facts.context).into(),
1024                exe_writable: crate::process::writable_by_others(event.pid),
1025                suggestions: suggestions
1026                    .iter()
1027                    .map(|s| v1::Suggestion {
1028                        label: s.label.clone(),
1029                        conditions: Some(convert::conditions(&s.conditions)),
1030                        broad: s.broad,
1031                    })
1032                    .collect(),
1033                ..v1::Prompt::default()
1034            },
1035            suggestions,
1036            event: event.clone(),
1037        }
1038    });
1039    if let Hold::Opened(prompt) = opened {
1040        shared
1041            .hub
1042            .publish(v1::Topic::Prompts, &v1::notice::Body::Prompt(prompt));
1043    }
1044}

1046/// Pushes recorded events to the subscribers of the events topic: one message each to those that want messages, and
1047/// the batch as one columnar block to those that want blocks.
1048fn publish_events(shared: &Shared, batch: &[Event], recorded: &[crate::store::Recorded]) {
1049    let (messages, blocks) = shared.hub.wanted(v1::Topic::Events);
1050    let plain: Vec<Outgoing> = if messages {
1051        batch
1052            .iter()
1053            .zip(recorded)
1054            .map(|(event, r)| {
1055                Outgoing::Notice(v1::notice::Body::Event(crate::api::event(
1056                    &event.stored(r.seq),
1057                )))
1058            })
1059            .collect()
1060    } else {
1061        Vec::new()
1062    };
1063    let block = if blocks {
1064        let rows: Vec<EventRow> = batch
1065            .iter()
1066            .zip(recorded)
1067            .map(|(e, r)| event_row(e, r))
1068            .collect();
1069        let mut strings: Vec<(u32, Arc<str>)> = Vec::new();
1070        for (event, r) in batch.iter().zip(recorded) {
1071            strings.push((r.exe, Arc::clone(&event.exe)));
1072            if let Some(host) = &event.host {
1073                strings.push((r.host, Arc::clone(host)));
1074            }
1075        }
1076        let mut columns = Vec::new();
1077        match columns::encode_events(&rows, &mut columns) {
1078            Ok(()) => Some(Outgoing::Block {
1079                columns: bytes::Bytes::from(columns),
1080                strings: Arc::from(strings),
1081            }),
1082            Err(error) => {
1083                error!(%error, "encoding an event block");
1084                None
1085            }
1086        }
1087    } else {
1088        None
1089    };
1090    if messages || block.is_some() {
1091        shared
1092            .hub
1093            .publish_all(v1::Topic::Events, &plain, block.as_ref());
1094    }
1095}

1097/// A recorded event as a block's row: the protocol's numbers, the strings' ids.
1098fn event_row(event: &Event, recorded: &crate::store::Recorded) -> EventRow {
1099    let address = match event.address {
1100        std::net::IpAddr::V4(a) => a.to_ipv6_mapped().octets(),
1101        std::net::IpAddr::V6(a) => a.octets(),
1102    };
1103    EventRow {
1104        seq: recorded.seq,
1105        time_ms: event.time_ms,
1106        exe: recorded.exe,
1107        host: recorded.host,
1108        uid: event.uid,
1109        pid: event.pid,
1110        address,
1111        port: event.port,
1112        protocol: event.protocol,
1113        context: event.context.saturating_add(1),
1114        action: u8::try_from(i32::from(convert::action(event.action))).unwrap_or(0),
1115        source: event.source as u8,
1116        rule: event.rule.unwrap_or(0),
1117        bytes_out: 0,
1118        bytes_in: 0,
1119    }
1120}

1152/// Packets the kernel dropped on queues `first..first + count` (`/proc/net/netfilter/nfnetlink_queue`: the queue
1153/// overflowed, or the daemon's socket couldn't take them).
1154fn queue_drops(first: u16, count: u16) -> u64 {
1155    std::fs::read_to_string("/proc/net/netfilter/nfnetlink_queue").map_or(0, |text| {
1156        text.lines()
1157            .filter_map(|line| {
1158                let fields: Vec<u64> = line
1159                    .split_whitespace()
1160                    .filter_map(|f| f.parse().ok())
1161                    .collect();
1162                let queue = u16::try_from(*fields.first()?).ok()?;
1163                (queue >= first && queue < first.saturating_add(count)).then(|| {
1164                    fields.get(5).copied().unwrap_or(0) + fields.get(6).copied().unwrap_or(0)
1165                })
1166            })
1167            .sum()
1168    })
1169}

1171/// Alerts about what the node lost since the last check: events the history couldn't keep, packets the kernel
1172/// dropped.
1173fn check_losses(shared: &Shared, store: &Store, seen: &mut (u64, u64)) {
1174    let dropped = shared.dropped_events.load(Ordering::Relaxed);
1175    if dropped > seen.0 {
1176        crate::alerts::raise(
1177            &shared.hub,
1178            store,
1179            &crate::alerts::NewAlert {
1180                kind: "dropped_events",
1181                title: "Connections weren't recorded".to_owned(),
1182                body: format!(
1183                    "{} connections were decided but not kept in the history: the database couldn't keep up.",
1184                    dropped - seen.0
1185                ),
1186                rule: None,
1187            },
1188        );
1189    }
1190    let queued = queue_drops(shared.config.queue, shared.config.queues);
1191    if queued > seen.1 {
1192        crate::alerts::raise(
1193            &shared.hub,
1194            store,
1195            &crate::alerts::NewAlert {
1196                kind: "queue_overflow",
1197                title: "The kernel dropped packets waiting for a decision".to_owned(),
1198                body: format!(
1199                    "{} packets were dropped on the daemon's queues: they overflowed. New connections were refused \
1200                     while that lasted.",
1201                    queued - seen.1
1202                ),
1203                rule: None,
1204            },
1205        );
1206    }
1207    *seen = (dropped, queued.max(seen.1));
1208}

1210fn write_loop(shared: &Shared, store: &Mutex<Store>, events: &Receiver<Event>) {
1211    let mut batch = Vec::with_capacity(BATCH);
1212    let mut engine = crate::alerts::Engine::default();
1213    let mut rules_seen = u64::MAX;
1214    let mut last_loss_check = Instant::now();
1215    let mut losses = (0, queue_drops(shared.config.queue, shared.config.queues));
1216    let mut recorded = Vec::with_capacity(BATCH);
1217    let mut last_prune = Instant::now();
1218    let mut last_bytes = Instant::now();
1219    let mut counted: HashMap<u64, (u64, u64)> = HashMap::new();
1220    let mut changed: Vec<(u64, u64, u64)> = Vec::new();
1221    loop {
1222        batch.clear();
1223        match events.recv_timeout(BYTES_EVERY) {
1224            Ok(event) => batch.push(event),
1225            Err(RecvTimeoutError::Timeout) => {}
1226            Err(RecvTimeoutError::Disconnected) => return,
1227        }
1228        let deadline = Instant::now() + BATCH_WAIT;
1229        while batch.len() < BATCH {
1230            let left = deadline.saturating_duration_since(Instant::now());
1231            match events.recv_timeout(left) {
1232                Ok(event) => batch.push(event),
1233                Err(_) => break,
1234            }
1235        }
1236        let mut store = store.lock().unwrap_or_else(PoisonError::into_inner);
1237        let generation = shared.alert_rules_changed.load(Ordering::Relaxed);
1238        if generation != rules_seen {
1239            rules_seen = generation;
1240            match store.alert_rules() {
1241                Ok(rules) => engine.load(rules),
1242                Err(error) => error!(%error, "loading the alert rules"),
1243            }
1244        }
1245        if !batch.is_empty() {
1246            match store.record(&batch, &mut recorded) {
1247                Ok(()) => {
1248                    publish_events(shared, &batch, &recorded);
1249                    for alert in engine.check(&batch, &recorded) {
1250                        crate::alerts::raise(&shared.hub, &store, &alert);
1251                    }
1252                }
1253                Err(error) => error!(%error, events = batch.len(), "recording events"),
1254            }
1255        }
1256        if last_loss_check.elapsed() >= LOSS_CHECK_EVERY {
1257            last_loss_check = Instant::now();
1258            check_losses(shared, &store, &mut losses);
1259        }
1260        if last_bytes.elapsed() >= BYTES_EVERY {
1261            last_bytes = Instant::now();
1262            changed.clear();
1263            let mut live: HashMap<u64, (u64, u64)> = HashMap::with_capacity(counted.len());
1264            for (cookie, sent, read) in shared.attribution.socket_bytes() {
1265                if counted.get(&cookie) != Some(&(sent, read)) {
1266                    changed.push((cookie, sent, read));
1267                }
1268                live.insert(cookie, (sent, read));
1269            }
1270            counted = live;
1271            if !changed.is_empty()
1272                && let Err(error) = store.record_bytes(&changed)
1273            {
1274                error!(%error, "recording byte counts");
1275            }
1276        }
1277        if last_prune.elapsed() >= PRUNE_EVERY {
1278            last_prune = Instant::now();
1279            let retention_ms = u64::from(shared.policy.load().retention_days) * 86_400_000;
1280            let cutoff = now_ms().saturating_sub(retention_ms);
1281            while let Ok(pruned) = store.prune(cutoff, PRUNE_CHUNK) {
1282                if pruned < PRUNE_CHUNK as usize {
1283                    break;
1284                }
1285            }
1286            if let Err(error) = store.prune_alerts(cutoff) {
1287                error!(%error, "pruning alerts");
1288            }
1289            if let Err(error) = store.prune_rollups(cutoff) {
1290                error!(%error, "pruning rollups");
1291            }
1292        }
1293    }
