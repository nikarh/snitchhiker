1//! The running daemon: the table, the attribution programs, one deciding thread per queue, and the writer.
2//!
3//! The packet path: receive → parse the headers → look the flow's owner up (eBPF) → the process's facts (cached) →
4//! the rules (an immutable set, swapped atomically on change) → verdict, and an event to the writer's queue. Nothing
5//! on it waits for the database, and a full event queue drops the event (counted), never the verdict.

39/// Events waiting for the writer; beyond this they are dropped (and counted). Allocated up front (128 bytes each):
40/// enough for 300 ms of 50 000 connections a second while the writer commits.
41const EVENT_QUEUE: usize = 16_384;
42/// Events written per transaction at most. Large, so that at a high rate the index pages a transaction changes are
43/// shared by many events (the writer still waits at most `BATCH_WAIT`).
44pub const BATCH: usize = 8192;
45/// How long the writer waits to fill a batch.
46const BATCH_WAIT: Duration = Duration::from_millis(100);
47/// How much of a DNS answer is copied: a whole UDP answer, as large as resolvers send without EDNS tricks.
48const DNS_COPY: u32 = 4096;
49const DNS_PORT: u16 = 53;
50/// How often byte counts are written to the history.
51const BYTES_EVERY: Duration = Duration::from_secs(5);
52/// How often old history is pruned, and how many rows per chunk.
53const PRUNE_EVERY: Duration = Duration::from_secs(600);
54const PRUNE_CHUNK: u32 = 10_000;

161    /// Opens the store, compiles the rules, installs the table, loads the attribution programs and starts the
162    /// deciding threads and the writer.
163    pub fn start(config: Config) -> Result<Arc<Self>, Error> {
164        if let Some(parent) = config.database.parent() {
165            std::fs::create_dir_all(parent)?;
166        }
167        let store = Store::open(&config.database)?;
168        let blocklists = load_blocklists(&config, &store)?;
169        let (rules, mut agents) = compile(&config, &store, &blocklists)?;
170        info!(rules = rules.len(), "rules loaded");
171        // Agents whose proposals wait, to expire them if the agent ends.
172        agents.extend(waiting_agents(&store)?);
173        // The programs first: from the moment the table is in place, every new flow's owner must be known.
174        let attribution = Attribution::load()?;
175        // The queues before the table: a queue another program holds fails the start with nothing put in place.
176        let (queues, writers) = bind_queues(&config)?;
177        let table = table_of(&config);
178        install(&table)?;
179        info!(queues = config.queues, mode = ?config.mode, "table installed");
180        // From here on a failed start takes the tables back out: a daemon that isn't running is never in the way.
181        let mut installed = Installed(true);
182
183        // Until-restart rules from before don't apply any more.
184        let mut store = store;
185        store.delete_until_restart()?;
186        let (sender, receiver) = sync_channel(EVENT_QUEUE);
187        let store = Arc::new(crate::locked::Locked::new(store));
188        let policy = Effective::of(&config, &store.lock().policy()?);
189        let broker = Broker::new(writers, Duration::from_secs(policy.prompt_timeout_secs));
190        let shared = Arc::new(Shared {
191            rules: ArcSwap::from_pointee(rules),
192            attribution,
193            names: RwLock::new(Names::default()),
194            netns: u32::try_from(std::fs::metadata("/proc/self/ns/net")?.ino()).unwrap_or(0),
195            host: HostNamespaces::read()?,
196            events: sender,
197            dropped_events: AtomicU64::new(0),
198            unknown: Arc::from(""),
199            decided: AtomicU64::new(0),
200            remote_pid: std::sync::atomic::AtomicU32::new(0),
201            hub: Arc::new(Hub::default()),
202            broker,
203            alert_rules_changed: AtomicU64::new(0),
204            policy: ArcSwap::from_pointee(policy),
205            blocked: crate::agents::Ring::default(),
206            agents: crate::agents::Watch::new()?,
207            proposals_changed: (Mutex::new(0), std::sync::Condvar::new()),
208            router: crate::router::Router::new(
209                config.prompt_routing,
210                Duration::from_secs(config.escalate_after_secs),
211            ),
212            contexts: crate::contexts::Contexts::default(),
213            namespaces: crate::namespaces::Namespaces::default(),
214            webhook: crate::webhook::Webhook::default(),
215            blocklists,
216            packages: Arc::new(crate::apps::Packages::load()),
217            muted_agents: Mutex::default(),
218            net_admin: Mutex::default(),
219            config,
220        });
221        for (offset, queue) in queues.into_iter().enumerate() {
222            let number = queue.number();
223            let shared = Arc::clone(&shared);
224            thread::Builder::new()
225                .name(format!("queue-{number}"))
226                .spawn(move || {
227                    if let Err(error) = decide_loop(&shared, queue, offset) {
228                        error!(%error, queue = number, "deciding thread stopped");
229                    }
230                })?;
231        }
232        if let Some(number) = shared.config.dns_queue() {
233            let queue = Queue::bind_copying(number, DNS_COPY).map_err(queue_error(number))?;
234            let shared = Arc::clone(&shared);
235            thread::Builder::new()
236                .name("dns".to_owned())
237                .spawn(move || {
238                    if let Err(error) = dns_loop(&shared, queue) {
239                        error!(%error, "learning names stopped");
240                    }
241                })?;
242        }
243        {
244            let store = Arc::clone(&store);
245            let shared = Arc::clone(&shared);
246            thread::Builder::new()
247                .name("writer".to_owned())
248                .spawn(move || write_loop(&shared, &store, &receiver))?;
249        }
250        check_marker(&shared, &store.lock());
251        // The system firewall as last set: a rule the kernel refuses mustn't keep the daemon from deciding.
252        let firewall = store.lock().system_firewall()?;
253        if let Err(error) = crate::system::apply(&firewall, &shared.config) {
254            error!(%error, "installing the system firewall");
255        }
256        let daemon = Arc::new(Self {
257            shared,
258            store,
259            table,
260            trial: Mutex::new(Trial::default()),
261            remote: crate::remote::Remote::default(),
262            approvals: crate::approvals::Approvals::default(),
263        });
264        daemon.spawn_watches(agents)?;
265        installed.0 = false;
266        Ok(daemon)
267    }

269    /// The threads that wait for something: prompts timing out, the tables, agents ending, prompts to escalate.
270    fn spawn_watches(self: &Arc<Self>, agents: HashSet<ProcessRef>) -> std::io::Result<()> {
271        self.spawn_prompt_timeouts()?;
272        self.spawn_table_watch()?;
273        self.spawn_agent_watch()?;
274        self.spawn_blocklist_refresh()?;
275        self.shared.packages.watch()?;
276        crate::container_events::spawn(self)?;
277        if self.shared.config.learn_dns {
278            // Names programs resolved through systemd-resolved's cache or its library interface: no packet shows them.
279            let shared = Arc::clone(&self.shared);
280            crate::resolved::spawn(move |answer| {
281                shared
282                    .names
283                    .write()
284                    .unwrap_or_else(PoisonError::into_inner)
285                    .learn(0, answer, now_ms());
286            })?;
287        }
288        self.watch_agents(agents);
289        crate::router::Router::spawn(&self.shared)?;
290        crate::namespaces::Namespaces::spawn(self)?;
291        if !self.shared.config.alert_webhook.is_empty() {
292            self.shared.webhook.start(
293                &self.shared.config.alert_webhook,
294                self.shared.config.alert_webhook_secret_file.as_deref(),
295            )?;
296        }
297        self.devices_changed();
298        Ok(())
299    }

301    /// Tells the router which devices are muted and where each is woken, from the store.
302    pub fn devices_changed(&self) {
303        let devices = self.store().devices();
304        match devices {
305            Ok(devices) => {
306                let router = &self.shared.router;
307                router.set_muted(devices.iter().filter(|d| d.muted).map(|d| d.id.clone()));
308                router.push.set_targets(
309                    devices
310                        .iter()
311                        .filter_map(|d| {
312                            Some((d.id.clone(), crate::push::target(d.push.as_deref()?)?))
313                        })
314                        .collect(),
315                );
316            }
317            Err(error) => error!(%error, "reading the paired devices"),
318        }
319    }

321    /// Announces a proposal made or changed: to subscribers, and to requests waiting for a change.
322    pub fn publish_proposal(&self, proposal: &v1::Proposal) {
323        self.shared.hub.publish(
324            v1::Topic::Proposals,
325            &v1::notice::Body::Proposal(Box::new(proposal.clone())),
326        );
327        let (count, changed) = &self.shared.proposals_changed;
328        *count.lock().unwrap_or_else(PoisonError::into_inner) += 1;
329        changed.notify_all();
330    }

332    /// Waits until `still` says no, checking it again whenever a proposal changes, for at most `timeout`.
333    pub fn wait_for_proposals(&self, timeout: Duration, mut still: impl FnMut() -> bool) {
334        let deadline = Instant::now() + timeout;
335        let (count, changed) = &self.shared.proposals_changed;
336        let mut seen = *count.lock().unwrap_or_else(PoisonError::into_inner);
337        while still() {
338            let left = deadline.saturating_duration_since(Instant::now());
339            if left.is_zero() {
340                return;
341            }
342            let guard = count.lock().unwrap_or_else(PoisonError::into_inner);
343            let (guard, _) = changed
344                .wait_timeout_while(guard, left, |now| *now == seen)
345                .unwrap_or_else(PoisonError::into_inner);
346            seen = *guard;
347        }
348    }

378    /// Ends what the agents that exited left: their session rules are removed (one batch of the history) and their
379    /// waiting proposals expire.
380    fn end_agents(&self, ended: &[ProcessRef]) {
381        let now = now_ms();
382        let removed = {
383            let mut store = self.store();
384            let rules = store.rules();
385            rules.and_then(|rules| {
386                let gone: Vec<RuleId> = rules
387                    .iter()
388                    .filter(|r| {
389                        crate::agents::sessions(r).is_some_and(|roots| {
390                            roots
391                                .iter()
392                                .all(|a| ended.contains(a) || !crate::agents::alive(*a))
393                        })
394                    })
395                    .map(|r| r.id)
396                    .collect();
397                if gone.is_empty() {
398                    return Ok(0);
399                }
400                let batch = store.batch("snitchhikerd (an agent's session ended)", now)?;
401                store.atomically(|store| {
402                    for id in &gone {
403                        store.remove_rule(*id, &batch)?;
404                    }
405                    Ok(gone.len())
406                })
407            })
408        };
409        match removed {
410            Ok(0) => {}
411            Ok(count) => {
412                info!(count, "an agent ended: its session's rules are removed");
413                if let Err(error) = self.reload_rules() {
414                    error!(%error, "reloading the rules");
415                }
416            }
417            Err(error) => error!(%error, "removing an ended agent's rules"),
418        }
419        let pending = self
420            .store()
421            .proposals(Some(v1::ProposalState::Pending), 1000);
422        for mut proposal in pending.unwrap_or_default() {
423            let Some(p) = &proposal.provenance else {
424                continue;
425            };
426            let agent = ProcessRef {
427                pid: p.agent_pid,
428                start: p.agent_start,
429            };
430            if !ended.contains(&agent) {
431                continue;
432            }
433            proposal.state = v1::ProposalState::Expired.into();
434            proposal.decided_ms = now;
435            let updated = self.store().update_proposal(&proposal);
436            match updated {
437                Ok(()) => self.publish_proposal(&proposal),
438                Err(error) => error!(%error, "expiring a proposal"),
439            }
440        }
441    }

443    /// Whenever the kernel announces that the daemon's table or the system firewall's was deleted, checks they are
444    /// still there: one removed by someone else is put back, with an alert. (The daemon's own reinstalls delete and
445    /// create its table in one transaction: by the time their announcement is read, it is there.)
446    fn spawn_table_watch(self: &Arc<Self>) -> std::io::Result<()> {
447        let mut monitor = snitchhiker_netfilter::nftables::Monitor::open()?;
448        let daemon = Arc::clone(self);
449        thread::Builder::new()
450            .name("table-watch".to_owned())
451            .spawn(move || {
452                let names = [daemon.table.name.as_str(), crate::system::TABLE];
453                loop {
454                    match monitor.deleted_any(&names) {
455                        Ok(true) => daemon.check_tables(),
456                        Ok(false) => {}
457                        // Announcements were lost (the socket's buffer overflowed): one may have been a deletion.
458                        Err(error) if error.raw_os_error() == Some(nix::libc::ENOBUFS) => {
459                            daemon.check_tables();
460                        }
461                        Err(error) => {
462                            warn!(%error, "following the nftables changes; removed tables are no longer put back");
463                            return;
464                        }
465                    }
466                }
467            })?;
468        Ok(())
469    }

471    fn check_tables(&self) {
472        let missing = |name: &str| {
473            let mut writer = Writer::new();
474            let sequence = encode_get_table(&mut writer, name, 1);
475            Socket::open()
476                .and_then(|mut socket| socket.transact(&writer, sequence, sequence))
477                .is_err_and(|e| e.raw_os_error() == Some(2))
478        };
479        if missing(&self.table.name) {
480            let put_back = install(&self.table);
481            self.raise_table_removed("the daemon's", &put_back);
482        }
483        let firewall = self.store().system_firewall();
484        if firewall.as_ref().is_ok_and(|f| f.enabled) && missing(crate::system::TABLE) {
485            let put_back = self
486                .apply_system()
487                .map_err(|e| std::io::Error::other(e.to_string()));
488            self.raise_table_removed("the system firewall's", &put_back);
489        }
490    }

523    /// Fetches blocklists when they are due.
524    fn spawn_blocklist_refresh(self: &Arc<Self>) -> std::io::Result<()> {
525        let daemon = Arc::clone(self);
526        thread::Builder::new()
527            .name("blocklists".to_owned())
528            .spawn(move || {
529                loop {
530                    let records = daemon.store().blocklists();
531                    let records = match records {
532                        Ok(records) => records,
533                        Err(error) => {
534                            error!(%error, "reading the blocklists; they are no longer refreshed");
535                            return;
536                        }
537                    };
538                    let now = now_ms();
539                    let due: Vec<_> = records
540                        .iter()
541                        .filter(|r| crate::blocklists::is_due(r, now))
542                        .cloned()
543                        .collect();
544                    if due.is_empty() {
545                        daemon
546                            .shared
547                            .blocklists
548                            .wait(crate::blocklists::next_due(&records), now);
549                        continue;
550                    }
551                    for mut record in due {
552                        daemon.refresh_blocklist(&mut record);
553                    }
554                }
555            })?;
556        Ok(())
557    }

559    /// Fetches a list now, records how it went and, when it changed, compiles the rules again.
560    pub fn refresh_blocklist(&self, record: &mut crate::blocklists::Record) {
561        let changed = self.shared.blocklists.fetch(record, now_ms());
562        let updated = self.store().update_blocklist(record);
563        if let Err(error) = updated {
564            warn!(%error, list = record.name, "recording a blocklist's fetch");
565        }
566        if changed && let Err(error) = self.reload_rules() {
567            warn!(%error, "compiling the rules with a changed blocklist");
568        }
569    }

571    /// Prompts nobody answered in time are decided by the policy's `on_timeout`.
572    fn spawn_prompt_timeouts(self: &Arc<Self>) -> std::io::Result<()> {
573        let daemon = Arc::clone(self);
574        thread::Builder::new()
575            .name("prompt-timeouts".to_owned())
576            .spawn(move || {
577                loop {
578                    for pending in daemon.shared.broker.wait_expired() {
579                        let action = daemon.shared.policy.load().on_timeout;
580                        daemon.resolve(&pending, action, Source::Timeout, None, "", true);
581                    }
582                }
583            })?;
584        Ok(())
585    }

599    /// The live TCP connections and connected UDP sockets, each with its program as the attribution programs
600    /// recorded it when it opened (unknown for inbound sockets, and for ones older than the daemon).
601    pub fn connections(&self) -> std::io::Result<Vec<v1::LiveConnection>> {
602        let sockets = SockDiag::open()?.list()?;
603        let bytes: HashMap<u64, (u64, u64)> = self
604            .shared
605            .attribution
606            .socket_bytes()
607            .map(|(cookie, sent, read)| (cookie, (sent, read)))
608            .collect();
609        let mut processes = ProcessCache::new(self.shared.host);
610        let names = self
611            .shared
612            .names
613            .read()
614            .unwrap_or_else(PoisonError::into_inner);
615        let now = now_ms();
616        let plain = |address: std::net::IpAddr| match address {
617            std::net::IpAddr::V6(v6) => v6.to_ipv4_mapped().map_or(address, std::net::IpAddr::V4),
618            v4 @ std::net::IpAddr::V4(_) => v4,
619        };
620        Ok(sockets
621            .iter()
622            .map(|socket| {
623                let remote = plain(socket.id.remote);
624                let key = FlowKey {
625                    netns: self.shared.netns,
626                    protocol: socket.id.protocol,
627                    source_port: socket.id.local_port,
628                    destination: remote,
629                    destination_port: socket.id.remote_port,
630                };
631                let owner = self.shared.attribution.owner(&key);
632                let process = owner.and_then(|o| processes.get(o.pid, o.start_boottime));
633                let pid = owner.map_or(0, |o| o.pid);
634                let (sent, read) = bytes.get(&socket.cookie).copied().unwrap_or_default();
635                v1::LiveConnection {
636                    protocol: u32::from(socket.id.protocol),
637                    local: convert::address(plain(socket.id.local)),
638                    local_port: u32::from(socket.id.local_port),
639                    remote: convert::address(remote),
640                    remote_port: u32::from(socket.id.remote_port),
641                    state: snitchhiker_netfilter::sockdiag::state_name(
642                        socket.id.protocol,
643                        socket.state,
644                    )
645                    .to_owned(),
646                    uid: owner.map_or(socket.uid, |o| o.uid),
647                    pid,
648                    exe: process
649                        .as_ref()
650                        .map_or_else(String::new, |p| p.exe.to_string()),
651                    host: names
652                        .lookup(pid, remote, now)
653                        .first()
654                        .map_or_else(String::new, ToString::to_string),
655                    bytes_out: sent,
656                    bytes_in: read,
657                    context: convert::context(
658                        process.as_ref().map_or(ContextKind::Host, |p| p.context),
659                    )
660                    .into(),
661                    container: owner
662                        .filter(|_| {
663                            process.as_ref().map(|p| p.context) == Some(ContextKind::Container)
664                        })
665                        .and_then(|o| self.shared.contexts.container(o.pid, o.cgroup))
666                        .map(|c| Box::new(convert::container(&c.container))),
667                }
668            })
669            .collect())
670    }

672    /// Why the recorded connection `seq` is decided as it is, by today's rules; `None` if there is no such event.
673    pub fn explain(&self, seq: i64) -> Result<Option<v1::Explanation>, Error> {
674        let Some(stored) = self.store().event(seq)? else {
675            return Ok(None);
676        };
677        let rules = self.shared.rules.load();
678        let hosts: Vec<Arc<str>> = stored.host.iter().map(|h| Arc::from(h.as_str())).collect();
679        let home = crate::process::home_of(stored.uid);
680        let facts = Facts {
681            exe: &stored.exe,
682            apps: &[],
683            sha256: None,
684            writable: false,
685            cmdline: "",
686            ancestors: &[],
687            lineage: &[],
688            uid: stored.uid,
689            home: home.as_deref(),
690            hosts: &hosts,
691            address: stored.address,
692            port: stored.port,
693            protocol: stored.protocol,
694            context: match stored.context {
695                1 => ContextKind::Container,
696                2 => ContextKind::Vm,
697                3 => ContextKind::Netns,
698                _ => ContextKind::Host,
699            },
700            now_ms: now_ms(),
701            container: None,
702            vm: None,
703            netns: None,
704        };
705        let matches = rules.explain(&facts);
706        // Names and whether a match depends on what the history doesn't keep.
707        let store = self.store();
708        let managed = crate::managed::Managed::load(
709            self.shared.config.default_rules,
710            &self.shared.config.rules_dir,
711        )
712        .map_err(Error::from)?;
713        // Rules on command lines or parents can't be checked against the history: their absence from the matches says
714        // nothing.
715        let needs_more = |r: &Rule| {
716            r.enabled
717                && r.conditions
718                    .iter()
719                    .any(|c| matches!(c, Condition::Cmdline(_) | Condition::Parent(_)))
720        };
721        let partial = store.rules()?.iter().any(needs_more) || managed.rules.iter().any(needs_more);
722        let mut explained = Vec::new();
723        for m in &matches {
724            let rule = match store.rule(m.rule)? {
725                Some(rule) => Some(rule),
726                None => managed.rules.iter().find(|r| r.id == m.rule).cloned(),
727            };
728            explained.push(v1::ExplainedRule {
729                id: m.rule,
730                name: rule.map_or_else(String::new, |r| r.name),
731                action: convert::action(m.action).into(),
732                tier: convert::tier(m.tier).into(),
733                specificity: m.specificity,
734            });
735        }
736        Ok(Some(v1::Explanation {
737            event: Some(crate::api::event(&stored)),
738            rules: explained,
739            partial,
740        }))
741    }

748    /// Answers a prompt: makes a rule of the chosen suggestion (or `conditions`) unless `duration` is once, then
749    /// decides every connection waiting on it. Returns the rule made, if any.
750    pub fn answer_prompt(
751        &self,
752        id: u64,
753        action: Action,
754        duration: v1::Duration,
755        suggestion: usize,
756        conditions: Option<&v1::Conditions>,
757        folder: Option<FolderId>,
758        by: &str,
759    ) -> Result<Option<RuleId>, AnswerError> {
760        let now = now_ms();
761        let expiry = match duration {
762            v1::Duration::Once => None,
763            v1::Duration::UntilRestart => Some(Expiry::UntilRestart),
764            v1::Duration::Hour => Some(Expiry::At(now + 3_600_000)),
765            v1::Duration::HalfDay => Some(Expiry::At(now + 43_200_000)),
766            v1::Duration::Forever | v1::Duration::Unspecified => Some(Expiry::Never),
767        };
768        // The rule is checked before the prompt is taken: a refused answer leaves it waiting.
769        let rule = match expiry {
770            None => None,
771            Some(expiry) => Some(
772                self.shared
773                    .broker
774                    .peek(id, |pending| -> Result<(Rule, String), AnswerError> {
775                        let (label, spec) = match conditions {
776                            Some(conditions) => (
777                                "Answer to a prompt".to_owned(),
778                                convert::conditions_from(conditions),
779                            ),
780                            None => pending
781                                .suggestions
782                                .get(suggestion)
783                                .map(|s| (s.label.clone(), s.conditions.clone()))
784                                .ok_or_else(|| {
785                                    AnswerError::Invalid(format!("no suggestion {suggestion}"))
786                                })?,
787                        };
788                        let conditions = spec
789                            .to_conditions()
790                            .map_err(|e| AnswerError::Invalid(e.to_string()))?;
791                        if conditions.is_empty() {
792                            return Err(AnswerError::Invalid(
793                                "the answer's rule has no conditions".to_owned(),
794                            ));
795                        }
796                        let program = pending
797                            .prompt
798                            .exe
799                            .rsplit('/')
800                            .next()
801                            .unwrap_or_default()
802                            .to_owned();
803                        let rule = Rule {
804                            id: 0,
805                            folder: None,
806                            name: label,
807                            enabled: true,
808                            action,
809                            tier: Tier::Normal,
810                            expiry,
811                            log: true,
812                            notes: String::new(),
813                            conditions,
814                        };
815                        Ok((rule, program))
816                    })
817                    .ok_or(AnswerError::NotFound)??,
818            ),
819        };
820        // Into the chosen folder, or the program's own.
821        let rule = match rule {
822            Some((rule, program)) => {
823                // A folder the rule can't go in leaves the prompt waiting.
824                crate::placement::check(&self.store(), folder).map_err(placement_error)?;
825                Some((rule, program))
826            }
827            None => None,
828        };
829        let pending = self.shared.broker.take(id).ok_or(AnswerError::NotFound)?;
830        let rule_id = match rule {
831            Some((mut rule, program)) => {
832                let id = {
833                    let mut store = self.store();
834                    let batch = store.batch(by, now).map_err(Error::from)?;
835                    store
836                        .atomically(|store| {
837                            crate::placement::place(store, &batch, &mut rule, folder, &program)?;
838                            Ok(store.put_rule(&rule, &batch)?)
839                        })
840                        .map_err(placement_error)?
841                };
842                self.reload_rules()?;
843                Some(id)
844            }
845            None => None,
846        };
847        self.resolve(&pending, action, Source::Prompt, rule_id, by, false);
848        Ok(rule_id)
849    }

851    /// Decides a prompt's connections, records the decision, and tells the clients.
852    fn resolve(
853        &self,
854        pending: &Pending,
855        action: Action,
856        source: Source,
857        rule: Option<RuleId>,
858        by: &str,
859        timed_out: bool,
860    ) {
861        let verdict = if action == Action::Allow {
862            Verdict::Accept
863        } else {
864            Verdict::Drop
865        };
866        self.shared.broker.release(pending, verdict);
867        let mut event = pending.event.clone();
868        event.time_ms = now_ms();
869        event.action = action;
870        event.source = source;
871        event.rule = rule;
872        if self.shared.events.try_send(event).is_err() {
873            self.shared.dropped_events.fetch_add(1, Ordering::Relaxed);
874        }
875        self.shared.router.closed(pending.prompt.id);
876        self.shared.hub.publish(
877            v1::Topic::Prompts,
878            &v1::notice::Body::PromptResolved(v1::PromptResolved {
879                id: pending.prompt.id,
880                action: convert::action(action).into(),
881                by: by.to_owned(),
882                rule,
883                timed_out,
884            }),
885        );
886    }

896    /// Replaces the system firewall with `firewall` (checked), recording it as `app`'s change. With `revert_after`,
897    /// the previous one comes back then unless [`Daemon::confirm_system`] is called first.
898    pub fn set_system(
899        self: &Arc<Self>,
900        firewall: &snitchhiker_core::system::Firewall,
901        app: &str,
902        revert_after: Option<Duration>,
903    ) -> Result<u64, Error> {
904        let before = self.store().system_firewall()?;
905        // The kernel first: a table it refuses is never recorded.
906        crate::system::apply(firewall, &self.shared.config)?;
907        let what = crate::system::describe(&before, firewall);
908        let recorded = {
909            let mut store = self.store();
910            store.batch(app, now_ms()).and_then(|batch| {
911                store.atomically(|s| s.put_system_firewall(firewall, &what, &batch))
912            })
913        };
914        if let Err(error) = recorded {
915            let _ = crate::system::apply(&before, &self.shared.config);
916            return Err(error.into());
917        }
918        let mut trial = self.trial.lock().unwrap_or_else(PoisonError::into_inner);
919        trial.generation += 1;
920        trial.pending = None;
921        let Some(after) = revert_after else {
922            return Ok(0);
923        };
924        let reverts_at_ms = now_ms() + u64::try_from(after.as_millis()).unwrap_or(u64::MAX);
925        trial.pending = Some((reverts_at_ms, before));
926        let generation = trial.generation;
927        drop(trial);
928        let daemon = Arc::clone(self);
929        thread::Builder::new()
930            .name("system-trial".to_owned())
931            .spawn(move || {
932                thread::sleep(after);
933                daemon.revert_system(generation);
934            })?;
935        Ok(reverts_at_ms)
936    }

938    /// Reverts trial `generation` if it is still waiting.
939    fn revert_system(self: &Arc<Self>, generation: u64) {
940        let previous = {
941            let mut trial = self.trial.lock().unwrap_or_else(PoisonError::into_inner);
942            if trial.generation != generation {
943                return;
944            }
945            trial.pending.take()
946        };
947        if let Some((_, previous)) = previous {
948            warn!("the system firewall change wasn't confirmed in time: reverting it");
949            if let Err(error) = self.set_system(&previous, "daemon (not confirmed)", None) {
950                error!(%error, "reverting the system firewall");
951            }
952        }
953    }

955    /// Confirms the change on trial; false when none waits.
956    pub fn confirm_system(&self) -> bool {
957        let mut trial = self.trial.lock().unwrap_or_else(PoisonError::into_inner);
958        trial.generation += 1;
959        trial.pending.take().is_some()
960    }

998    /// Stops, and takes its tables out, unless it stops cleanly (`clean`) with `fail = closed`: then they stay, so
999    /// nothing new gets out until the daemon is back. Serving that ended in an error always takes them out.
1000    pub fn stop(&self, clean: bool) {
1001        // Clients are told before the connections end, so a planned stop isn't taken for a crash.
1002        let reason = if shutting_down() {
1003            SHUTDOWN
1004        } else {
1005            "the daemon was stopped (by an administrator, or for an upgrade)"
1006        };
1007        self.shared.router.push.stopping(reason);
1008        self.shared.hub.publish(
1009            v1::Topic::Prompts,
1010            &v1::notice::Body::Stopping(v1::Stopping {
1011                reason: reason.to_owned(),
1012            }),
1013        );
1014        // A moment for the notices to leave.
1015        thread::sleep(Duration::from_millis(300));
1016        let _ = std::fs::remove_file(config_marker(&self.shared.config));
1017        if self.shared.config.fail == Fail::Open || !clean {
1018            remove_tables(&self.table.name);
1019        } else {
1020            info!("fail = closed: the tables stay until the daemon is back");
1021        }
1022        info!(
1023            decided = self.shared.decided.load(Ordering::Relaxed),
1024            dropped_events = self.shared.dropped_events.load(Ordering::Relaxed),
1025            "stopped"
1026        );
1027    }

1091/// The daemon's own cgroup, when it has one to itself: the service's (`…/snitchhikerd.service`), not a session's or a

1093fn own_cgroup() -> Option<snitchhiker_netfilter::nftables::Cgroup> {

1099    }

1101        .ok()?
1102        .ino();
1103    let level = u32::try_from(path.split('/').filter(|part| !part.is_empty()).count()).ok()?;
1104    Some(snitchhiker_netfilter::nftables::Cgroup { level, id })
1105}

1107/// The deciding queues, bound, with their verdict writers.
1108fn bind_queues(
1109    config: &Config,
1110) -> Result<(Vec<Queue>, Vec<snitchhiker_netfilter::queue::VerdictWriter>), Error> {
1111    let mut queues = Vec::with_capacity(usize::from(config.queues));
1112    let mut writers = Vec::with_capacity(usize::from(config.queues));
1113    for offset in 0..config.queues {
1114        let number = config.queue + offset;
1115        let queue = Queue::bind(number).map_err(queue_error(number))?;
1116        writers.push(queue.verdict_writer()?);
1117        queues.push(queue);
1118    }
1119    Ok((queues, writers))
1120}

1122/// The error of binding NFQUEUE `number`. The kernel refuses a queue another process has bound with EPERM, which
1123/// says nothing on its own: the error names the queue and the likely holder.
1124fn queue_error(number: u16) -> impl Fn(std::io::Error) -> Error {
1125    move |error| {
1126        if error.raw_os_error() == Some(nix::libc::EPERM) {
1127            Error::QueueTaken(number)
1128        } else {
1129            Error::Io(error)
1130        }
1131    }
1132}

1203/// Learns names from DNS answers: only those to sockets that sent a query to that resolver, and remembered for the
1204/// process that sent it. Every packet passes; this queue only observes.
1205fn dns_loop(shared: &Shared, mut queue: Queue) -> Result<(), Error> {
1206    let mut ids: Vec<u32> = Vec::with_capacity(64);
1207    loop {
1208        ids.clear();
1209        for queued in queue.receive()? {
1210            ids.push(queued.id);
1211            let Some(headers) = packet::parse(queued.payload) else {
1212                continue;
1213            };
1214            let Some(answer) = packet::udp_payload(queued.payload, &headers).and_then(parse_answer)
1215            else {
1216                continue;
1217            };
1218            let query = FlowKey {
1219                netns: shared.netns,
1220                protocol: packet::IPPROTO_UDP,
1221                source_port: headers.destination_port,
1222                destination: headers.source,
1223                destination_port: DNS_PORT,
1224            };
1225            // An answer nobody asked for teaches nothing.
1226            let Some(asker) = shared.attribution.owner(&query) else {
1227                continue;
1228            };
1229            shared
1230                .names
1231                .write()
1232                .unwrap_or_else(PoisonError::into_inner)
1233                .learn(asker.pid, &answer, now_ms());
1234        }
1235        for &id in &ids {
1236            queue.verdict(id, Verdict::Accept)?;
1237        }
1238    }
1239}

1241fn decide_loop(shared: &Shared, mut queue: Queue, offset: usize) -> Result<(), Error> {
1242    let mut processes =
1243        ProcessCache::with_packages(shared.host, Some(Arc::clone(&shared.packages)));
1244    let mut senders = crate::forward::Senders::default();
1245    let mut verdicts: Vec<(u32, Verdict)> = Vec::with_capacity(64);
1246    let mut rejected: Vec<SocketId> = Vec::with_capacity(8);
1247    let mut sockdiag: Option<SockDiag> = None;
1248    loop {
1249        verdicts.clear();
1250        rejected.clear();
1251        for packet in queue.receive()? {
1252            let (verdict, reject) = decide(
1253                shared,
1254                &mut processes,
1255                &mut senders,
1256                &packet,
1257                (offset, shared.netns),
1258            );
1259            // No verdict: the packet waits for a prompt's answer.
1260            if let Some(verdict) = verdict {
1261                verdicts.push((packet.id, verdict));
1262            }
1263            rejected.extend(reject);
1264        }
1265        for &(id, verdict) in &verdicts {
1266            queue.verdict(id, verdict)?;
1267        }
1268        // After the drop: the program's socket is closed, so it fails at once instead of retrying.
1269        for id in &rejected {
1270            if sockdiag.is_none() {
1271                sockdiag = SockDiag::open()
1272                    .inspect_err(|error| warn!(%error, "sock_diag: rejected connections will time out instead"))
1273                    .ok();
1274            }
1275            if let Some(diag) = sockdiag.as_mut()
1276                && let Err(error) = diag.destroy(id)
1277                && error.kind() != std::io::ErrorKind::NotFound
1278            {
1279                warn!(%error, "closing a rejected socket");
1280            }
1281        }
1282    }
1283}

1390/// The verdict for one packet (none: it is held for a prompt), recording what happened, and the socket to close for a
1391/// rejected connection.
1392/// The context a connection comes from: the process's own (its cgroup names its container), or, for a rootless
1393/// container's user-mode network helper, the container it connects for (`true`: the program in it is unknown).
1394fn context_of(
1395    shared: &Shared,
1396    owner: Option<snitchhiker_ebpf::FlowOwner>,
1397    process: Option<&crate::process::ProcessInfo>,
1398) -> (ContextKind, Option<Arc<crate::contexts::Identified>>, bool) {
1399    let (Some(owner), Some(process)) = (owner, process) else {
1400        return (ContextKind::Host, None, false);
1401    };
1402    // A helper makes its connections from the host's network namespace (rootless, it runs in the runtime's user
1403    // namespace): a program of that name inside a container is no helper.
1404    if crate::contexts::is_network_helper(&process.exe)
1405        && std::fs::metadata(format!("/proc/{}/ns/net", owner.pid))
1406            .is_ok_and(|m| u32::try_from(m.ino()).ok() == Some(shared.netns))
1407        && let Some(served) =
1408            shared
1409                .contexts
1410                .served_by(owner.pid, owner.start_boottime, &process.exe)
1411    {
1412        return (ContextKind::Container, Some(served), true);
1413    }
1414    let container = (process.context == ContextKind::Container)
1415        .then(|| shared.contexts.container(owner.pid, owner.cgroup))
1416        .flatten();
1417    (process.context, container, false)
1418}

1420/// The netfilter hook of forwarded packets.
1421const NF_INET_FORWARD: u8 = 2;
1422/// `NF_INET_LOCAL_OUT`: the netfilter hook of packets this host sends.
1423const NF_INET_LOCAL_OUT: u8 = 3;
1424
1425/// The owner of a forwarded packet's flow, as its sender's namespace recorded it (see [`crate::forward`]).
1426fn forwarded_owner(
1427    shared: &Shared,
1428    senders: &mut crate::forward::Senders,
1429    queued: &Packet<'_>,
1430    key: &FlowKey,
1431) -> Owner {
1432    let (Some(bridge), Some(mac)) = (queued.in_interface, queued.source_mac) else {
1433        debug!(in_interface = ?queued.in_interface, mac = ?queued.source_mac, "a forwarded packet without its sender's link");
1434        return Owner::Unknown;
1435    };
1436    for fresh in [false, true] {
1437        let sender = senders.sender(bridge, mac, fresh);
1438        debug!(bridge, ?mac, ?sender, fresh, "a forwarded packet's sender");
1439        match sender {
1440            Some(crate::forward::Sender::Netns(netns)) => {
1441                if let Some(owner) = shared.attribution.owner(&FlowKey { netns, ..*key }) {
1442                    return Owner::Process(owner);
1443                }
1444            }
1445            Some(crate::forward::Sender::Tap(tap)) => return Owner::Tap(tap),
1446            None => {}
1447        }
1448    }
1449    Owner::Unknown
1450}

1452/// Who sent a queued packet.
1453enum Owner {
1454    /// The process that opened its flow.
1455    Process(snitchhiker_ebpf::FlowOwner),
1456    /// A VM, through this tap.
1457    Tap(Arc<str>),
1458    Unknown,
1459}
1460
1461/// Who opened a queued packet's flow: on this host, or, for a forwarded packet, in its sender's namespace.
1462fn owner_of(
1463    shared: &Shared,
1464    senders: &mut crate::forward::Senders,
1465    queued: &Packet<'_>,
1466    (key, source): (&FlowKey, std::net::IpAddr),
1467) -> Owner {
1468    if queued.hook == NF_INET_FORWARD {
1469        forwarded_owner(shared, senders, queued, key)
1470    } else {
1471        // A flow the programs didn't record: its socket, if exactly one process holds it.
1472        shared
1473            .attribution
1474            .owner(key)
1475            .or_else(|| {
1476                crate::fallback::owner(
1477                    key.protocol,
1478                    (source, key.source_port),
1479                    (key.destination, key.destination_port),
1480                )
1481            })
1482            .map_or(Owner::Unknown, Owner::Process)
1483    }
1484}

1486/// Packets never decided, sent by `pid` (when a program was found):
1487///
1488/// - the daemon's own traffic, as marked packets aren't queued: sockets a library opened, which it can't mark, but the
1489///   kernel names their process;
1490/// - the node's remote service's (`snitchhiker-remote`, `remote_pid` while it is connected: the relay, devices, the
1491///   push gateway), the node's own as well: a prompt about it couldn't reach the devices it would cut off;
1492/// - ICMP and IGMP the kernel itself sends (its errors about a flow it has, such as "port unreachable"; its multicast
1493///   membership reports): no socket a program opened sent it (the queue names the owner of any socket backed by a
1494///   file, raw and ping sockets included), so there is no program to decide about or ask.
1495fn never_decided(shared: &Shared, pid: Option<u32>, queued: &Packet<'_>, protocol: u8) -> bool {
1496    match pid {
1497        Some(pid) => {
1498            pid == std::process::id()
1499                || (pid != 0 && pid == shared.remote_pid.load(Ordering::Relaxed))
1500        }
1501        None => {
1502            queued.hook == NF_INET_LOCAL_OUT
1503                && queued.uid.is_none()
1504                && matches!(
1505                    protocol,
1506                    packet::IPPROTO_ICMP | packet::IPPROTO_ICMPV6 | packet::IPPROTO_IGMP
1507                )
1508        }
1509    }
1510}

1512pub(crate) fn decide(
1513    shared: &Shared,
1514    processes: &mut ProcessCache,
1515    senders: &mut crate::forward::Senders,
1516    queued: &Packet<'_>,
1517    (queue, netns): (usize, u32),
1518) -> (Option<Verdict>, Option<SocketId>) {
1519    shared.decided.fetch_add(1, Ordering::Relaxed);
1520    let Some(headers) = packet::parse(queued.payload) else {
1521        // Not IPv4 or IPv6: nothing rules speak about.
1522        return (Some(Verdict::Accept), None);
1523    };
1524    let key = FlowKey {
1525        netns,
1526        protocol: headers.protocol,
1527        source_port: headers.source_port,
1528        destination: headers.destination,
1529        destination_port: headers.destination_port,
1530    };
1531    let (owner, vm) = match owner_of(shared, senders, queued, (&key, headers.source)) {
1532        Owner::Process(owner) => (Some(owner), None),
1533        Owner::Tap(tap) => (None, shared.contexts.vm_on_tap(&tap)),
1534        Owner::Unknown => (None, None),
1535    };
1536    if never_decided(shared, owner.map(|o| o.pid), queued, headers.protocol) {
1537        return (Some(Verdict::Accept), None);
1538    }
1539    let process = owner.and_then(|owner| processes.get(owner.pid, owner.start_boottime));
1540    let now = now_ms();
1541    // The user as the kernel recorded it when the socket was opened (the queue's uid is the socket's owner, for
1542    // packets the attribution programs didn't see).
1543    let uid = owner.map(|o| o.uid).or(queued.uid).unwrap_or(u32::MAX);
1544    let home = process.as_ref().and_then(|_| processes.home(uid));
1545    let pid = owner.map_or(0, |o| o.pid);
1546    let hosts = shared
1547        .names
1548        .read()
1549        .unwrap_or_else(PoisonError::into_inner)
1550        .lookup(pid, headers.destination, now);
1551    let empty: &[String] = &[];
1552    let (context, container, helper) = if vm.is_some() {
1553        (ContextKind::Vm, None, false)
1554    } else {
1555        context_of(shared, owner, process.as_deref())
1556    };
1557    // A helper's connection for a namespace decided inside was decided there already, with its program: deciding it
1558    // again here, knowing less, could only contradict that. Until then, the daemon starts deciding inside.
1559    if helper && let Some((pid, inode)) = container.as_ref().and_then(|c| c.served) {
1560        if shared.namespaces.is_managed(inode) {
1561            return (Some(Verdict::Accept), None);
1562        }
1563        shared.namespaces.request(pid, inode);
1564    }
1565    // A network helper isn't the program: the one in the container is, unknown from out here.
1566    let process = if helper { None } else { process };
1567    // A named namespace's connections (queued inside it) carry its name.
1568    let named = (netns != shared.netns)
1569        .then(|| shared.namespaces.named(u64::from(netns)))
1570        .flatten();
1571    // Hashed only while a rule pins a binary.
1572    let sha256 = process
1573        .as_ref()
1574        .filter(|_| shared.rules.load().needs_sha256())
1575        .and_then(|p| crate::process::sha256_of(p.pid));
1576    if let Some(found) = container.as_deref() {
1577        note_net_admin(shared, found);
1578    }
1579    let facts = Facts {
1580        exe: process.as_ref().map_or("", |p| &*p.exe),
1581        apps: process.as_ref().map_or(&[], |p| p.apps.as_slice()),
1582        sha256: sha256.as_deref(),
1583        writable: process.as_ref().is_some_and(|p| p.writable),
1584        cmdline: process.as_ref().map_or("", |p| p.cmdline.as_str()),
1585        ancestors: process.as_ref().map_or(empty, |p| p.ancestors.as_slice()),
1586        lineage: process.as_ref().map_or(&[], |p| p.lineage.as_slice()),
1587        uid,
1588        home: home.as_deref(),
1589        hosts: &hosts,
1590        address: headers.destination,
1591        port: headers.destination_port,
1592        protocol: headers.protocol,
1593        context,
1594        now_ms: now,
1595        container: container.as_deref().map(|c| &c.container),
1596        vm: vm.as_deref().map(|v| &v.vm),
1597        netns: named.as_ref().map(|(name, _)| &**name),
1598    };
1599    let exe = process
1600        .as_ref()
1601        .map_or_else(|| Arc::clone(&shared.unknown), |p| Arc::clone(&p.exe));
1602    let event = undecided_event(
1603        &facts,
1604        exe,
1605        pid,
1606        owner.map(|o| o.cookie),
1607        container
1608            .as_ref()
1609            .map(|c| Arc::clone(&c.key))
1610            .or_else(|| named.as_ref().map(|(_, key)| Arc::clone(key)))
1611            .or_else(|| vm.as_ref().map(|v| Arc::clone(&v.key))),
1612    );
1613    let rules = shared.rules.load();
1614    let (action, source, rule) = if let Some(decision) = rules.decide(&facts) {
1615        (decision.action, Source::Rule, Some(decision.rule))
1616    } else {
1617        let held = Held {
1618            queue,
1619            id: queued.id,
1620        };
1621        match undecided(shared, &facts, &event, held) {
1622            Some(decided) => decided,
1623            None => return (None, None),
1624        }
1625    };
1626    conclude(shared, &facts, event, &headers, (action, source, rule))
1627}

1629/// A decided connection: its verdict by the mode, the history's record, and the socket to close for a rejection.
1630fn conclude(
1631    shared: &Shared,
1632    facts: &Facts<'_>,
1633    mut event: Event,
1634    headers: &packet::Headers,
1635    (action, source, rule): (Action, Source, Option<RuleId>),
1636) -> (Option<Verdict>, Option<SocketId>) {
1637    let (verdict, source) = match shared.policy.load().mode {
1638        Mode::Monitor => (Verdict::Accept, Source::Monitor),
1639        Mode::Enforce => (
1640            if action == Action::Allow {
1641                Verdict::Accept
1642            } else {
1643                Verdict::Drop
1644            },
1645            source,
1646        ),
1647    };
1648    if action != Action::Allow {
1649        remember(shared, facts, &event, None);
1650    }
1651    event.action = action;
1652    event.source = source;
1653    event.rule = rule;
1654    if let Some(rule) = rule {
1655        shared.blocklists.decided(rule);
1656    }
1657    if let Err(TrySendError::Full(_) | TrySendError::Disconnected(_)) =
1658        shared.events.try_send(event)
1659    {
1660        shared.dropped_events.fetch_add(1, Ordering::Relaxed);
1661    }
1662    let reject = (verdict == Verdict::Drop && action == Action::Reject).then_some(SocketId {
1663        protocol: headers.protocol,
1664        local: headers.source,
1665        local_port: headers.source_port,
1666        remote: headers.destination,
1667        remote_port: headers.destination_port,
1668    });
1669    (Some(verdict), reject)
1670}

1691/// A connection no rule decides: held for a prompt (`None`), or decided by the policy (by `on_timeout` when asking
1692/// but too many prompts are open).
1693fn undecided(
1694    shared: &Shared,
1695    facts: &Facts<'_>,
1696    event: &Event,
1697    held: Held,
1698) -> Option<(Action, Source, Option<RuleId>)> {
1699    let asking = asks(shared);
1700    if asking && let Some(prompt) = hold(shared, facts, event, held) {
1701        remember(shared, facts, event, Some(prompt));
1702        return None;
1703    }
1704    let policy = shared.policy.load();
1705    Some(if asking {
1706        (policy.on_timeout, Source::Timeout, None)
1707    } else {
1708        let action = policy_action(policy.no_match, policy.no_client);
1709        (action, Source::Policy, None)
1710    })
1711}

1713/// Holds a packet for the prompt of its program and destination, opening (and announcing) one if none is open; the
1714/// prompt's id, or `None` when the broker is full (see [`crate::prompts::MAX_PROMPTS`]).
1715fn hold(shared: &Shared, facts: &Facts<'_>, event: &Event, packet: Held) -> Option<u64> {
1716    let destination = facts
1717        .hosts
1718        .first()
1719        .map_or_else(|| facts.address.to_string(), ToString::to_string);
1720    let key = PromptKey {
1721        exe: Arc::clone(&event.exe),
1722        uid: facts.uid,
1723        destination,
1724        port: facts.port,
1725        protocol: facts.protocol,
1726        container: event.container.clone(),
1727    };
1728    let opened = shared.broker.hold(key, packet, facts.now_ms, || {
1729        let suggestions = snitchhiker_core::suggest::in_context(
1730            snitchhiker_core::suggest::suggestions(
1731                facts.exe,
1732                facts.home,
1733                facts.hosts,
1734                facts.address,
1735                facts.port,
1736                facts.protocol,
1737            ),
1738            facts.context,
1739            facts.container,
1740            facts.vm,
1741        );
1742        NewPrompt {
1743            prompt: v1::Prompt {
1744                exe: facts.exe.to_owned(),
1745                cmdline: facts.cmdline.to_owned(),
1746                uid: facts.uid,
1747                user: crate::process::user_of(facts.uid).unwrap_or_default(),
1748                pid: event.pid,
1749                hosts: facts.hosts.iter().map(ToString::to_string).collect(),
1750                address: convert::address(facts.address),
1751                port: u32::from(facts.port),
1752                protocol: u32::from(facts.protocol),
1753                context: convert::context(facts.context).into(),
1754                // As found (with what it can do), else as the history keeps it.
1755                container: facts
1756                    .container
1757                    .map(|c| Box::new(convert::container(c)))
1758                    .or_else(|| {
1759                        event
1760                            .container
1761                            .as_deref()
1762                            .map(|key| Box::new(convert::container_from_key(key)))
1763                    }),
1764                exe_writable: crate::process::writable_by_others(event.pid),
1765                suggestions: suggestions
1766                    .iter()
1767                    .map(|s| v1::Suggestion {
1768                        label: s.label.clone(),
1769                        conditions: Some(convert::conditions(&s.conditions)),
1770                        broad: s.broad,
1771                    })
1772                    .collect(),
1773                ..v1::Prompt::default()
1774            },
1775            suggestions,
1776            event: event.clone(),
1777        }
1778    });
1779    match opened {
1780        Hold::Opened(prompt) => {
1781            shared.router.opened(&shared.hub, &prompt);
1782            Some(prompt.id)
1783        }
1784        Hold::Joined(id) => Some(id),
1785        Hold::Full => None,
1786    }
1787}

1789/// Pushes recorded events to the subscribers of the events topic: one message each to those that want messages, and
1790/// the batch as one columnar block to those that want blocks.
1791fn publish_events(shared: &Shared, batch: &[Event], recorded: &[crate::store::Recorded]) {
1792    let (messages, blocks) = shared.hub.wanted(v1::Topic::Events);
1793    let plain: Vec<Outgoing> = if messages {
1794        batch
1795            .iter()
1796            .zip(recorded)
1797            .map(|(event, r)| {
1798                Outgoing::Notice(v1::notice::Body::Event(crate::api::event(
1799                    &event.stored(r.seq),
1800                )))
1801            })
1802            .collect()
1803    } else {
1804        Vec::new()
1805    };
1806    let block = if blocks {
1807        let rows: Vec<EventRow> = batch
1808            .iter()
1809            .zip(recorded)
1810            .map(|(e, r)| event_row(e, r))
1811            .collect();
1812        let mut strings: Vec<(u32, Arc<str>)> = Vec::new();
1813        for (event, r) in batch.iter().zip(recorded) {
1814            strings.push((r.exe, Arc::clone(&event.exe)));
1815            if let Some(host) = &event.host {
1816                strings.push((r.host, Arc::clone(host)));
1817            }
1818            if let Some(container) = &event.container {
1819                strings.push((r.container, Arc::clone(container)));
1820            }
1821        }
1822        let mut columns = Vec::new();
1823        match columns::encode_events(&rows, &mut columns) {
1824            Ok(()) => Some(Outgoing::Block {
1825                columns: bytes::Bytes::from(columns),
1826                strings: Arc::from(strings),
1827            }),
1828            Err(error) => {
1829                error!(%error, "encoding an event block");
1830                None
1831            }
1832        }
1833    } else {
1834        None
1835    };
1836    if messages || block.is_some() {
1837        shared
1838            .hub
1839            .publish_all(v1::Topic::Events, &plain, block.as_ref());
1840    }
1841}

1843/// A recorded event as a block's row: the protocol's numbers, the strings' ids.
1844fn event_row(event: &Event, recorded: &crate::store::Recorded) -> EventRow {
1845    let address = match event.address {
1846        std::net::IpAddr::V4(a) => a.to_ipv6_mapped().octets(),
1847        std::net::IpAddr::V6(a) => a.octets(),
1848    };
1849    EventRow {
1850        seq: recorded.seq,
1851        time_ms: event.time_ms,
1852        exe: recorded.exe,
1853        host: recorded.host,
1854        uid: event.uid,
1855        pid: event.pid,
1856        address,
1857        port: event.port,
1858        protocol: event.protocol,
1859        context: event.context.saturating_add(1),
1860        action: u8::try_from(i32::from(convert::action(event.action))).unwrap_or(0),
1861        source: event.source as u8,
1862        rule: event.rule.unwrap_or(0),
1863        bytes_out: 0,
1864        bytes_in: 0,
1865        container: recorded.container,
1866    }
1867}

1869/// Raises an alert when the last daemon didn't stop cleanly (its marker is still there: it crashed, was killed, or the
1870/// machine lost power), then puts this one's marker in place.
1871fn check_marker(shared: &Shared, store: &Store) {
1872    let marker = config_marker(&shared.config);
1873    if marker.exists() {
1874        crate::alerts::raise(
1875            shared,
1876            store,
1877            &crate::alerts::NewAlert {
1878                kind: "crashed",
1879                title: "The daemon stopped unexpectedly".to_owned(),
1880                body: "It didn't stop cleanly last time: it crashed, was killed, or the machine lost power."
1881                    .to_owned(),
1882                rule: None,
1883            },
1884        );
1885    }
1886    let _ = std::fs::write(&marker, b"");
1887}

1894/// How often the tables are checked.
1895/// How often the counters of what was lost are checked.
1896const LOSS_CHECK_EVERY: Duration = Duration::from_secs(60);

1898/// Packets the kernel dropped on queues `first..first + count` (`/proc/net/netfilter/nfnetlink_queue`: the queue
1899/// overflowed, or the daemon's socket couldn't take them).
1900fn queue_drops(first: u16, count: u16) -> u64 {
1901    std::fs::read_to_string("/proc/net/netfilter/nfnetlink_queue").map_or(0, |text| {
1902        text.lines()
1903            .filter_map(|line| {
1904                let fields: Vec<u64> = line
1905                    .split_whitespace()
1906                    .filter_map(|f| f.parse().ok())
1907                    .collect();
1908                let queue = u16::try_from(*fields.first()?).ok()?;
1909                (queue >= first && queue < first.saturating_add(count)).then(|| {
1910                    fields.get(5).copied().unwrap_or(0) + fields.get(6).copied().unwrap_or(0)
1911                })
1912            })
1913            .sum()
1914    })
1915}

1917/// Alerts about what the node lost since the last check: events the history couldn't keep, packets the kernel
1918/// dropped.
1919fn check_losses(shared: &Shared, store: &Store, seen: &mut (u64, u64)) {
1920    let dropped = shared.dropped_events.load(Ordering::Relaxed);
1921    if dropped > seen.0 {
1922        crate::alerts::raise(
1923            shared,
1924            store,
1925            &crate::alerts::NewAlert {
1926                kind: "dropped_events",
1927                title: "Connections weren't recorded".to_owned(),
1928                body: format!(
1929                    "{} connections were decided but not kept in the history: the database couldn't keep up.",
1930                    dropped - seen.0
1931                ),
1932                rule: None,
1933            },
1934        );
1935    }
1936    let queued = queue_drops(shared.config.queue, shared.config.queues);
1937    if queued > seen.1 {
1938        crate::alerts::raise(
1939            shared,
1940            store,
1941            &crate::alerts::NewAlert {
1942                kind: "queue_overflow",
1943                title: "The kernel dropped packets waiting for a decision".to_owned(),
1944                body: format!(
1945                    "{} packets were dropped on the daemon's queues: they overflowed. New connections were refused \
1946                     while that lasted.",
1947                    queued - seen.1
1948                ),
1949                rule: None,
1950            },
1951        );
1952    }
1953    *seen = (dropped, queued.max(seen.1));
1954}

1956fn write_loop(shared: &Shared, store: &crate::locked::Locked, events: &Receiver<Event>) {
1957    let mut batch = Vec::with_capacity(BATCH);
1958    let mut engine = crate::alerts::Engine::default();
1959    let mut rules_seen = u64::MAX;
1960    let mut last_loss_check = Instant::now();
1961    let mut losses = (0, queue_drops(shared.config.queue, shared.config.queues));
1962    let mut recorded = Vec::with_capacity(BATCH);
1963    let mut last_prune = Instant::now();
1964    let mut last_bytes = Instant::now();
1965    let mut counted: HashMap<u64, (u64, u64)> = HashMap::new();
1966    let mut changed: Vec<(u64, u64, u64)> = Vec::new();
1967    loop {
1968        batch.clear();
1969        match events.recv_timeout(BYTES_EVERY) {
1970            Ok(event) => batch.push(event),
1971            Err(RecvTimeoutError::Timeout) => {}
1972            Err(RecvTimeoutError::Disconnected) => return,
1973        }
1974        let deadline = Instant::now() + BATCH_WAIT;
1975        while batch.len() < BATCH {
1976            let left = deadline.saturating_duration_since(Instant::now());
1977            match events.recv_timeout(left) {
1978                Ok(event) => batch.push(event),
1979                Err(_) => break,
1980            }
1981        }
1982        let mut store = store.lock();
1983        let generation = shared.alert_rules_changed.load(Ordering::Relaxed);
1984        if generation != rules_seen {
1985            rules_seen = generation;
1986            match store.alert_rules() {
1987                Ok(rules) => engine.load(rules),
1988                Err(error) => error!(%error, "loading the alert rules"),
1989            }
1990        }
1991        if !batch.is_empty() {
1992            match store.record(&batch, &mut recorded) {
1993                Ok(()) => {
1994                    publish_events(shared, &batch, &recorded);
1995                    for alert in engine.check(&batch, &recorded) {
1996                        crate::alerts::raise(shared, &store, &alert);
1997                    }
1998                    alert_net_admin(shared, &store);
1999                }
2000                Err(error) => error!(%error, events = batch.len(), "recording events"),
2001            }
2002        }
2003        if last_loss_check.elapsed() >= LOSS_CHECK_EVERY {
2004            last_loss_check = Instant::now();
2005            check_losses(shared, &store, &mut losses);
2006        }
2007        if last_bytes.elapsed() >= BYTES_EVERY {
2008            last_bytes = Instant::now();
2009            changed.clear();
2010            let mut live: HashMap<u64, (u64, u64)> = HashMap::with_capacity(counted.len());
2011            for (cookie, sent, read) in shared.attribution.socket_bytes() {
2012                if counted.get(&cookie) != Some(&(sent, read)) {
2013                    changed.push((cookie, sent, read));
2014                }
2015                live.insert(cookie, (sent, read));
2016            }
2017            counted = live;
2018            if !changed.is_empty()
2019                && let Err(error) = store.record_bytes(&changed)
2020            {
2021                error!(%error, "recording byte counts");
2022            }
2023        }
2024        if last_prune.elapsed() >= PRUNE_EVERY {
2025            last_prune = Instant::now();
2026            let retention_ms = u64::from(shared.policy.load().retention_days) * 86_400_000;
2027            let cutoff = now_ms().saturating_sub(retention_ms);
2028            while let Ok(pruned) = store.prune(cutoff, PRUNE_CHUNK) {
2029                if pruned < PRUNE_CHUNK as usize {
2030                    break;
2031                }
2032            }
2033            if let Err(error) = store.prune_alerts(cutoff) {
2034                error!(%error, "pruning alerts");
2035            }
2036            if let Err(error) = store.prune_proposals(cutoff) {
2037                error!(%error, "pruning proposals");
2038            }
2039            if let Err(error) = store.prune_rollups(cutoff) {
2040                error!(%error, "pruning rollups");
2041            }
2042        }
2043    }
