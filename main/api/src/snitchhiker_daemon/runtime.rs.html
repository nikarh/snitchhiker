1//! The running daemon: the table, the attribution programs, one deciding thread per queue, and the writer.
2//!
3//! The packet path: receive → parse the headers → look the flow's owner up (eBPF) → the process's facts (cached) →
4//! the rules (an immutable set, swapped atomically on change) → verdict, and an event to the writer's queue. Nothing
5//! on it waits for the database, and a full event queue drops the event (counted), never the verdict.
6

39/// Events waiting for the writer; beyond this they are dropped (and counted).
40const EVENT_QUEUE: usize = 65_536;
41/// Events written per transaction at most.
42const BATCH: usize = 1024;
43/// How long the writer waits to fill a batch.
44const BATCH_WAIT: Duration = Duration::from_millis(100);
45/// How much of a DNS answer is copied: a whole UDP answer, as large as resolvers send without EDNS tricks.
46const DNS_COPY: u32 = 4096;
47const DNS_PORT: u16 = 53;
48/// How often byte counts are written to the history.
49const BYTES_EVERY: Duration = Duration::from_secs(5);
50/// How often old history is pruned, and how many rows per chunk.
51const PRUNE_EVERY: Duration = Duration::from_secs(600);
52const PRUNE_CHUNK: u32 = 10_000;

68/// Why an answer to a prompt was refused.
69#[derive(Debug, thiserror::Error)]
70pub enum AnswerError {
71    #[error("no such prompt (answered already, or timed out)")]
72    NotFound,
73    #[error("{0}")]

75    #[error(transparent)]

80    match error {

137    /// Opens the store, compiles the rules, installs the table, loads the attribution programs and starts the
138    /// deciding threads and the writer.
139    pub fn start(config: Config) -> Result<Arc<Self>, Error> {
140        if let Some(parent) = config.database.parent() {
141            std::fs::create_dir_all(parent)?;
142        }
143        let store = Store::open(&config.database)?;
144        let (rules, mut agents) = compile(&config, &store)?;
145        info!(rules = rules.len(), "rules loaded");
146        // Agents whose proposals wait, to expire them if the agent ends.
147        agents.extend(waiting_agents(&store)?);
148        // The programs first: from the moment the table is in place, every new flow's owner must be known.
149        let attribution = Attribution::load()?;
150        let table = Table {
151            queue: config.queue,
152            queues: config.queues,
153            bypass: config.fail == Fail::Open,
154            dns_queue: config.dns_queue(),
155            ..Table::default()
156        };
157        install(&table)?;
158        info!(queues = config.queues, mode = ?config.mode, "table installed");
159
160        // Until-restart rules from before don't apply any more.
161        let mut store = store;
162        store.delete_until_restart()?;
163        let (sender, receiver) = sync_channel(EVENT_QUEUE);
164        let store = Arc::new(crate::locked::Locked::new(store));
165        let mut queues = Vec::with_capacity(usize::from(config.queues));
166        let mut writers = Vec::with_capacity(usize::from(config.queues));
167        for offset in 0..config.queues {
168            let queue = Queue::bind(config.queue + offset)?;
169            writers.push(queue.verdict_writer()?);
170            queues.push(queue);
171        }
172        let policy = Effective::of(&config, &store.lock().policy()?);
173        let broker = Broker::new(writers, Duration::from_secs(policy.prompt_timeout_secs));
174        let shared = Arc::new(Shared {
175            rules: ArcSwap::from_pointee(rules),
176            attribution,
177            names: RwLock::new(Names::default()),
178            netns: u32::try_from(std::fs::metadata("/proc/self/ns/net")?.ino()).unwrap_or(0),
179            host: HostNamespaces::read()?,
180            events: sender,
181            dropped_events: AtomicU64::new(0),
182            unknown: Arc::from(""),
183            decided: AtomicU64::new(0),
184            hub: Arc::new(Hub::default()),
185            broker,
186            alert_rules_changed: AtomicU64::new(0),
187            policy: ArcSwap::from_pointee(policy),
188            blocked: crate::agents::Ring::default(),
189            agents: crate::agents::Watch::new()?,
190            proposals_changed: (Mutex::new(0), std::sync::Condvar::new()),
191            router: crate::router::Router::new(
192                config.prompt_routing,
193                Duration::from_secs(config.escalate_after_secs),
194            ),
195            config,
196        });
197        for (offset, queue) in queues.into_iter().enumerate() {
198            let number = queue.number();
199            let shared = Arc::clone(&shared);
200            thread::Builder::new()
201                .name(format!("queue-{number}"))
202                .spawn(move || {
203                    if let Err(error) = decide_loop(&shared, queue, offset) {
204                        error!(%error, queue = number, "deciding thread stopped");
205                    }
206                })?;
207        }
208        if let Some(number) = shared.config.dns_queue() {
209            let queue = Queue::bind_copying(number, DNS_COPY)?;
210            let shared = Arc::clone(&shared);
211            thread::Builder::new()
212                .name("dns".to_owned())
213                .spawn(move || {
214                    if let Err(error) = dns_loop(&shared, queue) {
215                        error!(%error, "learning names stopped");
216                    }
217                })?;
218        }
219        {
220            let store = Arc::clone(&store);
221            let shared = Arc::clone(&shared);
222            thread::Builder::new()
223                .name("writer".to_owned())
224                .spawn(move || write_loop(&shared, &store, &receiver))?;
225        }
226        check_marker(&shared, &store.lock());
227        // The system firewall as last set: a rule the kernel refuses mustn't keep the daemon from deciding.
228        let firewall = store.lock().system_firewall()?;
229        if let Err(error) = crate::system::apply(&firewall, &shared.config) {
230            error!(%error, "installing the system firewall");
231        }
232        let daemon = Arc::new(Self {
233            shared,
234            store,
235            table,
236            trial: Mutex::new(Trial::default()),
237            remote: crate::remote::Remote::default(),
238            approvals: crate::approvals::Approvals::default(),
239        });
240        daemon.spawn_watches(agents)?;
241        Ok(daemon)
242    }

244    /// The threads that wait for something: prompts timing out, the tables, agents ending, prompts to escalate.
245    fn spawn_watches(self: &Arc<Self>, agents: HashSet<ProcessRef>) -> std::io::Result<()> {
246        self.spawn_prompt_timeouts()?;
247        self.spawn_table_watch()?;
248        self.spawn_agent_watch()?;
249        self.watch_agents(agents);
250        crate::router::Router::spawn(&self.shared)?;
251        self.devices_changed();
252        Ok(())
253    }

255    /// Tells the router which devices are muted and where each is woken, from the store.
256    pub fn devices_changed(&self) {
257        let devices = self.store().devices();
258        match devices {
259            Ok(devices) => {
260                let router = &self.shared.router;
261                router.set_muted(devices.iter().filter(|d| d.muted).map(|d| d.id.clone()));
262                router.push.set_targets(
263                    devices
264                        .iter()
265                        .filter_map(|d| {
266                            Some((d.id.clone(), crate::push::target(d.push.as_deref()?)?))
267                        })
268                        .collect(),
269                );
270            }
271            Err(error) => error!(%error, "reading the paired devices"),
272        }
273    }

275    /// Announces a proposal made or changed: to subscribers, and to requests waiting for a change.
276    pub fn publish_proposal(&self, proposal: &v1::Proposal) {
277        self.shared.hub.publish(
278            v1::Topic::Proposals,
279            &v1::notice::Body::Proposal(Box::new(proposal.clone())),
280        );
281        let (count, changed) = &self.shared.proposals_changed;
282        *count.lock().unwrap_or_else(PoisonError::into_inner) += 1;
283        changed.notify_all();
284    }

286    /// Waits until `still` says no, checking it again whenever a proposal changes, for at most `timeout`.
287    pub fn wait_for_proposals(&self, timeout: Duration, mut still: impl FnMut() -> bool) {
288        let deadline = Instant::now() + timeout;
289        let (count, changed) = &self.shared.proposals_changed;
290        let mut seen = *count.lock().unwrap_or_else(PoisonError::into_inner);
291        while still() {
292            let left = deadline.saturating_duration_since(Instant::now());
293            if left.is_zero() {
294                return;
295            }
296            let guard = count.lock().unwrap_or_else(PoisonError::into_inner);
297            let (guard, _) = changed
298                .wait_timeout_while(guard, left, |now| *now == seen)
299                .unwrap_or_else(PoisonError::into_inner);
300            seen = *guard;
301        }
302    }

304    /// Watches `agents` too: their session rules and waiting proposals end with them (at once, for those already
305    /// gone).
306    pub fn watch_agents(&self, agents: impl IntoIterator<Item = ProcessRef>) {
307        let ended: Vec<ProcessRef> = agents
308            .into_iter()
309            .filter(|a| !self.shared.agents.add(*a))
310            .collect();
311        if !ended.is_empty() {
312            self.end_agents(&ended);
313        }
314    }

316    /// Waits for watched agents to end, and ends what they left.
317    fn spawn_agent_watch(self: &Arc<Self>) -> std::io::Result<()> {
318        let daemon = Arc::clone(self);
319        thread::Builder::new()
320            .name("agent-watch".to_owned())
321            .spawn(move || {
322                loop {
323                    let ended = daemon.shared.agents.wait();
324                    if !ended.is_empty() {
325                        daemon.end_agents(&ended);
326                    }
327                }
328            })?;
329        Ok(())
330    }

332    /// Ends what the agents that exited left: their session rules are removed (one batch of the history) and their
333    /// waiting proposals expire.
334    fn end_agents(&self, ended: &[ProcessRef]) {
335        let now = now_ms();
336        let removed = {
337            let mut store = self.store();
338            let rules = store.rules();
339            rules.and_then(|rules| {
340                let gone: Vec<RuleId> = rules
341                    .iter()
342                    .filter(|r| {
343                        crate::agents::sessions(r).is_some_and(|roots| {
344                            roots
345                                .iter()
346                                .all(|a| ended.contains(a) || !crate::agents::alive(*a))
347                        })
348                    })
349                    .map(|r| r.id)
350                    .collect();
351                if gone.is_empty() {
352                    return Ok(0);
353                }
354                let batch = store.batch("snitchhikerd (an agent's session ended)", now)?;
355                store.atomically(|store| {
356                    for id in &gone {
357                        store.remove_rule(*id, &batch)?;
358                    }
359                    Ok(gone.len())
360                })
361            })
362        };
363        match removed {
364            Ok(0) => {}
365            Ok(count) => {
366                info!(count, "an agent ended: its session's rules are removed");
367                if let Err(error) = self.reload_rules() {
368                    error!(%error, "reloading the rules");
369                }
370            }
371            Err(error) => error!(%error, "removing an ended agent's rules"),
372        }
373        let pending = self
374            .store()
375            .proposals(Some(v1::ProposalState::Pending), 1000);
376        for mut proposal in pending.unwrap_or_default() {
377            let Some(p) = &proposal.provenance else {
378                continue;
379            };
380            let agent = ProcessRef {
381                pid: p.agent_pid,
382                start: p.agent_start,
383            };
384            if !ended.contains(&agent) {
385                continue;
386            }
387            proposal.state = v1::ProposalState::Expired.into();
388            proposal.decided_ms = now;
389            let updated = self.store().update_proposal(&proposal);
390            match updated {
391                Ok(()) => self.publish_proposal(&proposal),
392                Err(error) => error!(%error, "expiring a proposal"),
393            }
394        }
395    }

397    /// Every few seconds, checks that the tables are still there: one removed by someone else is put back, with an alert.
398    fn spawn_table_watch(self: &Arc<Self>) -> std::io::Result<()> {
399        let daemon = Arc::clone(self);
400        thread::Builder::new()
401            .name("table-watch".to_owned())
402            .spawn(move || {
403                loop {
404                    thread::sleep(TABLE_WATCH_EVERY);
405                    daemon.check_tables();
406                }
407            })?;
408        Ok(())
409    }

411    fn check_tables(&self) {
412        let missing = |name: &str| {
413            let mut writer = Writer::new();
414            let sequence = encode_get_table(&mut writer, name, 1);
415            Socket::open()
416                .and_then(|mut socket| socket.transact(&writer, sequence, sequence))
417                .is_err_and(|e| e.raw_os_error() == Some(2))
418        };
419        if missing(&self.table.name) {
420            let put_back = install(&self.table);
421            self.raise_table_removed("the daemon's", &put_back);
422        }
423        let firewall = self.store().system_firewall();
424        if firewall.as_ref().is_ok_and(|f| f.enabled) && missing(crate::system::TABLE) {
425            let put_back = self
426                .apply_system()
427                .map_err(|e| std::io::Error::other(e.to_string()));
428            self.raise_table_removed("the system firewall's", &put_back);
429        }
430    }

464    fn spawn_prompt_timeouts(self: &Arc<Self>) -> std::io::Result<()> {

469                loop {

474                }

491    /// The live TCP connections and connected UDP sockets, each with its program as the attribution programs
492    /// recorded it when it opened (unknown for inbound sockets, and for ones older than the daemon).
493    pub fn connections(&self) -> std::io::Result<Vec<v1::LiveConnection>> {
494        let sockets = SockDiag::open()?.list()?;
495        let bytes: HashMap<u64, (u64, u64)> = self
496            .shared
497            .attribution
498            .socket_bytes()
499            .map(|(cookie, sent, read)| (cookie, (sent, read)))
500            .collect();
501        let mut processes = ProcessCache::new(self.shared.host);
502        let names = self
503            .shared
504            .names
505            .read()
506            .unwrap_or_else(PoisonError::into_inner);
507        let now = now_ms();
508        let plain = |address: std::net::IpAddr| match address {
509            std::net::IpAddr::V6(v6) => v6.to_ipv4_mapped().map_or(address, std::net::IpAddr::V4),
510            v4 @ std::net::IpAddr::V4(_) => v4,
511        };
512        Ok(sockets
513            .iter()
514            .map(|socket| {
515                let remote = plain(socket.id.remote);
516                let key = FlowKey {
517                    netns: self.shared.netns,
518                    protocol: socket.id.protocol,
519                    source_port: socket.id.local_port,
520                    destination: remote,
521                    destination_port: socket.id.remote_port,
522                };
523                let owner = self.shared.attribution.owner(&key);
524                let process = owner.and_then(|o| processes.get(o.pid, o.start_boottime));
525                let pid = owner.map_or(0, |o| o.pid);
526                let (sent, read) = bytes.get(&socket.cookie).copied().unwrap_or_default();
527                v1::LiveConnection {
528                    protocol: u32::from(socket.id.protocol),
529                    local: convert::address(plain(socket.id.local)),
530                    local_port: u32::from(socket.id.local_port),
531                    remote: convert::address(remote),
532                    remote_port: u32::from(socket.id.remote_port),
533                    state: snitchhiker_netfilter::sockdiag::state_name(
534                        socket.id.protocol,
535                        socket.state,
536                    )
537                    .to_owned(),
538                    uid: owner.map_or(socket.uid, |o| o.uid),
539                    pid,
540                    exe: process
541                        .as_ref()
542                        .map_or_else(String::new, |p| p.exe.to_string()),
543                    host: names
544                        .lookup(pid, remote, now)
545                        .first()
546                        .map_or_else(String::new, ToString::to_string),
547                    bytes_out: sent,
548                    bytes_in: read,
549                    context: convert::context(
550                        process.as_ref().map_or(ContextKind::Host, |p| p.context),
551                    )
552                    .into(),
553                }
554            })
555            .collect())
556    }

558    /// Why the recorded connection `seq` is decided as it is, by today's rules; `None` if there is no such event.
559    pub fn explain(&self, seq: i64) -> Result<Option<v1::Explanation>, Error> {
560        let Some(stored) = self.store().event(seq)? else {
561            return Ok(None);
562        };
563        let rules = self.shared.rules.load();
564        let hosts: Vec<Arc<str>> = stored.host.iter().map(|h| Arc::from(h.as_str())).collect();
565        let home = crate::process::home_of(stored.uid);
566        let facts = Facts {
567            exe: &stored.exe,
568            apps: &[],
569            cmdline: "",
570            ancestors: &[],
571            lineage: &[],
572            uid: stored.uid,
573            home: home.as_deref(),
574            hosts: &hosts,
575            address: stored.address,
576            port: stored.port,
577            protocol: stored.protocol,
578            context: match stored.context {
579                1 => ContextKind::Container,
580                2 => ContextKind::Vm,
581                3 => ContextKind::Netns,
582                _ => ContextKind::Host,
583            },
584            now_ms: now_ms(),
585        };
586        let matches = rules.explain(&facts);
587        // Names and whether a match depends on what the history doesn't keep.
588        let store = self.store();
589        let managed = crate::managed::Managed::load(
590            self.shared.config.default_rules,
591            &self.shared.config.rules_dir,
592        )
593        .map_err(Error::from)?;
594        // Rules on command lines or parents can't be checked against the history: their absence from the matches says
595        // nothing.
596        let needs_more = |r: &Rule| {
597            r.enabled
598                && r.conditions
599                    .iter()
600                    .any(|c| matches!(c, Condition::Cmdline(_) | Condition::Parent(_)))
601        };
602        let partial = store.rules()?.iter().any(needs_more) || managed.rules.iter().any(needs_more);
603        let mut explained = Vec::new();
604        for m in &matches {
605            let rule = match store.rule(m.rule)? {
606                Some(rule) => Some(rule),
607                None => managed.rules.iter().find(|r| r.id == m.rule).cloned(),
608            };
609            explained.push(v1::ExplainedRule {
610                id: m.rule,
611                name: rule.map_or_else(String::new, |r| r.name),
612                action: convert::action(m.action).into(),
613                tier: convert::tier(m.tier).into(),
614                specificity: m.specificity,
615            });
616        }
617        Ok(Some(v1::Explanation {
618            event: Some(crate::api::event(&stored)),
619            rules: explained,
620            partial,
621        }))
622    }

629    /// Answers a prompt: makes a rule of the chosen suggestion (or `conditions`) unless `duration` is once, then
630    /// decides every connection waiting on it. Returns the rule made, if any.
631    pub fn answer_prompt(
632        &self,
633        id: u64,
634        action: Action,
635        duration: v1::Duration,
636        suggestion: usize,
637        conditions: Option<&v1::Conditions>,
638        folder: Option<FolderId>,
639        by: &str,
640    ) -> Result<Option<RuleId>, AnswerError> {
641        let now = now_ms();
642        let expiry = match duration {
643            v1::Duration::Once => None,
644            v1::Duration::UntilRestart => Some(Expiry::UntilRestart),
645            v1::Duration::Hour => Some(Expiry::At(now + 3_600_000)),
646            v1::Duration::HalfDay => Some(Expiry::At(now + 43_200_000)),
647            v1::Duration::Forever | v1::Duration::Unspecified => Some(Expiry::Never),
648        };
649        // The rule is checked before the prompt is taken: a refused answer leaves it waiting.
650        let rule = match expiry {
651            None => None,
652            Some(expiry) => Some(
653                self.shared
654                    .broker
655                    .peek(id, |pending| -> Result<(Rule, String), AnswerError> {
656                        let (label, spec) = match conditions {
657                            Some(conditions) => (
658                                "Answer to a prompt".to_owned(),
659                                convert::conditions_from(conditions),
660                            ),
661                            None => pending
662                                .suggestions
663                                .get(suggestion)
664                                .map(|s| (s.label.clone(), s.conditions.clone()))
665                                .ok_or_else(|| {
666                                    AnswerError::Invalid(format!("no suggestion {suggestion}"))
667                                })?,
668                        };
669                        let conditions = spec
670                            .to_conditions()
671                            .map_err(|e| AnswerError::Invalid(e.to_string()))?;
672                        if conditions.is_empty() {
673                            return Err(AnswerError::Invalid(
674                                "the answer's rule has no conditions".to_owned(),
675                            ));
676                        }
677                        let program = pending
678                            .prompt
679                            .exe
680                            .rsplit('/')
681                            .next()
682                            .unwrap_or_default()
683                            .to_owned();
684                        let rule = Rule {
685                            id: 0,
686                            folder: None,
687                            name: label,
688                            enabled: true,
689                            action,
690                            tier: Tier::Normal,
691                            expiry,
692                            log: true,
693                            notes: String::new(),
694                            conditions,
695                        };
696                        Ok((rule, program))
697                    })
698                    .ok_or(AnswerError::NotFound)??,
699            ),
700        };
701        // Into the chosen folder, or the program's own.
702        let rule = match rule {
703            Some((rule, program)) => {
704                // A folder the rule can't go in leaves the prompt waiting.
705                crate::placement::check(&self.store(), folder).map_err(placement_error)?;
706                Some((rule, program))
707            }
708            None => None,
709        };
710        let pending = self.shared.broker.take(id).ok_or(AnswerError::NotFound)?;
711        let rule_id = match rule {
712            Some((mut rule, program)) => {
713                let id = {
714                    let mut store = self.store();
715                    let batch = store.batch(by, now).map_err(Error::from)?;
716                    store
717                        .atomically(|store| {
718                            crate::placement::place(store, &batch, &mut rule, folder, &program)?;
719                            Ok(store.put_rule(&rule, &batch)?)
720                        })
721                        .map_err(placement_error)?
722                };
723                self.reload_rules()?;
724                Some(id)
725            }
726            None => None,
727        };
728        self.resolve(&pending, action, Source::Prompt, rule_id, by, false);
729        Ok(rule_id)
730    }

732    /// Decides a prompt's connections, records the decision, and tells the clients.
733    fn resolve(
734        &self,
735        pending: &Pending,
736        action: Action,
737        source: Source,
738        rule: Option<RuleId>,
739        by: &str,
740        timed_out: bool,
741    ) {
742        let verdict = if action == Action::Allow {
743            Verdict::Accept
744        } else {
745            Verdict::Drop
746        };
747        self.shared.broker.release(pending, verdict);
748        let mut event = pending.event.clone();
749        event.time_ms = now_ms();
750        event.action = action;
751        event.source = source;
752        event.rule = rule;
753        if self.shared.events.try_send(event).is_err() {
754            self.shared.dropped_events.fetch_add(1, Ordering::Relaxed);
755        }
756        self.shared.router.closed(pending.prompt.id);
757        self.shared.hub.publish(
758            v1::Topic::Prompts,
759            &v1::notice::Body::PromptResolved(v1::PromptResolved {
760                id: pending.prompt.id,
761                action: convert::action(action).into(),
762                by: by.to_owned(),
763                rule,
764                timed_out,
765            }),
766        );
767    }

777    /// Replaces the system firewall with `firewall` (checked), recording it as `app`'s change. With `revert_after`,
778    /// the previous one comes back then unless [`Daemon::confirm_system`] is called first.
779    pub fn set_system(
780        self: &Arc<Self>,
781        firewall: &snitchhiker_core::system::Firewall,
782        app: &str,
783        revert_after: Option<Duration>,
784    ) -> Result<u64, Error> {
785        let before = self.store().system_firewall()?;
786        // The kernel first: a table it refuses is never recorded.
787        crate::system::apply(firewall, &self.shared.config)?;
788        let what = crate::system::describe(&before, firewall);
789        let recorded = {
790            let mut store = self.store();
791            store.batch(app, now_ms()).and_then(|batch| {
792                store.atomically(|s| s.put_system_firewall(firewall, &what, &batch))
793            })
794        };
795        if let Err(error) = recorded {
796            let _ = crate::system::apply(&before, &self.shared.config);
797            return Err(error.into());
798        }
799        let mut trial = self.trial.lock().unwrap_or_else(PoisonError::into_inner);
800        trial.generation += 1;
801        trial.pending = None;
802        let Some(after) = revert_after else {
803            return Ok(0);
804        };
805        let reverts_at_ms = now_ms() + u64::try_from(after.as_millis()).unwrap_or(u64::MAX);
806        trial.pending = Some((reverts_at_ms, before));
807        let generation = trial.generation;
808        drop(trial);
809        let daemon = Arc::clone(self);
810        thread::Builder::new()
811            .name("system-trial".to_owned())
812            .spawn(move || {
813                thread::sleep(after);
814                daemon.revert_system(generation);
815            })?;
816        Ok(reverts_at_ms)
817    }
818
819    /// Reverts trial `generation` if it is still waiting.
820    fn revert_system(self: &Arc<Self>, generation: u64) {
821        let previous = {
822            let mut trial = self.trial.lock().unwrap_or_else(PoisonError::into_inner);
823            if trial.generation != generation {
824                return;
825            }
826            trial.pending.take()
827        };
828        if let Some((_, previous)) = previous {
829            warn!("the system firewall change wasn't confirmed in time: reverting it");
830            if let Err(error) = self.set_system(&previous, "daemon (not confirmed)", None) {
831                error!(%error, "reverting the system firewall");
832            }
833        }
834    }
835
836    /// Confirms the change on trial; false when none waits.
837    pub fn confirm_system(&self) -> bool {
838        let mut trial = self.trial.lock().unwrap_or_else(PoisonError::into_inner);
839        trial.generation += 1;
840        trial.pending.take().is_some()
841    }
842
843    /// When the change on trial reverts (Unix ms); 0 for none.
844    pub fn system_reverts_at(&self) -> u64 {
845        self.trial
846            .lock()
847            .unwrap_or_else(PoisonError::into_inner)
848            .pending
849            .as_ref()
850            .map_or(0, |(at, _)| *at)
851    }

879    /// Stops: with `fail = open` the table is removed, so traffic flows freely; with `fail = closed` it stays, so
880    /// nothing new gets out until the daemon is back.
881    pub fn stop(&self) {
882        // Clients are told before the connections end, so a planned stop isn't taken for a crash.
883        let reason = if std::process::Command::new("systemctl")
884            .arg("is-system-running")
885            .output()
886            .is_ok_and(|o| String::from_utf8_lossy(&o.stdout).trim() == "stopping")
887        {
888            "the machine is shutting down or restarting"
889        } else {
890            "the daemon was stopped (by an administrator, or for an upgrade)"
891        };
892        self.shared.router.push.stopping(reason);
893        self.shared.hub.publish(
894            v1::Topic::Prompts,
895            &v1::notice::Body::Stopping(v1::Stopping {
896                reason: reason.to_owned(),
897            }),
898        );
899        // A moment for the notices to leave.
900        thread::sleep(Duration::from_millis(300));
901        let _ = std::fs::remove_file(config_marker(&self.shared.config));
902        if self.shared.config.fail == Fail::Open {
903            if let Err(error) = remove(&self.table.name) {
904                warn!(%error, "removing the table");
905            }
906            if let Err(error) = crate::system::remove() {
907                warn!(%error, "removing the system firewall");
908            }
909        }
910        info!(
911            decided = self.shared.decided.load(Ordering::Relaxed),
912            dropped_events = self.shared.dropped_events.load(Ordering::Relaxed),
913            "stopped"
914        );
915    }
916}

962/// Learns names from DNS answers: only those to sockets that sent a query to that resolver, and remembered for the
963/// process that sent it. Every packet passes; this queue only observes.
964fn dns_loop(shared: &Shared, mut queue: Queue) -> Result<(), Error> {
965    let mut ids: Vec<u32> = Vec::with_capacity(64);
966    loop {
967        ids.clear();
968        for queued in queue.receive()? {
969            ids.push(queued.id);
970            let Some(headers) = packet::parse(queued.payload) else {
971                continue;
972            };
973            let Some(answer) = packet::udp_payload(queued.payload, &headers).and_then(parse_answer)
974            else {
975                continue;
976            };
977            let query = FlowKey {
978                netns: shared.netns,
979                protocol: packet::IPPROTO_UDP,
980                source_port: headers.destination_port,
981                destination: headers.source,
982                destination_port: DNS_PORT,
983            };
984            // An answer nobody asked for teaches nothing.
985            let Some(asker) = shared.attribution.owner(&query) else {
986                continue;
987            };
988            shared
989                .names
990                .write()
991                .unwrap_or_else(PoisonError::into_inner)
992                .learn(asker.pid, &answer, now_ms());
993        }
994        for &id in &ids {
995            queue.verdict(id, Verdict::Accept)?;
996        }
997    }
998}

1000fn decide_loop(shared: &Shared, mut queue: Queue, offset: usize) -> Result<(), Error> {
1001    let mut processes = ProcessCache::new(shared.host);
1002    let mut verdicts: Vec<(u32, Verdict)> = Vec::with_capacity(64);
1003    let mut rejected: Vec<SocketId> = Vec::with_capacity(8);
1004    let mut sockdiag: Option<SockDiag> = None;
1005    loop {
1006        verdicts.clear();
1007        rejected.clear();
1008        for packet in queue.receive()? {
1009            let (verdict, reject) = decide(shared, &mut processes, &packet, offset);
1010            // No verdict: the packet waits for a prompt's answer.
1011            if let Some(verdict) = verdict {
1012                verdicts.push((packet.id, verdict));
1013            }
1014            rejected.extend(reject);
1015        }
1016        for &(id, verdict) in &verdicts {
1017            queue.verdict(id, verdict)?;
1018        }
1019        // After the drop: the program's socket is closed, so it fails at once instead of retrying.
1020        for id in &rejected {
1021            if sockdiag.is_none() {
1022                sockdiag = SockDiag::open()
1023                    .inspect_err(|error| warn!(%error, "sock_diag: rejected connections will time out instead"))
1024                    .ok();
1025            }
1026            if let Some(diag) = sockdiag.as_mut()
1027                && let Err(error) = diag.destroy(id)
1028                && error.kind() != std::io::ErrorKind::NotFound
1029            {
1030                warn!(%error, "closing a rejected socket");
1031            }
1032        }
1033    }
1034}

1072/// Whether a connection no rule decides is held for a prompt: the policy asks, the daemon enforces, and some client
1073/// listens for prompts, or a device can be woken to.
1074pub fn asks(shared: &Shared) -> bool {
1075    let policy = shared.policy.load();
1076    policy.no_match == Policy::Ask
1077        && policy.mode == Mode::Enforce
1078        && (shared.hub.has_subscribers(v1::Topic::Prompts) || shared.router.can_wake_someone())
1079}

1081/// What a policy does without asking (`ask` falls back to `no_client`).
1082#[must_use]

1084    match policy {
1085        Policy::Allow => Action::Allow,
1086        Policy::Deny => Action::Deny,
1087        Policy::Reject => Action::Reject,
1088        Policy::Ask => no_client,

1092/// The verdict for one packet (none: it is held for a prompt), recording what happened, and the socket to close for a
1093/// rejected connection.
1094fn decide(
1095    shared: &Shared,
1096    processes: &mut ProcessCache,
1097    queued: &Packet<'_>,
1098    queue: usize,
1099) -> (Option<Verdict>, Option<SocketId>) {
1100    shared.decided.fetch_add(1, Ordering::Relaxed);
1101    let Some(headers) = packet::parse(queued.payload) else {
1102        // Not IPv4 or IPv6: nothing rules speak about.
1103        return (Some(Verdict::Accept), None);
1104    };
1105    let key = FlowKey {
1106        netns: shared.netns,
1107        protocol: headers.protocol,
1108        source_port: headers.source_port,
1109        destination: headers.destination,
1110        destination_port: headers.destination_port,
1111    };
1112    let owner = shared.attribution.owner(&key);
1113    // The daemon's own traffic (its remote endpoint: the relay, devices) is never decided, as marked packets aren't
1114    // queued: its sockets come from a library that can't mark them, but the kernel names their process.
1115    if owner.is_some_and(|o| o.pid == std::process::id()) {
1116        return (Some(Verdict::Accept), None);
1117    }
1118    let process = owner.and_then(|owner| processes.get(owner.pid, owner.start_boottime));
1119    let now = now_ms();
1120    // The user as the kernel recorded it when the socket was opened (the queue's uid is the socket's owner, for
1121    // packets the attribution programs didn't see).
1122    let uid = owner.map(|o| o.uid).or(queued.uid).unwrap_or(u32::MAX);
1123    let home = process.as_ref().and_then(|_| processes.home(uid));
1124    let pid = owner.map_or(0, |o| o.pid);
1125    let hosts = shared
1126        .names
1127        .read()
1128        .unwrap_or_else(PoisonError::into_inner)
1129        .lookup(pid, headers.destination, now);
1130    let empty: &[String] = &[];
1131    let facts = Facts {
1132        exe: process.as_ref().map_or("", |p| &*p.exe),
1133        apps: empty,
1134        cmdline: process.as_ref().map_or("", |p| p.cmdline.as_str()),
1135        ancestors: process.as_ref().map_or(empty, |p| p.ancestors.as_slice()),
1136        lineage: process.as_ref().map_or(&[], |p| p.lineage.as_slice()),
1137        uid,
1138        home: home.as_deref(),
1139        hosts: &hosts,
1140        address: headers.destination,
1141        port: headers.destination_port,
1142        protocol: headers.protocol,
1143        context: process.as_ref().map_or(ContextKind::Host, |p| p.context),
1144        now_ms: now,
1145    };
1146    let exe = process
1147        .as_ref()
1148        .map_or_else(|| Arc::clone(&shared.unknown), |p| Arc::clone(&p.exe));
1149    let mut event = undecided_event(&facts, exe, pid, owner.map(|o| o.cookie));
1150    let rules = shared.rules.load();
1151    let (action, source, rule) = if let Some(decision) = rules.decide(&facts) {
1152        (decision.action, Source::Rule, Some(decision.rule))
1153    } else {
1154        if asks(shared) {
1155            let prompt = hold(
1156                shared,
1157                &facts,
1158                &event,
1159                Held {
1160                    queue,
1161                    id: queued.id,
1162                },
1163            );
1164            remember(shared, &facts, &event, Some(prompt));
1165            return (None, None);
1166        }
1167        let policy = shared.policy.load();
1168        let action = policy_action(policy.no_match, policy.no_client);
1169        (action, Source::Policy, None)
1170    };
1171    let (verdict, source) = match shared.policy.load().mode {
1172        Mode::Monitor => (Verdict::Accept, Source::Monitor),
1173        Mode::Enforce => (
1174            if action == Action::Allow {
1175                Verdict::Accept
1176            } else {
1177                Verdict::Drop
1178            },
1179            source,
1180        ),
1181    };
1182    if action != Action::Allow {
1183        remember(shared, &facts, &event, None);
1184    }
1185    event.action = action;
1186    event.source = source;
1187    event.rule = rule;
1188    if let Err(TrySendError::Full(_) | TrySendError::Disconnected(_)) =
1189        shared.events.try_send(event)
1190    {
1191        shared.dropped_events.fetch_add(1, Ordering::Relaxed);
1192    }
1193    let reject = (verdict == Verdict::Drop && action == Action::Reject).then_some(SocketId {
1194        protocol: headers.protocol,
1195        local: headers.source,
1196        local_port: headers.source_port,
1197        remote: headers.destination,
1198        remote_port: headers.destination_port,
1199    });
1200    (Some(verdict), reject)
1201}

1222/// Holds a packet for the prompt of its program and destination, opening (and announcing) one if none is open; the
1223/// prompt's id.
1224fn hold(shared: &Shared, facts: &Facts<'_>, event: &Event, packet: Held) -> u64 {
1225    let destination = facts
1226        .hosts
1227        .first()
1228        .map_or_else(|| facts.address.to_string(), ToString::to_string);
1229    let key = PromptKey {
1230        exe: Arc::clone(&event.exe),
1231        uid: facts.uid,
1232        destination,
1233        port: facts.port,
1234        protocol: facts.protocol,
1235    };
1236    let opened = shared.broker.hold(key, packet, facts.now_ms, || {
1237        let suggestions = snitchhiker_core::suggest::suggestions(
1238            facts.exe,
1239            facts.home,
1240            facts.hosts,
1241            facts.address,
1242            facts.port,
1243            facts.protocol,
1244        );
1245        NewPrompt {
1246            prompt: v1::Prompt {
1247                exe: facts.exe.to_owned(),
1248                cmdline: facts.cmdline.to_owned(),
1249                uid: facts.uid,
1250                user: crate::process::user_of(facts.uid).unwrap_or_default(),
1251                pid: event.pid,
1252                hosts: facts.hosts.iter().map(ToString::to_string).collect(),
1253                address: convert::address(facts.address),
1254                port: u32::from(facts.port),
1255                protocol: u32::from(facts.protocol),
1256                context: convert::context(facts.context).into(),
1257                exe_writable: crate::process::writable_by_others(event.pid),
1258                suggestions: suggestions
1259                    .iter()
1260                    .map(|s| v1::Suggestion {
1261                        label: s.label.clone(),
1262                        conditions: Some(convert::conditions(&s.conditions)),
1263                        broad: s.broad,
1264                    })
1265                    .collect(),
1266                ..v1::Prompt::default()
1267            },
1268            suggestions,
1269            event: event.clone(),
1270        }
1271    });
1272    match opened {
1273        Hold::Opened(prompt) => {
1274            shared.router.opened(&shared.hub, &prompt);
1275            prompt.id
1276        }
1277        Hold::Joined(id) => id,
1278    }
1279}

1281/// Pushes recorded events to the subscribers of the events topic: one message each to those that want messages, and
1282/// the batch as one columnar block to those that want blocks.
1283fn publish_events(shared: &Shared, batch: &[Event], recorded: &[crate::store::Recorded]) {
1284    let (messages, blocks) = shared.hub.wanted(v1::Topic::Events);
1285    let plain: Vec<Outgoing> = if messages {
1286        batch
1287            .iter()
1288            .zip(recorded)
1289            .map(|(event, r)| {
1290                Outgoing::Notice(v1::notice::Body::Event(crate::api::event(
1291                    &event.stored(r.seq),
1292                )))
1293            })
1294            .collect()
1295    } else {
1296        Vec::new()
1297    };
1298    let block = if blocks {
1299        let rows: Vec<EventRow> = batch
1300            .iter()
1301            .zip(recorded)
1302            .map(|(e, r)| event_row(e, r))
1303            .collect();
1304        let mut strings: Vec<(u32, Arc<str>)> = Vec::new();
1305        for (event, r) in batch.iter().zip(recorded) {
1306            strings.push((r.exe, Arc::clone(&event.exe)));
1307            if let Some(host) = &event.host {
1308                strings.push((r.host, Arc::clone(host)));
1309            }
1310        }
1311        let mut columns = Vec::new();
1312        match columns::encode_events(&rows, &mut columns) {
1313            Ok(()) => Some(Outgoing::Block {
1314                columns: bytes::Bytes::from(columns),
1315                strings: Arc::from(strings),
1316            }),
1317            Err(error) => {
1318                error!(%error, "encoding an event block");
1319                None
1320            }
1321        }
1322    } else {
1323        None
1324    };
1325    if messages || block.is_some() {
1326        shared
1327            .hub
1328            .publish_all(v1::Topic::Events, &plain, block.as_ref());
1329    }
1330}

1332/// A recorded event as a block's row: the protocol's numbers, the strings' ids.
1333fn event_row(event: &Event, recorded: &crate::store::Recorded) -> EventRow {
1334    let address = match event.address {
1335        std::net::IpAddr::V4(a) => a.to_ipv6_mapped().octets(),
1336        std::net::IpAddr::V6(a) => a.octets(),
1337    };
1338    EventRow {
1339        seq: recorded.seq,
1340        time_ms: event.time_ms,
1341        exe: recorded.exe,
1342        host: recorded.host,
1343        uid: event.uid,
1344        pid: event.pid,
1345        address,
1346        port: event.port,
1347        protocol: event.protocol,
1348        context: event.context.saturating_add(1),
1349        action: u8::try_from(i32::from(convert::action(event.action))).unwrap_or(0),
1350        source: event.source as u8,
1351        rule: event.rule.unwrap_or(0),
1352        bytes_out: 0,
1353        bytes_in: 0,
1354    }
1355}

1357/// Raises an alert when the last daemon didn't stop cleanly (its marker is still there: it crashed, was killed, or the
1358/// machine lost power), then puts this one's marker in place.
1359fn check_marker(shared: &Shared, store: &Store) {
1360    let marker = config_marker(&shared.config);
1361    if marker.exists() {
1362        crate::alerts::raise(
1363            &shared.hub,
1364            store,
1365            &crate::alerts::NewAlert {
1366                kind: "crashed",
1367                title: "The daemon stopped unexpectedly".to_owned(),
1368                body: "It didn't stop cleanly last time: it crashed, was killed, or the machine lost power."
1369                    .to_owned(),
1370                rule: None,
1371            },
1372        );
1373    }
1374    let _ = std::fs::write(&marker, b"");
1375}

1377/// Where a running daemon keeps its marker: next to its database, so a crash is noticed after a reboot too.
1378fn config_marker(config: &Config) -> std::path::PathBuf {
1379    config.database.with_extension("running")

1382/// How often the tables are checked.
1383const TABLE_WATCH_EVERY: Duration = Duration::from_secs(10);
1384/// How often the counters of what was lost are checked.
1385const LOSS_CHECK_EVERY: Duration = Duration::from_secs(60);

1387/// Packets the kernel dropped on queues `first..first + count` (`/proc/net/netfilter/nfnetlink_queue`: the queue
1388/// overflowed, or the daemon's socket couldn't take them).
1389fn queue_drops(first: u16, count: u16) -> u64 {
1390    std::fs::read_to_string("/proc/net/netfilter/nfnetlink_queue").map_or(0, |text| {
1391        text.lines()
1392            .filter_map(|line| {
1393                let fields: Vec<u64> = line
1394                    .split_whitespace()
1395                    .filter_map(|f| f.parse().ok())
1396                    .collect();
1397                let queue = u16::try_from(*fields.first()?).ok()?;
1398                (queue >= first && queue < first.saturating_add(count)).then(|| {
1399                    fields.get(5).copied().unwrap_or(0) + fields.get(6).copied().unwrap_or(0)
1400                })
1401            })
1402            .sum()
1403    })
1404}

1406/// Alerts about what the node lost since the last check: events the history couldn't keep, packets the kernel
1407/// dropped.
1408fn check_losses(shared: &Shared, store: &Store, seen: &mut (u64, u64)) {
1409    let dropped = shared.dropped_events.load(Ordering::Relaxed);
1410    if dropped > seen.0 {
1411        crate::alerts::raise(
1412            &shared.hub,
1413            store,
1414            &crate::alerts::NewAlert {
1415                kind: "dropped_events",
1416                title: "Connections weren't recorded".to_owned(),
1417                body: format!(
1418                    "{} connections were decided but not kept in the history: the database couldn't keep up.",
1419                    dropped - seen.0
1420                ),
1421                rule: None,
1422            },
1423        );
1424    }
1425    let queued = queue_drops(shared.config.queue, shared.config.queues);
1426    if queued > seen.1 {
1427        crate::alerts::raise(
1428            &shared.hub,
1429            store,
1430            &crate::alerts::NewAlert {
1431                kind: "queue_overflow",
1432                title: "The kernel dropped packets waiting for a decision".to_owned(),
1433                body: format!(
1434                    "{} packets were dropped on the daemon's queues: they overflowed. New connections were refused \
1435                     while that lasted.",
1436                    queued - seen.1
1437                ),
1438                rule: None,
1439            },
1440        );
1441    }
1442    *seen = (dropped, queued.max(seen.1));
1443}

1445fn write_loop(shared: &Shared, store: &crate::locked::Locked, events: &Receiver<Event>) {
1446    let mut batch = Vec::with_capacity(BATCH);
1447    let mut engine = crate::alerts::Engine::default();
1448    let mut rules_seen = u64::MAX;
1449    let mut last_loss_check = Instant::now();
1450    let mut losses = (0, queue_drops(shared.config.queue, shared.config.queues));
1451    let mut recorded = Vec::with_capacity(BATCH);
1452    let mut last_prune = Instant::now();
1453    let mut last_bytes = Instant::now();
1454    let mut counted: HashMap<u64, (u64, u64)> = HashMap::new();
1455    let mut changed: Vec<(u64, u64, u64)> = Vec::new();
1456    loop {
1457        batch.clear();
1458        match events.recv_timeout(BYTES_EVERY) {
1459            Ok(event) => batch.push(event),
1460            Err(RecvTimeoutError::Timeout) => {}
1461            Err(RecvTimeoutError::Disconnected) => return,
1462        }
1463        let deadline = Instant::now() + BATCH_WAIT;
1464        while batch.len() < BATCH {
1465            let left = deadline.saturating_duration_since(Instant::now());
1466            match events.recv_timeout(left) {
1467                Ok(event) => batch.push(event),
1468                Err(_) => break,
1469            }
1470        }
1471        let mut store = store.lock();
1472        let generation = shared.alert_rules_changed.load(Ordering::Relaxed);
1473        if generation != rules_seen {
1474            rules_seen = generation;
1475            match store.alert_rules() {
1476                Ok(rules) => engine.load(rules),
1477                Err(error) => error!(%error, "loading the alert rules"),
1478            }
1479        }
1480        if !batch.is_empty() {
1481            match store.record(&batch, &mut recorded) {
1482                Ok(()) => {
1483                    publish_events(shared, &batch, &recorded);
1484                    for alert in engine.check(&batch, &recorded) {
1485                        crate::alerts::raise(&shared.hub, &store, &alert);
1486                    }
1487                }
1488                Err(error) => error!(%error, events = batch.len(), "recording events"),
1489            }
1490        }
1491        if last_loss_check.elapsed() >= LOSS_CHECK_EVERY {
1492            last_loss_check = Instant::now();
1493            check_losses(shared, &store, &mut losses);
1494        }
1495        if last_bytes.elapsed() >= BYTES_EVERY {
1496            last_bytes = Instant::now();
1497            changed.clear();
1498            let mut live: HashMap<u64, (u64, u64)> = HashMap::with_capacity(counted.len());
1499            for (cookie, sent, read) in shared.attribution.socket_bytes() {
1500                if counted.get(&cookie) != Some(&(sent, read)) {
1501                    changed.push((cookie, sent, read));
1502                }
1503                live.insert(cookie, (sent, read));
1504            }
1505            counted = live;
1506            if !changed.is_empty()
1507                && let Err(error) = store.record_bytes(&changed)
1508            {
1509                error!(%error, "recording byte counts");
1510            }
1511        }
1512        if last_prune.elapsed() >= PRUNE_EVERY {
1513            last_prune = Instant::now();
1514            let retention_ms = u64::from(shared.policy.load().retention_days) * 86_400_000;
1515            let cutoff = now_ms().saturating_sub(retention_ms);
1516            while let Ok(pruned) = store.prune(cutoff, PRUNE_CHUNK) {
1517                if pruned < PRUNE_CHUNK as usize {
1518                    break;
1519                }
1520            }
1521            if let Err(error) = store.prune_alerts(cutoff) {
1522                error!(%error, "pruning alerts");
1523            }
1524            if let Err(error) = store.prune_proposals(cutoff) {
1525                error!(%error, "pruning proposals");
1526            }
1527            if let Err(error) = store.prune_rollups(cutoff) {
1528                error!(%error, "pruning rollups");
1529            }
1530        }
1531    }
