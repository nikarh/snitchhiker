1//! Blocklists from URLs and files (plan §5.8): fetched on a schedule, checked before they replace what was there, kept
2//! on disk so a restart without a network still has them, and named by rules' `list` condition.
3//!
4//! A list's source is an `https://` URL (plain HTTP isn't taken: anyone on the path could change it) or a file in
5//! [`LOCAL_DIR`], a directory only root writes: a client may add lists, and must not make the daemon (root) read any
6//! other file on the node, which the list's entries, counts and errors would then show. A fetch is refused, and the version before kept, when the file is larger than
7//! `blocklist_max_bytes`, isn't text, has no entries, or has more than a tenth of its lines that aren't entries (an
8//! error page, a list in a format this doesn't read). Unchanged lists aren't downloaded again (`If-None-Match`).
9

26/// How often a list is fetched unless it says otherwise: daily.
27pub const DEFAULT_REFRESH_SECS: u64 = 86_400;
28/// Lists aren't fetched more often than this.
29pub const MIN_REFRESH_SECS: u64 = 900;
30
31/// A list as the store keeps it: what it is, and how its last fetch went.
32#[derive(Debug, Clone, Default, PartialEq, Eq, Serialize, Deserialize)]
33#[serde(default)]
34pub struct Record {
35    #[serde(skip)]
36    pub id: u64,
37    #[serde(skip)]
38    pub name: String,
39    pub source: String,
40    pub refresh_secs: u64,
41    /// When the content last changed.
42    pub updated_ms: u64,
43    /// When it was last fetched, changed or not.
44    pub checked_ms: u64,
45    pub sha256: String,
46    pub etag: String,
47    pub bytes: u64,
48    pub entries: u64,
49    pub invalid: u64,
50    /// The last fetch's error; empty when it worked.
51    pub error: String,
52}
53
54impl Record {
55    /// When it is fetched next: an hour after a failure, if that's sooner.
56    fn due_ms(&self) -> u64 {
57        let mut every = self.refresh_secs.max(MIN_REFRESH_SECS);
58        if !self.error.is_empty() {
59            every = every.min(3600);
60        }
61        self.checked_ms.saturating_add(every.saturating_mul(1000))
62    }
63}

65/// The lists in use, and what decided by them.
66pub struct Blocklists {
67    dir: PathBuf,
68    /// Where local lists are read from: [`LOCAL_DIR`].
69    local_dir: PathBuf,
70    max_bytes: u64,
71    lists: RwLock<Lists>,
72    /// Connections decided by rules naming each list, since the daemon started.
73    matched: RwLock<HashMap<Box<str>, Arc<AtomicU64>>>,
74    /// Each rule naming lists, with their counters.
75    by_rule: ArcSwap<HashMap<RuleId, Vec<Arc<AtomicU64>>>>,
76    /// One fetch at a time.
77    fetching: Mutex<()>,
78    /// Counts changes of the lists, to wake the refresher.
79    changed: (Mutex<u64>, Condvar),
80}

82impl Blocklists {
83    /// The lists of `records`, as last fetched into `dir`.
84    #[must_use]
85    pub fn load(dir: PathBuf, max_bytes: u64, records: &[Record]) -> Self {
86        let mut lists = Lists::new();
87        for record in records {
88            match std::fs::read_to_string(file(&dir, record.id)) {
89                Ok(text) => {
90                    lists.insert(
91                        record.name.as_str().into(),
92                        Arc::new(Blocklist::parse(&text).list),
93                    );
94                }
95                Err(error) if error.kind() == std::io::ErrorKind::NotFound => {}
96                Err(error) => warn!(%error, list = record.name, "reading a blocklist"),
97            }
98        }
99        Self {
100            dir,
101            local_dir: PathBuf::from(LOCAL_DIR),
102            max_bytes,
103            lists: RwLock::new(lists),
104            matched: RwLock::default(),
105            by_rule: ArcSwap::default(),
106            fetching: Mutex::new(()),
107            changed: (Mutex::new(0), Condvar::new()),
108        }
109    }
110
111    /// The lists, for compiling rules.
112    #[must_use]
113    pub fn lists(&self) -> Lists {
114        self.lists
115            .read()
116            .unwrap_or_else(PoisonError::into_inner)
117            .clone()
118    }
119
120    /// Which lists `rules` name, for counting what they decide.
121    pub fn rules_compiled(&self, rules: &[Rule]) {
122        let mut matched = self.matched.write().unwrap_or_else(PoisonError::into_inner);
123        let mut by_rule = HashMap::new();
124        for rule in rules {
125            let counters: Vec<Arc<AtomicU64>> = names(&rule.conditions)
126                .map(|name| Arc::clone(matched.entry(name.into()).or_default()))
127                .collect();
128            if !counters.is_empty() {
129                by_rule.insert(rule.id, counters);
130            }
131        }
132        drop(matched);
133        self.by_rule.store(Arc::new(by_rule));
134    }
135
136    /// Counts a connection `rule` decided.
137    pub fn decided(&self, rule: RuleId) {
138        if let Some(counters) = self.by_rule.load().get(&rule) {
139            for counter in counters {
140                counter.fetch_add(1, Ordering::Relaxed);
141            }
142        }
143    }
144
145    fn matched(&self, name: &str) -> u64 {
146        self.matched
147            .read()
148            .unwrap_or_else(PoisonError::into_inner)
149            .get(name)
150            .map_or(0, |c| c.load(Ordering::Relaxed))
151    }

153    /// The list as clients see it; `rules` are the rules naming it.
154    #[must_use]
155    pub fn wire(&self, record: &Record, rules: Vec<u64>) -> v1::Blocklist {
156        v1::Blocklist {
157            name: record.name.clone(),
158            source: record.source.clone(),
159            refresh_secs: record.refresh_secs,
160            updated_ms: record.updated_ms,
161            checked_ms: record.checked_ms,
162            sha256: record.sha256.clone(),
163            bytes: record.bytes,
164            entries: record.entries,
165            invalid: record.invalid,
166            error: record.error.clone(),
167            matched: self.matched(&record.name),
168            rules,
169        }
170    }

172    /// Wakes the refresher: lists were added, removed or changed.
173    pub fn changed(&self) {
174        let (lock, wake) = &self.changed;
175        *lock.lock().unwrap_or_else(PoisonError::into_inner) += 1;
176        wake.notify_all();
177    }
178
179    /// Waits until the next list is due (`next_ms` from now, or forever) or the lists change.
180    pub fn wait(&self, next_ms: Option<u64>, now_ms: u64) {
181        let (lock, wake) = &self.changed;
182        let guard = lock.lock().unwrap_or_else(PoisonError::into_inner);
183        let seen = *guard;
184        let timeout =
185            Duration::from_millis(next_ms.map_or(u64::MAX / 2, |at| at.saturating_sub(now_ms)));
186        let _ = wake
187            .wait_timeout_while(guard, timeout, |count| *count == seen)
188            .unwrap_or_else(PoisonError::into_inner);
189    }
190
191    /// The list removed: its file and its set.
192    pub fn removed(&self, record: &Record) {
193        self.lists
194            .write()
195            .unwrap_or_else(PoisonError::into_inner)
196            .remove(record.name.as_str());
197        let _ = std::fs::remove_file(file(&self.dir, record.id));
198        self.changed();
199    }
200
201    /// Fetches a list; `record` is updated with how it went. True when its content changed (the rules then need
202    /// compiling again).
203    pub fn fetch(&self, record: &mut Record, now_ms: u64) -> bool {
204        let _one = self.fetching.lock().unwrap_or_else(PoisonError::into_inner);
205        record.checked_ms = now_ms;
206        match self.download(record) {
207            Ok(None) => {
208                record.error.clear();
209                false
210            }
211            Ok(Some((text, etag))) => match self.accept(record, &text) {
212                Ok(changed) => {
213                    record.etag = etag;
214                    record.error.clear();
215                    if changed {
216                        record.updated_ms = now_ms;
217                        info!(
218                            list = record.name,
219                            entries = record.entries,
220                            "a blocklist changed"
221                        );
222                    }
223                    changed
224                }
225                Err(error) => {
226                    warn!(list = record.name, %error, "a blocklist was refused; the version before stays");
227                    record.error = error;
228                    false
229                }
230            },
231            Err(error) => {
232                warn!(list = record.name, %error, "fetching a blocklist");
233                record.error = error;
234                false
235            }
236        }
237    }

239    /// The content, or `None` when the server says it is unchanged.
240    fn download(&self, record: &Record) -> Result<Option<(String, String)>, String> {
241        let limit = self.max_bytes;
242        let too_large = || format!("larger than {limit} bytes (`blocklist_max_bytes`)");
243        let mut bytes = Vec::new();
244        let mut etag = String::new();
245        if record.source.starts_with('/') {
246            // Checked again here, after resolving links: a list added before files were confined to the directory, or a
247            // link put there.
248            let file = std::fs::File::open(local_file(&self.local_dir, &record.source)?)
249                .map_err(|e| e.to_string())?;
250            file.take(limit + 1)
251                .read_to_end(&mut bytes)
252                .map_err(|e| e.to_string())?;
253        } else {
254            let client = reqwest::blocking::Client::builder()
255                .timeout(Duration::from_secs(120))
256                .https_only(true)
257                .build()
258                .map_err(|e| e.to_string())?;
259            let mut request = client.get(&record.source).header(
260                "user-agent",
261                concat!("snitchhiker/", env!("CARGO_PKG_VERSION")),
262            );
263            // Only when the file is still there to fall back on.
264            if !record.etag.is_empty() && file(&self.dir, record.id).exists() {
265                request = request.header("if-none-match", &record.etag);
266            }
267            let response = request
268                .send()
269                .and_then(reqwest::blocking::Response::error_for_status)
270                .map_err(|e| e.to_string())?;
271            if response.status() == reqwest::StatusCode::NOT_MODIFIED {
272                return Ok(None);
273            }
274            if response.content_length().is_some_and(|n| n > limit) {
275                return Err(too_large());
276            }
277            response
278                .headers()
279                .get("etag")
280                .and_then(|v| v.to_str().ok())
281                .unwrap_or_default()
282                .clone_into(&mut etag);
283            response
284                .take(limit + 1)
285                .read_to_end(&mut bytes)
286                .map_err(|e| e.to_string())?;
287        }
288        if u64::try_from(bytes.len()).unwrap_or(u64::MAX) > limit {
289            return Err(too_large());
290        }
291        let text = String::from_utf8(bytes).map_err(|_| "not a text file".to_owned())?;
292        Ok(Some((text, etag)))
293    }

295    /// Checks `text` and puts it in use. True when it differs from what was there.
296    fn accept(&self, record: &mut Record, text: &str) -> Result<bool, String> {
297        let digest = ring::digest::digest(&ring::digest::SHA256, text.as_bytes());
298        let sha256 = digest
299            .as_ref()
300            .iter()
301            .fold(String::with_capacity(64), |mut hex, b| {
302                let _ = write!(hex, "{b:02x}");
303                hex
304            });
305        let present = self
306            .lists
307            .read()
308            .unwrap_or_else(PoisonError::into_inner)
309            .contains_key(record.name.as_str());
310        if sha256 == record.sha256 && present {
311            return Ok(false);
312        }
313        let parsed = Blocklist::parse(text);
314        if parsed.entries == 0 {
315            return Err("no entries: not a list this reads (hosts, domains, adblock domain rules, addresses)".to_owned());
316        }
317        let lines = parsed.entries + parsed.invalid;
318        if parsed.invalid * 10 > lines {
319            return Err(format!(
320                "{} of its lines aren't entries (the first: {:?})",
321                parsed.invalid,
322                parsed.first_invalid.unwrap_or_default()
323            ));
324        }
325        write(&self.dir, record.id, text).map_err(|e| format!("saving it: {e}"))?;
326        record.sha256 = sha256;
327        record.bytes = u64::try_from(text.len()).unwrap_or(u64::MAX);
328        record.entries = u64::try_from(parsed.entries).unwrap_or(u64::MAX);
329        record.invalid = u64::try_from(parsed.invalid).unwrap_or(u64::MAX);
330        self.lists
331            .write()
332            .unwrap_or_else(PoisonError::into_inner)
333            .insert(record.name.as_str().into(), Arc::new(parsed.list));
334        Ok(true)
335    }
336}

338/// The lists a rule's conditions name.
339pub fn names(conditions: &[Condition]) -> impl Iterator<Item = &str> {
340    conditions.iter().flat_map(|c| match c {
341        Condition::List(names) => names.iter().map(String::as_str).collect::<Vec<_>>(),
342        Condition::Not(inner) => names(std::slice::from_ref(inner.as_ref())).collect(),
343        _ => Vec::new(),
344    })
345}

359/// The directory local lists are read from: regular files in it, never through a symbolic link out of it.
360pub const LOCAL_DIR: &str = "/etc/snitchhiker/lists";
361
362/// A local list's file, after resolving links: one that is still directly in `local_dir` and a regular file.
363fn local_file(local_dir: &Path, source: &str) -> Result<PathBuf, String> {
364    let outside = || format!("{source:?}: a local list is a file in {LOCAL_DIR}");
365    let real = std::fs::canonicalize(source).map_err(|e| format!("{source:?}: {e}"))?;
366    let dir =
367        std::fs::canonicalize(local_dir).map_err(|e| format!("{}: {e}", local_dir.display()))?;
368    if real.parent() != Some(dir.as_path()) {
369        return Err(outside());
370    }
371    let metadata = std::fs::symlink_metadata(&real).map_err(|e| e.to_string())?;
372    if !metadata.is_file() {
373        return Err(outside());
374    }
375    Ok(real)
376}

378/// Checks what a client asks for: a name rules can use, and a source this fetches.
379pub fn check(name: &str, source: &str) -> Result<(), String> {
380    if name.is_empty()
381        || name.len() > 64
382        || !name
383            .bytes()
384            .all(|b| b.is_ascii_alphanumeric() || matches!(b, b'-' | b'_' | b'.'))
385    {
386        return Err(format!(
387            "{name:?}: a list's name is 1 to 64 letters, digits, '-', '_' and '.'"
388        ));
389    }
390    let local = Path::new(source);
391    let in_dir = local.parent() == Some(Path::new(LOCAL_DIR))
392        && local.file_name().is_some_and(|n| n != ".." && n != ".");
393    if !(source.starts_with("https://") || in_dir) {
394        return Err(format!(
395            "{source:?}: a list comes from an https:// URL or a file in {LOCAL_DIR} on the node"
396        ));
397    }
398    Ok(())
399}

401fn file(dir: &Path, id: u64) -> PathBuf {
402    dir.join(format!("{id}.list"))
403}

405fn write(dir: &Path, id: u64, text: &str) -> std::io::Result<()> {
406    std::fs::create_dir_all(dir)?;
407    let path = file(dir, id);
408    let partial = path.with_extension("partial");
409    std::fs::write(&partial, text)?;
410    std::fs::rename(partial, path)
411}

417    #[test]
418    fn lists_are_checked_before_they_are_used() {
419        let dir = std::env::temp_dir().join(format!("snitchhiker-lists-{}", std::process::id()));
420        let source = dir.join("source.txt");
421        std::fs::create_dir_all(&dir).unwrap();
422        let mut lists = Blocklists::load(dir.join("lists"), 1024, &[]);
423        lists.local_dir.clone_from(&dir);
424        let mut record = Record {
425            id: 1,
426            name: "ads".to_owned(),
427            source: source.to_string_lossy().into_owned(),
428            ..Record::default()
429        };
430        std::fs::write(&source, "ads.example\n198.51.100.0/24\n").unwrap();
431        assert!(lists.fetch(&mut record, 10));
432        assert_eq!((record.entries, record.error.as_str()), (2, ""));
433        assert!(lists.lists()["ads"].has_host("x.ads.example"));
434        // The same again: nothing changed.
435        assert!(!lists.fetch(&mut record, 20));
436        assert_eq!((record.updated_ms, record.checked_ms), (10, 20));
437        // An error page instead of the list: refused, the list before stays.
438        std::fs::write(&source, "<html>\n<body>\nNot found\n</body>\n</html>\n").unwrap();
439        assert!(!lists.fetch(&mut record, 30));
440        assert!(record.error.starts_with("no entries"), "{}", record.error);
441        std::fs::write(&source, "a.example\n<b>\n").unwrap();
442        assert!(!lists.fetch(&mut record, 30));
443        assert!(record.error.contains("aren't entries"), "{}", record.error);
444        std::fs::write(&source, "x".repeat(2000)).unwrap();
445        assert!(!lists.fetch(&mut record, 30));
446        assert!(record.error.starts_with("larger than"), "{}", record.error);
447        assert!(lists.lists()["ads"].has_host("ads.example"));
448        // What was fetched is there after a restart.
449        let again = Blocklists::load(dir.join("lists"), 1024, std::slice::from_ref(&record));
450        assert!(again.lists()["ads"].has_host("ads.example"));
451        // Counting what rules naming it decide.
452        let rule = Rule {
453            id: 7,
454            folder: None,
455            name: "ads".to_owned(),
456            enabled: true,
457            action: snitchhiker_core::rule::Action::Deny,
458            tier: snitchhiker_core::rule::Tier::Normal,
459            expiry: snitchhiker_core::rule::Expiry::Never,
460            log: true,
461            notes: String::new(),
462            conditions: vec![Condition::List(vec!["ads".to_owned()])],
463        };
464        lists.rules_compiled(std::slice::from_ref(&rule));
465        lists.decided(7);
466        lists.decided(8);
467        assert_eq!(lists.wire(&record, vec![7]).matched, 1);
468        // A link from the directory to a file elsewhere isn't read.
469        let elsewhere =
470            std::env::temp_dir().join(format!("snitchhiker-elsewhere-{}", std::process::id()));
471        std::fs::write(&elsewhere, "ads.example\n").unwrap();
472        let link = dir.join("link.txt");
473        std::os::unix::fs::symlink(&elsewhere, &link).unwrap();
474        let mut linked = Record {
475            id: 2,
476            name: "linked".to_owned(),
477            source: link.to_string_lossy().into_owned(),
478            ..Record::default()
479        };
480        assert!(!lists.fetch(&mut linked, 40));
481        assert!(
482            linked.error.contains("a local list is a file in"),
483            "{}",
484            linked.error
485        );
486        std::fs::remove_file(elsewhere).unwrap();
487        std::fs::remove_dir_all(dir).unwrap();
488    }
