1//! Forwarded traffic (plan §5.10): a container on a bridge network, or a VM on a tap, sends packets that are forwarded
2//! from its bridge port; they never pass the host's `output` hook. Their sender is found exactly:
3//!
4//! 1. the packet's link-layer source address, looked up in the bridge's forwarding database, gives the bridge port it
5//!    came in through;
6//! 2. a veth port names the namespace of its peer (by a namespace id local to this namespace), which is mapped to the
7//!    namespace's inode through the processes in it; a tap port is a VM's;
8//! 3. the flow is then looked up as its sender recorded it in that namespace: the tuple before NAT, which is what the
9//!    forward hook sees.
10//!
11//! Each step is the kernel's own table. A stale cache can only make the lookup miss (the flow must exist in the
12//! namespace it names, with that very tuple), never name the wrong sender; a miss is retried once afresh.
13
14use std::collections::HashMap;
15use std::os::unix::fs::MetadataExt as _;
16
17use snitchhiker_netfilter::route::{Link, Route};
18use tracing::debug;
19
20/// Who sent a forwarded packet.
21#[derive(Debug, Clone, PartialEq, Eq)]
22pub enum Sender {
23    /// A network namespace (a container's), by its inode.
24    Netns(u32),
25    /// A tap interface (a VM's), by its name.
26    Tap(std::sync::Arc<str>),
27}
28
29/// The lookups, cached per deciding thread.
30#[derive(Default)]
31pub struct Senders {
32    route: Option<Route>,
33    /// Bridge port by (bridge, source address).
34    ports: HashMap<(u32, [u8; 6]), u32>,
35    links: HashMap<u32, Link>,
36    /// Namespace inode by this namespace's id for it.
37    namespaces: HashMap<i32, u32>,
38}
39
40impl Senders {
41    /// The sender of a packet that came in through `bridge` from `mac`; `fresh` skips the caches (after a miss).
42    pub fn sender(&mut self, bridge: u32, mac: [u8; 6], fresh: bool) -> Option<Sender> {
43        if fresh {
44            self.ports.remove(&(bridge, mac));
45        }
46        let port = if let Some(port) = self.ports.get(&(bridge, mac)) {
47            *port
48        } else {
49            let port = self.route()?.bridge_port(bridge, mac).ok()??;
50            self.ports.insert((bridge, mac), port);
51            port
52        };
53        if fresh {
54            self.links.remove(&port);
55        }
56        let link = if let Some(link) = self.links.get(&port) {
57            link.clone()
58        } else {
59            let link = self.route()?.link(port).ok()??;
60            self.links.insert(port, link.clone());
61            link
62        };
63        match link.kind.as_deref() {
64            Some("tun") => Some(Sender::Tap(std::sync::Arc::from(link.name.as_str()))),
65            Some("veth") => {
66                let id = link.peer_netns_id?;
67                if fresh || !self.namespaces.contains_key(&id) {
68                    self.scan_namespaces();
69                }
70                self.namespaces.get(&id).copied().map(Sender::Netns)
71            }
72            _ => None,
73        }
74    }
75
76    fn route(&mut self) -> Option<&mut Route> {
77        if self.route.is_none() {
78            self.route = Route::open()
79                .inspect_err(|error| debug!(%error, "opening a route netlink socket"))
80                .ok();
81        }
82        self.route.as_mut()
83    }
84
85    /// Maps the namespace ids this namespace has given to the inodes of the namespaces they name, from one process in
86    /// each namespace there is.
87    fn scan_namespaces(&mut self) {
88        let Ok(processes) = std::fs::read_dir("/proc") else {
89            return;
90        };
91        let mut seen = std::collections::HashSet::new();
92        let mut found = HashMap::new();
93        for entry in processes.flatten() {
94            let Some(pid) = entry
95                .file_name()
96                .to_str()
97                .and_then(|n| n.parse::<u32>().ok())
98            else {
99                continue;
100            };
101            let Ok(inode) = std::fs::metadata(format!("/proc/{pid}/ns/net")).map(|m| m.ino())
102            else {
103                continue;
104            };
105            let Ok(inode) = u32::try_from(inode) else {
106                continue;
107            };
108            if !seen.insert(inode) {
109                continue;
110            }
111            if let Some(Ok(Some(id))) = self.route().map(|route| route.netns_id_of_pid(pid)) {
112                found.insert(id, inode);
113            }
114        }
115        self.namespaces = found;
116    }
117}
