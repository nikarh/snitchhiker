1//! Agents (decision 0008): who an agent is, what its own processes were refused, and its proposals.
2//!
3//! An agent is the process that started the MCP server on the agent socket; its *tree* is it and everything it
4//! started, each process held by pid and start time. The node remembers, in memory, the lineage of every connection
5//! it didn't allow over the last hour, so it can tell an agent what its own processes were refused, and nothing else.

24/// How long refused connections are remembered.
25const WINDOW_MS: u64 = 3_600_000;
26/// Refused connections remembered, at most.
27const CAPACITY: usize = 4096;
28/// Limits on what an agent sends.
29pub const MAX_JUSTIFICATION: usize = 2000;
30pub const MAX_RULES: usize = 16;
31pub const MAX_PENDING: usize = 10;
32pub const MAX_NOTE: usize = 500;
33/// The folder agents' rules go in.
34pub const AGENTS: &str = "Agents";

36/// A connection the node didn't allow, with the processes it came from.
37#[derive(Debug, Clone)]
38pub struct Blocked {
39    pub time_ms: u64,
40    /// The process and its ancestors, nearest first.
41    pub lineage: Vec<ProcessRef>,
42    pub exe: Arc<str>,
43    pub host: Option<String>,
44    pub address: IpAddr,
45    pub port: u16,
46    pub protocol: u8,
47    pub uid: u32,
48    pub context: ContextKind,
49    /// The prompt it waits on (or waited on); none: a rule or the policy refused it.
50    pub prompt: Option<u64>,
51}

53/// The refused connections of the last hour.
54#[derive(Default)]
55pub struct Ring {
56    entries: Mutex<VecDeque<Blocked>>,
57}
58
59impl Ring {
60    pub fn record(&self, blocked: Blocked) {
61        let mut entries = self.entries.lock().unwrap_or_else(PoisonError::into_inner);
62        let cutoff = blocked.time_ms.saturating_sub(WINDOW_MS);
63        while entries
64            .front()
65            .is_some_and(|e| e.time_ms < cutoff || entries.len() >= CAPACITY)
66        {
67            entries.pop_front();
68        }
69        entries.push_back(blocked);
70    }
71
72    /// The refused connections of `agent`'s tree since `since_ms`.
73    #[must_use]
74    pub fn of(&self, agent: ProcessRef, now_ms: u64) -> Vec<Blocked> {
75        let cutoff = now_ms.saturating_sub(WINDOW_MS);
76        self.entries
77            .lock()
78            .unwrap_or_else(PoisonError::into_inner)
79            .iter()
80            .filter(|e| e.time_ms >= cutoff && e.lineage.contains(&agent))
81            .cloned()
82            .collect()
83    }
84}

93/// Who the agent behind the MCP server `server` is: its parent, a process of the same user. `programs` are the
94/// node's `agent_programs`: when there are any, only those may propose.
95pub fn identify(server: &Peer, programs: &[String]) -> Result<Agent, String> {
96    let pid = crate::process::parent(server.pid).ok_or("the MCP server has no parent")?;
97    let start = crate::process::start_time(pid).ok_or("the MCP server's parent exited")?;
98    if pid <= 1 {
99        return Err("the MCP server's parent exited".to_owned());
100    }
101    let status = fs::read_to_string(format!("/proc/{pid}/status")).map_err(|e| e.to_string())?;
102    let uid = status
103        .lines()
104        .find_map(|l| l.strip_prefix("Uid:"))
105        .and_then(|rest| rest.split_whitespace().next())
106        .and_then(|uid| uid.parse::<u32>().ok());
107    if uid != Some(server.uid) {
108        return Err("the MCP server's parent is another user's process".to_owned());
109    }
110    let exe = fs::read_link(format!("/proc/{pid}/exe"))
111        .map(|p| p.to_string_lossy().into_owned())
112        .map_err(|e| e.to_string())?;
113    let cmdline = fs::read(format!("/proc/{pid}/cmdline")).map_or_else(
114        |_| String::new(),
115        |bytes| {
116            bytes
117                .split(|&b| b == 0)
118                .filter(|part| !part.is_empty())
119                .map(String::from_utf8_lossy)
120                .collect::<Vec<_>>()
121                .join(" ")
122        },
123    );
124    let cwd = fs::read_link(format!("/proc/{pid}/cwd"))
125        .map_or_else(|_| String::new(), |p| p.to_string_lossy().into_owned());
126    let session = fs::read_to_string(format!("/proc/{pid}/sessionid"))
127        .ok()
128        .and_then(|s| s.trim().parse::<u32>().ok())
129        .unwrap_or(u32::MAX);
130    let (ancestors, _) = crate::process::lineage(pid);
131    // Everything read was the process that started then: a pid reused meanwhile is caught here.
132    if crate::process::start_time(pid) != Some(start) {
133        return Err("the MCP server's parent exited".to_owned());
134    }
135    let home = crate::process::home_of(uid.unwrap_or(u32::MAX));
136    let known_agent = !programs.is_empty()
137        && programs.iter().any(|p| {
138            Pattern::compile(&pattern(p))
139                .is_ok_and(|pattern| pattern.matches(&exe, home.as_deref()))
140        });
141    if !programs.is_empty() && !known_agent {
142        return Err(format!("{exe} isn't one of the node's agent programs"));
143    }
144    Ok(Agent {
145        anchor: ProcessRef { pid, start },
146        provenance: v1::Provenance {
147            uid: server.uid,
148            user: crate::process::user_of(server.uid).unwrap_or_default(),
149            agent_pid: pid,
150            agent_start: start,
151            agent_exe: exe,
152            agent_cmdline: truncate(&cmdline, 1000),
153            cwd,
154            session,
155            ancestors,
156            server_pid: server.pid,
157            server_exe: server.exe.to_string_lossy().into_owned(),
158            known_agent,
159        },
160    })
161}

163fn pattern(text: &str) -> PatternSource {
164    if let Some(glob) = text.strip_prefix("glob:") {
165        PatternSource::Glob(glob.to_owned())
166    } else if let Some(regex) = text.strip_prefix("re:") {
167        PatternSource::Regex(regex.to_owned())
168    } else {
169        PatternSource::Exact(text.to_owned())
170    }
171}
172
173fn truncate(text: &str, max: usize) -> String {
174    match text.char_indices().nth(max) {
175        Some((at, _)) => format!("{}…", &text[..at]),
176        None => text.to_owned(),
177    }
178}

180/// The agents whose session rules or waiting proposals end with them, each held by a pidfd: it becomes readable when
181/// the process exits, so nothing is polled.
182pub struct Watch {
183    agents: Mutex<HashMap<ProcessRef, Arc<std::os::fd::OwnedFd>>>,
184    wake_reader: std::io::PipeReader,
185    wake: Mutex<std::io::PipeWriter>,
186}
187
188impl Watch {
189    pub fn new() -> std::io::Result<Self> {
190        let (wake_reader, wake) = std::io::pipe()?;
191        Ok(Self {
192            agents: Mutex::new(HashMap::new()),
193            wake_reader,
194            wake: Mutex::new(wake),
195        })
196    }
197
198    /// Watches `agent`; false when it has ended already.
199    pub fn add(&self, agent: ProcessRef) -> bool {
200        use std::io::Write as _;
201        let mut agents = self.agents.lock().unwrap_or_else(PoisonError::into_inner);
202        if agents.contains_key(&agent) {
203            return true;
204        }
205        let pidfd = rustix::process::Pid::from_raw(i32::try_from(agent.pid).unwrap_or(-1))
206            .and_then(|pid| {
207                rustix::process::pidfd_open(pid, rustix::process::PidfdFlags::empty()).ok()
208            });
209        // The pid held is the agent's only if it still started when the agent did.
210        let Some(pidfd) = pidfd.filter(|_| alive(agent)) else {
211            return false;
212        };
213        agents.insert(agent, Arc::new(pidfd));
214        drop(agents);
215        let _ = self
216            .wake
217            .lock()
218            .unwrap_or_else(PoisonError::into_inner)
219            .write_all(&[1]);
220        true
221    }
222
223    /// Waits until agents end (or the list changes); the agents that ended, no longer watched.
224    pub fn wait(&self) -> Vec<ProcessRef> {
225        use nix::poll::{PollFd, PollFlags, PollTimeout, poll};
226        use std::io::Read as _;
227        use std::os::fd::AsFd as _;
228        let watched: Vec<(ProcessRef, Arc<std::os::fd::OwnedFd>)> = self
229            .agents
230            .lock()
231            .unwrap_or_else(PoisonError::into_inner)
232            .iter()
233            .map(|(agent, fd)| (*agent, Arc::clone(fd)))
234            .collect();
235        let mut fds: Vec<PollFd<'_>> =
236            std::iter::once(PollFd::new(self.wake_reader.as_fd(), PollFlags::POLLIN))
237                .chain(
238                    watched
239                        .iter()
240                        .map(|(_, fd)| PollFd::new(fd.as_fd(), PollFlags::POLLIN)),
241                )
242                .collect();
243        if poll(&mut fds, PollTimeout::NONE).is_err() {
244            return Vec::new();
245        }
246        let ready = |fd: &PollFd<'_>| fd.revents().is_some_and(|r| !r.is_empty());
247        if fds.first().is_some_and(ready) {
248            let _ = (&self.wake_reader).read(&mut [0; 64]);
249        }
250        let ended: Vec<ProcessRef> = watched
251            .iter()
252            .zip(fds.iter().skip(1))
253            .filter(|(_, fd)| ready(fd))
254            .map(|((agent, _), _)| *agent)
255            .collect();
256        let mut agents = self.agents.lock().unwrap_or_else(PoisonError::into_inner);
257        for agent in &ended {
258            agents.remove(agent);
259        }
260        ended
261    }
262}

270/// Refused connections, grouped (program, destination, port, protocol, outcome), most recent first. `waiting` says
271/// whether a prompt still waits.
272#[must_use]
273pub fn summarize(blocked: &[Blocked], waiting: &dyn Fn(u64) -> bool) -> Vec<v1::BlockedConnection> {
274    let mut groups: HashMap<(Arc<str>, String, u16, u8, bool), v1::BlockedConnection> =
275        HashMap::new();
276    for b in blocked {
277        let destination = b.host.clone().unwrap_or_else(|| b.address.to_string());
278        let key = (
279            Arc::clone(&b.exe),
280            destination,
281            b.port,
282            b.protocol,
283            b.prompt.is_some(),
284        );
285        let entry = groups.entry(key).or_insert_with(|| v1::BlockedConnection {
286            exe: b.exe.to_string(),
287            host: b.host.clone().unwrap_or_default(),
288            address: convert::address(b.address),
289            port: u32::from(b.port),
290            protocol: u32::from(b.protocol),
291            context: convert::context(b.context).into(),
292            outcome: if b.prompt.is_some() {
293                v1::BlockedOutcome::Asked
294            } else {
295                v1::BlockedOutcome::Denied
296            }
297            .into(),
298            ..v1::BlockedConnection::default()
299        });
300        entry.count += 1;
301        if b.time_ms >= entry.last_ms {
302            entry.last_ms = b.time_ms;
303            entry.prompt = b.prompt.filter(|&id| waiting(id));
304        }
305    }
306    let mut list: Vec<_> = groups.into_values().collect();
307    list.sort_by_key(|b| std::cmp::Reverse(b.last_ms));
308    list
309}

320fn risk(r: &Risk) -> v1::Risk {
321    let kind = match r {
322        Risk::AnyHost => "any_host",
323        Risk::PublicSuffix(_) => "public_suffix",
324        Risk::AnyProgram => "any_program",
325        Risk::AnyPort => "any_port",
326        Risk::WideNetwork(_) => "wide_network",
327        Risk::SensitiveAlias(_) => "sensitive_alias",
328        Risk::WritableBinary(_) => "writable_binary",
329        Risk::Speculative => "speculative",
330    };
331    v1::Risk {
332        kind: kind.to_owned(),
333        description: r.describe(),
334    }
335}

366/// Shapes one proposed rule against the agent's refused connections: its conditions, risks and evidence.
367/// `allowed` tells whether today's rules let a refused connection through already.
368pub fn shape(
369    proposed: &Proposed,
370    blocked: &[Blocked],
371    waiting: &dyn Fn(u64) -> bool,
372    allowed: &dyn Fn(&Blocked, &Conditions) -> bool,
373) -> Result<v1::ShapedRule, proposal::Refusal> {
374    let observed: Vec<Observation> = blocked.iter().map(observation).collect();
375    let shaped = proposal::shape(
376        proposed,
377        &observed,
378        &crate::process::writable_by_others_path,
379    )?;
380    let exe = shaped.conditions.exe.first().cloned().unwrap_or_default();
381    let evidence: Vec<Blocked> = blocked
382        .iter()
383        .filter(|b| proposal::grounds(&observation(b), &shaped.conditions, &exe))
384        .cloned()
385        .collect();
386    let already_allowed =
387        !evidence.is_empty() && evidence.iter().all(|b| allowed(b, &shaped.conditions));
388    Ok(v1::ShapedRule {
389        conditions: Some(convert::conditions(&shaped.conditions)),
390        risks: shaped.risks.iter().map(risk).collect(),
391        evidence: summarize(&evidence, waiting),
392        already_allowed,
393        included: false,
394    })
395}

397/// Facts for a refused connection, to decide it again by today's rules.
398#[must_use]
399pub fn decide_again(rules: &snitchhiker_core::engine::RuleSet, b: &Blocked, now_ms: u64) -> bool {
400    let hosts: Vec<Arc<str>> = b.host.iter().map(|h| Arc::from(h.as_str())).collect();
401    let home = crate::process::home_of(b.uid);
402    let facts = Facts {
403        exe: &b.exe,
404        apps: &[],
405        cmdline: "",
406        ancestors: &[],
407        lineage: &b.lineage,
408        uid: b.uid,
409        home: home.as_deref(),
410        hosts: &hosts,
411        address: b.address,
412        port: b.port,
413        protocol: b.protocol,
414        context: b.context,
415        now_ms,
416    };
417    rules
418        .decide(&facts)
419        .is_some_and(|d| d.action == Action::Allow)
420}

422/// A short name for a rule made from a proposal: the program and where to.
423#[must_use]
424pub fn rule_name(conditions: &Conditions) -> String {
425    let program = conditions
426        .exe
427        .first()
428        .and_then(|e| e.rsplit('/').next())
429        .unwrap_or("any program");
430    let mut to: Vec<&str> = conditions
431        .host
432        .iter()
433        .chain(&conditions.net)
434        .map(String::as_str)
435        .collect();
436    if to.len() > 3 {
437        to.truncate(3);
438        to.push("…");
439    }
440    let mut name = if to.is_empty() {
441        program.to_owned()
442    } else {
443        format!("{program} → {}", to.join(", "))
444    };
445    if !conditions.port.is_empty() {
446        name.push_str(" :");
447        name.push_str(&conditions.port.join(","));
448    }
449    name
450}

452/// The folder for an agent's rules: `Agents/<agent>/<project>`, made in `batch` where missing.
453pub fn folder(
454    store: &Store,
455    batch: &Batch,
456    provenance: &v1::Provenance,
457) -> Result<FolderId, StoreError> {
458    let agent = provenance
459        .agent_exe
460        .rsplit('/')
461        .next()
462        .filter(|n| !n.is_empty())
463        .unwrap_or("agent");
464    let project = provenance
465        .cwd
466        .rsplit('/')
467        .find(|n| !n.is_empty())
468        .unwrap_or("/");
469    let mut parent: Option<FolderId> = None;
470    for name in [AGENTS, agent, project] {
471        let folders = store.folders()?;
472        let found = folders
473            .iter()
474            .find(|f| f.parent == parent && f.name == name && f.scope.is_empty())
475            .map(|f| f.id)
476            .filter(|id| id & crate::managed::MANAGED_ID == 0);
477        parent = Some(match found {
478            Some(id) => id,
479            None => store.put_folder(
480                &Folder {
481                    id: 0,
482                    parent,
483                    name: name.to_owned(),
484                    enabled: true,
485                    scope: Vec::new(),
486                },
487                batch,
488            )?,
489        });
490    }
491    Ok(parent.unwrap_or(0))
492}

494/// A rule from an approved proposal's rule.
495pub fn rule(
496    conditions: &Conditions,
497    folder: FolderId,
498    duration: v1::ProposalDuration,
499    agent: ProcessRef,
500    proposal: u64,
501    now_ms: u64,
502) -> Result<Rule, snitchhiker_core::spec::SpecError> {
503    let mut conditions_model = conditions.to_conditions()?;
504    let expiry = match duration {
505        v1::ProposalDuration::Hour => Expiry::At(now_ms + 3_600_000),
506        v1::ProposalDuration::TwelveHours => Expiry::At(now_ms + 12 * 3_600_000),
507        v1::ProposalDuration::Forever => Expiry::Never,
508        v1::ProposalDuration::Session | v1::ProposalDuration::Unspecified => {
509            conditions_model.push(Condition::Session(vec![agent]));
510            Expiry::Never
511        }
512    };
513    Ok(Rule {
514        id: 0,
515        folder: Some(folder),
516        name: rule_name(conditions),
517        enabled: true,
518        action: Action::Allow,
519        tier: Tier::Normal,
520        expiry,
521        log: true,
522        notes: format!("From agent proposal {proposal}."),
523        conditions: conditions_model,
524    })
525}
