1//! The daemon's local API: the client socket (the desktop app, the MCP server: verified by the kernel, see `peer`) and
2//! the admin socket (root, the command line), both speaking the client protocol (`snitchhiker-proto`).
3//!
4//! Each connection has a thread reading its requests and a writer thread draining its outbound queue (responses and
5//! subscription notices), so a slow client never blocks the node.
6

28pub const RUN_DIR: &str = "/run/snitchhiker";
29pub const CLIENT_SOCKET: &str = "/run/snitchhiker/client.sock";

35/// How many clients are connected.
36pub fn connected() -> u64 {
37    OPEN.load(Ordering::Relaxed)
38}

40/// Binds both sockets and serves them on threads of their own.
41pub fn spawn(daemon: &Arc<Daemon>) -> std::io::Result<()> {
42    std::fs::create_dir_all(RUN_DIR)?;
43    std::fs::set_permissions(RUN_DIR, std::fs::Permissions::from_mode(0o755))?;
44    let config = &daemon.shared.config;
45    let group = group_id(&config.client_group);
46    if group.is_none() {
47        warn!(group = %config.client_group, "the client group doesn't exist: the desktop app can't connect");
48    }
49
50    let client = bind(CLIENT_SOCKET, 0o660, group)?;
51    {
52        let daemon = Arc::clone(daemon);
53        let access = Access::Client {
54            group: group.unwrap_or(u32::MAX),
55        };
56        thread::Builder::new()
57            .name("api-client".to_owned())
58            .spawn(move || accept(&daemon, &client, &access))?;
59    }
60    let admin = bind(ADMIN_SOCKET, 0o600, Some(0))?;
61    let daemon = Arc::clone(daemon);
62    thread::Builder::new()
63        .name("api-admin".to_owned())
64        .spawn(move || accept(&daemon, &admin, &Access::Admin))?;
65    Ok(())
66}

68fn bind(path: &str, mode: u32, group: Option<u32>) -> std::io::Result<UnixListener> {
69    let _ = std::fs::remove_file(path);
70    let listener = UnixListener::bind(path)?;
71    std::os::unix::fs::chown(path, Some(0), group)?;
72    std::fs::set_permissions(path, std::fs::Permissions::from_mode(mode))?;
73    Ok(listener)
74}

76/// The gid of a group, from the group database.
77#[must_use]
78pub fn group_id(name: &str) -> Option<u32> {
79    let groups = std::fs::read_to_string("/etc/group").ok()?;
80    groups.lines().find_map(|line| {
81        let mut fields = line.split(':');
82        (fields.next()? == name)
83            .then(|| fields.nth(1)?.parse().ok())
84            .flatten()
85    })
86}

97fn accept(daemon: &Arc<Daemon>, listener: &UnixListener, access: &Access) {
98    for stream in listener.incoming() {
99        let stream = match stream {
100            Ok(stream) => stream,
101            Err(error) => {
102                warn!(%error, "accepting a client");
103                continue;
104            }
105        };
106        let (caller, pidfd) = match admit(&stream, access, &daemon.shared.config) {
107            Ok(admitted) => admitted,
108            Err(reason) => {
109                info!(%reason, "refused a client");
110                refuse(&stream, &reason);
111                continue;
112            }
113        };
114        let daemon = Arc::clone(daemon);
115        let _ = thread::Builder::new()
116            .name("api-connection".to_owned())
117            .spawn(move || {
118                // Held until the connection ends; dropping it stops the watch.
119                let _watch = match pidfd.map(|pidfd| end_with_process(pidfd, &stream)) {
120                    Some(Err(error)) => {
121                        warn!(%error, "watching a client's process; refusing it");
122                        return;
123                    }
124                    Some(Ok(guard)) => Some(guard),
125                    None => None,
126                };
127                OPEN.fetch_add(1, Ordering::Relaxed);
128                if let Err(error) = serve(&daemon, &stream, &caller) {
129                    debug!(%error, "client connection ended");
130                }
131                OPEN.fetch_sub(1, Ordering::Relaxed);
132            });
133    }
134}

136/// Who is on the other end of a connection.
137struct Caller {
138    role: v1::Role,
139    uid: u32,
140    pid: u32,
141    /// A client's process, for step-up (root's connections need none).
142    pidfd: Option<OwnedFd>,
143}

145/// Who a connection is, or why it is refused. For a client, also its process's pidfd: the connection ends when that
146/// process does.
147fn admit(
148    stream: &UnixStream,
149    access: &Access,
150    config: &Config,
151) -> Result<(Caller, Option<OwnedFd>), String> {
152    match access {
153        Access::Admin => {
154            let credentials =
155                getsockopt(stream, sockopt::PeerCredentials).map_err(|e| e.to_string())?;
156            if credentials.uid() == 0 {
157                let caller = Caller {
158                    role: v1::Role::Root,
159                    uid: 0,
160                    pid: u32::try_from(credentials.pid()).unwrap_or(0),
161                    pidfd: None,
162                };
163                Ok((caller, None))
164            } else {
165                Err("the admin socket is root's".to_owned())
166            }
167        }
168        Access::Client { group } => {
169            let policy = crate::trust::policy(config, *group)
170                .map_err(|e| format!("the trusted-client registry: {e}"))?;
171            let (peer, pidfd) = identify(stream).map_err(|e| e.to_string())?;
172            verify(&peer, &policy).map_err(|e| e.to_string())?;
173            let caller = Caller {
174                role: v1::Role::Admin,
175                uid: peer.uid,
176                pid: peer.pid,
177                pidfd: Some(pidfd.try_clone().map_err(|e| e.to_string())?),
178            };
179            Ok((caller, Some(pidfd)))
180        }
181    }
182}

184fn refuse(stream: &UnixStream, reason: &str) {
185    let frame = v1::ServerFrame {
186        body: Some(v1::server_frame::Body::Refusal(v1::Refusal {
187            reason: reason.to_owned(),
188        })),
189    };
190    let _ = frame::write(&mut &*stream, &frame, &mut Vec::new());
191}

193fn serve(daemon: &Arc<Daemon>, stream: &UnixStream, caller: &Caller) -> std::io::Result<()> {
194    let role = caller.role;
195    let connection = CONNECTIONS.fetch_add(1, Ordering::Relaxed) + 1;
196    let mut reader = BufReader::new(stream.try_clone()?);
197    let mut buffer = Vec::new();
198    let major = frame::read_preamble(&mut reader)?;
199    if major != MAJOR {
200        refuse(
201            stream,
202            &format!("this node speaks protocol {MAJOR}, not {major}"),
203        );
204        return Ok(());
205    }
206    let hello: v1::ClientFrame = frame::read(&mut reader, &mut buffer)?;
207    let Some(v1::client_frame::Body::Hello(hello)) = hello.body else {
208        refuse(stream, "expected a hello");
209        return Ok(());
210    };
211    debug!(app = %hello.app, version = %hello.app_version, ?role, "client connected");
212
213    let (outbound, frames) = sync_channel::<v1::ServerFrame>(OUTBOUND);
214    let writer = {
215        let stream = stream.try_clone()?;
216        thread::Builder::new()
217            .name("api-writer".to_owned())
218            .spawn(move || write_frames(stream, &frames))?
219    };
220    let welcome = v1::ServerFrame {
221        body: Some(v1::server_frame::Body::Welcome(v1::Welcome {
222            minor: MINOR,
223            daemon_version: daemon.version().to_owned(),
224            node_name: hostname(),
225            node_epoch: 0,
226            capabilities: vec!["prompts".to_owned()],
227            role: role.into(),
228        })),
229    };
230    let _ = outbound.send(welcome);
231    let result = loop {
232        let incoming: v1::ClientFrame = match frame::read(&mut reader, &mut buffer) {
233            Ok(frame) => frame,
234            Err(error) if error.kind() == std::io::ErrorKind::UnexpectedEof => break Ok(()),
235            Err(error) => break Err(error),
236        };
237        let Some(v1::client_frame::Body::Request(request)) = incoming.body else {
238            continue;
239        };
240        let id = request.id;
241        let result = handle(daemon, connection, caller, &hello.app, &outbound, request);
242        let response = v1::ServerFrame {
243            body: Some(v1::server_frame::Body::Response(v1::Response {
244                id,
245                result: Some(result),
246            })),
247        };
248        if outbound.send(response).is_err() {
249            break Ok(());
250        }
251    };
252    daemon.shared.hub.forget(connection);
253    drop(outbound);
254    let _ = writer.join();
255    result
256}

258fn write_frames(stream: UnixStream, frames: &Receiver<v1::ServerFrame>) {
259    let mut out = BufWriter::new(stream);
260    let mut buffer = Vec::new();
261    // Everything already waiting is written before one flush, so bursts go out together.
262    while let Ok(first) = frames.recv() {
263        let mut next = Some(first);
264        while let Some(frame) = next {
265            if frame::write(&mut out, &frame, &mut buffer).is_err() {
266                return;
267            }
268            next = frames.try_recv().ok();
269        }
270        if out.flush().is_err() {
271            return;
272        }
273    }
274}

292/// Whether `role` may call `method`.
293fn allowed(role: v1::Role, method: &v1::request::Method) -> bool {
294    use v1::request::Method;
295    let reads = matches!(
296        method,
297        Method::GetStatus(_)
298            | Method::ListRules(_)
299            | Method::ListEvents(_)
300            | Method::Subscribe(_)
301            | Method::Unsubscribe(_)
302            | Method::ListPrompts(_)
303            | Method::ExportRules(_)
304            | Method::ListAudit(_)
305    );
306    match role {
307        v1::Role::Root | v1::Role::Admin => true,
308        v1::Role::Viewer => reads,
309        v1::Role::Agent | v1::Role::Unspecified => false,
310    }
311}

313fn handle(
314    daemon: &Arc<Daemon>,
315    connection: u64,
316    caller: &Caller,
317    app: &str,
318    outbound: &SyncSender<v1::ServerFrame>,
319    request: v1::Request,
320) -> v1::response::Result {
321    use v1::request::Method;
322    use v1::response::Result as R;
323    let role = caller.role;
324    let Some(method) = request.method else {
325        return error(v1::ErrorCode::Unsupported, "unknown method");
326    };
327    if !allowed(role, &method) {
328        return error(v1::ErrorCode::Forbidden, "not allowed for this connection");
329    }
330    match method {
331        Method::GetStatus(_) => R::Status(status(daemon)),
332        Method::ListRules(_) => list_rules(daemon),
333        Method::AddRule(add) => add_rule(daemon, caller, app, add),
334        Method::DeleteRule(delete) => delete_rule(daemon, delete.id),
335        Method::SetRuleEnabled(set) => set_rule_enabled(daemon, &set),
336        Method::ListEvents(list) => list_events(daemon, &list),
337        Method::Subscribe(subscribe) => self::subscribe(daemon, connection, outbound, &subscribe),
338        Method::Unsubscribe(unsubscribe) => {
339            if daemon
340                .shared
341                .hub
342                .unsubscribe(connection, unsubscribe.subscription)
343            {
344                R::Done(v1::Done {})
345            } else {
346                error(v1::ErrorCode::NotFound, "no such subscription")
347            }
348        }
349        Method::ListPrompts(_) => R::Prompts(v1::PromptList {
350            prompts: daemon.shared.broker.list(),
351        }),
352        Method::AnswerPrompt(answer) => answer_prompt(daemon, caller, app, &answer),
353        Method::Learn(learn) => self::learn(daemon, &learn),
354        Method::ImportRules(import) => import_rules(daemon, caller, app, &import),
355        Method::ListAudit(list) => list_audit(daemon, &list),
356        Method::ExportRules(_) => export_rules(daemon),
357    }
358}

360fn add_rule(daemon: &Daemon, caller: &Caller, app: &str, add: v1::AddRule) -> v1::response::Result {
361    let Some(wire) = add.rule else {
362        return error(v1::ErrorCode::Invalid, "no rule");
363    };
364    let mut rule = match convert::rule_from(&wire) {
365        Ok(rule) => rule,
366        Err(e) => return error(v1::ErrorCode::Invalid, e.to_string()),
367    };
368    rule.id = 0;
369    let scope = match scope_of(daemon, rule.folder) {
370        Ok(scope) => scope,
371        Err(e) => return internal(e),
372    };
373    if allows_everything(rule.action, rule.conditions.iter().chain(&scope))
374        && let Err(refused) = step_up(
375            daemon,
376            caller,
377            app,
378            &format!(
379                "add the rule \"{}\", which allows every program to connect anywhere",
380                rule.name
381            ),
382        )
383    {
384        return refused;
385    }
386    let added = daemon.store().insert_rule(&rule, now_ms());
387    match added
388        .map_err(|e| e.to_string())
389        .and_then(|id| daemon.reload_rules().map(|_| id).map_err(|e| e.to_string()))
390    {
391        Ok(id) => v1::response::Result::RuleAdded(v1::RuleAdded { id }),
392        Err(e) => internal(e),
393    }
394}

396fn delete_rule(daemon: &Daemon, id: u64) -> v1::response::Result {
397    if id & crate::managed::MANAGED_ID != 0 {
398        return error(v1::ErrorCode::Locked, "a default or managed rule");
399    }
400    let deleted = daemon.store().delete_rule(id);
401    match deleted {
402        Ok(true) => reload(daemon),
403        Ok(false) => error(v1::ErrorCode::NotFound, format!("no rule {id}")),
404        Err(e) => internal(e),
405    }
406}

408fn set_rule_enabled(daemon: &Daemon, set: &v1::SetRuleEnabled) -> v1::response::Result {
409    if set.id & crate::managed::MANAGED_ID != 0 {
410        return error(v1::ErrorCode::Locked, "a default or managed rule");
411    }
412    let changed = daemon
413        .store()
414        .set_rule_enabled(set.id, set.enabled, now_ms());
415    match changed {
416        Ok(true) => reload(daemon),
417        Ok(false) => error(v1::ErrorCode::NotFound, format!("no rule {}", set.id)),
418        Err(e) => internal(e),
419    }
420}

437fn answer_prompt(
438    daemon: &Daemon,
439    caller: &Caller,
440    app: &str,
441    answer: &v1::AnswerPrompt,
442) -> v1::response::Result {
443    let Some(action) = convert::action_from(answer.action()) else {
444        return error(v1::ErrorCode::Invalid, "no action");
445    };
446    // The suggestions all name a program or a destination; conditions of the client's own may not.
447    if let Some(conditions) = &answer.conditions
448        && answer.duration() != v1::Duration::Once
449    {
450        let conditions = match convert::conditions_from(conditions).to_conditions() {
451            Ok(conditions) => conditions,
452            Err(e) => return error(v1::ErrorCode::Invalid, e.to_string()),
453        };
454        if allows_everything(action, &conditions)
455            && let Err(refused) = step_up(
456                daemon,
457                caller,
458                app,
459                "answer a prompt with a rule that allows every program to connect anywhere",
460            )
461        {
462            return refused;
463        }
464    }
465    let suggestion = usize::try_from(answer.suggestion).unwrap_or(usize::MAX);
466    match daemon.answer_prompt(
467        answer.id,
468        action,
469        answer.duration(),
470        suggestion,
471        answer.conditions.as_ref(),
472        answer.folder,
473        app,
474    ) {
475        Ok(_) => v1::response::Result::Done(v1::Done {}),
476        Err(AnswerError::NotFound) => {
477            error(v1::ErrorCode::NotFound, AnswerError::NotFound.to_string())
478        }
479        Err(AnswerError::Invalid(message)) => error(v1::ErrorCode::Invalid, message),
480        Err(AnswerError::Runtime(e)) => internal(e),
481    }
482}

484fn learn(daemon: &Daemon, learn: &v1::Learn) -> v1::response::Result {
485    let since = now_ms().saturating_sub(u64::from(learn.hours.max(1)) * 3_600_000);
486    let folder = if learn.folder.is_empty() {
487        "Learned"
488    } else {
489        &learn.folder
490    };
491    let seen = daemon.store().undecided_since(since);
492    match seen {
493        Ok(seen) => rule_file(&crate::learn::suggest(
494            &seen,
495            folder,
496            crate::process::home_of,
497        )),
498        Err(e) => internal(e),
499    }
500}

530fn status(daemon: &Daemon) -> v1::Status {
531    let config: &Config = &daemon.shared.config;
532    v1::Status {
533        daemon_version: daemon.version().to_owned(),
534        mode: match config.mode {
535            Mode::Enforce => v1::Mode::Enforce,
536            Mode::Monitor => v1::Mode::Monitor,
537        }
538        .into(),
539        no_match: match config.no_match {
540            Policy::Allow => v1::Policy::Allow,
541            Policy::Deny => v1::Policy::Deny,
542            Policy::Reject => v1::Policy::Reject,
543            Policy::Ask => v1::Policy::Ask,
544        }
545        .into(),
546        fail_open: config.fail == crate::config::Fail::Open,
547        decided: daemon.shared.decided.load(Ordering::Relaxed),
548        dropped_events: daemon.shared.dropped_events.load(Ordering::Relaxed),
549        rules: u32::try_from(daemon.shared.rules.load().len()).unwrap_or(u32::MAX),
550        pending_prompts: u32::try_from(daemon.shared.broker.count()).unwrap_or(u32::MAX),
551        clients: u32::try_from(connected()).unwrap_or(u32::MAX),
552    }
553}

555fn list_rules(daemon: &Daemon) -> v1::response::Result {
556    let config = &daemon.shared.config;
557    let managed = match crate::managed::Managed::load(config.default_rules, &config.rules_dir) {
558        Ok(managed) => managed,
559        Err(e) => return internal(e),
560    };
561    let store = daemon.store();
562    let (folders, rules) = match (store.folders(), store.rules()) {
563        (Ok(folders), Ok(rules)) => (folders, rules),
564        (Err(e), _) | (_, Err(e)) => return internal(e),
565    };
566    drop(store);
567    v1::response::Result::Rules(v1::RuleList {
568        folders: folders
569            .iter()
570            .map(|f| convert::folder(f, false))
571            .chain(managed.folders.iter().map(|f| convert::folder(f, true)))
572            .collect(),
573        rules: rules
574            .iter()
575            .map(|r| convert::rule(r, false))
576            .chain(managed.rules.iter().map(|r| convert::rule(r, true)))
577            .collect(),
578    })
579}

581fn import_rules(
582    daemon: &Daemon,
583    caller: &Caller,
584    app: &str,
585    import: &v1::ImportRules,
586) -> v1::response::Result {
587    let file: snitchhiker_core::spec::RuleFile = match toml::from_str(&import.toml) {
588        Ok(file) => file,
589        Err(e) => return error(v1::ErrorCode::Invalid, e.to_string()),
590    };
591    let name = file.folder.as_ref().map_or_else(
592        || {
593            if import.folder.is_empty() {
594                "Imported".to_owned()
595            } else {
596                import.folder.clone()
597            }
598        },
599        |f| f.name.clone(),
600    );
601    let scope = match &file.folder {
602        Some(folder) => match folder.scope.to_conditions() {
603            Ok(scope) => scope,
604            Err(e) => return error(v1::ErrorCode::Invalid, e.to_string()),
605        },
606        None => Vec::new(),
607    };
608    // Every rule is checked before anything is stored.
609    let rules = match file
610        .rule
611        .iter()
612        .map(|spec| spec.to_rule(0, None))
613        .collect::<Result<Vec<_>, _>>()
614    {
615        Ok(rules) => rules,
616        Err(e) => return error(v1::ErrorCode::Invalid, e.to_string()),
617    };
618    let everything = rules
619        .iter()
620        .filter(|rule| {
621            rule.enabled && allows_everything(rule.action, rule.conditions.iter().chain(&scope))
622        })
623        .count();
624    if everything > 0
625        && let Err(refused) = step_up(
626            daemon,
627            caller,
628            app,
629            &format!(
630                "import {everything} rule(s) into \"{name}\" that allow every program to connect anywhere"
631            ),
632        )
633    {
634        return refused;
635    }
636    let stored = {
637        let mut store = daemon.store();
638        store
639            .insert_folder(&snitchhiker_core::rule::Folder {
640                id: 0,
641                parent: None,
642                name,
643                enabled: true,
644                scope,
645            })
646            .and_then(|folder| {
647                let now = now_ms();
648                rules.into_iter().try_for_each(|mut rule| {
649                    rule.folder = Some(folder);
650                    store.insert_rule(&rule, now).map(|_| ())
651                })
652            })
653    };
654    match stored {
655        Ok(()) => reload(daemon),
656        Err(e) => internal(e),
657    }
658}

661    v1::ConnectionEvent {
662        seq: stored.seq,

664        exe: stored.exe.clone(),
665        host: stored.host.clone().unwrap_or_default(),
666        uid: stored.uid,
667        pid: stored.pid,
668        protocol: u32::from(stored.protocol),
669        address: convert::address(stored.address),
670        port: u32::from(stored.port),
671        context: match stored.context {
672            1 => v1::Context::Container,
673            2 => v1::Context::Vm,
674            3 => v1::Context::Netns,
675            _ => v1::Context::Host,
676        }
677        .into(),
678        action: convert::action(stored.action).into(),
679        source: match stored.source {
680            1 => v1::Source::Rule,
681            2 => v1::Source::Policy,
682            3 => v1::Source::Monitor,

688        rule: stored.rule,
689        bytes_out: stored.bytes_out,
690        bytes_in: stored.bytes_in,
691    }

694/// The conditions the folder `folder` and those above it add to their rules.
695fn scope_of(daemon: &Daemon, folder: Option<u64>) -> Result<Vec<Condition>, crate::store::Error> {
696    let Some(mut next) = folder else {
697        return Ok(Vec::new());
698    };
699    let folders = daemon.store().folders()?;
700    let mut scope = Vec::new();
701    // A cycle can't be stored, but the walk is bounded anyway.
702    for _ in 0..folders.len() {
703        let Some(folder) = folders.iter().find(|f| f.id == next) else {
704            break;
705        };
706        scope.extend(folder.scope.iter().cloned());
707        match folder.parent {
708            Some(parent) => next = parent,
709            None => break,
710        }
711    }
712    Ok(scope)
713}

715/// A dangerous action asked for by `caller`: goes ahead when root asks, when step-up is off, or when polkit authorizes
716/// the client's process; otherwise the error to answer with. Every attempt is audited.
717fn step_up(
718    daemon: &Daemon,
719    caller: &Caller,
720    app: &str,
721    action: &str,
722) -> Result<(), v1::response::Result> {
723    let unchecked = caller.role == v1::Role::Root || daemon.shared.config.step_up == StepUp::Off;
724    let outcome = if unchecked {
725        v1::AuditOutcome::Unchecked
726    } else {
727        let message = format!("Authentication is required to {action}.");
728        match caller
729            .pidfd
730            .as_ref()
731            .map(|pidfd| crate::stepup::authorize(pidfd, caller.uid, &message))
732        {
733            Some(Ok(true)) => v1::AuditOutcome::Authorized,
734            Some(Ok(false)) | None => v1::AuditOutcome::Refused,
735            Some(Err(reason)) => {
736                warn!(%reason, "step-up authorization couldn't be asked for; refused");
737                v1::AuditOutcome::Refused
738            }
739        }
740    };
741    let entry = AuditEntry {
742        id: 0,
743        time_ms: now_ms(),
744        app: app.to_owned(),
745        uid: caller.uid,
746        pid: caller.pid,
747        action: action.to_owned(),
748        outcome: u8::try_from(i32::from(outcome)).unwrap_or(0),
749    };
750    let recorded = daemon.store().audit(&entry);
751    if let Err(error) = recorded {
752        // An action that can't be audited doesn't happen.
753        return Err(internal(format!("auditing: {error}")));
754    }
755    info!(
756        app,
757        uid = caller.uid,
758        pid = caller.pid,
759        action,
760        ?outcome,
761        "dangerous action"
762    );
763    if outcome == v1::AuditOutcome::Refused {
764        return Err(error(
765            v1::ErrorCode::NotAuthorized,
766            format!("not authorized to {action}"),
767        ));
768    }
769    Ok(())
770}

772fn list_audit(daemon: &Daemon, list: &v1::ListAudit) -> v1::response::Result {
773    let before = if list.before == 0 {
774        u64::MAX
775    } else {
776        list.before
777    };
778    let page = daemon.store().audit_page(before, list.limit.clamp(1, 1000));
779    match page {
780        Ok(entries) => v1::response::Result::Audit(v1::AuditPage {
781            entries: entries
782                .into_iter()
783                .map(|e| v1::AuditEntry {
784                    id: e.id,
785                    time_ms: e.time_ms,
786                    app: e.app,
787                    uid: e.uid,
788                    pid: e.pid,
789                    action: e.action,
790                    outcome: i32::from(e.outcome),
791                })
792                .collect(),
793        }),
794        Err(e) => internal(e),
795    }
796}

798/// The most stored events a subscription catches up with; further behind, the client pages with `ListEvents`.
799const BACKFILL: u32 = 4000;
800
801fn subscribe(
802    daemon: &Daemon,
803    connection: u64,
804    outbound: &SyncSender<v1::ServerFrame>,
805    subscribe: &v1::Subscribe,
806) -> v1::response::Result {
807    let topic = subscribe.topic();
808    let backfill = || -> Option<Vec<v1::notice::Body>> {
809        match (topic, subscribe.after) {
810            (v1::Topic::Prompts, _) => Some(
811                daemon
812                    .shared
813                    .broker
814                    .list()
815                    .into_iter()
816                    .map(v1::notice::Body::Prompt)
817                    .collect(),
818            ),
819            (v1::Topic::Events, Some(after)) => {
820                let events = match daemon.store().events_after(after, BACKFILL + 1) {
821                    Ok(events) => events,
822                    Err(error) => {
823                        warn!(%error, "reading events for a subscription");
824                        return None;
825                    }
826                };
827                (events.len() <= BACKFILL as usize).then(|| {
828                    events
829                        .iter()
830                        .map(|e| v1::notice::Body::Event(event(e)))
831                        .collect()
832                })
833            }
834            _ => Some(Vec::new()),
835        }
836    };
837    if topic == v1::Topic::Unspecified {
838        return error(v1::ErrorCode::Invalid, "no topic");
839    }
840    match daemon
841        .shared
842        .hub
843        .subscribe_with(connection, topic, outbound.clone(), backfill)
844    {
845        Ok(subscription) => v1::response::Result::Subscribed(v1::Subscribed { subscription }),
846        Err(crate::hub::BackfillError::TooMuch) => error(
847            v1::ErrorCode::Invalid,
848            format!(
849                "more than {BACKFILL} events since {}: page them with ListEvents, then subscribe from the newest",
850                subscribe.after.unwrap_or_default()
851            ),
852        ),
853        Err(crate::hub::BackfillError::Closed) => {
854            error(v1::ErrorCode::Internal, "the connection closed")
855        }
856    }
857}
