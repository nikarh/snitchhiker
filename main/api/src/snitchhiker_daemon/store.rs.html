1//! The daemon's state: rules, folders and the connection history, in one SQLite database.
2//!
3//! Every statement the daemon runs is in [`STATEMENTS`]; a test runs `EXPLAIN QUERY PLAN` on each against a populated
4//! database and fails on a table scan or a temporary sort, so every query stays an index seek. Strings that repeat
5//! (executables, hosts) are stored once and referred to by id.
6

142    (

146    (

152        "INSERT INTO folders (parent, name, enabled, scope) VALUES (?1, ?2, ?3, ?4)",
153    ),
154    (
155        "strings.find",
156        "SELECT id FROM strings WHERE kind = ?1 AND value = ?2",
157    ),
158    (
159        "strings.insert",
160        "INSERT INTO strings (kind, value) VALUES (?1, ?2)",
161    ),
162    (
163        "events.insert",
164        "INSERT INTO events (time_ms, exe, host, uid, pid, protocol, address, port, context, action, source, rule,
165                             cookie)

171                e.action, e.source, e.rule, e.bytes_out, e.bytes_in

183    // Connections no rule decided, since a time: what `learn` turns into rules.
184    (
185        "events.undecided_since",
186        "SELECT x.value, h.value, e.uid, e.protocol, e.address, e.port

297fn sql(name: &'static str) -> Result<&'static str, Error> {
298    STATEMENTS
299        .iter()
300        .find(|(n, _)| *n == name)
301        .map(|(_, sql)| *sql)
302        .ok_or(Error::UnknownStatement(name))
303}

316    fn init(connection: Connection) -> Result<Self, Error> {
317        connection.execute_batch(
318            "PRAGMA journal_mode = WAL;
319             PRAGMA synchronous = NORMAL;
320             PRAGMA foreign_keys = ON;
321             PRAGMA temp_store = MEMORY;
322             PRAGMA busy_timeout = 5000;",
323        )?;
324        let version: i64 = connection.query_row("PRAGMA user_version", [], |row| row.get(0))?;
325        let version = usize::try_from(version).unwrap_or(0);
326        for (index, migration) in MIGRATIONS.iter().enumerate().skip(version) {
327            let transaction = connection.unchecked_transaction()?;
328            transaction.execute_batch(migration)?;
329            transaction.pragma_update(
330                None,
331                "user_version",
332                i64::try_from(index + 1).unwrap_or(i64::MAX),
333            )?;
334            transaction.commit()?;
335        }
336        Ok(Self {
337            connection,
338            strings: HashMap::new(),
339        })
340    }

464    /// Records events in one transaction.
465    /// Records `events`; their sequence numbers replace what `seqs` held, in the same order.
466    pub fn record(&mut self, events: &[Event], seqs: &mut Vec<i64>) -> Result<(), Error> {
467        seqs.clear();
468        let transaction = self.connection.unchecked_transaction()?;
469        for event in events {
470            let exe = intern(
471                &self.connection,
472                &mut self.strings,
473                StringKind::Exe,
474                &event.exe,
475            )?;
476            let host = match &event.host {
477                Some(host) => Some(intern(
478                    &self.connection,
479                    &mut self.strings,
480                    StringKind::Host,
481                    host,
482                )?),
483                None => None,
484            };
485            let address = match event.address {
486                std::net::IpAddr::V4(a) => a.to_ipv6_mapped().octets(),
487                std::net::IpAddr::V6(a) => a.octets(),
488            };
489            self.connection
490                .prepare_cached(sql("events.insert")?)?
491                .execute(params![
492                    db_id(event.time_ms),
493                    exe,
494                    host,
495                    event.uid,
496                    event.pid,
497                    event.protocol,
498                    &address[..],
499                    event.port,
500                    event.context,
501                    action_code(event.action),
502                    event.source as u8,
503                    event.rule.map(db_id),
504                    event.cookie.map(db_id),
505                ])?;
506            seqs.push(self.connection.last_insert_rowid());
507        }
508        transaction.commit()?;
509        Ok(())
510    }

605/// The id of a string, adding it if new. Cached, so repeated strings cost no query.
606fn intern(
607    connection: &Connection,
608    cache: &mut HashMap<(StringKind, std::sync::Arc<str>), i64>,
609    kind: StringKind,
610    value: &std::sync::Arc<str>,
611) -> Result<i64, Error> {
612    if let Some(&id) = cache.get(&(kind, std::sync::Arc::clone(value))) {
613        return Ok(id);
614    }
615    let found: Option<i64> = connection
616        .prepare_cached(sql("strings.find")?)?
617        .query_row(params![kind as u8, &**value], |row| row.get(0))
618        .optional()?;
619    let id = if let Some(id) = found {
620        id
621    } else {
622        connection
623            .prepare_cached(sql("strings.insert")?)?
624            .execute(params![kind as u8, &**value])?;
625        connection.last_insert_rowid()
626    };
627    cache.insert((kind, std::sync::Arc::clone(value)), id);
628    Ok(id)
629}

636/// An address as stored: 16 bytes, IPv4 mapped into IPv6.
637fn address_from(bytes: &[u8]) -> std::net::IpAddr {
638    <[u8; 16]>::try_from(bytes).map_or(
639        std::net::IpAddr::V6(std::net::Ipv6Addr::UNSPECIFIED),
640        |octets| {
641            let v6 = std::net::Ipv6Addr::from(octets);
642            v6.to_ipv4_mapped()
643                .map_or(std::net::IpAddr::V6(v6), std::net::IpAddr::V4)
644        },
645    )
646}
647
648fn action_code(action: Action) -> u8 {
649    match action {
650        Action::Allow => 1,
651        Action::Deny => 2,
652        Action::Reject => 3,
653    }
654}
655
656fn action_from(code: u8) -> Action {
657    match code {
658        1 => Action::Allow,
659        3 => Action::Reject,
660        _ => Action::Deny,
661    }
662}

664/// Ids and times are unsigned in the model and signed in SQLite; both stay far below 2^63.
665fn db_id(id: u64) -> i64 {
666    i64::try_from(id).unwrap_or(i64::MAX)
667}

669/// An event as `events.page` and `events.after` select it.
670fn stored_event(row: &rusqlite::Row<'_>) -> rusqlite::Result<StoredEvent> {
671    let address: Vec<u8> = row.get(7)?;
672    Ok(StoredEvent {
673        seq: row.get(0)?,
674        time_ms: u64::try_from(row.get::<_, i64>(1)?).unwrap_or(0),
675        exe: row.get(2)?,
676        host: row.get(3)?,
677        uid: row.get(4)?,
678        pid: row.get(5)?,
679        protocol: row.get(6)?,
680        address: address_from(&address),
681        port: row.get(8)?,
682        context: row.get(9)?,
683        action: action_from(row.get(10)?),
684        source: row.get(11)?,
685        rule: row.get::<_, Option<i64>>(12)?.map(id_of),
686        bytes_out: id_of(row.get(13)?),
687        bytes_in: id_of(row.get(14)?),
688    })
689}
690
691fn id_of(id: i64) -> u64 {
692    u64::try_from(id).unwrap_or(0)
693}
