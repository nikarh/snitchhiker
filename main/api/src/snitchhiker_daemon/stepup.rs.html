1//! Step-up: a dangerous action asked for by a client needs more than the client's session. polkit authorizes the
2//! client's process for `org.snitchhiker.manage` (dist/polkit), which by default asks for an administrator's password
3//! through the user's polkit agent. The process is named to polkit by its pidfd, so its pid can't be reused meanwhile.
4
5use std::collections::HashMap;
6use std::os::fd::OwnedFd;
7
8use zbus_polkit::policykit1::{AuthorityProxy, CheckAuthorizationFlags, Subject};
9
10/// The polkit action of dangerous actions.
11pub const ACTION_ID: &str = "org.snitchhiker.manage";
12
13/// Asks polkit whether the process of `pidfd` (running as `uid`) may do what `message` says; this waits while the
14/// user authenticates. An error means polkit couldn't be asked (no polkit, no system bus) or couldn't ask the user
15/// (no polkit agent in their session): refused, then, with the reason to tell them.
16///
17/// Called from the API's own threads, never from inside zbus, so `block_on` waits on this thread alone.
18pub fn authorize(pidfd: &OwnedFd, uid: u32, message: &str) -> Result<bool, String> {
19    zbus::block_on(async {
20        let connection = zbus::Connection::system()
21            .await
22            .map_err(|e| format!("the system bus: {e}"))?;
23        let authority = AuthorityProxy::new(&connection)
24            .await
25            .map_err(|e| format!("polkit: {e}"))?;
26        let subject =
27            Subject::new_for_owner(pidfd, uid).map_err(|e| format!("polkit subject: {e}"))?;
28        let details = HashMap::from([("polkit.message", message)]);
29        let result = authority
30            .check_authorization(
31                &subject,
32                ACTION_ID,
33                &details,
34                CheckAuthorizationFlags::ALLOW_USER_INTERACTION,
35                "",
36            )
37            .await
38            .map_err(|e| format!("polkit: {e}"))?;
39        // Not authorized, yet authorization could be had by authenticating: no agent was there to ask.
40        if !result.is_authorized && result.is_challenge {
41            return Err("no polkit agent is running in this session to ask for a password (desktops start one; \
42                 under a bare window manager, start one such as polkit-gnome or lxpolkit)"
43                .to_owned());
44        }
45        Ok(result.is_authorized)
46    })
47}
