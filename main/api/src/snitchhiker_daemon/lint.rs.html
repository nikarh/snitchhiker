13/// The node's side of the lint: the filesystem as root sees it, and the rules' times from the database.
14pub struct Facts {
15    /// For each rule: when it was last changed, and when it last decided a connection.
16    pub times: HashMap<RuleId, (u64, Option<u64>)>,
17}
18
19/// Paths a glob may expand to before the search gives up (a pattern over a huge directory).
20const EXPANSION_LIMIT: usize = 4096;
21
22impl NodeFacts for Facts {
23    fn exists(&self, path: &str) -> bool {
24        Path::new(path).exists()
25    }
26
27    fn writable_by_user(&self, path: &str) -> bool {
28        writable_by_user(Path::new(path))
29    }
30
31    fn exists_like(&self, glob: &str) -> bool {
32        let mut paths = vec![PathBuf::from("/")];
33        for segment in glob.trim_start_matches('/').split('/') {
34            if segment.is_empty() {
35                continue;
36            }
37            let wild = segment.contains(['*', '?', '{']);
38            let pattern = Pattern::compile(&PatternSource::Glob(segment.to_owned())).ok();
39            let mut next = Vec::new();
40            for path in &paths {
41                if !wild {
42                    let joined = path.join(segment);
43                    if joined.exists() {
44                        next.push(joined);
45                    }
46                    continue;
47                }
48                let Some(pattern) = &pattern else {
49                    return false;
50                };
51                let Ok(entries) = std::fs::read_dir(path) else {
52                    continue;
53                };
54                for entry in entries.flatten() {
55                    if entry
56                        .file_name()
57                        .to_str()
58                        .is_some_and(|name| pattern.matches(name, None))
59                    {
60                        next.push(entry.path());
61                    }
62                }
63            }
64            if next.len() > EXPANSION_LIMIT {
65                return false;
66            }
67            paths = next;
68            if paths.is_empty() {
69                return false;
70            }
71        }
72        true
73    }
74
75    fn last_hit_ms(&self, rule: RuleId) -> Option<u64> {
76        self.times.get(&rule).and_then(|(_, hit)| *hit)
77    }
78
79    fn changed_ms(&self, rule: RuleId) -> Option<u64> {
80        self.times.get(&rule).map(|(changed, _)| *changed)
81    }
82}

84/// Whether a user other than root can change the file or put another in its place: the file, or a directory above it,
85/// is owned by a user other than root, or writable by a group other than root's or by everyone (a sticky directory
86/// lets others replace only their own files).
87#[must_use]
88pub fn writable_by_user(path: &Path) -> bool {
89    let Ok(file) = std::fs::metadata(path) else {
90        return false;
91    };
92    let others_write = |meta: &std::fs::Metadata| {
93        let mode = meta.permissions().mode();
94        meta.uid() != 0 || (mode & 0o020 != 0 && meta.gid() != 0) || mode & 0o002 != 0
95    };
96    if others_write(&file) {
97        return true;
98    }
99    path.ancestors().skip(1).any(|dir| {
100        std::fs::metadata(dir).is_ok_and(|meta| {
101            let sticky = meta.permissions().mode() & 0o1000 != 0;
102            meta.uid() != 0 || (!sticky && others_write(&meta))
103        })
104    })
105}

107/// Whether a folder and every folder above it is enabled, and their scopes.
108fn folder_state(folders: &[Folder], mut next: Option<u64>) -> (bool, Vec<&Condition>) {
109    let mut enabled = true;
110    let mut scope = Vec::new();
111    for _ in 0..=folders.len() {
112        let Some(folder) = next.and_then(|id| folders.iter().find(|f| f.id == id)) else {
113            break;
114        };
115        enabled &= folder.enabled;
116        scope.extend(folder.scope.iter());
117        next = folder.parent;
118    }
119    (enabled, scope)
120}

122/// The report over `rules` (the database's, then the managed ones: `locked`), as the protocol has it.
123#[must_use]
124pub fn report(
125    rules: &[Rule],
126    managed: &[Rule],
127    folders: &[Folder],
128    facts: &dyn NodeFacts,
129    now_ms: u64,
130    unused_days: u32,
131) -> Vec<v1::LintFinding> {
132    let linted: Vec<Linted<'_>> = rules
133        .iter()
134        .map(|r| (r, false))
135        .chain(managed.iter().map(|r| (r, true)))
136        .map(|(rule, locked)| {
137            let (folders_enabled, scope) = folder_state(folders, rule.folder);
138            Linted {
139                rule,
140                conditions: rule.conditions.iter().chain(scope).collect(),
141                effective: rule.enabled && folders_enabled,
142                locked,
143            }
144        })
145        .collect();
146    let name = |id: RuleId| {
147        rules
148            .iter()
149            .chain(managed)
150            .find(|r| r.id == id)
151            .map_or_else(|| format!("rule {id}"), |r| format!("\"{}\"", r.name))
152    };
153    lint::lint(&linted, facts, now_ms, unused_days)
154        .into_iter()
155        .map(|finding| {
156            let description = match (finding.kind, finding.other) {
157                (Kind::Duplicate, Some(other)) => format!("duplicates {}", name(other)),
158                (Kind::Redundant, Some(other)) => format!(
159                    "{} already matches everything it does, with the same action",
160                    name(other)
161                ),
162                (Kind::Shadowed, Some(other)) => format!("{} always beats it", name(other)),
163                (Kind::NeverHit, _) => format!("decided no connection in {unused_days} days"),
164                (kind, _) => kind.describe().to_owned(),
165            };
166            v1::LintFinding {
167                kind: kind(finding.kind).into(),
168                rule: finding.rule,
169                other: finding.other,
170                suggestion: finding
171                    .suggestion
172                    .map_or_else(String::new, |s| s.to_string()),
173                description,
174            }
175        })
176        .collect()
177}
178
179fn kind(kind: Kind) -> v1::LintKind {
180    match kind {
181        Kind::Duplicate => v1::LintKind::Duplicate,
182        Kind::Redundant => v1::LintKind::Redundant,
183        Kind::Shadowed => v1::LintKind::Shadowed,
184        Kind::Expired => v1::LintKind::Expired,
185        Kind::NeverHit => v1::LintKind::NeverHit,
186        Kind::WritableBinary => v1::LintKind::WritableBinary,
187        Kind::MissingBinary => v1::LintKind::MissingBinary,
188    }
189}

191#[cfg(test)]
192mod tests {
193    use super::*;
194
195    #[test]
196    fn globs_expand_over_the_filesystem() {
197        let facts = Facts {
198            times: HashMap::new(),
199        };
200        let dir = std::env::temp_dir().join(format!("snitchhiker-lint-{}", std::process::id()));
201        std::fs::create_dir_all(dir.join("versions/2.0.40")).unwrap();
202        std::fs::write(dir.join("versions/2.0.40/app"), b"").unwrap();
203        let base = dir.to_str().unwrap();
204        assert!(facts.exists_like(&format!("{base}/versions/*/app")));
205        assert!(!facts.exists_like(&format!("{base}/versions/*/other")));
206        assert!(!facts.exists_like(&format!("{base}/nothing/*/app")));
207        std::fs::remove_dir_all(&dir).unwrap();
208    }
209
210    #[test]
211    fn a_users_file_is_writable_by_a_user() {
212        // The test runs as a user (in CI, and here): its own temporary file is theirs.
213        let path =
214            std::env::temp_dir().join(format!("snitchhiker-writable-{}", std::process::id()));
215        std::fs::write(&path, b"").unwrap();
216        if std::fs::metadata(&path).unwrap().uid() != 0 {
217            assert!(writable_by_user(&path));
218        }
219        std::fs::remove_file(&path).unwrap();
220        assert!(!writable_by_user(Path::new("/nonexistent/binary")));
221    }
