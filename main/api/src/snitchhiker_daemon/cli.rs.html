75#[derive(Debug, thiserror::Error)]
76pub enum Error {
77    #[error(transparent)]
78    Args(#[from] pico_args::Error),
79    #[error("{0}")]
80    Usage(String),
81    #[error("{0} (is the daemon running, and is this root?)")]
82    Connect(snitchhiker_client::Error),
83    #[error(transparent)]
84    Client(#[from] snitchhiker_client::Error),
85    #[error("writing to stdout: {0}")]
86    Stdout(io::Error),
87}
88
89fn out(text: &str) -> Result<(), Error> {
90    let mut stdout = io::stdout().lock();
91    stdout
92        .write_all(text.as_bytes())
93        .and_then(|()| stdout.flush())
94        .map_err(Error::Stdout)
95}

101fn reject_rest(args: pico_args::Arguments) -> Result<(), Error> {
102    let rest = args.finish();
103    if rest.is_empty() {
104        Ok(())
105    } else {
106        Err(Error::Usage(format!("unexpected arguments: {rest:?}")))
107    }
108}
109
110fn unexpected() -> Error {
111    Error::Usage("the daemon answered something unexpected".to_owned())
112}

114/// Runs a command; `version` is the binary's version string.
115pub fn run(mut args: pico_args::Arguments, version: &str) -> Result<(), Error> {
116    use v1::request::Method;
117    if args.contains(["-h", "--help"]) {
118        return out(HELP);
119    }
120    match args.subcommand()?.as_deref() {
121        None => out(HELP),
122        Some("version") => out(&format!("snitchhiker {version}\n")),
123        Some("status") => {
124            reject_rest(args)?;
125            let mut client = connect(version)?;
126            match client.call(Method::GetStatus(v1::GetStatus {}))? {
127                v1::response::Result::Status(status) => out(&format_status(&status)),
128                _ => Err(unexpected()),
129            }
130        }
131        Some("rules") => rules(args, version),
132        Some("audit") => {
133            let limit: u32 = args.opt_value_from_str("--limit")?.unwrap_or(50);
134            reject_rest(args)?;
135            let mut client = connect(version)?;
136            match client.call(Method::ListAudit(v1::ListAudit { before: 0, limit }))? {
137                v1::response::Result::Audit(page) => out(&format_audit(&page.entries)),
138                _ => Err(unexpected()),
139            }
140        }
141        Some("events") => {
142            let limit: u32 = args.opt_value_from_str("--limit")?.unwrap_or(50);
143            reject_rest(args)?;
144            let mut client = connect(version)?;
145            match client.call(Method::ListEvents(v1::ListEvents { before: 0, limit }))? {
146                v1::response::Result::Events(page) => out(&format_events(&page.events)),
147                _ => Err(unexpected()),
148            }
149        }
150        Some("learn") => {
151            let hours: u32 = args.opt_value_from_str("--hours")?.unwrap_or(24);
152            let folder: String = args
153                .opt_value_from_str("--folder")?
154                .unwrap_or_else(|| "Learned".to_owned());
155            reject_rest(args)?;
156            let mut client = connect(version)?;
157            print_file(client.call(Method::Learn(v1::Learn { hours, folder }))?)
158        }
159        Some("prompts") => prompts(args, version),
160        Some(other) => Err(Error::Usage(format!(
161            "unknown command `{other}`; see --help"
162        ))),
163    }
164}
165
166fn print_file(result: v1::response::Result) -> Result<(), Error> {
167    match result {
168        v1::response::Result::RuleFile(file) => out(&file.toml),
169        _ => Err(unexpected()),
170    }
171}

173/// `snitchhiker rules ...`.
174fn rules(mut args: pico_args::Arguments, version: &str) -> Result<(), Error> {
175    use v1::request::Method;
176    let command = args.subcommand()?;
177    let mut client = connect(version)?;
178    match command.as_deref() {
179        None | Some("list" | "ls") => {
180            reject_rest(args)?;
181            match client.call(Method::ListRules(v1::ListRules {}))? {
182                v1::response::Result::Rules(list) => out(&format_rules(&list)),
183                _ => Err(unexpected()),
184            }
185        }
186        Some("add") => {
187            let spec = spec_from_args(&mut args)?;
188            reject_rest(args)?;
189            let rule = spec
190                .to_rule(0, None)
191                .map_err(|e| Error::Usage(e.to_string()))?;
192            match client.call(Method::AddRule(v1::AddRule {
193                rule: Some(convert::rule(&rule, false)),
194            }))? {
195                v1::response::Result::RuleAdded(added) => {
196                    out(&format!("added rule {}\n", added.id))
197                }
198                _ => Err(unexpected()),
199            }
200        }
201        Some("rm" | "delete") => {
202            let id: u64 = args.free_from_str()?;
203            reject_rest(args)?;
204            client.call(Method::DeleteRule(v1::DeleteRule { id }))?;
205            out(&format!("deleted rule {id}\n"))
206        }
207        Some(which @ ("enable" | "disable")) => {
208            let id: u64 = args.free_from_str()?;
209            reject_rest(args)?;
210            client.call(Method::SetRuleEnabled(v1::SetRuleEnabled {
211                id,
212                enabled: which == "enable",
213            }))?;
214            out(&format!("{which}d rule {id}\n"))
215        }
216        Some("import") => {
217            let path: String = args.free_from_str()?;
218            reject_rest(args)?;
219            let toml =
220                std::fs::read_to_string(&path).map_err(|e| Error::Usage(format!("{path}: {e}")))?;
221            let folder = Path::new(&path).file_stem().map_or_else(
222                || "Imported".to_owned(),
223                |s| s.to_string_lossy().into_owned(),
224            );
225            client.call(Method::ImportRules(v1::ImportRules { toml, folder }))?;
226            out(&format!("imported {path}\n"))
227        }
228        Some("export") => {
229            reject_rest(args)?;
230            print_file(client.call(Method::ExportRules(v1::ExportRules {}))?)
231        }
232        Some(other) => Err(Error::Usage(format!("unknown rules command `{other}`"))),
233    }
234}

236/// `snitchhiker prompts ...`.
237fn prompts(mut args: pico_args::Arguments, version: &str) -> Result<(), Error> {
238    use v1::request::Method;
239    let command = args.subcommand()?;
240    let mut client = connect(version)?;
241    match command.as_deref() {
242        None | Some("list" | "ls") => {
243            reject_rest(args)?;
244            match client.call(Method::ListPrompts(v1::ListPrompts {}))? {
245                v1::response::Result::Prompts(list) => {
246                    out(&list.prompts.iter().map(format_prompt).collect::<String>())
247                }
248                _ => Err(unexpected()),
249            }
250        }
251        Some("watch") => {
252            reject_rest(args)?;
253            client.subscribe(v1::Topic::Prompts)?;
254            loop {
255                match client.notice()?.body {
256                    Some(v1::notice::Body::Prompt(prompt)) => out(&format_prompt(&prompt))?,
257                    Some(v1::notice::Body::PromptResolved(resolved)) => {
258                        out(&format_resolved(&resolved))?;
259                    }
260                    _ => {}
261                }
262            }
263        }
264        Some("answer") => {
265            let id: u64 = args.free_from_str()?;
266            let action = action(&args.free_from_str::<String>()?)?;
267            let suggestion: u32 = args.opt_value_from_str("--suggestion")?.unwrap_or(0);
268            let folder: Option<u64> = args.opt_value_from_str("--folder")?;
269            let duration = match args
270                .opt_value_from_str::<_, String>("--duration")?
271                .as_deref()
272            {
273                None | Some("forever") => v1::Duration::Forever,
274                Some("once") => v1::Duration::Once,
275                Some("restart") => v1::Duration::UntilRestart,
276                Some("hour") => v1::Duration::Hour,
277                Some("12h") => v1::Duration::HalfDay,
278                Some(other) => return Err(Error::Usage(format!("unknown duration `{other}`"))),
279            };
280            reject_rest(args)?;
281            client.call(Method::AnswerPrompt(v1::AnswerPrompt {
282                id,
283                action: convert::action(action).into(),
284                duration: duration.into(),
285                suggestion,
286                conditions: None,
287                folder,
288            }))?;
289            out(&format!("answered prompt {id}\n"))
290        }
291        Some(other) => Err(Error::Usage(format!("unknown prompts command `{other}`"))),
292    }
293}

295fn action(text: &str) -> Result<Action, Error> {
296    match text {
297        "allow" => Ok(Action::Allow),
298        "deny" => Ok(Action::Deny),
299        "reject" => Ok(Action::Reject),
300        other => Err(Error::Usage(format!("unknown action `{other}`"))),
301    }
302}

304fn spec_from_args(args: &mut pico_args::Arguments) -> Result<RuleSpec, Error> {
305    let action = action(&args.free_from_str::<String>()?)?;
306    let name: Option<String> = args.opt_value_from_str("--name")?;
307    let tier = match args.opt_value_from_str::<_, String>("--tier")?.as_deref() {
308        None | Some("normal") => snitchhiker_core::rule::Tier::Normal,
309        Some("override") => snitchhiker_core::rule::Tier::Override,
310        Some("fallback") => snitchhiker_core::rule::Tier::Fallback,
311        Some(other) => return Err(Error::Usage(format!("unknown tier `{other}`"))),
312    };
313    let conditions = Conditions {
314        exe: args.values_from_str("--exe")?,
315        host: args.values_from_str("--host")?,
316        net: args.values_from_str("--net")?,
317        port: args.values_from_str("--port")?,
318        uid: args.values_from_str("--uid")?,
319        protocol: args.values_from_str("--protocol")?,
320        context: args
321            .contains("--any-context")
322            .then_some(snitchhiker_core::rule::ContextKind::Any),
323        ..Conditions::default()
324    };
325    Ok(RuleSpec {
326        name: name.unwrap_or_else(|| format!("{action:?}").to_lowercase()),
327        action,
328        tier,
329        enabled: true,
330        notes: String::new(),
331        conditions,
332    })
333}

335fn format_status(status: &v1::Status) -> String {
336    format!(
337        "version          {}\nmode             {:?}\nno match         {:?}\nfail open        {}\nrules            {}\n\
338         decided          {}\ndropped events   {}\npending prompts  {}\nclients          {}\n",
339        status.daemon_version,
340        status.mode(),
341        status.no_match(),
342        status.fail_open,
343        status.rules,
344        status.decided,
345        status.dropped_events,
346        status.pending_prompts,
347        status.clients,
348    )
349}

351/// `Apps/curl/`: the folders a rule is in, from the top.
352fn folder_path(list: &v1::RuleList, folder: Option<u64>) -> String {
353    let mut names = Vec::new();
354    let mut next = folder;
355    for _ in 0..list.folders.len() {
356        let Some(found) = next.and_then(|id| list.folders.iter().find(|f| f.id == id)) else {
357            break;
358        };
359        names.push(found.name.as_str());
360        next = found.parent;
361    }
362    names.reverse();
363    names.iter().fold(String::new(), |mut path, name| {
364        path.push_str(name);
365        path.push('/');
366        path
367    })
368}

370fn format_rules(list: &v1::RuleList) -> String {
371    let mut text = String::new();
372    for rule in &list.rules {
373        let spec = rule
374            .conditions
375            .as_ref()
376            .map_or_else(Conditions::default, convert::conditions_from);
377        let _ = writeln!(
378            text,
379            "{:>5}  {:<7} {:<8} {}{}{}{}  {}",
380            if rule.locked {
381                "-".to_owned()
382            } else {
383                rule.id.to_string()
384            },
385            format!("{:?}", rule.action()).to_lowercase(),
386            format!("{:?}", rule.tier()).to_lowercase(),
387            if rule.enabled { "" } else { "(disabled) " },
388            if rule.locked { "(locked) " } else { "" },
389            folder_path(list, rule.folder),
390            rule.name,
391            toml::to_string(&spec)
392                .unwrap_or_default()
393                .replace('\n', " "),
394        );
395    }
396    text
397}

399fn format_audit(entries: &[v1::AuditEntry]) -> String {
400    let mut text = String::new();
401    for entry in entries {
402        let _ = writeln!(
403            text,
404            "{:>6}  {}  {:<10} {} uid {} pid {}: {}",
405            entry.id,
406            entry.time_ms,
407            format!("{:?}", entry.outcome()).to_lowercase(),
408            entry.app,
409            entry.uid,
410            entry.pid,
411            entry.action,
412        );
413    }
414    text
415}

417fn format_events(events: &[v1::ConnectionEvent]) -> String {
418    let mut text = String::new();
419    for event in events {
420        let address =
421            convert::address_from(&event.address).map_or_else(String::new, |a| a.to_string());
422        let _ = writeln!(
423            text,
424            "{:>8}  {:<6} uid {:<6} {} -> {}:{} proto {} out {} in {}{}",
425            event.seq,
426            format!("{:?}", event.action()).to_lowercase(),
427            event.uid,
428            if event.exe.is_empty() {
429                "?"
430            } else {
431                &event.exe
432            },
433            if event.host.is_empty() {
434                &address
435            } else {
436                &event.host
437            },
438            event.port,
439            event.protocol,
440            event.bytes_out,
441            event.bytes_in,
442            event
443                .rule
444                .map_or_else(String::new, |r| format!(" (rule {r})")),
445        );
446    }
447    text
448}

450fn format_prompt(prompt: &v1::Prompt) -> String {
451    let address =
452        convert::address_from(&prompt.address).map_or_else(String::new, |a| a.to_string());
453    let mut text = format!(
454        "prompt {}: {} (uid {}{}) -> {}:{} proto {}, {} waiting{}\n",
455        prompt.id,
456        prompt.exe,
457        prompt.uid,
458        if prompt.user.is_empty() {
459            String::new()
460        } else {
461            format!(" {}", prompt.user)
462        },
463        prompt.hosts.first().unwrap_or(&address),
464        prompt.port,
465        prompt.protocol,
466        prompt.waiting,
467        if prompt.exe_writable {
468            " (its executable is writable by a user)"
469        } else {
470            ""
471        },
472    );
473    for (index, suggestion) in prompt.suggestions.iter().enumerate() {
474        let _ = writeln!(
475            text,
476            "  {index}: {}{}",
477            suggestion.label,
478            if suggestion.broad { " (broad)" } else { "" }
479        );
480    }
481    text
482}

484fn format_resolved(resolved: &v1::PromptResolved) -> String {
485    format!(
486        "prompt {} resolved: {:?}{}\n",
487        resolved.id,
488        resolved.action(),
489        if resolved.timed_out {
490            " (timed out)".to_owned()
491        } else {
492            format!(" by {}", resolved.by)
493        }
494    )
495}
