92#[derive(Debug, thiserror::Error)]
93pub enum Error {
94    #[error(transparent)]
95    Args(#[from] pico_args::Error),
96    #[error("{0}")]
97    Usage(String),
98    #[error("{0} (is the daemon running, and is this root?)")]
99    Connect(snitchhiker_client::Error),
100    #[error(transparent)]
101    Client(#[from] snitchhiker_client::Error),
102    #[error("writing to stdout: {0}")]
103    Stdout(io::Error),
104}
105
106fn out(text: &str) -> Result<(), Error> {
107    let mut stdout = io::stdout().lock();
108    stdout
109        .write_all(text.as_bytes())
110        .and_then(|()| stdout.flush())
111        .map_err(Error::Stdout)
112}

114fn connect(version: &str) -> Result<Client, Error> {
115    Client::connect(Path::new(ADMIN_SOCKET), "cli", version).map_err(Error::Connect)
116}

118fn reject_rest(args: pico_args::Arguments) -> Result<(), Error> {
119    let rest = args.finish();
120    if rest.is_empty() {
121        Ok(())
122    } else {
123        Err(Error::Usage(format!("unexpected arguments: {rest:?}")))
124    }
125}
126
127fn unexpected() -> Error {
128    Error::Usage("the daemon answered something unexpected".to_owned())
129}

131/// Runs a command; `version` is the binary's version string.
132pub fn run(mut args: pico_args::Arguments, version: &str) -> Result<(), Error> {
133    use v1::request::Method;
134    if args.contains(["-h", "--help"]) {
135        return out(HELP);
136    }
137    match args.subcommand()?.as_deref() {
138        None => out(HELP),
139        Some("version") => out(&format!("snitchhiker {version}\n")),
140        Some("status") => {
141            reject_rest(args)?;
142            let mut client = connect(version)?;
143            match client.call(Method::GetStatus(v1::GetStatus {}))? {
144                v1::response::Result::Status(status) => out(&format_status(&status)),
145                _ => Err(unexpected()),
146            }
147        }
148        Some("rules") => rules(args, version),
149        Some("folders") => folders(args, version),
150        Some("explore") => explore(args, version),
151        Some("history") => history(args, version),
152        Some("connections") => connections(args, version),
153        Some("kill") => kill_connection(args, version),
154        Some("undo") => {
155            let change: u64 = args.free_from_str()?;
156            reject_rest(args)?;
157            connect(version)?.call(Method::Undo(v1::Undo { change }))?;
158            out(&format!("undid the changes of change {change}'s batch\n"))
159        }
160        Some("audit") => {
161            let limit: u32 = args.opt_value_from_str("--limit")?.unwrap_or(50);
162            reject_rest(args)?;
163            let mut client = connect(version)?;
164            match client.call(Method::ListAudit(v1::ListAudit { before: 0, limit }))? {
165                v1::response::Result::Audit(page) => out(&format_audit(&page.entries)),
166                _ => Err(unexpected()),
167            }
168        }
169        Some("events") => {
170            let limit: u32 = args.opt_value_from_str("--limit")?.unwrap_or(50);
171            reject_rest(args)?;
172            let mut client = connect(version)?;
173            match client.call(Method::ListEvents(v1::ListEvents {
174                before: 0,
175                limit,
176                columnar: false,
177            }))? {
178                v1::response::Result::Events(page) => out(&format_events(&page.events)),
179                _ => Err(unexpected()),
180            }
181        }
182        Some("learn") => {
183            let hours: u32 = args.opt_value_from_str("--hours")?.unwrap_or(24);
184            let folder: String = args
185                .opt_value_from_str("--folder")?
186                .unwrap_or_else(|| "Learned".to_owned());
187            reject_rest(args)?;
188            let mut client = connect(version)?;
189            print_file(client.call(Method::Learn(v1::Learn { hours, folder }))?)
190        }
191        Some("prompts") => prompts(args, version),
192        Some(other) => Err(Error::Usage(format!(
193            "unknown command `{other}`; see --help"
194        ))),
195    }
196}

198fn print_file(result: v1::response::Result) -> Result<(), Error> {
199    match result {
200        v1::response::Result::RuleFile(file) => out(&file.toml),
201        _ => Err(unexpected()),
202    }
203}

205/// `snitchhiker rules ...`.
206/// `explore`.
207fn explore(mut args: pico_args::Arguments, version: &str) -> Result<(), Error> {
208    use v1::request::Method;
209    let by: String = args
210        .opt_value_from_str("--by")?
211        .unwrap_or_else(|| "program".to_owned());
212    let hours: u64 = args.opt_value_from_str("--hours")?.unwrap_or(24);
213    let filter: Option<String> = args.opt_value_from_str("--where")?;
214    reject_rest(args)?;
215    let dimension = |name: &str| match name {
216        "program" => Ok(v1::Dimension::Program),
217        "host" => Ok(v1::Dimension::Host),
218        "port" => Ok(v1::Dimension::Port),
219        "user" => Ok(v1::Dimension::User),
220        other => Err(Error::Usage(format!(
221            "unknown dimension `{other}` (program, host, port, user)"
222        ))),
223    };
224    let filter = match filter {
225        None => None,
226        Some(text) => {
227            let (name, value) = text
228                .split_once('=')
229                .ok_or_else(|| Error::Usage("--where <dimension>=<value>".to_owned()))?;
230            Some(v1::ExploreFilter {
231                dimension: dimension(name)?.into(),
232                value: value.to_owned(),
233            })
234        }
235    };
236    let now = std::time::SystemTime::now()
237        .duration_since(std::time::UNIX_EPOCH)
238        .map_or(0, |d| u64::try_from(d.as_millis()).unwrap_or(u64::MAX));
239    let mut client = connect(version)?;
240    match client.call(Method::Explore(v1::Explore {
241        from_ms: now.saturating_sub(hours * 3_600_000),
242        to_ms: now,
243        group_by: dimension(&by)?.into(),
244        filter,
245        limit: 50,
246    }))? {
247        v1::response::Result::Explored(page) => {
248            let mut text = format!(
249                "{:>9} {:>9} {:>11} {:>11}  {by}\n",
250                "allowed", "blocked", "sent", "received"
251            );
252            for row in &page.rows {
253                let _ = writeln!(
254                    text,
255                    "{:>9} {:>9} {:>11} {:>11}  {}",
256                    row.allowed,
257                    row.blocked,
258                    row.bytes_out,
259                    row.bytes_in,
260                    if row.key.is_empty() {
261                        "(no name)"
262                    } else {
263                        &row.key
264                    }
265                );
266            }
267            out(&text)
268        }
269        _ => Err(unexpected()),
270    }
271}

273/// `history`.
274fn history(mut args: pico_args::Arguments, version: &str) -> Result<(), Error> {
275    use v1::request::Method;
276    let limit: u32 = args.opt_value_from_str("--limit")?.unwrap_or(50);
277    reject_rest(args)?;
278    let mut client = connect(version)?;
279    match client.call(Method::ListChanges(v1::ListChanges { before: 0, limit }))? {
280        v1::response::Result::Changes(page) => {
281            let mut text = String::new();
282            for change in &page.changes {
283                let _ = writeln!(
284                    text,
285                    "{:>6}  batch {:<5} {}  {:<8} {}",
286                    change.id, change.batch, change.time_ms, change.app, change.what
287                );
288            }
289            out(&text)
290        }
291        _ => Err(unexpected()),
292    }
293}

295/// `connections`.
296fn connections(args: pico_args::Arguments, version: &str) -> Result<(), Error> {
297    use v1::request::Method;
298    reject_rest(args)?;
299    match connect(version)?.call(Method::ListConnections(v1::ListConnections {}))? {
300        v1::response::Result::Connections(list) => {
301            let mut text = String::new();
302            for c in &list.connections {
303                let address = |bytes: &[u8], port: u32| {
304                    convert::address_from(bytes).map_or_else(String::new, |a| {
305                        std::net::SocketAddr::new(a, u16::try_from(port).unwrap_or(0)).to_string()
306                    })
307                };
308                let _ = writeln!(
309                    text,
310                    "{:<4} {:<11} {:<22} -> {:<22} uid {:<6} {}{}  out {} in {}",
311                    if c.protocol == 6 { "tcp" } else { "udp" },
312                    c.state,
313                    address(&c.local, c.local_port),
314                    address(&c.remote, c.remote_port),
315                    c.uid,
316                    if c.exe.is_empty() { "?" } else { &c.exe },
317                    if c.host.is_empty() {
318                        String::new()
319                    } else {
320                        format!(" ({})", c.host)
321                    },
322                    c.bytes_out,
323                    c.bytes_in,
324                );
325            }
326            out(&text)
327        }
328        _ => Err(unexpected()),
329    }
330}

332/// `kill`.
333fn kill_connection(mut args: pico_args::Arguments, version: &str) -> Result<(), Error> {
334    use v1::request::Method;
335    let protocol: String = args.free_from_str()?;
336    let local: std::net::SocketAddr = args.free_from_str()?;
337    let remote: std::net::SocketAddr = args.free_from_str()?;
338    reject_rest(args)?;
339    let protocol = match protocol.as_str() {
340        "tcp" => 6,
341        "udp" => 17,
342        other => {
343            return Err(Error::Usage(format!(
344                "unknown protocol `{other}` (tcp, udp)"
345            )));
346        }
347    };
348    connect(version)?.call(Method::KillConnection(v1::KillConnection {
349        protocol,
350        local: convert::address(local.ip()),
351        local_port: u32::from(local.port()),
352        remote: convert::address(remote.ip()),
353        remote_port: u32::from(remote.port()),
354    }))?;
355    out("closed\n")
356}

358/// `folders`: list, add, rename and delete folders.
359fn folders(mut args: pico_args::Arguments, version: &str) -> Result<(), Error> {
360    use v1::request::Method;
361    let mut client = connect(version)?;
362    match args.subcommand()?.as_deref() {
363        None => {
364            reject_rest(args)?;
365            let v1::response::Result::Rules(list) =
366                client.call(Method::ListRules(v1::ListRules {}))?
367            else {
368                return Err(unexpected());
369            };
370            let mut text = String::new();
371            for folder in &list.folders {
372                let count = list
373                    .rules
374                    .iter()
375                    .filter(|r| r.folder == Some(folder.id))
376                    .count();
377                let _ = writeln!(
378                    text,
379                    "{:>5}  {}{}  ({count} rules){}{}",
380                    if folder.locked {
381                        "-".to_owned()
382                    } else {
383                        folder.id.to_string()
384                    },
385                    folder_path(&list, folder.parent),
386                    folder.name,
387                    if folder.enabled { "" } else { " (disabled)" },
388                    if folder.locked { " (locked)" } else { "" },
389                );
390            }
391            out(&text)
392        }
393        Some("add") => {
394            let name: String = args.free_from_str()?;
395            let parent: Option<u64> = args.opt_value_from_str("--parent")?;
396            reject_rest(args)?;
397            match client.call(Method::AddFolder(v1::AddFolder {
398                folder: Some(v1::Folder {
399                    name,
400                    parent,
401                    enabled: true,
402                    ..v1::Folder::default()
403                }),
404            }))? {
405                v1::response::Result::FolderAdded(added) => {
406                    out(&format!("added folder {}\n", added.id))
407                }
408                _ => Err(unexpected()),
409            }
410        }
411        Some("rm" | "delete") => {
412            let id: u64 = args.free_from_str()?;
413            reject_rest(args)?;
414            client.call(Method::DeleteFolder(v1::DeleteFolder { id }))?;
415            out(&format!(
416                "deleted folder {id}, the folders in it and their rules\n"
417            ))
418        }
419        Some(other) => Err(Error::Usage(format!("unknown folders command `{other}`"))),
420    }
421}

423/// `rules import-opensnitch`: OpenSnitch's rule files in `dir`, converted and imported (or printed, `dry_run`).
424fn import_opensnitch(dir: &str, dry_run: bool, version: &str) -> Result<(), Error> {
425    let mut files = Vec::new();
426    let entries = std::fs::read_dir(dir).map_err(|e| Error::Usage(format!("{dir}: {e}")))?;
427    for entry in entries.flatten() {
428        let path = entry.path();
429        if path.extension().is_some_and(|e| e == "json") {
430            let text = std::fs::read_to_string(&path)
431                .map_err(|e| Error::Usage(format!("{}: {e}", path.display())))?;
432            files.push((path.display().to_string(), text));
433        }
434    }
435    files.sort();
436    let imported = snitchhiker_core::opensnitch::import(&files, "OpenSnitch");
437    let mut report = String::new();
438    for (name, reason) in &imported.skipped {
439        let _ = writeln!(report, "skipped {name}: {reason}");
440    }
441    let toml = toml::to_string(&imported.file)
442        .map_err(|e| Error::Usage(format!("writing the rule file: {e}")))?;
443    if dry_run {
444        // The rule file, and what was skipped as comments: the output stays a valid rule file.
445        let comments = report.lines().fold(String::new(), |mut text, line| {
446            let _ = writeln!(text, "# {line}");
447            text
448        });
449        return out(&format!("{toml}{comments}"));
450    }
451    let mut client = connect(version)?;
452    if !imported.file.rule.is_empty() {
453        client.call(v1::request::Method::ImportRules(v1::ImportRules {
454            toml,
455            folder: "OpenSnitch".to_owned(),
456        }))?;
457    }
458    let _ = writeln!(
459        report,
460        "imported {} of {} rules into the folder OpenSnitch; they apply to the host only \
461         (add context = \"any\" to a rule for containers too)",
462        imported.file.rule.len(),
463        imported.file.rule.len() + imported.skipped.len()
464    );
465    out(&report)
466}

468fn rules(mut args: pico_args::Arguments, version: &str) -> Result<(), Error> {
469    use v1::request::Method;
470    let command = args.subcommand()?;
471    let mut client = connect(version)?;
472    match command.as_deref() {
473        None | Some("list" | "ls") => {
474            reject_rest(args)?;
475            match client.call(Method::ListRules(v1::ListRules {}))? {
476                v1::response::Result::Rules(list) => out(&format_rules(&list)),
477                _ => Err(unexpected()),
478            }
479        }
480        Some("add") => {
481            let spec = spec_from_args(&mut args)?;
482            reject_rest(args)?;
483            let rule = spec
484                .to_rule(0, None)
485                .map_err(|e| Error::Usage(e.to_string()))?;
486            match client.call(Method::AddRule(v1::AddRule {
487                rule: Some(convert::rule(&rule, false)),
488            }))? {
489                v1::response::Result::RuleAdded(added) => {
490                    out(&format!("added rule {}\n", added.id))
491                }
492                _ => Err(unexpected()),
493            }
494        }
495        Some("rm" | "delete") => {
496            let id: u64 = args.free_from_str()?;
497            reject_rest(args)?;
498            client.call(Method::DeleteRule(v1::DeleteRule { id }))?;
499            out(&format!("deleted rule {id}\n"))
500        }
501        Some("move") => {
502            let id: u64 = args.free_from_str()?;
503            let folder: u64 = args.free_from_str()?;
504            reject_rest(args)?;
505            let v1::response::Result::Rules(list) =
506                client.call(Method::ListRules(v1::ListRules {}))?
507            else {
508                return Err(unexpected());
509            };
510            let mut rule = list
511                .rules
512                .into_iter()
513                .find(|r| r.id == id)
514                .ok_or_else(|| Error::Usage(format!("no rule {id}")))?;
515            rule.folder = (folder != 0).then_some(folder);
516            client.call(Method::UpdateRule(v1::UpdateRule { rule: Some(rule) }))?;
517            out(&format!("moved rule {id}\n"))
518        }
519        Some(which @ ("enable" | "disable")) => {
520            let id: u64 = args.free_from_str()?;
521            reject_rest(args)?;
522            client.call(Method::SetRuleEnabled(v1::SetRuleEnabled {
523                id,
524                enabled: which == "enable",
525            }))?;
526            out(&format!("{which}d rule {id}\n"))
527        }
528        Some("import-opensnitch") => {
529            let dry_run = args.contains("--dry-run");
530            let dir: String = args
531                .opt_free_from_str()?
532                .unwrap_or_else(|| "/etc/opensnitchd/rules".to_owned());
533            reject_rest(args)?;
534            import_opensnitch(&dir, dry_run, version)
535        }
536        Some("import") => {
537            let path: String = args.free_from_str()?;
538            reject_rest(args)?;
539            let toml =
540                std::fs::read_to_string(&path).map_err(|e| Error::Usage(format!("{path}: {e}")))?;
541            let folder = Path::new(&path).file_stem().map_or_else(
542                || "Imported".to_owned(),
543                |s| s.to_string_lossy().into_owned(),
544            );
545            client.call(Method::ImportRules(v1::ImportRules { toml, folder }))?;
546            out(&format!("imported {path}\n"))
547        }
548        Some("export") => {
549            reject_rest(args)?;
550            print_file(client.call(Method::ExportRules(v1::ExportRules {}))?)
551        }
552        Some(other) => Err(Error::Usage(format!("unknown rules command `{other}`"))),
553    }
554}

556/// `snitchhiker prompts ...`.
557fn prompts(mut args: pico_args::Arguments, version: &str) -> Result<(), Error> {
558    use v1::request::Method;
559    let command = args.subcommand()?;
560    let mut client = connect(version)?;
561    match command.as_deref() {
562        None | Some("list" | "ls") => {
563            reject_rest(args)?;
564            match client.call(Method::ListPrompts(v1::ListPrompts {}))? {
565                v1::response::Result::Prompts(list) => {
566                    out(&list.prompts.iter().map(format_prompt).collect::<String>())
567                }
568                _ => Err(unexpected()),
569            }
570        }
571        Some("watch") => {
572            reject_rest(args)?;
573            client.subscribe(v1::Topic::Prompts)?;
574            loop {
575                match client.notice()?.body {
576                    Some(v1::notice::Body::Prompt(prompt)) => out(&format_prompt(&prompt))?,
577                    Some(v1::notice::Body::PromptResolved(resolved)) => {
578                        out(&format_resolved(&resolved))?;
579                    }
580                    _ => {}
581                }
582            }
583        }
584        Some("answer") => {
585            let id: u64 = args.free_from_str()?;
586            let action = action(&args.free_from_str::<String>()?)?;
587            let suggestion: u32 = args.opt_value_from_str("--suggestion")?.unwrap_or(0);
588            let folder: Option<u64> = args.opt_value_from_str("--folder")?;
589            let duration = match args
590                .opt_value_from_str::<_, String>("--duration")?
591                .as_deref()
592            {
593                None | Some("forever") => v1::Duration::Forever,
594                Some("once") => v1::Duration::Once,
595                Some("restart") => v1::Duration::UntilRestart,
596                Some("hour") => v1::Duration::Hour,
597                Some("12h") => v1::Duration::HalfDay,
598                Some(other) => return Err(Error::Usage(format!("unknown duration `{other}`"))),
599            };
600            reject_rest(args)?;
601            client.call(Method::AnswerPrompt(v1::AnswerPrompt {
602                id,
603                action: convert::action(action).into(),
604                duration: duration.into(),
605                suggestion,
606                conditions: None,
607                folder,
608            }))?;
609            out(&format!("answered prompt {id}\n"))
610        }
611        Some(other) => Err(Error::Usage(format!("unknown prompts command `{other}`"))),
612    }
613}
614
615fn action(text: &str) -> Result<Action, Error> {
616    match text {
617        "allow" => Ok(Action::Allow),
618        "deny" => Ok(Action::Deny),
619        "reject" => Ok(Action::Reject),
620        other => Err(Error::Usage(format!("unknown action `{other}`"))),
621    }
622}
623
624fn spec_from_args(args: &mut pico_args::Arguments) -> Result<RuleSpec, Error> {
625    let action = action(&args.free_from_str::<String>()?)?;
626    let name: Option<String> = args.opt_value_from_str("--name")?;
627    let tier = match args.opt_value_from_str::<_, String>("--tier")?.as_deref() {
628        None | Some("normal") => snitchhiker_core::rule::Tier::Normal,
629        Some("override") => snitchhiker_core::rule::Tier::Override,
630        Some("fallback") => snitchhiker_core::rule::Tier::Fallback,
631        Some(other) => return Err(Error::Usage(format!("unknown tier `{other}`"))),
632    };
633    let conditions = Conditions {
634        exe: args.values_from_str("--exe")?,
635        host: args.values_from_str("--host")?,
636        net: args.values_from_str("--net")?,
637        port: args.values_from_str("--port")?,
638        uid: args.values_from_str("--uid")?,
639        protocol: args.values_from_str("--protocol")?,
640        context: args
641            .contains("--any-context")
642            .then_some(snitchhiker_core::rule::ContextKind::Any),
643        ..Conditions::default()
644    };
645    Ok(RuleSpec {
646        name: name.unwrap_or_else(|| format!("{action:?}").to_lowercase()),
647        action,
648        tier,
649        enabled: true,
650        notes: String::new(),
651        conditions,
652    })
653}

671/// `Apps/curl/`: the folders a rule is in, from the top.
672fn folder_path(list: &v1::RuleList, folder: Option<u64>) -> String {
673    let mut names = Vec::new();
674    let mut next = folder;
675    for _ in 0..list.folders.len() {
676        let Some(found) = next.and_then(|id| list.folders.iter().find(|f| f.id == id)) else {
677            break;
678        };
679        names.push(found.name.as_str());
680        next = found.parent;
681    }
682    names.reverse();
683    names.iter().fold(String::new(), |mut path, name| {
684        path.push_str(name);
685        path.push('/');
686        path
687    })
688}

690fn format_rules(list: &v1::RuleList) -> String {
691    let mut text = String::new();
692    for rule in &list.rules {
693        let spec = rule
694            .conditions
695            .as_ref()
696            .map_or_else(Conditions::default, convert::conditions_from);
697        let _ = writeln!(
698            text,
699            "{:>5}  {:<7} {:<8} {}{}{}{}  {}",
700            if rule.locked {
701                "-".to_owned()
702            } else {
703                rule.id.to_string()
704            },
705            format!("{:?}", rule.action()).to_lowercase(),
706            format!("{:?}", rule.tier()).to_lowercase(),
707            if rule.enabled { "" } else { "(disabled) " },
708            if rule.locked { "(locked) " } else { "" },
709            folder_path(list, rule.folder),
710            rule.name,
711            toml::to_string(&spec)
712                .unwrap_or_default()
713                .replace('\n', " "),
714        );
715    }
716    text
717}

737fn format_events(events: &[v1::ConnectionEvent]) -> String {
738    let mut text = String::new();
739    for event in events {
740        let address =
741            convert::address_from(&event.address).map_or_else(String::new, |a| a.to_string());
742        let _ = writeln!(
743            text,
744            "{:>8}  {:<6} uid {:<6} {} -> {}:{} proto {} out {} in {}{}",
745            event.seq,
746            format!("{:?}", event.action()).to_lowercase(),
747            event.uid,
748            if event.exe.is_empty() {
749                "?"
750            } else {
751                &event.exe
752            },
753            if event.host.is_empty() {
754                &address
755            } else {
756                &event.host
757            },
758            event.port,
759            event.protocol,
760            event.bytes_out,
761            event.bytes_in,
762            event
763                .rule
764                .map_or_else(String::new, |r| format!(" (rule {r})")),
765        );
766    }
767    text
768}

770fn format_prompt(prompt: &v1::Prompt) -> String {
771    let address =
772        convert::address_from(&prompt.address).map_or_else(String::new, |a| a.to_string());
773    let mut text = format!(
774        "prompt {}: {} (uid {}{}) -> {}:{} proto {}, {} waiting{}\n",
775        prompt.id,
776        prompt.exe,
777        prompt.uid,
778        if prompt.user.is_empty() {
779            String::new()
780        } else {
781            format!(" {}", prompt.user)
782        },
783        prompt.hosts.first().unwrap_or(&address),
784        prompt.port,
785        prompt.protocol,
786        prompt.waiting,
787        if prompt.exe_writable {
788            " (its executable is writable by a user)"
789        } else {
790            ""
791        },
792    );
793    for (index, suggestion) in prompt.suggestions.iter().enumerate() {
794        let _ = writeln!(
795            text,
796            "  {index}: {}{}",
797            suggestion.label,
798            if suggestion.broad { " (broad)" } else { "" }
799        );
800    }
801    text
802}
