105#[derive(Debug, thiserror::Error)]
106pub enum Error {
107    #[error(transparent)]
108    Args(#[from] pico_args::Error),
109    #[error("{0}")]
110    Usage(String),
111    #[error("{0} (is the daemon running, and is this root?)")]
112    Connect(snitchhiker_client::Error),
113    #[error(transparent)]
114    Client(#[from] snitchhiker_client::Error),
115    #[error("writing to stdout: {0}")]
116    Stdout(io::Error),
117}

119fn out(text: &str) -> Result<(), Error> {
120    let mut stdout = io::stdout().lock();
121    stdout
122        .write_all(text.as_bytes())
123        .and_then(|()| stdout.flush())
124        .map_err(Error::Stdout)
125}

131fn reject_rest(args: pico_args::Arguments) -> Result<(), Error> {
132    let rest = args.finish();
133    if rest.is_empty() {
134        Ok(())
135    } else {
136        Err(Error::Usage(format!("unexpected arguments: {rest:?}")))
137    }
138}

144/// Runs a command; `version` is the binary's version string.
145pub fn run(mut args: pico_args::Arguments, version: &str) -> Result<(), Error> {
146    use v1::request::Method;
147    if args.contains(["-h", "--help"]) {
148        return out(HELP);
149    }
150    match args.subcommand()?.as_deref() {
151        None => out(HELP),
152        Some("version") => out(&format!("snitchhiker {version}\n")),
153        Some("status") => {
154            reject_rest(args)?;
155            let mut client = connect(version)?;
156            match client.call(Method::GetStatus(v1::GetStatus {}))? {
157                v1::response::Result::Status(status) => out(&format_status(&status)),
158                _ => Err(unexpected()),
159            }
160        }
161        Some("rules") => rules(args, version),
162        Some("policy") => policy(args, version),
163        Some("folders") => folders(args, version),
164        Some("explore") => explore(args, version),
165        Some("history") => history(args, version),
166        Some("connections") => connections(args, version),
167        Some("kill") => kill_connection(args, version),
168        Some("explain") => {
169            let seq: i64 = args.free_from_str()?;
170            reject_rest(args)?;
171            match connect(version)?.call(Method::Explain(v1::Explain { seq }))? {
172                v1::response::Result::Explanation(explanation) => {
173                    out(&format_explanation(&explanation))
174                }
175                _ => Err(unexpected()),
176            }
177        }
178        Some("undo") => {
179            let change: u64 = args.free_from_str()?;
180            reject_rest(args)?;
181            connect(version)?.call(Method::Undo(v1::Undo { change }))?;
182            out(&format!("undid the changes of change {change}'s batch\n"))
183        }
184        Some("audit") => {
185            let limit: u32 = args.opt_value_from_str("--limit")?.unwrap_or(50);
186            reject_rest(args)?;
187            let mut client = connect(version)?;
188            match client.call(Method::ListAudit(v1::ListAudit { before: 0, limit }))? {
189                v1::response::Result::Audit(page) => out(&format_audit(&page.entries)),
190                _ => Err(unexpected()),
191            }
192        }
193        Some("events") => {
194            let limit: u32 = args.opt_value_from_str("--limit")?.unwrap_or(50);
195            let query: String = args.opt_value_from_str("--query")?.unwrap_or_default();
196            reject_rest(args)?;
197            let mut client = connect(version)?;
198            match client.call(Method::ListEvents(v1::ListEvents {
199                before: 0,
200                limit,
201                columnar: false,
202                query,
203            }))? {
204                v1::response::Result::Events(page) => out(&format_events(&page.events)),
205                _ => Err(unexpected()),
206            }
207        }
208        Some("learn") => {
209            let hours: u32 = args.opt_value_from_str("--hours")?.unwrap_or(24);
210            let folder: String = args
211                .opt_value_from_str("--folder")?
212                .unwrap_or_else(|| "Learned".to_owned());
213            reject_rest(args)?;
214            let mut client = connect(version)?;
215            print_file(client.call(Method::Learn(v1::Learn { hours, folder }))?)
216        }
217        Some("prompts") => prompts(args, version),
218        Some(other) => Err(Error::Usage(format!(
219            "unknown command `{other}`; see --help"
220        ))),
221    }
222}

224fn format_lint(report: &v1::LintReport) -> String {
225    let mut text = String::new();
226    for finding in &report.findings {
227        let _ = writeln!(
228            text,
229            "rule {}: {}{}",
230            finding.rule,
231            finding.description,
232            if finding.suggestion.is_empty() {
233                String::new()
234            } else {
235                format!(" (the program now matches {})", finding.suggestion)
236            }
237        );
238    }
239    if report.findings.is_empty() {
240        text.push_str("nothing to report\n");
241    }
242    text
243}

245/// The rules and the change of `rules edit`.
246fn rules_edit(args: &mut pico_args::Arguments) -> Result<v1::EditRules, Error> {
247    use v1::edit_rules::Edit;
248    let ids: String = args.free_from_str()?;
249    let ids = ids
250        .split(',')
251        .map(|id| {
252            id.trim()
253                .parse()
254                .map_err(|_| Error::Usage(format!("`{id}` isn't a rule id")))
255        })
256        .collect::<Result<Vec<u64>, _>>()?;
257    let what: String = args.free_from_str()?;
258    let edit = match what.as_str() {
259        "enable" => Edit::Enabled(true),
260        "disable" => Edit::Enabled(false),
261        "delete" => Edit::Delete(true),
262        "move" => {
263            let folder: u64 = args.free_from_str()?;
264            Edit::Move(v1::MoveRules {
265                folder: (folder != 0).then_some(folder),
266            })
267        }
268        "action" => {
269            let action: String = args.free_from_str()?;
270            Edit::Action(
271                match action.as_str() {
272                    "allow" => v1::Action::Allow,
273                    "deny" => v1::Action::Deny,
274                    "reject" => v1::Action::Reject,
275                    other => return Err(Error::Usage(format!("unknown action `{other}`"))),
276                }
277                .into(),
278            )
279        }
280        "expires" => {
281            let when: String = args.free_from_str()?;
282            Edit::Expires(
283                match when.as_str() {
284                    "hour" => v1::Duration::Hour,
285                    "half-day" => v1::Duration::HalfDay,
286                    "until-restart" => v1::Duration::UntilRestart,
287                    "never" => v1::Duration::Forever,
288                    other => return Err(Error::Usage(format!("unknown expiry `{other}`"))),
289                }
290                .into(),
291            )
292        }
293        other => return Err(Error::Usage(format!("unknown edit `{other}`"))),
294    };
295    Ok(v1::EditRules {
296        ids,
297        edit: Some(edit),
298    })
299}

301/// `policy`, `policy set`, `policy reset`.
302fn policy(mut args: pico_args::Arguments, version: &str) -> Result<(), Error> {
303    use v1::request::Method;
304    let command = args.subcommand()?;
305    let rest: Vec<String> = args
306        .finish()
307        .into_iter()
308        .map(|a| a.to_string_lossy().into_owned())
309        .collect();
310    let method = match command.as_deref() {
311        None | Some("show") if rest.is_empty() => Method::GetPolicy(v1::GetPolicy {}),
312        Some("set") if !rest.is_empty() => Method::SetPolicy(policy_set(&rest)?),
313        Some("reset") => Method::SetPolicy(v1::SetPolicy {
314            reset: if rest.is_empty() {
315                crate::policy::SETTINGS
316                    .iter()
317                    .map(|s| (*s).to_owned())
318                    .collect()
319            } else {
320                rest
321            },
322            ..v1::SetPolicy::default()
323        }),
324        _ => {
325            return Err(Error::Usage(
326                "policy [show | set <setting>=<value>... | reset [<setting>...]]".to_owned(),
327            ));
328        }
329    };
330    match connect(version)?.call(method)? {
331        v1::response::Result::Policy(policy) => out(&format_policy(&policy)),
332        _ => Err(unexpected()),
333    }
334}

336fn policy_set(pairs: &[String]) -> Result<v1::SetPolicy, Error> {
337    let action = |value: &str| match value {
338        "allow" => Ok(v1::Action::Allow.into()),
339        "deny" => Ok(v1::Action::Deny.into()),
340        "reject" => Ok(v1::Action::Reject.into()),
341        other => Err(Error::Usage(format!("unknown action `{other}`"))),
342    };
343    let number = |value: &str| {
344        value
345            .parse::<u32>()
346            .map_err(|_| Error::Usage(format!("`{value}` isn't a number")))
347    };
348    let mut set = v1::SetPolicy::default();
349    for pair in pairs {
350        let (key, value) = pair
351            .split_once('=')
352            .ok_or_else(|| Error::Usage(format!("`{pair}`: expected <setting>=<value>")))?;
353        match key {
354            "mode" => {
355                set.mode = Some(match value {
356                    "enforce" => v1::Mode::Enforce.into(),
357                    "monitor" => v1::Mode::Monitor.into(),
358                    other => return Err(Error::Usage(format!("unknown mode `{other}`"))),
359                });
360            }
361            "no_match" => {
362                set.no_match = Some(match value {
363                    "ask" => v1::Policy::Ask.into(),
364                    "allow" => v1::Policy::Allow.into(),
365                    "deny" => v1::Policy::Deny.into(),
366                    "reject" => v1::Policy::Reject.into(),
367                    other => return Err(Error::Usage(format!("unknown policy `{other}`"))),
368                });
369            }
370            "no_client" => set.no_client = Some(action(value)?),
371            "on_timeout" => set.on_timeout = Some(action(value)?),
372            "prompt_timeout_secs" => set.prompt_timeout_secs = Some(number(value)?),
373            "retention_days" => set.retention_days = Some(number(value)?),
374            other => return Err(Error::Usage(format!("unknown setting `{other}`"))),
375        }
376    }
377    Ok(set)
378}

380fn format_policy(policy: &v1::NodePolicy) -> String {
381    let name = |text: &str, prefix: &str| text.trim_start_matches(prefix).to_lowercase();
382    let mark = |setting: &str| {
383        if policy.changed.iter().any(|c| c == setting) {
384            "  (changed by a client)"
385        } else {
386            ""
387        }
388    };
389    let mut text = String::new();
390    for (setting, value) in [
391        ("mode", name(policy.mode().as_str_name(), "MODE_")),
392        ("no_match", name(policy.no_match().as_str_name(), "POLICY_")),
393        (
394            "no_client",
395            name(policy.no_client().as_str_name(), "ACTION_"),
396        ),
397        (
398            "prompt_timeout_secs",
399            policy.prompt_timeout_secs.to_string(),
400        ),
401        (
402            "on_timeout",
403            name(policy.on_timeout().as_str_name(), "ACTION_"),
404        ),
405        ("retention_days", policy.retention_days.to_string()),
406    ] {
407        let _ = writeln!(text, "{setting:<20} {value}{}", mark(setting));
408    }
409    text
410}

419/// `snitchhiker rules ...`.
420/// `explore`.
421fn explore(mut args: pico_args::Arguments, version: &str) -> Result<(), Error> {
422    use v1::request::Method;
423    let by: String = args
424        .opt_value_from_str("--by")?
425        .unwrap_or_else(|| "program".to_owned());
426    let hours: u64 = args.opt_value_from_str("--hours")?.unwrap_or(24);
427    let filter: Option<String> = args.opt_value_from_str("--where")?;
428    reject_rest(args)?;
429    let dimension = |name: &str| match name {
430        "program" => Ok(v1::Dimension::Program),
431        "host" => Ok(v1::Dimension::Host),
432        "port" => Ok(v1::Dimension::Port),
433        "user" => Ok(v1::Dimension::User),
434        other => Err(Error::Usage(format!(
435            "unknown dimension `{other}` (program, host, port, user)"
436        ))),
437    };
438    let filter = match filter {
439        None => None,
440        Some(text) => {
441            let (name, value) = text
442                .split_once('=')
443                .ok_or_else(|| Error::Usage("--where <dimension>=<value>".to_owned()))?;
444            Some(v1::ExploreFilter {
445                dimension: dimension(name)?.into(),
446                value: value.to_owned(),
447            })
448        }
449    };
450    let now = std::time::SystemTime::now()
451        .duration_since(std::time::UNIX_EPOCH)
452        .map_or(0, |d| u64::try_from(d.as_millis()).unwrap_or(u64::MAX));
453    let mut client = connect(version)?;
454    match client.call(Method::Explore(v1::Explore {
455        from_ms: now.saturating_sub(hours * 3_600_000),
456        to_ms: now,
457        group_by: dimension(&by)?.into(),
458        filter,
459        limit: 50,
460    }))? {
461        v1::response::Result::Explored(page) => {
462            let mut text = format!(
463                "{:>9} {:>9} {:>11} {:>11}  {by}\n",
464                "allowed", "blocked", "sent", "received"
465            );
466            for row in &page.rows {
467                let _ = writeln!(
468                    text,
469                    "{:>9} {:>9} {:>11} {:>11}  {}",
470                    row.allowed,
471                    row.blocked,
472                    row.bytes_out,
473                    row.bytes_in,
474                    if row.key.is_empty() {
475                        "(no name)"
476                    } else {
477                        &row.key
478                    }
479                );
480            }
481            out(&text)
482        }
483        _ => Err(unexpected()),
484    }
485}

509/// `connections`.
510fn connections(args: pico_args::Arguments, version: &str) -> Result<(), Error> {
511    use v1::request::Method;
512    reject_rest(args)?;
513    match connect(version)?.call(Method::ListConnections(v1::ListConnections {}))? {
514        v1::response::Result::Connections(list) => {
515            let mut text = String::new();
516            for c in &list.connections {
517                let address = |bytes: &[u8], port: u32| {
518                    convert::address_from(bytes).map_or_else(String::new, |a| {
519                        std::net::SocketAddr::new(a, u16::try_from(port).unwrap_or(0)).to_string()
520                    })
521                };
522                let _ = writeln!(
523                    text,
524                    "{:<4} {:<11} {:<22} -> {:<22} uid {:<6} {}{}  out {} in {}",
525                    if c.protocol == 6 { "tcp" } else { "udp" },
526                    c.state,
527                    address(&c.local, c.local_port),
528                    address(&c.remote, c.remote_port),
529                    c.uid,
530                    if c.exe.is_empty() { "?" } else { &c.exe },
531                    if c.host.is_empty() {
532                        String::new()
533                    } else {
534                        format!(" ({})", c.host)
535                    },
536                    c.bytes_out,
537                    c.bytes_in,
538                );
539            }
540            out(&text)
541        }
542        _ => Err(unexpected()),
543    }
544}

546/// `kill`.
547fn kill_connection(mut args: pico_args::Arguments, version: &str) -> Result<(), Error> {
548    use v1::request::Method;
549    let protocol: String = args.free_from_str()?;
550    let local: std::net::SocketAddr = args.free_from_str()?;
551    let remote: std::net::SocketAddr = args.free_from_str()?;
552    reject_rest(args)?;
553    let protocol = match protocol.as_str() {
554        "tcp" => 6,
555        "udp" => 17,
556        other => {
557            return Err(Error::Usage(format!(
558                "unknown protocol `{other}` (tcp, udp)"
559            )));
560        }
561    };
562    connect(version)?.call(Method::KillConnection(v1::KillConnection {
563        protocol,
564        local: convert::address(local.ip()),
565        local_port: u32::from(local.port()),
566        remote: convert::address(remote.ip()),
567        remote_port: u32::from(remote.port()),
568    }))?;
569    out("closed\n")
570}

572/// `folders`: list, add, rename and delete folders.
573fn folders(mut args: pico_args::Arguments, version: &str) -> Result<(), Error> {
574    use v1::request::Method;
575    let mut client = connect(version)?;
576    match args.subcommand()?.as_deref() {
577        None => {
578            reject_rest(args)?;
579            let v1::response::Result::Rules(list) =
580                client.call(Method::ListRules(v1::ListRules {}))?
581            else {
582                return Err(unexpected());
583            };
584            let mut text = String::new();
585            for folder in &list.folders {
586                let count = list
587                    .rules
588                    .iter()
589                    .filter(|r| r.folder == Some(folder.id))
590                    .count();
591                let _ = writeln!(
592                    text,
593                    "{:>5}  {}{}  ({count} rules){}{}",
594                    if folder.locked {
595                        "-".to_owned()
596                    } else {
597                        folder.id.to_string()
598                    },
599                    folder_path(&list, folder.parent),
600                    folder.name,
601                    if folder.enabled { "" } else { " (disabled)" },
602                    if folder.locked { " (locked)" } else { "" },
603                );
604            }
605            out(&text)
606        }
607        Some("add") => {
608            let name: String = args.free_from_str()?;
609            let parent: Option<u64> = args.opt_value_from_str("--parent")?;
610            reject_rest(args)?;
611            match client.call(Method::AddFolder(v1::AddFolder {
612                folder: Some(v1::Folder {
613                    name,
614                    parent,
615                    enabled: true,
616                    ..v1::Folder::default()
617                }),
618            }))? {
619                v1::response::Result::FolderAdded(added) => {
620                    out(&format!("added folder {}\n", added.id))
621                }
622                _ => Err(unexpected()),
623            }
624        }
625        Some("rm" | "delete") => {
626            let id: u64 = args.free_from_str()?;
627            reject_rest(args)?;
628            client.call(Method::DeleteFolder(v1::DeleteFolder { id }))?;
629            out(&format!(
630                "deleted folder {id}, the folders in it and their rules\n"
631            ))
632        }
633        Some(other) => Err(Error::Usage(format!("unknown folders command `{other}`"))),
634    }
635}

637/// `rules import-opensnitch`: OpenSnitch's rule files in `dir`, converted and imported (or printed, `dry_run`).
638fn import_opensnitch(dir: &str, dry_run: bool, version: &str) -> Result<(), Error> {
639    let mut files = Vec::new();
640    let entries = std::fs::read_dir(dir).map_err(|e| Error::Usage(format!("{dir}: {e}")))?;
641    for entry in entries.flatten() {
642        let path = entry.path();
643        if path.extension().is_some_and(|e| e == "json") {
644            let text = std::fs::read_to_string(&path)
645                .map_err(|e| Error::Usage(format!("{}: {e}", path.display())))?;
646            files.push((path.display().to_string(), text));
647        }
648    }
649    files.sort();
650    let imported = snitchhiker_core::opensnitch::import(&files, "OpenSnitch");
651    let mut report = String::new();
652    for (name, reason) in &imported.skipped {
653        let _ = writeln!(report, "skipped {name}: {reason}");
654    }
655    let toml = toml::to_string(&imported.file)
656        .map_err(|e| Error::Usage(format!("writing the rule file: {e}")))?;
657    if dry_run {
658        // The rule file, and what was skipped as comments: the output stays a valid rule file.
659        let comments = report.lines().fold(String::new(), |mut text, line| {
660            let _ = writeln!(text, "# {line}");
661            text
662        });
663        return out(&format!("{toml}{comments}"));
664    }
665    let mut client = connect(version)?;
666    if !imported.file.rule.is_empty() {
667        client.call(v1::request::Method::ImportRules(v1::ImportRules {
668            toml,
669            folder: "OpenSnitch".to_owned(),
670        }))?;
671    }
672    let _ = writeln!(
673        report,
674        "imported {} of {} rules into the folder OpenSnitch; they apply to the host only \
675         (add context = \"any\" to a rule for containers too)",
676        imported.file.rule.len(),
677        imported.file.rule.len() + imported.skipped.len()
678    );
679    out(&report)
680}

682fn rules(mut args: pico_args::Arguments, version: &str) -> Result<(), Error> {
683    use v1::request::Method;
684    let command = args.subcommand()?;
685    let mut client = connect(version)?;
686    match command.as_deref() {
687        None | Some("list" | "ls") => {
688            reject_rest(args)?;
689            match client.call(Method::ListRules(v1::ListRules {}))? {
690                v1::response::Result::Rules(list) => out(&format_rules(&list)),
691                _ => Err(unexpected()),
692            }
693        }
694        Some("add") => {
695            let spec = spec_from_args(&mut args)?;
696            reject_rest(args)?;
697            let rule = spec
698                .to_rule(0, None)
699                .map_err(|e| Error::Usage(e.to_string()))?;
700            match client.call(Method::AddRule(v1::AddRule {
701                rule: Some(convert::rule(&rule, false)),
702            }))? {
703                v1::response::Result::RuleAdded(added) => {
704                    out(&format!("added rule {}\n", added.id))
705                }
706                _ => Err(unexpected()),
707            }
708        }
709        Some("rm" | "delete") => {
710            let id: u64 = args.free_from_str()?;
711            reject_rest(args)?;
712            client.call(Method::DeleteRule(v1::DeleteRule { id }))?;
713            out(&format!("deleted rule {id}\n"))
714        }
715        Some("move") => {
716            let id: u64 = args.free_from_str()?;
717            let folder: u64 = args.free_from_str()?;
718            reject_rest(args)?;
719            let v1::response::Result::Rules(list) =
720                client.call(Method::ListRules(v1::ListRules {}))?
721            else {
722                return Err(unexpected());
723            };
724            let mut rule = list
725                .rules
726                .into_iter()
727                .find(|r| r.id == id)
728                .ok_or_else(|| Error::Usage(format!("no rule {id}")))?;
729            rule.folder = (folder != 0).then_some(folder);
730            client.call(Method::UpdateRule(v1::UpdateRule { rule: Some(rule) }))?;
731            out(&format!("moved rule {id}\n"))
732        }
733        Some("edit") => {
734            let edit = rules_edit(&mut args)?;
735            reject_rest(args)?;
736            let count = edit.ids.len();
737            client.call(Method::EditRules(edit))?;
738            out(&format!("changed {count} rules\n"))
739        }
740        Some("lint") => {
741            let unused_days: u32 = args.opt_value_from_str("--days")?.unwrap_or(0);
742            reject_rest(args)?;
743            match client.call(Method::LintRules(v1::LintRules { unused_days }))? {
744                v1::response::Result::LintReport(report) => out(&format_lint(&report)),
745                _ => Err(unexpected()),
746            }
747        }
748        Some(which @ ("enable" | "disable")) => {
749            let id: u64 = args.free_from_str()?;
750            reject_rest(args)?;
751            client.call(Method::SetRuleEnabled(v1::SetRuleEnabled {
752                id,
753                enabled: which == "enable",
754            }))?;
755            out(&format!("{which}d rule {id}\n"))
756        }
757        Some("import-opensnitch") => {
758            let dry_run = args.contains("--dry-run");
759            let dir: String = args
760                .opt_free_from_str()?
761                .unwrap_or_else(|| "/etc/opensnitchd/rules".to_owned());
762            reject_rest(args)?;
763            import_opensnitch(&dir, dry_run, version)
764        }
765        Some("import") => {
766            let path: String = args.free_from_str()?;
767            reject_rest(args)?;
768            let toml =
769                std::fs::read_to_string(&path).map_err(|e| Error::Usage(format!("{path}: {e}")))?;
770            let folder = Path::new(&path).file_stem().map_or_else(
771                || "Imported".to_owned(),
772                |s| s.to_string_lossy().into_owned(),
773            );
774            client.call(Method::ImportRules(v1::ImportRules { toml, folder }))?;
775            out(&format!("imported {path}\n"))
776        }
777        Some("export") => {
778            reject_rest(args)?;
779            print_file(client.call(Method::ExportRules(v1::ExportRules {}))?)
780        }
781        Some(other) => Err(Error::Usage(format!("unknown rules command `{other}`"))),
782    }
783}

785/// `snitchhiker prompts ...`.
786fn prompts(mut args: pico_args::Arguments, version: &str) -> Result<(), Error> {
787    use v1::request::Method;
788    let command = args.subcommand()?;
789    let mut client = connect(version)?;
790    match command.as_deref() {
791        None | Some("list" | "ls") => {
792            reject_rest(args)?;
793            match client.call(Method::ListPrompts(v1::ListPrompts {}))? {
794                v1::response::Result::Prompts(list) => {
795                    out(&list.prompts.iter().map(format_prompt).collect::<String>())
796                }
797                _ => Err(unexpected()),
798            }
799        }
800        Some("watch") => {
801            reject_rest(args)?;
802            client.subscribe(v1::Topic::Prompts)?;
803            loop {
804                match client.notice()?.body {
805                    Some(v1::notice::Body::Prompt(prompt)) => out(&format_prompt(&prompt))?,
806                    Some(v1::notice::Body::PromptResolved(resolved)) => {
807                        out(&format_resolved(&resolved))?;
808                    }
809                    _ => {}
810                }
811            }
812        }
813        Some("answer") => {
814            let id: u64 = args.free_from_str()?;
815            let action = action(&args.free_from_str::<String>()?)?;
816            let suggestion: u32 = args.opt_value_from_str("--suggestion")?.unwrap_or(0);
817            let folder: Option<u64> = args.opt_value_from_str("--folder")?;
818            let duration = match args
819                .opt_value_from_str::<_, String>("--duration")?
820                .as_deref()
821            {
822                None | Some("forever") => v1::Duration::Forever,
823                Some("once") => v1::Duration::Once,
824                Some("restart") => v1::Duration::UntilRestart,
825                Some("hour") => v1::Duration::Hour,
826                Some("12h") => v1::Duration::HalfDay,
827                Some(other) => return Err(Error::Usage(format!("unknown duration `{other}`"))),
828            };
829            reject_rest(args)?;
830            client.call(Method::AnswerPrompt(v1::AnswerPrompt {
831                id,
832                action: convert::action(action).into(),
833                duration: duration.into(),
834                suggestion,
835                conditions: None,
836                folder,
837            }))?;
838            out(&format!("answered prompt {id}\n"))
839        }
840        Some(other) => Err(Error::Usage(format!("unknown prompts command `{other}`"))),
841    }
842}

844fn action(text: &str) -> Result<Action, Error> {
845    match text {
846        "allow" => Ok(Action::Allow),
847        "deny" => Ok(Action::Deny),
848        "reject" => Ok(Action::Reject),
849        other => Err(Error::Usage(format!("unknown action `{other}`"))),
850    }

853fn spec_from_args(args: &mut pico_args::Arguments) -> Result<RuleSpec, Error> {
854    let action = action(&args.free_from_str::<String>()?)?;
855    let name: Option<String> = args.opt_value_from_str("--name")?;
856    let tier = match args.opt_value_from_str::<_, String>("--tier")?.as_deref() {
857        None | Some("normal") => snitchhiker_core::rule::Tier::Normal,
858        Some("override") => snitchhiker_core::rule::Tier::Override,
859        Some("fallback") => snitchhiker_core::rule::Tier::Fallback,
860        Some(other) => return Err(Error::Usage(format!("unknown tier `{other}`"))),
861    };
862    let conditions = Conditions {
863        exe: args.values_from_str("--exe")?,
864        host: args.values_from_str("--host")?,
865        net: args.values_from_str("--net")?,
866        port: args.values_from_str("--port")?,
867        uid: args.values_from_str("--uid")?,
868        protocol: args.values_from_str("--protocol")?,
869        context: args
870            .contains("--any-context")
871            .then_some(snitchhiker_core::rule::ContextKind::Any),
872        ..Conditions::default()
873    };
874    Ok(RuleSpec {
875        name: name.unwrap_or_else(|| format!("{action:?}").to_lowercase()),
876        action,
877        tier,
878        enabled: true,
879        notes: String::new(),
880        conditions,
881    })
882}

900/// `Apps/curl/`: the folders a rule is in, from the top.
901fn folder_path(list: &v1::RuleList, folder: Option<u64>) -> String {
902    let mut names = Vec::new();
903    let mut next = folder;
904    for _ in 0..list.folders.len() {
905        let Some(found) = next.and_then(|id| list.folders.iter().find(|f| f.id == id)) else {
906            break;
907        };
908        names.push(found.name.as_str());
909        next = found.parent;
910    }
911    names.reverse();
912    names.iter().fold(String::new(), |mut path, name| {
913        path.push_str(name);
914        path.push('/');
915        path
916    })
917}

919fn format_rules(list: &v1::RuleList) -> String {
920    let mut text = String::new();
921    for rule in &list.rules {
922        let spec = rule
923            .conditions
924            .as_ref()
925            .map_or_else(Conditions::default, convert::conditions_from);
926        let _ = writeln!(
927            text,
928            "{:>5}  {:<7} {:<8} {}{}{}{}  {}",
929            if rule.locked {
930                "-".to_owned()
931            } else {
932                rule.id.to_string()
933            },
934            format!("{:?}", rule.action()).to_lowercase(),
935            format!("{:?}", rule.tier()).to_lowercase(),
936            if rule.enabled { "" } else { "(disabled) " },
937            if rule.locked { "(locked) " } else { "" },
938            folder_path(list, rule.folder),
939            rule.name,
940            toml::to_string(&spec)
941                .unwrap_or_default()
942                .replace('\n', " "),
943        );
944    }
945    text
946}

966fn format_events(events: &[v1::ConnectionEvent]) -> String {
967    let mut text = String::new();
968    for event in events {
969        let address =
970            convert::address_from(&event.address).map_or_else(String::new, |a| a.to_string());
971        let _ = writeln!(
972            text,
973            "{:>8}  {:<6} uid {:<6} {} -> {}:{} proto {} out {} in {}{}",
974            event.seq,
975            format!("{:?}", event.action()).to_lowercase(),
976            event.uid,
977            if event.exe.is_empty() {
978                "?"
979            } else {
980                &event.exe
981            },
982            if event.host.is_empty() {
983                &address
984            } else {
985                &event.host
986            },
987            event.port,
988            event.protocol,
989            event.bytes_out,
990            event.bytes_in,
991            event
992                .rule
993                .map_or_else(String::new, |r| format!(" (rule {r})")),
994        );
995    }
996    text
997}

999fn format_explanation(explanation: &v1::Explanation) -> String {
1000    let mut text = format_events(explanation.event.as_slice());
1001    if explanation.rules.is_empty() {
1002        text.push_str("  no rule matches it now: it would be prompted for (or decided by the default action)\n");
1003    }
1004    for (i, rule) in explanation.rules.iter().enumerate() {
1005        let _ = writeln!(
1006            text,
1007            "  {} rule {} {:?}: {} ({:?}, specificity {})",
1008            if i == 0 { "decides" } else { "        " },
1009            rule.id,
1010            rule.name,
1011            format!("{:?}", rule.action()).to_lowercase(),
1012            rule.tier(),
1013            rule.specificity,
1014        );
1015    }
1016    if explanation.partial {
1017        text.push_str(
1018            "  rules on command lines or parent processes aren't checked: the history doesn't keep them\n",
1019        );
1020    }
1021    text
1022}

1024fn format_prompt(prompt: &v1::Prompt) -> String {
1025    let address =
1026        convert::address_from(&prompt.address).map_or_else(String::new, |a| a.to_string());
1027    let mut text = format!(
1028        "prompt {}: {} (uid {}{}) -> {}:{} proto {}, {} waiting{}\n",
1029        prompt.id,
1030        prompt.exe,
1031        prompt.uid,
1032        if prompt.user.is_empty() {
1033            String::new()
1034        } else {
1035            format!(" {}", prompt.user)
1036        },
1037        prompt.hosts.first().unwrap_or(&address),
1038        prompt.port,
1039        prompt.protocol,
1040        prompt.waiting,
1041        if prompt.exe_writable {
1042            " (its executable is writable by a user)"
1043        } else {
1044            ""
1045        },
1046    );
1047    for (index, suggestion) in prompt.suggestions.iter().enumerate() {
1048        let _ = writeln!(
1049            text,
1050            "  {index}: {}{}",
1051            suggestion.label,
1052            if suggestion.broad { " (broad)" } else { "" }
1053        );
1054    }
1055    text
1056}
