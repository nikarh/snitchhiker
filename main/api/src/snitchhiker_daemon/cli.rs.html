124#[derive(Debug, thiserror::Error)]
125pub enum Error {
126    #[error(transparent)]
127    Args(#[from] pico_args::Error),
128    #[error("{0}")]
129    Usage(String),
130    #[error("{0} (is the daemon running, and is this root?)")]
131    Connect(snitchhiker_client::Error),
132    #[error(transparent)]
133    Client(#[from] snitchhiker_client::Error),
134    #[error("writing to stdout: {0}")]
135    Stdout(io::Error),
136}

138fn out(text: &str) -> Result<(), Error> {
139    let mut stdout = io::stdout().lock();
140    stdout
141        .write_all(text.as_bytes())
142        .and_then(|()| stdout.flush())
143        .map_err(Error::Stdout)
144}

150fn reject_rest(args: pico_args::Arguments) -> Result<(), Error> {
151    let rest = args.finish();
152    if rest.is_empty() {
153        Ok(())
154    } else {
155        Err(Error::Usage(format!("unexpected arguments: {rest:?}")))
156    }
157}

159fn unexpected() -> Error {
160    Error::Usage("the daemon answered something unexpected".to_owned())
161}

163/// Runs a command; `version` is the binary's version string.
164pub fn run(mut args: pico_args::Arguments, version: &str) -> Result<(), Error> {
165    use v1::request::Method;
166    if args.contains(["-h", "--help"]) {
167        return out(HELP);
168    }
169    match args.subcommand()?.as_deref() {
170        None => out(HELP),
171        Some("version") => out(&format!("snitchhiker {version}\n")),
172        Some("status") => {
173            reject_rest(args)?;
174            let mut client = connect(version)?;
175            match client.call(Method::GetStatus(v1::GetStatus {}))? {
176                v1::response::Result::Status(status) => out(&format_status(&status)),
177                _ => Err(unexpected()),
178            }
179        }
180        Some("rules") => rules(args, version),
181        Some("policy") => policy(args, version),
182        Some("system") => system(args, version),
183        Some("alerts") => alerts(args, version),
184        Some("folders") => folders(args, version),
185        Some("explore") => explore(args, version),
186        Some("history") => history(args, version),
187        Some("connections") => connections(args, version),
188        Some("kill") => kill_connection(args, version),
189        Some("explain") => {
190            let seq: i64 = args.free_from_str()?;
191            reject_rest(args)?;
192            match connect(version)?.call(Method::Explain(v1::Explain { seq }))? {
193                v1::response::Result::Explanation(explanation) => {
194                    out(&format_explanation(&explanation))
195                }
196                _ => Err(unexpected()),
197            }
198        }
199        Some("undo") => {
200            let change: u64 = args.free_from_str()?;
201            reject_rest(args)?;
202            connect(version)?.call(Method::Undo(v1::Undo { change }))?;
203            out(&format!("undid the changes of change {change}'s batch\n"))
204        }
205        Some("audit") => {
206            let limit: u32 = args.opt_value_from_str("--limit")?.unwrap_or(50);
207            reject_rest(args)?;
208            let mut client = connect(version)?;
209            match client.call(Method::ListAudit(v1::ListAudit { before: 0, limit }))? {
210                v1::response::Result::Audit(page) => out(&format_audit(&page.entries)),
211                _ => Err(unexpected()),
212            }
213        }
214        Some("events") => {
215            let limit: u32 = args.opt_value_from_str("--limit")?.unwrap_or(50);
216            let query: String = args.opt_value_from_str("--query")?.unwrap_or_default();
217            reject_rest(args)?;
218            let mut client = connect(version)?;
219            match client.call(Method::ListEvents(v1::ListEvents {
220                before: 0,
221                limit,
222                columnar: false,
223                query,
224            }))? {
225                v1::response::Result::Events(page) => out(&format_events(&page.events)),
226                _ => Err(unexpected()),
227            }
228        }
229        Some("learn") => {
230            let hours: u32 = args.opt_value_from_str("--hours")?.unwrap_or(24);
231            let folder: String = args
232                .opt_value_from_str("--folder")?
233                .unwrap_or_else(|| "Learned".to_owned());
234            reject_rest(args)?;
235            let mut client = connect(version)?;
236            print_file(client.call(Method::Learn(v1::Learn { hours, folder }))?)
237        }
238        Some("prompts") => prompts(args, version),
239        Some(other) => Err(Error::Usage(format!(
240            "unknown command `{other}`; see --help"
241        ))),
242    }
243}

245fn format_lint(report: &v1::LintReport) -> String {
246    let mut text = String::new();
247    for finding in &report.findings {
248        let _ = writeln!(
249            text,
250            "rule {}: {}{}",
251            finding.rule,
252            finding.description,
253            if finding.suggestion.is_empty() {
254                String::new()
255            } else {
256                format!(" (the program now matches {})", finding.suggestion)
257            }
258        );
259    }
260    if report.findings.is_empty() {
261        text.push_str("nothing to report\n");
262    }
263    text
264}

266/// The rules and the change of `rules edit`.
267fn rules_edit(args: &mut pico_args::Arguments) -> Result<v1::EditRules, Error> {
268    use v1::edit_rules::Edit;
269    let ids: String = args.free_from_str()?;
270    let ids = ids
271        .split(',')
272        .map(|id| {
273            id.trim()
274                .parse()
275                .map_err(|_| Error::Usage(format!("`{id}` isn't a rule id")))
276        })
277        .collect::<Result<Vec<u64>, _>>()?;
278    let what: String = args.free_from_str()?;
279    let edit = match what.as_str() {
280        "enable" => Edit::Enabled(true),
281        "disable" => Edit::Enabled(false),
282        "delete" => Edit::Delete(true),
283        "move" => {
284            let folder: u64 = args.free_from_str()?;
285            Edit::Move(v1::MoveRules {
286                folder: (folder != 0).then_some(folder),
287            })
288        }
289        "action" => {
290            let action: String = args.free_from_str()?;
291            Edit::Action(
292                match action.as_str() {
293                    "allow" => v1::Action::Allow,
294                    "deny" => v1::Action::Deny,
295                    "reject" => v1::Action::Reject,
296                    other => return Err(Error::Usage(format!("unknown action `{other}`"))),
297                }
298                .into(),
299            )
300        }
301        "expires" => {
302            let when: String = args.free_from_str()?;
303            Edit::Expires(
304                match when.as_str() {
305                    "hour" => v1::Duration::Hour,
306                    "half-day" => v1::Duration::HalfDay,
307                    "until-restart" => v1::Duration::UntilRestart,
308                    "never" => v1::Duration::Forever,
309                    other => return Err(Error::Usage(format!("unknown expiry `{other}`"))),
310                }
311                .into(),
312            )
313        }
314        other => return Err(Error::Usage(format!("unknown edit `{other}`"))),
315    };
316    Ok(v1::EditRules {
317        ids,
318        edit: Some(edit),
319    })
320}

322/// `alerts` and its subcommands.
323fn alerts(mut args: pico_args::Arguments, version: &str) -> Result<(), Error> {
324    use v1::request::Method;
325    let all = args.contains("--all");
326    let command = args.subcommand()?;
327    let mut client = connect(version)?;
328    let print_rules = |result| match result {
329        v1::response::Result::AlertRules(list) => {
330            let mut text = String::new();
331            for rule in &list.rules {
332                let _ = writeln!(
333                    text,
334                    "{:>4} {} {:?}: {} in {} s{}{}",
335                    rule.id,
336                    if rule.enabled { "on " } else { "off" },
337                    rule.name,
338                    rule.threshold,
339                    rule.window_secs,
340                    if rule.query.is_empty() {
341                        String::new()
342                    } else {
343                        format!(" matching `{}`", rule.query)
344                    },
345                    if rule.first_program {
346                        ", programs never seen before"
347                    } else {
348                        ""
349                    },
350                );
351            }
352            out(&text)
353        }
354        _ => Err(unexpected()),
355    };
356    match command.as_deref() {
357        None | Some("list") => {
358            reject_rest(args)?;
359            match client.call(Method::ListAlerts(v1::ListAlerts {
360                before: 0,
361                limit: 50,
362                open_only: !all,
363            }))? {
364                v1::response::Result::Alerts(page) => {
365                    let mut text = String::new();
366                    for alert in &page.alerts {
367                        let _ = writeln!(
368                            text,
369                            "{:>6} {} {}{}: {}",
370                            alert.id,
371                            alert.kind,
372                            alert.title,
373                            if alert.acknowledged {
374                                format!(" (acknowledged by {})", alert.acknowledged_by)
375                            } else {
376                                String::new()
377                            },
378                            alert.body,
379                        );
380                    }
381                    let _ = writeln!(text, "{} not acknowledged", page.open);
382                    out(&text)
383                }
384                _ => Err(unexpected()),
385            }
386        }
387        Some("ack") => {
388            let ids: Vec<u64> = args
389                .finish()
390                .iter()
391                .map(|a| {
392                    a.to_string_lossy().parse().map_err(|_| {
393                        Error::Usage(format!("`{}` isn't an alert id", a.to_string_lossy()))
394                    })
395                })
396                .collect::<Result<_, _>>()?;
397            if ids.is_empty() && !all {
398                return Err(Error::Usage("alerts ack <id>... | --all".to_owned()));
399            }
400            client.call(Method::AcknowledgeAlerts(v1::AcknowledgeAlerts {
401                ids,
402                all,
403            }))?;
404            out("acknowledged\n")
405        }
406        Some("rules") => alert_rules(args, &mut client, print_rules),
407        Some(other) => Err(Error::Usage(format!("unknown command `alerts {other}`"))),
408    }
409}

411/// `alerts rules` and its subcommands.
412fn alert_rules(
413    mut args: pico_args::Arguments,
414    client: &mut Client,
415    print_rules: impl Fn(v1::response::Result) -> Result<(), Error>,
416) -> Result<(), Error> {
417    use v1::request::Method;
418    match args.subcommand()?.as_deref() {
419        None => {
420            reject_rest(args)?;
421            print_rules(client.call(Method::ListAlertRules(v1::ListAlertRules {}))?)
422        }
423        Some("add") => {
424            let query: String = args.opt_value_from_str("--query")?.unwrap_or_default();
425            let threshold: u32 = args.opt_value_from_str("--threshold")?.unwrap_or(1);
426            let window_secs: u32 = args.opt_value_from_str("--window")?.unwrap_or(0);
427            let first_program = args.contains("--first-program");
428            let name: String = args.free_from_str()?;
429            reject_rest(args)?;
430            print_rules(client.call(Method::PutAlertRule(v1::PutAlertRule {
431                rule: Some(v1::AlertRule {
432                    id: 0,
433                    name,
434                    enabled: true,
435                    query,
436                    threshold,
437                    window_secs,
438                    first_program,
439                }),
440            }))?)
441        }
442        Some("rm") => {
443            let id: u64 = args.free_from_str()?;
444            reject_rest(args)?;
445            print_rules(client.call(Method::DeleteAlertRule(v1::DeleteAlertRule { id }))?)
446        }
447        Some(other) => Err(Error::Usage(format!(
448            "unknown command `alerts rules {other}`"
449        ))),
450    }
451}

453/// `system` and its subcommands.
454fn system(mut args: pico_args::Arguments, version: &str) -> Result<(), Error> {
455    use v1::request::Method;
456    let command = args.subcommand()?;
457    let mut client = connect(version)?;
458    let current = |client: &mut Client| -> Result<v1::SystemFirewall, Error> {
459        match client.call(Method::GetSystemFirewall(v1::GetSystemFirewall {}))? {
460            v1::response::Result::SystemFirewall(firewall) => Ok(firewall),
461            _ => Err(unexpected()),
462        }
463    };
464    let set = |client: &mut Client, firewall, revert_after_secs| -> Result<(), Error> {
465        match client.call(Method::SetSystemFirewall(v1::SetSystemFirewall {
466            firewall: Some(firewall),
467            revert_after_secs,
468        }))? {
469            v1::response::Result::SystemFirewall(firewall) => out(&format_system(&firewall)),
470            _ => Err(unexpected()),
471        }
472    };
473    match command.as_deref() {
474        None | Some("show") => {
475            reject_rest(args)?;
476            out(&format_system(&current(&mut client)?))
477        }
478        Some(which @ ("on" | "off")) => {
479            reject_rest(args)?;
480            let firewall = v1::SystemFirewall {
481                enabled: which == "on",
482                ..current(&mut client)?
483            };
484            set(&mut client, firewall, 0)
485        }
486        Some(chain @ ("input" | "forward")) => {
487            let policy: String = args.free_from_str()?;
488            reject_rest(args)?;
489            let policy = match policy.as_str() {
490                "accept" => v1::ChainPolicy::Accept,
491                "drop" => v1::ChainPolicy::Drop,
492                other => return Err(Error::Usage(format!("unknown policy `{other}`"))),
493            };
494            let mut firewall = current(&mut client)?;
495            if chain == "input" {
496                firewall.input = policy.into();
497            } else {
498                firewall.forward = policy.into();
499            }
500            set(&mut client, firewall, 0)
501        }
502        Some("export") => {
503            reject_rest(args)?;
504            let firewall =
505                crate::system::from_wire(&current(&mut client)?).map_err(Error::Usage)?;
506            out(&toml::to_string(&firewall).map_err(|e| Error::Usage(e.to_string()))?)
507        }
508        Some("import") => {
509            let trial: u32 = args.opt_value_from_str("--trial")?.unwrap_or(0);
510            let path: std::path::PathBuf = args.free_from_str()?;
511            reject_rest(args)?;
512            let text = std::fs::read_to_string(&path)
513                .map_err(|e| Error::Usage(format!("{}: {e}", path.display())))?;
514            let firewall: snitchhiker_core::system::Firewall = toml::from_str(&text)
515                .map_err(|e| Error::Usage(format!("{}: {e}", path.display())))?;
516            set(&mut client, crate::system::wire(&firewall, 0), trial)
517        }
518        Some("confirm") => {
519            reject_rest(args)?;
520            client.call(Method::ConfirmSystemFirewall(v1::ConfirmSystemFirewall {}))?;
521            out("kept\n")
522        }
523        Some("import-opensnitch") => system_import_opensnitch(args, &mut client, set),
524        Some(other) => Err(Error::Usage(format!("unknown command `system {other}`"))),
525    }
526}

528/// `system import-opensnitch`.
529fn system_import_opensnitch(
530    mut args: pico_args::Arguments,
531    client: &mut Client,
532    set: impl Fn(&mut Client, v1::SystemFirewall, u32) -> Result<(), Error>,
533) -> Result<(), Error> {
534    let dry_run = args.contains("--dry-run");
535    let path: String = args
536        .opt_free_from_str()?
537        .unwrap_or_else(|| "/etc/opensnitchd/system-fw.json".to_owned());
538    reject_rest(args)?;
539    let text = std::fs::read_to_string(&path).map_err(|e| Error::Usage(format!("{path}: {e}")))?;
540    let imported = snitchhiker_core::system::import_opensnitch(&text);
541    let mut report = String::new();
542    for (name, reason) in &imported.skipped {
543        let _ = writeln!(report, "skipped {name}: {reason}");
544    }
545    for note in &imported.notes {
546        let _ = writeln!(report, "note: {note}");
547    }
548    let Some(firewall) = imported.firewall else {
549        return Err(Error::Usage(format!("nothing to import:\n{report}")));
550    };
551    if dry_run {
552        let toml = toml::to_string(&firewall).map_err(|e| Error::Usage(e.to_string()))?;
553        let comments = report.lines().fold(String::new(), |mut text, line| {
554            let _ = writeln!(text, "# {line}");
555            text
556        });
557        return out(&format!("{toml}{comments}"));
558    }
559    let rules = firewall.rules.len();
560    set(client, crate::system::wire(&firewall, 0), 0)?;
561    let _ = writeln!(
562        report,
563        "imported {rules} rules of OpenSnitch's system firewall"
564    );
565    out(&report)
566}

568fn format_system(firewall: &v1::SystemFirewall) -> String {
569    let policy = |p: v1::ChainPolicy| {
570        p.as_str_name()
571            .trim_start_matches("CHAIN_POLICY_")
572            .to_lowercase()
573    };
574    let mut text = format!(
575        "system firewall {}, input {}, forward {}\n",
576        if firewall.enabled { "on" } else { "off" },
577        policy(firewall.input()),
578        policy(firewall.forward()),
579    );
580    if firewall.reverts_at_ms != 0 {
581        let _ = writeln!(
582            text,
583            "a change reverts in {} s unless confirmed (snitchhiker system confirm)",
584            firewall
585                .reverts_at_ms
586                .saturating_sub(crate::runtime::now_ms())
587                / 1000
588        );
589    }
590    for (i, rule) in firewall.rules.iter().enumerate() {
591        let mut parts = vec![
592            rule.chain()
593                .as_str_name()
594                .trim_start_matches("SYSTEM_CHAIN_")
595                .to_lowercase(),
596        ];
597        for (key, value) in [
598            ("proto", rule.protocol.clone()),
599            ("from", rule.source.join(",")),
600            ("to", rule.destination.join(",")),
601            ("sport", rule.source_ports.clone()),
602            ("dport", rule.destination_ports.clone()),
603            ("in", rule.in_interface.clone()),
604            ("out", rule.out_interface.clone()),
605            ("ct", rule.ct_state.join(",")),
606            ("icmp", rule.icmp_types.join(",")),
607        ] {
608            if !value.is_empty() {
609                parts.push(format!("{key} {value}"));
610            }
611        }
612        let _ = writeln!(
613            text,
614            "{:>3} {} {:<7} {}  {}",
615            i + 1,
616            if rule.enabled { "on " } else { "off" },
617            rule.target()
618                .as_str_name()
619                .trim_start_matches("SYSTEM_TARGET_")
620                .to_lowercase(),
621            parts.join(" "),
622            rule.description,
623        );
624    }
625    text
626}

628/// `policy`, `policy set`, `policy reset`.
629fn policy(mut args: pico_args::Arguments, version: &str) -> Result<(), Error> {
630    use v1::request::Method;
631    let command = args.subcommand()?;
632    let rest: Vec<String> = args
633        .finish()
634        .into_iter()
635        .map(|a| a.to_string_lossy().into_owned())
636        .collect();
637    let method = match command.as_deref() {
638        None | Some("show") if rest.is_empty() => Method::GetPolicy(v1::GetPolicy {}),
639        Some("set") if !rest.is_empty() => Method::SetPolicy(policy_set(&rest)?),
640        Some("reset") => Method::SetPolicy(v1::SetPolicy {
641            reset: if rest.is_empty() {
642                crate::policy::SETTINGS
643                    .iter()
644                    .map(|s| (*s).to_owned())
645                    .collect()
646            } else {
647                rest
648            },
649            ..v1::SetPolicy::default()
650        }),
651        _ => {
652            return Err(Error::Usage(
653                "policy [show | set <setting>=<value>... | reset [<setting>...]]".to_owned(),
654            ));
655        }
656    };
657    match connect(version)?.call(method)? {
658        v1::response::Result::Policy(policy) => out(&format_policy(&policy)),
659        _ => Err(unexpected()),
660    }
661}

663fn policy_set(pairs: &[String]) -> Result<v1::SetPolicy, Error> {
664    let action = |value: &str| match value {
665        "allow" => Ok(v1::Action::Allow.into()),
666        "deny" => Ok(v1::Action::Deny.into()),
667        "reject" => Ok(v1::Action::Reject.into()),
668        other => Err(Error::Usage(format!("unknown action `{other}`"))),
669    };
670    let number = |value: &str| {
671        value
672            .parse::<u32>()
673            .map_err(|_| Error::Usage(format!("`{value}` isn't a number")))
674    };
675    let mut set = v1::SetPolicy::default();
676    for pair in pairs {
677        let (key, value) = pair
678            .split_once('=')
679            .ok_or_else(|| Error::Usage(format!("`{pair}`: expected <setting>=<value>")))?;
680        match key {
681            "mode" => {
682                set.mode = Some(match value {
683                    "enforce" => v1::Mode::Enforce.into(),
684                    "monitor" => v1::Mode::Monitor.into(),
685                    other => return Err(Error::Usage(format!("unknown mode `{other}`"))),
686                });
687            }
688            "no_match" => {
689                set.no_match = Some(match value {
690                    "ask" => v1::Policy::Ask.into(),
691                    "allow" => v1::Policy::Allow.into(),
692                    "deny" => v1::Policy::Deny.into(),
693                    "reject" => v1::Policy::Reject.into(),
694                    other => return Err(Error::Usage(format!("unknown policy `{other}`"))),
695                });
696            }
697            "no_client" => set.no_client = Some(action(value)?),
698            "on_timeout" => set.on_timeout = Some(action(value)?),
699            "prompt_timeout_secs" => set.prompt_timeout_secs = Some(number(value)?),
700            "retention_days" => set.retention_days = Some(number(value)?),
701            other => return Err(Error::Usage(format!("unknown setting `{other}`"))),
702        }
703    }
704    Ok(set)
705}

707fn format_policy(policy: &v1::NodePolicy) -> String {
708    let name = |text: &str, prefix: &str| text.trim_start_matches(prefix).to_lowercase();
709    let mark = |setting: &str| {
710        if policy.changed.iter().any(|c| c == setting) {
711            "  (changed by a client)"
712        } else {
713            ""
714        }
715    };
716    let mut text = String::new();
717    for (setting, value) in [
718        ("mode", name(policy.mode().as_str_name(), "MODE_")),
719        ("no_match", name(policy.no_match().as_str_name(), "POLICY_")),
720        (
721            "no_client",
722            name(policy.no_client().as_str_name(), "ACTION_"),
723        ),
724        (
725            "prompt_timeout_secs",
726            policy.prompt_timeout_secs.to_string(),
727        ),
728        (
729            "on_timeout",
730            name(policy.on_timeout().as_str_name(), "ACTION_"),
731        ),
732        ("retention_days", policy.retention_days.to_string()),
733    ] {
734        let _ = writeln!(text, "{setting:<20} {value}{}", mark(setting));
735    }
736    text
737}

746/// `snitchhiker rules ...`.
747/// `explore`.
748fn explore(mut args: pico_args::Arguments, version: &str) -> Result<(), Error> {
749    use v1::request::Method;
750    let by: String = args
751        .opt_value_from_str("--by")?
752        .unwrap_or_else(|| "program".to_owned());
753    let hours: u64 = args.opt_value_from_str("--hours")?.unwrap_or(24);
754    let filter: Option<String> = args.opt_value_from_str("--where")?;
755    reject_rest(args)?;
756    let dimension = |name: &str| match name {
757        "program" => Ok(v1::Dimension::Program),
758        "host" => Ok(v1::Dimension::Host),
759        "port" => Ok(v1::Dimension::Port),
760        "user" => Ok(v1::Dimension::User),
761        other => Err(Error::Usage(format!(
762            "unknown dimension `{other}` (program, host, port, user)"
763        ))),
764    };
765    let filter = match filter {
766        None => None,
767        Some(text) => {
768            let (name, value) = text
769                .split_once('=')
770                .ok_or_else(|| Error::Usage("--where <dimension>=<value>".to_owned()))?;
771            Some(v1::ExploreFilter {
772                dimension: dimension(name)?.into(),
773                value: value.to_owned(),
774            })
775        }
776    };
777    let now = std::time::SystemTime::now()
778        .duration_since(std::time::UNIX_EPOCH)
779        .map_or(0, |d| u64::try_from(d.as_millis()).unwrap_or(u64::MAX));
780    let mut client = connect(version)?;
781    match client.call(Method::Explore(v1::Explore {
782        from_ms: now.saturating_sub(hours * 3_600_000),
783        to_ms: now,
784        group_by: dimension(&by)?.into(),
785        filter,
786        limit: 50,
787    }))? {
788        v1::response::Result::Explored(page) => {
789            let mut text = format!(
790                "{:>9} {:>9} {:>11} {:>11}  {by}\n",
791                "allowed", "blocked", "sent", "received"
792            );
793            for row in &page.rows {
794                let _ = writeln!(
795                    text,
796                    "{:>9} {:>9} {:>11} {:>11}  {}",
797                    row.allowed,
798                    row.blocked,
799                    row.bytes_out,
800                    row.bytes_in,
801                    if row.key.is_empty() {
802                        "(no name)"
803                    } else {
804                        &row.key
805                    }
806                );
807            }
808            out(&text)
809        }
810        _ => Err(unexpected()),
811    }
812}

964/// `rules import-opensnitch`: OpenSnitch's rule files in `dir`, converted and imported (or printed, `dry_run`).
965fn import_opensnitch(dir: &str, dry_run: bool, version: &str) -> Result<(), Error> {
966    let mut files = Vec::new();
967    let entries = std::fs::read_dir(dir).map_err(|e| Error::Usage(format!("{dir}: {e}")))?;
968    for entry in entries.flatten() {
969        let path = entry.path();
970        if path.extension().is_some_and(|e| e == "json") {
971            let text = std::fs::read_to_string(&path)
972                .map_err(|e| Error::Usage(format!("{}: {e}", path.display())))?;
973            files.push((path.display().to_string(), text));
974        }
975    }
976    files.sort();
977    let imported = snitchhiker_core::opensnitch::import(&files, "OpenSnitch");
978    let mut report = String::new();
979    for (name, reason) in &imported.skipped {
980        let _ = writeln!(report, "skipped {name}: {reason}");
981    }
982    let toml = toml::to_string(&imported.file)
983        .map_err(|e| Error::Usage(format!("writing the rule file: {e}")))?;
984    if dry_run {
985        // The rule file, and what was skipped as comments: the output stays a valid rule file.
986        let comments = report.lines().fold(String::new(), |mut text, line| {
987            let _ = writeln!(text, "# {line}");
988            text
989        });
990        return out(&format!("{toml}{comments}"));
991    }
992    let mut client = connect(version)?;
993    if !imported.file.rule.is_empty() {
994        client.call(v1::request::Method::ImportRules(v1::ImportRules {
995            toml,
996            folder: "OpenSnitch".to_owned(),
997        }))?;
998    }
999    let _ = writeln!(
1000        report,
1001        "imported {} of {} rules into the folder OpenSnitch; they apply to the host only \
1002         (add context = \"any\" to a rule for containers too)",
1003        imported.file.rule.len(),
1004        imported.file.rule.len() + imported.skipped.len()
1005    );
1006    out(&report)
1007}

1009fn rules(mut args: pico_args::Arguments, version: &str) -> Result<(), Error> {
1010    use v1::request::Method;
1011    let command = args.subcommand()?;
1012    let mut client = connect(version)?;
1013    match command.as_deref() {
1014        None | Some("list" | "ls") => {
1015            reject_rest(args)?;
1016            match client.call(Method::ListRules(v1::ListRules {}))? {
1017                v1::response::Result::Rules(list) => out(&format_rules(&list)),
1018                _ => Err(unexpected()),
1019            }
1020        }
1021        Some("add") => {
1022            let spec = spec_from_args(&mut args)?;
1023            reject_rest(args)?;
1024            let rule = spec
1025                .to_rule(0, None)
1026                .map_err(|e| Error::Usage(e.to_string()))?;
1027            match client.call(Method::AddRule(v1::AddRule {
1028                rule: Some(convert::rule(&rule, false)),
1029            }))? {
1030                v1::response::Result::RuleAdded(added) => {
1031                    out(&format!("added rule {}\n", added.id))
1032                }
1033                _ => Err(unexpected()),
1034            }
1035        }
1036        Some("rm" | "delete") => {
1037            let id: u64 = args.free_from_str()?;
1038            reject_rest(args)?;
1039            client.call(Method::DeleteRule(v1::DeleteRule { id }))?;
1040            out(&format!("deleted rule {id}\n"))
1041        }
1042        Some("move") => {
1043            let id: u64 = args.free_from_str()?;
1044            let folder: u64 = args.free_from_str()?;
1045            reject_rest(args)?;
1046            let v1::response::Result::Rules(list) =
1047                client.call(Method::ListRules(v1::ListRules {}))?
1048            else {
1049                return Err(unexpected());
1050            };
1051            let mut rule = list
1052                .rules
1053                .into_iter()
1054                .find(|r| r.id == id)
1055                .ok_or_else(|| Error::Usage(format!("no rule {id}")))?;
1056            rule.folder = (folder != 0).then_some(folder);
1057            client.call(Method::UpdateRule(v1::UpdateRule { rule: Some(rule) }))?;
1058            out(&format!("moved rule {id}\n"))
1059        }
1060        Some("edit") => {
1061            let edit = rules_edit(&mut args)?;
1062            reject_rest(args)?;
1063            let count = edit.ids.len();
1064            client.call(Method::EditRules(edit))?;
1065            out(&format!("changed {count} rules\n"))
1066        }
1067        Some("lint") => {
1068            let unused_days: u32 = args.opt_value_from_str("--days")?.unwrap_or(0);
1069            reject_rest(args)?;
1070            match client.call(Method::LintRules(v1::LintRules { unused_days }))? {
1071                v1::response::Result::LintReport(report) => out(&format_lint(&report)),
1072                _ => Err(unexpected()),
1073            }
1074        }
1075        Some(which @ ("enable" | "disable")) => {
1076            let id: u64 = args.free_from_str()?;
1077            reject_rest(args)?;
1078            client.call(Method::SetRuleEnabled(v1::SetRuleEnabled {
1079                id,
1080                enabled: which == "enable",
1081            }))?;
1082            out(&format!("{which}d rule {id}\n"))
1083        }
1084        Some("import-opensnitch") => {
1085            let dry_run = args.contains("--dry-run");
1086            let dir: String = args
1087                .opt_free_from_str()?
1088                .unwrap_or_else(|| "/etc/opensnitchd/rules".to_owned());
1089            reject_rest(args)?;
1090            import_opensnitch(&dir, dry_run, version)
1091        }
1092        Some("import") => {
1093            let path: String = args.free_from_str()?;
1094            reject_rest(args)?;
1095            let toml =
1096                std::fs::read_to_string(&path).map_err(|e| Error::Usage(format!("{path}: {e}")))?;
1097            let folder = Path::new(&path).file_stem().map_or_else(
1098                || "Imported".to_owned(),
1099                |s| s.to_string_lossy().into_owned(),
1100            );
1101            client.call(Method::ImportRules(v1::ImportRules { toml, folder }))?;
1102            out(&format!("imported {path}\n"))
1103        }
1104        Some("export") => {
1105            reject_rest(args)?;
1106            print_file(client.call(Method::ExportRules(v1::ExportRules {}))?)
1107        }
1108        Some(other) => Err(Error::Usage(format!("unknown rules command `{other}`"))),
1109    }
1110}

1112/// `snitchhiker prompts ...`.
1113fn prompts(mut args: pico_args::Arguments, version: &str) -> Result<(), Error> {
1114    use v1::request::Method;
1115    let command = args.subcommand()?;
1116    let mut client = connect(version)?;
1117    match command.as_deref() {
1118        None | Some("list" | "ls") => {
1119            reject_rest(args)?;
1120            match client.call(Method::ListPrompts(v1::ListPrompts {}))? {
1121                v1::response::Result::Prompts(list) => {
1122                    out(&list.prompts.iter().map(format_prompt).collect::<String>())
1123                }
1124                _ => Err(unexpected()),
1125            }
1126        }
1127        Some("watch") => {
1128            reject_rest(args)?;
1129            client.subscribe(v1::Topic::Prompts)?;
1130            loop {
1131                match client.notice()?.body {
1132                    Some(v1::notice::Body::Prompt(prompt)) => out(&format_prompt(&prompt))?,
1133                    Some(v1::notice::Body::PromptResolved(resolved)) => {
1134                        out(&format_resolved(&resolved))?;
1135                    }
1136                    _ => {}
1137                }
1138            }
1139        }
1140        Some("answer") => {
1141            let id: u64 = args.free_from_str()?;
1142            let action = action(&args.free_from_str::<String>()?)?;
1143            let suggestion: u32 = args.opt_value_from_str("--suggestion")?.unwrap_or(0);
1144            let folder: Option<u64> = args.opt_value_from_str("--folder")?;
1145            let duration = match args
1146                .opt_value_from_str::<_, String>("--duration")?
1147                .as_deref()
1148            {
1149                None | Some("forever") => v1::Duration::Forever,
1150                Some("once") => v1::Duration::Once,
1151                Some("restart") => v1::Duration::UntilRestart,
1152                Some("hour") => v1::Duration::Hour,
1153                Some("12h") => v1::Duration::HalfDay,
1154                Some(other) => return Err(Error::Usage(format!("unknown duration `{other}`"))),
1155            };
1156            reject_rest(args)?;
1157            client.call(Method::AnswerPrompt(v1::AnswerPrompt {
1158                id,
1159                action: convert::action(action).into(),
1160                duration: duration.into(),
1161                suggestion,
1162                conditions: None,
1163                folder,
1164            }))?;
1165            out(&format!("answered prompt {id}\n"))
1166        }
1167        Some(other) => Err(Error::Usage(format!("unknown prompts command `{other}`"))),
1168    }
1169}

1180fn spec_from_args(args: &mut pico_args::Arguments) -> Result<RuleSpec, Error> {
1181    let action = action(&args.free_from_str::<String>()?)?;
1182    let name: Option<String> = args.opt_value_from_str("--name")?;
1183    let tier = match args.opt_value_from_str::<_, String>("--tier")?.as_deref() {
1184        None | Some("normal") => snitchhiker_core::rule::Tier::Normal,
1185        Some("override") => snitchhiker_core::rule::Tier::Override,
1186        Some("fallback") => snitchhiker_core::rule::Tier::Fallback,
1187        Some(other) => return Err(Error::Usage(format!("unknown tier `{other}`"))),
1188    };
1189    let conditions = Conditions {
1190        exe: args.values_from_str("--exe")?,
1191        host: args.values_from_str("--host")?,
1192        net: args.values_from_str("--net")?,
1193        port: args.values_from_str("--port")?,
1194        uid: args.values_from_str("--uid")?,
1195        protocol: args.values_from_str("--protocol")?,
1196        context: args
1197            .contains("--any-context")
1198            .then_some(snitchhiker_core::rule::ContextKind::Any),
1199        ..Conditions::default()
1200    };
1201    Ok(RuleSpec {
1202        name: name.unwrap_or_else(|| format!("{action:?}").to_lowercase()),
1203        action,
1204        tier,
1205        enabled: true,
1206        notes: String::new(),
1207        conditions,
1208    })
1209}

1228/// `Apps/curl/`: the folders a rule is in, from the top.
1229fn folder_path(list: &v1::RuleList, folder: Option<u64>) -> String {
1230    let mut names = Vec::new();
1231    let mut next = folder;
1232    for _ in 0..list.folders.len() {
1233        let Some(found) = next.and_then(|id| list.folders.iter().find(|f| f.id == id)) else {
1234            break;
1235        };
1236        names.push(found.name.as_str());
1237        next = found.parent;
1238    }
1239    names.reverse();
1240    names.iter().fold(String::new(), |mut path, name| {
1241        path.push_str(name);
1242        path.push('/');
1243        path
1244    })
1245}

1247fn format_rules(list: &v1::RuleList) -> String {
1248    let mut text = String::new();
1249    for rule in &list.rules {
1250        let spec = rule
1251            .conditions
1252            .as_ref()
1253            .map_or_else(Conditions::default, convert::conditions_from);
1254        let _ = writeln!(
1255            text,
1256            "{:>5}  {:<7} {:<8} {}{}{}{}  {}",
1257            if rule.locked {
1258                "-".to_owned()
1259            } else {
1260                rule.id.to_string()
1261            },
1262            format!("{:?}", rule.action()).to_lowercase(),
1263            format!("{:?}", rule.tier()).to_lowercase(),
1264            if rule.enabled { "" } else { "(disabled) " },
1265            if rule.locked { "(locked) " } else { "" },
1266            folder_path(list, rule.folder),
1267            rule.name,
1268            toml::to_string(&spec)
1269                .unwrap_or_default()
1270                .replace('\n', " "),
1271        );
1272    }
1273    text
1274}

1294fn format_events(events: &[v1::ConnectionEvent]) -> String {
1295    let mut text = String::new();
1296    for event in events {
1297        let address =
1298            convert::address_from(&event.address).map_or_else(String::new, |a| a.to_string());
1299        let _ = writeln!(
1300            text,
1301            "{:>8}  {:<6} uid {:<6} {} -> {}:{} proto {} out {} in {}{}",
1302            event.seq,
1303            format!("{:?}", event.action()).to_lowercase(),
1304            event.uid,
1305            if event.exe.is_empty() {
1306                "?"
1307            } else {
1308                &event.exe
1309            },
1310            if event.host.is_empty() {
1311                &address
1312            } else {
1313                &event.host
1314            },
1315            event.port,
1316            event.protocol,
1317            event.bytes_out,
1318            event.bytes_in,
1319            event
1320                .rule
1321                .map_or_else(String::new, |r| format!(" (rule {r})")),
1322        );
1323    }
1324    text
1325}

1327fn format_explanation(explanation: &v1::Explanation) -> String {
1328    let mut text = format_events(explanation.event.as_slice());
1329    if explanation.rules.is_empty() {
1330        text.push_str("  no rule matches it now: it would be prompted for (or decided by the default action)\n");
1331    }
1332    for (i, rule) in explanation.rules.iter().enumerate() {
1333        let _ = writeln!(
1334            text,
1335            "  {} rule {} {:?}: {} ({:?}, specificity {})",
1336            if i == 0 { "decides" } else { "        " },
1337            rule.id,
1338            rule.name,
1339            format!("{:?}", rule.action()).to_lowercase(),
1340            rule.tier(),
1341            rule.specificity,
1342        );
1343    }
1344    if explanation.partial {
1345        text.push_str(
1346            "  rules on command lines or parent processes aren't checked: the history doesn't keep them\n",
1347        );
1348    }
1349    text
1350}

1352fn format_prompt(prompt: &v1::Prompt) -> String {
1353    let address =
1354        convert::address_from(&prompt.address).map_or_else(String::new, |a| a.to_string());
1355    let mut text = format!(
1356        "prompt {}: {} (uid {}{}) -> {}:{} proto {}, {} waiting{}\n",
1357        prompt.id,
1358        prompt.exe,
1359        prompt.uid,
1360        if prompt.user.is_empty() {
1361            String::new()
1362        } else {
1363            format!(" {}", prompt.user)
1364        },
1365        prompt.hosts.first().unwrap_or(&address),
1366        prompt.port,
1367        prompt.protocol,
1368        prompt.waiting,
1369        if prompt.exe_writable {
1370            " (its executable is writable by a user)"
1371        } else {
1372            ""
1373        },
1374    );
1375    for (index, suggestion) in prompt.suggestions.iter().enumerate() {
1376        let _ = writeln!(
1377            text,
1378            "  {index}: {}{}",
1379            suggestion.label,
1380            if suggestion.broad { " (broad)" } else { "" }
1381        );
1382    }
1383    text
1384}
