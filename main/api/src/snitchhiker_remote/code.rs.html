1//! Pairing codes (decision 0010): `<nameplate>-<word>-<word>-<word>`, to type where a link can't be pasted or scanned.
2//!
3//! The nameplate (1 to 999) names a meeting point: an endpoint whose key is derived from it, which anyone can derive
4//! too. The node waits there, on the LAN (mDNS) and on its relays, for one attempt. Both sides run SPAKE2 with the whole
5//! code as the password: only someone who knows the words derives the same key, and an attacker at the meeting point
6//! (or on the relay) gets one online guess, against 2048³ words, after which the code is spent. The device proves it
7//! derived the key first; only then does the node send its pairing link, sealed with it. The device then pairs as a
8//! link does, with the real node's key proven by TLS.
9

21/// The ALPN of the meeting point.
22pub const CODE_ALPN: &[u8] = b"snitchhiker-code/1";

24/// BIP-39's English word list (MIT): 2048 words, no word a prefix of another's first four letters.
25const WORDS: &str = include_str!("words.txt");
26
27fn words() -> impl Iterator<Item = &'static str> {
28    WORDS.lines().filter(|w| !w.is_empty())
29}

31/// A pairing code.
32#[derive(Debug, Clone, PartialEq, Eq)]
33pub struct Code {
34    pub nameplate: u16,
35    pub words: [String; 3],
36}
37
38impl std::fmt::Display for Code {
39    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
40        let [a, b, c] = &self.words;
41        write!(f, "{}-{a}-{b}-{c}", self.nameplate)
42    }
43}
44
45impl FromStr for Code {
46    type Err = String;
47
48    fn from_str(text: &str) -> Result<Self, String> {
49        let lower = text.trim().to_ascii_lowercase();
50        let mut parts = lower.split(['-', ' ']).filter(|p| !p.is_empty());
51        let nameplate: u16 = parts
52            .next()
53            .and_then(|n| n.parse().ok())
54            .filter(|n| (1..=999).contains(n))
55            .ok_or("a code starts with a number from 1 to 999")?;
56        let mut found = Vec::new();
57        for part in parts {
58            let word = words()
59                .find(|w| *w == part)
60                .ok_or_else(|| format!("`{part}` isn't one of the code's words"))?;
61            found.push(word.to_owned());
62        }
63        let words: [String; 3] = found
64            .try_into()
65            .map_err(|_| "a code is a number and three words".to_owned())?;
66        Ok(Self { nameplate, words })
67    }
68}

70fn random(bytes: &mut [u8]) -> io::Result<()> {
71    use std::io::Read as _;
72    std::fs::File::open("/dev/urandom")?.read_exact(bytes)
73}
74
75impl Code {
76    /// A new code: a random nameplate and three random words (33 bits).
77    pub fn generate() -> io::Result<Self> {
78        let mut bytes = [0_u8; 8];
79        random(&mut bytes)?;
80        let list: Vec<&str> = words().collect();
81        let pick = |pair: [u8; 2]| {
82            let index = usize::from(u16::from_le_bytes(pair) & 0x7ff);
83            list.get(index).copied().unwrap_or("abandon").to_owned()
84        };
85        Ok(Self {
86            nameplate: 1 + u16::from_le_bytes([bytes[0], bytes[1]]) % 999,
87            words: [
88                pick([bytes[2], bytes[3]]),
89                pick([bytes[4], bytes[5]]),
90                pick([bytes[6], bytes[7]]),
91            ],
92        })
93    }
94}

96/// The meeting point's key: anyone may derive it from the nameplate; it is no secret.
97fn meeting_key(nameplate: u16) -> SecretKey {
98    let digest: [u8; 32] =
99        Sha256::digest(format!("snitchhiker pairing meeting point v1: {nameplate}")).into();
100    SecretKey::from_bytes(&digest)
101}
102
103/// The meeting point's id, as a device looks for it.
104fn meeting_id(nameplate: u16) -> PublicKey {
105    meeting_key(nameplate).public()
106}

108fn subkey(key: &[u8], label: &str) -> LessSafeKey {
109    let digest: [u8; 32] = Sha256::new()
110        .chain_update(label.as_bytes())
111        .chain_update(key)
112        .finalize()
113        .into();
114    // A 32-byte key is what ChaCha20-Poly1305 takes: this can't fail.
115    LessSafeKey::new(
116        UnboundKey::new(&CHACHA20_POLY1305, &digest).unwrap_or_else(|_| unreachable!()),
117    )
118}
119
120fn seal(key: &LessSafeKey, mut message: Vec<u8>) -> io::Result<Vec<u8>> {
121    // Each key seals one message: a zero nonce is fine.
122    key.seal_in_place_append_tag(
123        Nonce::assume_unique_for_key([0; 12]),
124        Aad::empty(),
125        &mut message,
126    )
127    .map_err(|_| other("sealing"))?;
128    Ok(message)
129}
130
131fn open(key: &LessSafeKey, mut sealed: Vec<u8>) -> Option<Vec<u8>> {
132    let opened = key
133        .open_in_place(
134            Nonce::assume_unique_for_key([0; 12]),
135            Aad::empty(),
136            &mut sealed,
137        )
138        .ok()?
139        .to_vec();
140    Some(opened)
141}

143async fn write_frame(send: &mut SendStream, bytes: &[u8]) -> io::Result<()> {
144    let length = u32::try_from(bytes.len()).map_err(other)?;
145    send.write_all(&length.to_le_bytes()).await.map_err(other)?;
146    send.write_all(bytes).await.map_err(other)
147}
148
149async fn read_frame(receive: &mut RecvStream) -> io::Result<Vec<u8>> {
150    let mut length = [0; 4];
151    receive.read_exact(&mut length).await.map_err(other)?;
152    let length = u32::from_le_bytes(length) as usize;
153    if length > 4096 {
154        return Err(other("a frame too long"));
155    }
156    let mut bytes = vec![0; length];
157    receive.read_exact(&mut bytes).await.map_err(other)?;
158    Ok(bytes)
159}

161const DEVICE_PROOF: &[u8] = b"snitchhiker: the device knows the code";

163fn spake(code: &Code) -> (Password, Identity, Identity) {
164    (
165        Password::new(code.to_string()),
166        Identity::new(b"snitchhiker device"),
167        Identity::new(b"snitchhiker node"),
168    )
169}

171/// The node's side: one attempt at the meeting point; its outcome (`true`: the device knew the code and was sent the
172/// link), or none when the meeting point closed before anyone came.
173async fn serve_one(endpoint: &Endpoint, code: &Code, link: &str) -> Option<bool> {
174    let incoming = endpoint.accept().await?;
175    Some(attempt(incoming, code, link).await.unwrap_or(false))
176}
177
178async fn attempt(incoming: iroh::endpoint::Incoming, code: &Code, link: &str) -> io::Result<bool> {
179    let connection = incoming.await.map_err(other)?;
180    let (mut send, mut receive) = connection.accept_bi().await.map_err(other)?;
181    let (password, device, node) = spake(code);
182    let (state, message) = Spake2::<Ed25519Group>::start_b(&password, &device, &node);
183    let theirs = read_frame(&mut receive).await?;
184    write_frame(&mut send, &message).await?;
185    let Ok(key) = state.finish(&theirs) else {
186        return Ok(false);
187    };
188    let proof = read_frame(&mut receive).await?;
189    if open(&subkey(&key, "device"), proof).as_deref() != Some(DEVICE_PROOF) {
190        return Ok(false);
191    }
192    write_frame(
193        &mut send,
194        &seal(&subkey(&key, "node"), link.as_bytes().to_vec())?,
195    )
196    .await?;
197    let _ = send.finish();
198    let _ = connection.closed().await;
199    Ok(true)
200}

202/// A code the node offers: its meeting point, open until the one attempt is made (or this is dropped).
203pub struct Offer {
204    endpoint: Endpoint,
205    runtime: tokio::runtime::Runtime,
206}
207
208impl Offer {
209    /// Opens the meeting point of `code`, on the LAN and on `options`' relays; `link` is what a device that knows the
210    /// code is sent. `attempted` is called once if a device makes the attempt, with its outcome (not when the offer
211    /// is dropped unused).
212    pub fn open(
213        code: &Code,
214        options: &Options,
215        link: String,
216        attempted: impl FnOnce(bool) + Send + 'static,
217    ) -> io::Result<Self> {
218        let runtime = runtime("pairing-code")?;
219        let key = crate::Key(meeting_key(code.nameplate));
220        let mut options = options.clone();
221        options.port = 0;
222        let endpoint = runtime.block_on(async {
223            builder(&key, &options)?
224                .alpns(vec![CODE_ALPN.to_vec()])
225                .bind()
226                .await
227                .map_err(other)
228        })?;
229        let serving = endpoint.clone();
230        let code = code.clone();
231        runtime.spawn(async move {
232            if let Some(outcome) = serve_one(&serving, &code, &link).await {
233                attempted(outcome);
234            }
235            serving.close().await;
236        });
237        Ok(Self { endpoint, runtime })
238    }
239}
240
241impl Drop for Offer {
242    fn drop(&mut self) {
243        let endpoint = self.endpoint.clone();
244        self.runtime.block_on(async move { endpoint.close().await });
245    }
246}

248/// The device's exchange at the meeting point `addr`: proves it knows `code`, and opens the node's link.
249async fn exchange(endpoint: &Endpoint, addr: EndpointAddr, code: &Code) -> io::Result<PairLink> {
250    let connection = endpoint.connect(addr, CODE_ALPN).await.map_err(other)?;
251    let (mut send, mut receive) = connection.open_bi().await.map_err(other)?;
252    let (password, device, node) = spake(code);
253    let (state, message) = Spake2::<Ed25519Group>::start_a(&password, &device, &node);
254    write_frame(&mut send, &message).await?;
255    let theirs = read_frame(&mut receive).await?;
256    let shared = state
257        .finish(&theirs)
258        .map_err(|_| other("the pairing exchange failed"))?;
259    write_frame(
260        &mut send,
261        &seal(&subkey(&shared, "device"), DEVICE_PROOF.to_vec())?,
262    )
263    .await?;
264    let wrong = || other("the code is wrong (it is spent now: start pairing again)");
265    let sealed = read_frame(&mut receive).await.map_err(|_| wrong())?;
266    let link = open(&subkey(&shared, "node"), sealed).ok_or_else(wrong)?;
267    connection.close(0_u32.into(), b"paired");
268    String::from_utf8(link)
269        .map_err(other)?
270        .parse::<PairLink>()
271        .map_err(other)
272}

274/// The device's side: finds the meeting point of `code` (on the LAN, or through `options`' relays), proves it knows
275/// the code, and returns the node's pairing link.
276pub fn redeem(code: &Code, options: &Options) -> io::Result<PairLink> {
277    let runtime = runtime("pairing-code")?;
278    let key = crate::Key::generate();
279    runtime.block_on(async {
280        let endpoint = builder(&key, options)?.bind().await.map_err(other)?;
281        let mut addrs = Vec::new();
282        for relay in &options.relays {
283            addrs.push(TransportAddr::Relay(
284                RelayUrl::from_str(relay).map_err(|e| other(format!("relay {relay}: {e}")))?,
285            ));
286        }
287        let addr = EndpointAddr::from_parts(meeting_id(code.nameplate), addrs);
288        let link = tokio::time::timeout(
289            std::time::Duration::from_secs(20),
290            exchange(&endpoint, addr, code),
291        )
292        .await
293        .map_err(|_| {
294            io::Error::new(
295                io::ErrorKind::TimedOut,
296                "no node offers this code nearby or on the relay",
297            )
298        })?;
299        endpoint.close().await;
300        link
301    })
302}

308    #[test]
309    fn codes_round_trip_and_reject_nonsense() {
310        let code = Code::generate().unwrap();
311        assert!((1..=999).contains(&code.nameplate));
312        assert_eq!(code.to_string().parse::<Code>().unwrap(), code);
313        assert_eq!(
314            "7 Orbit VELVET canyon".parse::<Code>().unwrap().to_string(),
315            "7-orbit-velvet-canyon"
316        );
317        assert!("0-orbit-velvet-canyon".parse::<Code>().is_err());
318        assert!("7-orbit-velvet".parse::<Code>().is_err());
319        assert!("7-orbit-velvet-notaword".parse::<Code>().is_err());
320        assert_eq!(words().count(), 2048);
321    }
