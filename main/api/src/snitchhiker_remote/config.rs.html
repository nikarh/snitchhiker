1//! `/etc/snitchhiker/remote.toml`: the node's remote endpoint. What devices may do once connected is the daemon's
2//! business (`daemon.toml`, the devices it paired); this file says only how the node is reached.
3
4use std::path::{Path, PathBuf};
5
6use serde::{Deserialize, Serialize};
7
8pub const DEFAULT_PATH: &str = "/etc/snitchhiker/remote.toml";
9
10#[derive(Debug, Clone, PartialEq, Eq, Deserialize, Serialize)]
11#[serde(default, deny_unknown_fields)]
12pub struct Config {
13    /// The relays (the user's own `snitchhiker-relay`) the node keeps a connection to, so devices reach it behind NAT.
14    /// None: direct connections only.
15    pub relays: Vec<String>,
16    /// The UDP port of the endpoint (0: any free port).
17    pub port: u16,
18    /// Announce the node on the local network (mDNS), so devices there find it without a relay, and pairing codes
19    /// work there.
20    pub lan_discovery: bool,
21    /// The push gateway's endpoint id (a `snitchhiker-relay` with `[push]`, reached through `relays`): it wakes the
22    /// devices with prompts while they are asleep, and tells them when the node goes offline. Empty: none, devices
23    /// learn of prompts only while connected.
24    pub push_gateway: String,
25    /// How long the node may be gone from its push gateway before its devices are told it is offline (a restart
26    /// doesn't make them wonder).
27    pub offline_after_secs: u32,
28    /// The node's key: made the first time, readable by the service's user only. Once sealed (`snitchhiker-remote
29    /// seal-key`), systemd gives it to the service instead, and this file is gone.
30    pub key: PathBuf,
31    /// The daemon's socket for its remote endpoint.
32    pub socket: PathBuf,
33}
34
35impl Default for Config {
36    fn default() -> Self {
37        Self {
38            relays: Vec::new(),
39            port: 0,
40            lan_discovery: true,
41            push_gateway: String::new(),
42            offline_after_secs: 120,
43            key: PathBuf::from("/var/lib/snitchhiker-remote/node.key"),
44            socket: PathBuf::from("/run/snitchhiker/remote.sock"),
45        }
46    }
47}
48
49#[derive(Debug, thiserror::Error)]
50pub enum Error {
51    #[error("reading {path}: {source}")]
52    Read {
53        path: PathBuf,
54        source: std::io::Error,
55    },
56    #[error("{path}: {source}")]
57    Parse {
58        path: PathBuf,
59        source: Box<toml::de::Error>,
60    },
61}
62
63impl Config {
64    /// The configuration in `path`; the defaults when there is no file.
65    pub fn load(path: &Path) -> Result<Self, Error> {
66        let text = match std::fs::read_to_string(path) {
67            Ok(text) => text,
68            Err(error) if error.kind() == std::io::ErrorKind::NotFound => {
69                return Ok(Self::default());
70            }
71            Err(source) => {
72                return Err(Error::Read {
73                    path: path.to_owned(),
74                    source,
75                });
76            }
77        };
78        toml::from_str(&text).map_err(|source| Error::Parse {
79            path: path.to_owned(),
80            source: Box::new(source),
81        })
82    }
83}

85#[cfg(test)]
86mod tests {
87    use super::*;
88
89    #[test]
90    fn the_packaged_file_is_the_defaults() {
91        let packaged: Config =
92            toml::from_str(include_str!("../../../dist/remote/remote.toml")).unwrap();
93        assert_eq!(packaged, Config::default());
94    }
95
96    #[test]
97    fn every_setting_is_documented() {
98        let defaults = toml::to_string(&Config::default()).unwrap();
99        let table: toml::Table = toml::from_str(&defaults).unwrap();
100        let page = include_str!("../../../site/src/content/docs/reference/configuration.mdx");
101        let section = page.split("## remote.toml").nth(1).unwrap_or_default();
102        let missing: Vec<&String> = table
103            .keys()
104            .filter(|key| !section.contains(&format!("### `{key}`")))
105            .collect();
106        assert!(
107            missing.is_empty(),
108            "settings missing from the configuration page: {missing:?}"
109        );
110    }
111
112    #[test]
113    fn unknown_keys_are_refused() {
114        assert!(toml::from_str::<Config>("relay = []").is_err());
115    }
