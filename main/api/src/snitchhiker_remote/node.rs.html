1//! The node's bridge (decision 0016): the iroh endpoint devices connect to, beside the daemon, as an unprivileged user
2//! of its own. It connects to the daemon's remote socket, says who the node is, and hands over each stream a peer
3//! opens, as a file descriptor with the peer's id (which TLS proved): the daemon decides whether the peer is a paired
4//! device, and serves it as it serves a local client. The bridge decides nothing.
5//!
6//! It also does what only the endpoint can: open a stream to the push gateway, offer a pairing code, say where the
7//! node is (for a pairing link).
8

24/// The node's key: the sealed one systemd decrypted for the service (`ImportCredential=`), else the file (made if
25/// there is none).
26fn node_key(config: &Config) -> Result<Key, Error> {
27    if let Some(dir) = std::env::var_os("CREDENTIALS_DIRECTORY") {
28        let sealed = std::path::PathBuf::from(dir).join(KEY_CREDENTIAL);
29        if let Ok(bytes) = std::fs::read(&sealed) {
30            let bytes: [u8; 32] = bytes
31                .try_into()
32                .map_err(|_| Error::Key(format!("{}: not a key", sealed.display())))?;
33            info!("the node's key is the sealed credential");
34            return Ok(Key::from_bytes(&bytes));
35        }
36    }
37    Key::load_or_create(&config.key)
38        .map_err(|e| Error::Key(format!("{}: {e}", config.key.display())))
39}

50fn lock<T>(mutex: &Mutex<T>) -> std::sync::MutexGuard<'_, T> {
51    mutex.lock().unwrap_or_else(PoisonError::into_inner)
52}

54struct Node {
55    daemon: Packets,
56    listener: Listener,
57    config: Config,
58    /// The pairing codes offered, by their pairing's secret.
59    offers: Mutex<HashMap<String, Offer>>,
60}
61
62impl Node {
63    fn send(&self, body: v1::bridge_frame::Body, fd: Option<std::os::fd::BorrowedFd<'_>>) {
64        let frame = v1::BridgeFrame { body: Some(body) };
65        if let Err(error) = self.daemon.send(&frame, fd) {
66            warn!(%error, "writing to the daemon");
67        }
68    }
69
70    fn reply(
71        &self,
72        id: u64,
73        answer: Result<Option<v1::BridgeOffered>, String>,
74        fd: Option<std::os::fd::BorrowedFd<'_>>,
75    ) {
76        let (error, offered) = match answer {
77            Ok(offered) => (String::new(), offered),
78            Err(error) => (error, None),
79        };
80        self.send(
81            v1::bridge_frame::Body::Reply(v1::BridgeReply { id, error, offered }),
82            fd,
83        );
84    }
85
86    fn open_gateway(&self, id: u64) {
87        if self.config.push_gateway.is_empty() {
88            self.reply(id, Err("no push gateway is configured".to_owned()), None);
89            return;
90        }
91        let gateway = NodeAddress {
92            id: self.config.push_gateway.clone(),
93            relays: self.config.relays.clone(),
94            addrs: Vec::new(),
95        };
96        // The gateway is reached through the relays: once the endpoint has one.
97        let _ = self.listener.address();
98        match self.listener.open(&gateway, PUSH_ALPN) {
99            Ok(stream) => self.reply(id, Ok(None), Some(stream.as_fd())),
100            Err(error) => self.reply(id, Err(error.to_string()), None),
101        }
102    }
103
104    fn offer(self: &Arc<Self>, id: u64, offer: &v1::Offer) {
105        let node = self.listener.address();
106        let link = PairLink {
107            node: node.clone(),
108            name: offer.name.clone(),
109            secret: offer.secret.clone(),
110            role: offer.role.clone(),
111        };
112        // The code is offered when it can be; the link works without it.
113        let code = Code::generate().and_then(|code| {
114            let weak = Arc::downgrade(self);
115            let secret = offer.secret.clone();
116            let opened = Offer::open(
117                &code,
118                &options(&self.config),
119                link.to_string(),
120                // On the offer's own runtime: the daemon withdraws it (dropping it here would wait for this task).
121                move |right| {
122                    if let Some(node) = weak.upgrade() {
123                        node.send(
124                            v1::bridge_frame::Body::CodeTried(v1::BridgeCodeTried {
125                                secret,
126                                right,
127                            }),
128                            None,
129                        );
130                    }
131                },
132            )?;
133            lock(&self.offers).insert(offer.secret.clone(), opened);
134            Ok(code)
135        });
136        let code = code.map_or_else(
137            |error| {
138                warn!(%error, "offering a pairing code");
139                String::new()
140            },
141            |code| code.to_string(),
142        );
143        self.reply(
144            id,
145            Ok(Some(v1::BridgeOffered {
146                link: link.to_string(),
147                node_id: node.id,
148                code,
149            })),
150            None,
151        );
152    }
153
154    fn withdraw(&self, secret: &str) {
155        // Dropped with no lock held: closing it waits for its task.
156        let offer = lock(&self.offers).remove(secret);
157        drop(offer);
158    }
159}

161/// The daemon closed the connection rather than serve this process: it serves one user alone, one service at a time,
162/// and says which it refused in its log.
163fn refused(error: impl std::fmt::Display) -> Error {
164    Error::Daemon(format!(
165        "refused ({error}): the daemon serves the snitchhiker-remote user alone, one service at a time"
166    ))
167}

169/// What ended the bridge.
170pub enum Ended {
171    /// SIGTERM or SIGINT: the daemon was told, and had its say.
172    Stopped,
173    /// The daemon closed the connection (it stopped, or refused this bridge).
174    DaemonGone,
175}
176
177/// Runs the bridge until it is stopped or the daemon goes. `stop` is called with a function that ends it gracefully,
178/// for the signal thread.
179pub fn run(config: Config, on_stop: impl FnOnce(Box<dyn FnOnce() + Send>)) -> Result<Ended, Error> {
180    let key = node_key(&config)?;
181    let daemon = Packets::connect(&config.socket)
182        .map_err(|e| Error::Daemon(format!("{}: {e}", config.socket.display())))?;
183    daemon
184        .send(
185            &v1::BridgeFrame {
186                body: Some(v1::bridge_frame::Body::Hello(v1::BridgeHello {
187                    node_id: key.id(),
188                    push_gateway: !config.push_gateway.is_empty(),
189                    offline_after_secs: config.offline_after_secs,
190                })),
191            },
192            None,
193        )
194        .map_err(refused)?;
195    // Served: from now on the daemon lets this process's traffic through undecided.
196    match daemon.receive::<v1::NodeFrame>() {
197        Ok(Some((
198            v1::NodeFrame {
199                body: Some(v1::node_frame::Body::Ready(_)),
200                ..
201            },
202            _,
203        ))) => {}
204        Ok(_) => return Err(refused("the connection closed")),
205        Err(error) => return Err(refused(error)),
206    }
207    let sender = daemon
208        .try_clone()
209        .map_err(|e| Error::Daemon(e.to_string()))?;
210    let listener = Listener::bind(&key, &options(&config), move |accepted| {
211        let kind = if accepted.alpn == PAIR_ALPN {
212            v1::StreamKind::Pairing
213        } else {
214            v1::StreamKind::Device
215        };
216        let frame = v1::BridgeFrame {
217            body: Some(v1::bridge_frame::Body::Accepted(v1::BridgeAccepted {
218                peer: accepted.peer,
219                kind: kind.into(),
220            })),
221        };
222        // The daemon has the stream now; this end's copy closes when `accepted` is dropped.
223        if let Err(error) = sender.send(&frame, Some(accepted.stream.as_fd())) {
224            warn!(%error, "handing a stream to the daemon");
225        }
226    })
227    .map_err(|e| Error::Endpoint(e.to_string()))?;
228    info!(id = %listener.id(), relays = ?config.relays, "remote endpoint started");
229    let node = Arc::new(Node {
230        daemon,
231        listener,
232        config,
233        offers: Mutex::default(),
234    });
235    {
236        let node = Arc::clone(&node);
237        on_stop(Box::new(move || {
238            // The daemon says its last word to the gateway; the read loop ends at its answer.
239            node.send(v1::bridge_frame::Body::Leaving(v1::BridgeLeaving {}), None);
240        }));
241    }
242    let ended = loop {
243        let frame = match node.daemon.receive::<v1::NodeFrame>() {
244            Ok(Some((frame, _))) => frame,
245            Ok(None) => break Ended::DaemonGone,
246            Err(error) => {
247                warn!(%error, "reading from the daemon");
248                break Ended::DaemonGone;
249            }
250        };
251        let id = frame.id;
252        match frame.body {
253            Some(v1::node_frame::Body::OpenGateway(_)) => {
254                let node = Arc::clone(&node);
255                let _ = thread::Builder::new()
256                    .name("open-gateway".to_owned())
257                    .spawn(move || node.open_gateway(id));
258            }
259            Some(v1::node_frame::Body::Offer(offer)) => {
260                let node = Arc::clone(&node);
261                let _ = thread::Builder::new()
262                    .name("offer".to_owned())
263                    .spawn(move || node.offer(id, &offer));
264            }
265            Some(v1::node_frame::Body::Withdraw(withdraw)) => node.withdraw(&withdraw.secret),
266            Some(v1::node_frame::Body::Left(_)) => break Ended::Stopped,
267            Some(v1::node_frame::Body::Ready(_)) | None => {}
268        }
269    };
270    // What the daemon wrote to the gateway last reaches it before the endpoint closes.
271    node.listener.finish_opened(Duration::from_secs(5));
272    lock(&node.offers).clear();
273    Ok(ended)
274}
